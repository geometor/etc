X(115) = CENTER OF KIEPERT HYPERBOLA — Code¶

The circumcircle of the incentral triangle intersects the nine-point circle at 2 points, X(11) and X(115), and X(115) lies on the incentral circle and the cevian circle of every point on the Kiepert hyperbola. Let A’B’C’ be the orthic triangle. The Brocard axes of AB’C’, BC’A’, CA’B’ concur in X(115). Let P be a point on the Brocard circle, and let L be the line tangent to the Brocard circle at P. Let P’ be the trilinear pole of L, and let P” be the isotomic conjugate of P’. As P varies, P” traces an ellipse with center at X(115). (Randy Hutson, July 23, 2015)
"""GEOMETOR construction model for X(115) = CENTER OF KIEPERT HYPERBOLA."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            nine_point_center_of,
            orthocenter_of,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00115(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X(115) = CENTER OF KIEPERT HYPERBOLA on reference triangle ABC."""
            X_115 = eval_trilinear_point(model, A, B, C, 'b*c*(b**(2) - c**(2))**(2)', ID="X(115)", classes=["selected", "target"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X(6)", classes=["center", "incident"])
            X_13 = eval_trilinear_point(model, A, B, C, 'csc(A + pi /3)', ID="X(13)", classes=["center", "incident"])
            central_line(model, X_6, X_13, 6, 13, incident=True)
            X_4 = orthocenter_of(model, A, B, C, ID="X(4)", classes=["center", "incident"])
            X_32 = eval_trilinear_point(model, A, B, C, 'a**(3)', ID="X(32)", classes=["center", "incident"])
            central_line(model, X_4, X_32, 4, 32, incident=True)
            X_5 = nine_point_center_of(model, A, B, C, ID="X(5)", classes=["center", "incident"])
            X_39 = eval_trilinear_point(model, A, B, C, 'a*(b**(2) + c**(2))', ID="X(39)", classes=["center", "incident"])
            central_line(model, X_5, X_39, 5, 39, incident=True)
            return promote_point_to_top(model, X_115)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X(115) on the given reference triangle."""
            model = create_model('X(115) = CENTER OF KIEPERT HYPERBOLA')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00115(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00115.json"))
            write_center_svg(
                model_acute,
                here / "x-00115.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
