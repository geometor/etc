X2047¶

EULER LINE INTERCEPT OF LINE X10X_485

"""GEOMETOR construction model for X_2047 = EULER LINE INTERCEPT OF LINE X_10X_485."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            point_from_barycentrics,
            promote_point_to_top,
            spieker_center_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02047(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2047 = EULER LINE INTERCEPT OF LINE X_10X_485 on reference triangle ABC."""
            X_2047 = eval_trilinear_point(model, A, B, C, '((s - r)*cos(A) + (s + r)*cos(B -C))', ID="X_2047", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center"])
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center"])
            central_line(model, X_2, X_3, 2, 3)
            X_10 = spieker_center_of(model, A, B, C, ID="X_10", classes=["center", "incident"])
            X_485 = eval_trilinear_point(model, A, B, C, 'sec(A -  pi  /4)', ID="X_485", classes=["center", "incident"])
            central_line(model, X_10, X_485, 10, 485, incident=True)
            X_86 = eval_trilinear_point(model, A, B, C, 'b*c/(b + c)', ID="X_86", classes=["center", "incident"])
            X_637 = point_from_barycentrics(model, A, B, C, 0.40143148, 0.29769606, 0.30087247, ID="X_637", classes=["center", "incident"])
            central_line(model, X_86, X_637, 86, 637, incident=True)
            return promote_point_to_top(model, X_2047)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2047 on the given reference triangle."""
            model = create_model('X_2047 = EULER LINE INTERCEPT OF LINE X_10X_485')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02047(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02047.json"))
            write_center_svg(
                model_acute,
                here / "x-02047.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
