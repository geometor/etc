X2017¶

INSIMILICENTER(GALLATLY CIRCLE, BEVAN CIRCLE)

X2017 is the insimilicenter(gallatly circle, Bevan circle).
"""GEOMETOR construction model for X_2017 = INSIMILICENTER(GALLATLY CIRCLE, BEVAN CIRCLE)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02017(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2017 = INSIMILICENTER(GALLATLY CIRCLE, BEVAN CIRCLE) on reference triangle ABC."""
            X_2017 = eval_trilinear_point(model, A, B, C, '(1 + cos(A) - cos(B) - cos(C) + 2*sin(A +  omega  ))', ID="X_2017", classes=["selected", "target"])
            X_39 = eval_trilinear_point(model, A, B, C, 'a*(b**(2) + c**(2))', ID="X_39", classes=["center", "incident"])
            X_40 = eval_trilinear_point(model, A, B, C, 'cos(B) + cos(C) - cos(A) - 1', ID="X_40", classes=["center", "incident"])
            central_line(model, X_39, X_40, 39, 40, incident=True)
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_1690 = eval_trilinear_point(model, A, B, C, 'cos(A +  omega  /2 -  pi  /4)', ID="X_1690", classes=["center", "incident"])
            central_line(model, X_1, X_1690, 1, 1690, incident=True)
            X_165 = eval_trilinear_point(model, A, B, C, 'tan(B/2) + tan(C/2) - tan(A/2)', ID="X_165", classes=["center", "incident"])
            X_1689 = eval_trilinear_point(model, A, B, C, 'cos(A +  omega  /2 +  pi  /4)', ID="X_1689", classes=["center", "incident"])
            central_line(model, X_165, X_1689, 165, 1689, incident=True)
            return promote_point_to_top(model, X_2017)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2017 on the given reference triangle."""
            model = create_model('X_2017 = INSIMILICENTER(GALLATLY CIRCLE, BEVAN CIRCLE)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02017(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02017.json"))
            write_center_svg(
                model_acute,
                here / "x-02017.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
