X330¶

ISOTOMIC CONJUGATE OF X192

X330 is the isotomic conjugate of X192.
"""GEOMETOR construction model for X_330 = ISOTOMIC CONJUGATE OF X_192."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            nagel_point_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00330(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_330 = ISOTOMIC CONJUGATE OF X_192 on reference triangle ABC."""
            X_330 = eval_trilinear_point(model, A, B, C, 'b*c/(a*b + a*c - b*c)', ID="X_330", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_87 = eval_trilinear_point(model, A, B, C, '1/(a*b + a*c - b*c)', ID="X_87", classes=["center", "incident"])
            central_line(model, X_1, X_87, 1, 87, incident=True)
            X_57 = eval_trilinear_point(model, A, B, C, '1/(b + c - a)', ID="X_57", classes=["center", "incident"])
            X_239 = eval_trilinear_point(model, A, B, C, 'b*c*(a**(2) - b*c)', ID="X_239", classes=["center", "incident"])
            central_line(model, X_57, X_239, 57, 239, incident=True)
            X_8 = nagel_point_of(model, A, B, C, ID="X_8", classes=["center", "incident"])
            X_291 = eval_trilinear_point(model, A, B, C, '1/(a**(2) - b*c)', ID="X_291", classes=["center", "incident"])
            central_line(model, X_8, X_291, 8, 291, incident=True)
            return promote_point_to_top(model, X_330)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_330 on the given reference triangle."""
            model = create_model('X_330 = ISOTOMIC CONJUGATE OF X_192')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00330(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00330.json"))
            write_center_svg(
                model_acute,
                here / "x-00330.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
