X332¶

ISOTOMIC CONJUGATE OF X225

X332 is the isotomic conjugate of X225.
"""GEOMETOR construction model for X_332 = ISOTOMIC CONJUGATE OF X_225."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            isotomic_conjugate,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00332(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_332 = ISOTOMIC CONJUGATE OF X_225 on reference triangle ABC."""
            X_225 = eval_trilinear_point(model, A, B, C, '(sec(A))*(cos(B) + cos(C))', ID="X_225", classes=["center"])
            X_332 = isotomic_conjugate(model, A, B, C, X_225, ID="X_332", classes=["selected", "target"])
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center", "incident"])
            X_69 = eval_trilinear_point(model, A, B, C, '(cos(A))/a**(2)', ID="X_69", classes=["center", "incident"])
            central_line(model, X_3, X_69, 3, 69, incident=True)
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_75 = eval_trilinear_point(model, A, B, C, '1/a**(2)', ID="X_75", classes=["center", "incident"])
            central_line(model, X_1, X_75, 1, 75, incident=True)
            X_99 = eval_trilinear_point(model, A, B, C, 'b*c/(b**(2) - c**(2))', ID="X_99", classes=["center", "incident"])
            X_102 = eval_trilinear_point(model, A, B, C, '1/(sin(B)*(sec(A) - sec(B)) + sin(C)*(sec(A) - sec(C)))', ID="X_102", classes=["center", "incident"])
            central_line(model, X_99, X_102, 99, 102, incident=True)
            return promote_point_to_top(model, X_332)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_332 on the given reference triangle."""
            model = create_model('X_332 = ISOTOMIC CONJUGATE OF X_225')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00332(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00332.json"))
            write_center_svg(
                model_acute,
                here / "x-00332.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
