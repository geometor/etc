X(157) = X(6)-OF-TANGENTIAL-TRIANGLE — Code¶

"""GEOMETOR construction model for X(157) = X(6)-OF-TANGENTIAL-TRIANGLE."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            intersect_lines,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00157(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X(157) = X(6)-OF-TANGENTIAL-TRIANGLE on reference triangle ABC."""
            X_3 = circumcenter_of(model, A, B, C, ID="X(3)", classes=["center"])
            X_66 = eval_trilinear_point(model, A, B, C, 'b*c/(b**(4) + c**(4) - a**(4))', ID="X(66)", classes=["center"])
            X_25 = eval_trilinear_point(model, A, B, C, 'sin(A)*tan(A)', ID="X(25)", classes=["center"])
            X_53 = eval_trilinear_point(model, A, B, C, 'tan(A)*cos(B - C)', ID="X(53)", classes=["center"])
            central_line(model, X_3, X_66, 3, 66)
            central_line(model, X_25, X_53, 25, 53)
            X_157 = intersect_lines(model, X_3, X_66, X_25, X_53, ID="X(157)", classes=["selected", "target"])
            X_22 = eval_trilinear_point(model, A, B, C, 'a*(b**(4) + c**(4) - a**(4))', ID="X(22)", classes=["center", "incident"])
            X_183 = eval_trilinear_point(model, A, B, C, 'b**(2)*c**(2)*cos(A- omega )', ID="X(183)", classes=["center", "incident"])
            central_line(model, X_22, X_183, 22, 183, incident=True)
            return promote_point_to_top(model, X_157)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X(157) on the given reference triangle."""
            model = create_model('X(157) = X(6)-OF-TANGENTIAL-TRIANGLE')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00157(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00157.json"))
            write_center_svg(
                model_acute,
                here / "x-00157.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
