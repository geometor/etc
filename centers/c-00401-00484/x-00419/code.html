X419¶

X4-HIRST INVERSE OF X25

X419 is the X4-hirst inverse of X25.
"""GEOMETOR construction model for X_419 = X_4-HIRST INVERSE OF X_25."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00419(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_419 = X_4-HIRST INVERSE OF X_25 on reference triangle ABC."""
            X_419 = eval_barycentric_point(model, A, B, C, '(a**(2) - b*c) * (a**(2) + b*c) * (a**(2) + b**(2) - c**(2)) * (a**(2) - b**(2) + c**(2))', ID="X_419", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center"])
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center"])
            central_line(model, X_2, X_3, 2, 3)
            X_238 = eval_trilinear_point(model, A, B, C, 'a**(2) - b*c', ID="X_238", classes=["center", "incident"])
            X_242 = eval_trilinear_point(model, A, B, C, '((sec(A))*((sin(A))**(2) - sin(B)*sin(C)))', ID="X_242", classes=["center", "incident"])
            central_line(model, X_238, X_242, 238, 242, incident=True)
            X_264 = eval_trilinear_point(model, A, B, C, 'csc(A)*csc(2*A)', ID="X_264", classes=["center", "incident"])
            X_1974 = eval_trilinear_point(model, A, B, C, 'a**(4)*cos(B)*cos(C)', ID="X_1974", classes=["center", "incident"])
            central_line(model, X_264, X_1974, 264, 1974, incident=True)
            return promote_point_to_top(model, X_419)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_419 on the given reference triangle."""
            model = create_model('X_419 = X_4-HIRST INVERSE OF X_25')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00419(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00419.json"))
            write_center_svg(
                model_acute,
                here / "x-00419.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
