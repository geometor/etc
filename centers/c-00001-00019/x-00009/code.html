X9¶

MITTENPUNKT

X9 is the symmedian point of the excentral triangle.
"""Exact symbolic GEOMETOR construction model for ``X_9 = MITTENPUNKT``."""

        from __future__ import annotations

        from pathlib import Path

        import sympy as sp
        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid,
            clawson_point,
            construct_baseline_triangle,
            construct_reference_triangle,
            create_model,
            gergonne_point,
            incenter,
            mittenpunkt,
            orthocenter,
            promote_point_to_top,
            spieker_center,
            symmedian_point,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00009(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct ``X_9 = MITTENPUNKT`` and its classical incident central lines on ``ABC``.

            Constructed elements include:

            - Excentral triangle ``< J_a J_b J_c >``, medial triangle ``< M_a M_b M_c >``,
              and excenter-midpoint lines ``[ J_a M_a ]``, ``[ J_b M_b ]``, ``[ J_c M_c ]``
              concurring at ``X_9``.
            - Incident central lines: Incenter-Symmedian ``{ 1, 6 }``, Centroid-Gergonne
              ``{ 2, 7 }``, and Orthocenter-Spieker-Clawson ``{ 4, 10 }`` (with ``X_4, X_10, X_19``).
            """
            X_9 = mittenpunkt(
                model,
                A,
                B,
                C,
                ID="X_9",
                classes=["selected", "target"],
                draw_symmedians=True,
            )

            X_1 = incenter(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_6 = symmedian_point(model, A, B, C, ID="X_6", classes=["center", "incident"])
            central_line(model, X_1, X_6, 1, 6, incident=True)

            X_2 = centroid(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_7 = gergonne_point(model, A, B, C, ID="X_7", classes=["center", "incident"])
            central_line(model, X_2, X_7, 2, 7, incident=True)

            X_4 = orthocenter(model, A, B, C, ID="X_4", classes=["center", "incident"])
            X_10 = spieker_center(model, A, B, C, ID="X_10", classes=["center", "incident"])
            clawson_point(model, A, B, C, ID="X_19", classes=["center", "incident"])
            central_line(model, X_4, X_10, 4, 10, incident=True)

            return promote_point_to_top(model, X_9)


        def build_model(
            preset: str | tuple[sp.Expr | int, sp.Expr | int, sp.Expr | int] = "heronian",
            show_guides: bool = False,
        ) -> Model:
            """Build an exact symbolic :class:`geometor.model.Model` for ``X_9 = MITTENPUNKT``."""
            model = create_model("X_9 = MITTENPUNKT")
            if isinstance(preset, str):
                A, B, C = construct_reference_triangle(model, preset, show_guides=show_guides)
            else:
                A, B, C = construct_baseline_triangle(model, *preset, show_guides=show_guides)
            construct_x_00009(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            plate_specs = [
                ("heronian", "x-00009.svg", "Heronian (15, 13, 14) | Δ = 84, 45 : 52 : 49"),
                ("right", "x-00009-right.svg", "Right (3, 4, 5) | Δ = 6, 9 : 8 : 5"),
                ("kepler", "x-00009-kepler.svg", "Kepler (1, √φ, φ) | Δ = √φ / 2"),
                ("kimberling", "x-00009-kimberling.svg", "Kimberling (6, 9, 13) | 48 : 45 : 13"),
            ]
            for preset_key, svg_name, sub_text in plate_specs:
                mdl = build_model(preset_key)
                if preset_key == "heronian":
                    mdl.save(str(here / "x-00009.json"))
                write_center_svg(mdl, here / svg_name, subtitle=sub_text)
