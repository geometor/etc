X(674) = CROSSDIFFERENCE OF X(6) AND X(514) — Code¶

As the isogonal conjugate of a point on the circumcircle, X(674) lies on the line at infinity.
"""GEOMETOR construction model for X(674) = CROSSDIFFERENCE OF X(6) AND X(514)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            intersect_lines,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00674(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X(674) = CROSSDIFFERENCE OF X(6) AND X(514) on reference triangle ABC."""
            X_6 = symmedian_point_of(model, A, B, C, ID="X(6)", classes=["center"])
            X_31 = eval_trilinear_point(model, A, B, C, 'a**(2)', ID="X(31)", classes=["center"])
            X_51 = eval_trilinear_point(model, A, B, C, 'a**(2)*cos(B - C)', ID="X(51)", classes=["center"])
            X_210 = eval_trilinear_point(model, A, B, C, '(b + c)*(b + c - a)', ID="X(210)", classes=["center"])
            central_line(model, X_6, X_31, 6, 31)
            central_line(model, X_51, X_210, 51, 210)
            X_674 = intersect_lines(model, X_6, X_31, X_51, X_210, ID="X(674)", classes=["selected", "target"])
            X_30 = eval_trilinear_point(model, A, B, C, 'cos(A) - 2*cos(B)*cos(C)', ID="X(30)", classes=["center", "incident"])
            X_511 = eval_trilinear_point(model, A, B, C, 'cos(A + omega )', ID="X(511)", classes=["center", "incident"])
            central_line(model, X_30, X_511, 30, 511, incident=True)
            return promote_point_to_top(model, X_674)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X(674) on the given reference triangle."""
            model = create_model('X(674) = CROSSDIFFERENCE OF X(6) AND X(514)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00674(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00674.json"))
            write_center_svg(
                model_acute,
                here / "x-00674.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
