X273¶

ISOGONAL CONJUGATE OF X212

X273 is the isogonal conjugate of X212.
"""GEOMETOR construction model for X_273 = ISOGONAL CONJUGATE OF X_212."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            gergonne_point_of,
            isogonal_conjugate,
            orthocenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00273(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_273 = ISOGONAL CONJUGATE OF X_212 on reference triangle ABC."""
            X_212 = eval_trilinear_point(model, A, B, C, '(cos(A))*(1 + cos(A))', ID="X_212", classes=["center"])
            X_273 = isogonal_conjugate(model, A, B, C, X_212, ID="X_273", classes=["selected", "target"])
            X_4 = orthocenter_of(model, A, B, C, ID="X_4", classes=["center", "incident"])
            X_7 = gergonne_point_of(model, A, B, C, ID="X_7", classes=["center", "incident"])
            central_line(model, X_4, X_7, 4, 7, incident=True)
            X_29 = eval_trilinear_point(model, A, B, C, '(sec(A))/(cos(B) + cos(C))', ID="X_29", classes=["center", "incident"])
            X_34 = eval_trilinear_point(model, A, B, C, '1 - sec(A)', ID="X_34", classes=["center", "incident"])
            central_line(model, X_29, X_34, 29, 34, incident=True)
            X_27 = eval_trilinear_point(model, A, B, C, '(sec(A))/(b + c)', ID="X_27", classes=["center", "incident"])
            X_57 = eval_trilinear_point(model, A, B, C, '1/(b + c - a)', ID="X_57", classes=["center", "incident"])
            central_line(model, X_27, X_57, 27, 57, incident=True)
            return promote_point_to_top(model, X_273)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_273 on the given reference triangle."""
            model = create_model('X_273 = ISOGONAL CONJUGATE OF X_212')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00273(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00273.json"))
            write_center_svg(
                model_acute,
                here / "x-00273.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
