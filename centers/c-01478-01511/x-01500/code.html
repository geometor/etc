X1500¶

INSIMILICENTER OF MOSES CIRCLE AND INCIRCLE

The circle having center X39 and radius R tan ω sin 2ω, where R denotes the circumradius of triangle ABC, is here introduced as the Moses circle. It is tangent to the nine-point circle at X115, and its internal and external centers of similitude with the incircle are X1500 and X1015, respectively. (Peter J. C. Moses, 5/29/03)
"""GEOMETOR construction model for X_1500 = INSIMILICENTER OF MOSES CIRCLE AND INCIRCLE."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
            spieker_center_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01500(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1500 = INSIMILICENTER OF MOSES CIRCLE AND INCIRCLE on reference triangle ABC."""
            X_1500 = eval_trilinear_point(model, A, B, C, 'a*(b + c)**(2)', ID="X_1500", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_39 = eval_trilinear_point(model, A, B, C, 'a*(b**(2) + c**(2))', ID="X_39", classes=["center", "incident"])
            central_line(model, X_1, X_39, 1, 39, incident=True)
            X_10 = spieker_center_of(model, A, B, C, ID="X_10", classes=["center", "incident"])
            X_37 = eval_trilinear_point(model, A, B, C, 'b + c', ID="X_37", classes=["center", "incident"])
            central_line(model, X_10, X_37, 10, 37, incident=True)
            X_32 = eval_trilinear_point(model, A, B, C, 'a**(3)', ID="X_32", classes=["center", "incident"])
            X_55 = eval_trilinear_point(model, A, B, C, 'a*(b + c - a)', ID="X_55", classes=["center", "incident"])
            central_line(model, X_32, X_55, 32, 55, incident=True)
            return promote_point_to_top(model, X_1500)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1500 on the given reference triangle."""
            model = create_model('X_1500 = INSIMILICENTER OF MOSES CIRCLE AND INCIRCLE')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01500(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01500.json"))
            write_center_svg(
                model_acute,
                here / "x-01500.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
