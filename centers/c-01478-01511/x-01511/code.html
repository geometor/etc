X1511¶

FERMAT CROSSSUM

X1511 = complement of X265
"""GEOMETOR construction model for X_1511 = FERMAT CROSSSUM."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            circumcenter_of,
            complement,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01511(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1511 = FERMAT CROSSSUM on reference triangle ABC."""
            X_265 = eval_trilinear_point(model, A, B, C, 'sin(2*A)*csc(3*A)', ID="X_265", classes=["center"])
            X_1511 = complement(model, A, B, C, X_265, ID="X_1511", classes=["selected", "target"])
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center", "incident"])
            X_74 = eval_trilinear_point(model, A, B, C, '1/(cos(A) - 2*cos(B)*cos(C))', ID="X_74", classes=["center", "incident"])
            central_line(model, X_3, X_74, 3, 74, incident=True)
            X_30 = eval_trilinear_point(model, A, B, C, 'cos(A) - 2*cos(B)*cos(C)', ID="X_30", classes=["center", "incident"])
            X_113 = eval_trilinear_point(model, A, B, C, 'sin(B)*sin(C)*((sin(C))/(cos(C) - 2*cos(A)*cos(B)) + (sin(B))/(cos(B) - 2*cos(A)*cos(C)))', ID="X_113", classes=["center", "incident"])
            central_line(model, X_30, X_113, 30, 113, incident=True)
            X_125 = eval_trilinear_point(model, A, B, C, 'cos(A)*(sin(B - C))**(2)', ID="X_125", classes=["center", "incident"])
            X_128 = eval_trilinear_point(model, A, B, C, '((sec(A))*(cos(2*B) + cos(2*C))*(1 + 2*cos(2*A))*(cos(2*A) + 2*cos(2*B)*cos(2*C)))', ID="X_128", classes=["center", "incident"])
            central_line(model, X_125, X_128, 125, 128, incident=True)
            return promote_point_to_top(model, X_1511)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1511 on the given reference triangle."""
            model = create_model('X_1511 = FERMAT CROSSSUM')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01511(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01511.json"))
            write_center_svg(
                model_acute,
                here / "x-01511.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
