X1478¶

CENTER OF JOHNSON-YFF CIRCLE

The Johnson Circle Theorem is the fact that if three congruent circles intersect in a point, then the circle passing through the other three intersections is congruent to them. This fourth circle is the Johnson circle of the three given circles. There are three congruent circles each tangent to two sides of triangle ABC. Peter Yff proved that their Johnson circle has center X1478. The circle is here named the Johnson-Yff circle of the triangle.
"""GEOMETOR construction model for X_1478 = CENTER OF JOHNSON-YFF CIRCLE."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            anticomplement,
            central_line,
            centroid_of,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            orthocenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01478(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1478 = CENTER OF JOHNSON-YFF CIRCLE on reference triangle ABC."""
            X_993 = eval_trilinear_point(model, A, B, C, '(a**(3) - a*(b**(2) + c**(2)) - b*c*(b + c))', ID="X_993", classes=["center"])
            X_1478 = anticomplement(model, A, B, C, X_993, ID="X_1478", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_4 = orthocenter_of(model, A, B, C, ID="X_4", classes=["center", "incident"])
            central_line(model, X_1, X_4, 1, 4, incident=True)
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center", "incident"])
            X_12 = eval_trilinear_point(model, A, B, C, '1 + cos(B - C)', ID="X_12", classes=["center", "incident"])
            central_line(model, X_3, X_12, 3, 12, incident=True)
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_36 = eval_trilinear_point(model, A, B, C, '1 - 2*cos(A)', ID="X_36", classes=["center", "incident"])
            central_line(model, X_2, X_36, 2, 36, incident=True)
            return promote_point_to_top(model, X_1478)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1478 on the given reference triangle."""
            model = create_model('X_1478 = CENTER OF JOHNSON-YFF CIRCLE')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01478(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01478.json"))
            write_center_svg(
                model_acute,
                here / "x-01478.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
