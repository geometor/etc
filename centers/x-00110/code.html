X(110) = FOCUS OF KIEPERT PARABOLA — Code¶

X(110) is the center of the Stammler hyperbola, SH, which is the rectangular hyperbola that passes through X(1), X(3), X(6), X(155), X(159), X(195), X(399), X(1498), X(2916), X(2917), X(2918), X(2929), X(2930), X(2931), X(2935), X(2948), X(3511), the excenters, and the vertices of the tangential triangle. SH is the bianticevian conic of X(1) and X(6) and the antipedal-anticevian conic of X(3). SH is the tangential isogonal conjugate of the Euler line, the tangential isotomic conjugate of the van Aubel line, the excentral isogonal conjugate of line X(30)X(40), and the excentral isotomic conjugate of line X(191)X(2938). SH is tangent to the Euler line at X(3) and meets the line at infinity (and the Jerabek hyperbola, other than at X(3) and X(6)) at X(2574) and X(2575). SH is the locus of a point P for which the P-Brocard triangle is perspective to ABC. (Randy Hutson, 9/23/2011, 1/29/2015)
"""GEOMETOR construction model for X(110) = FOCUS OF KIEPERT PARABOLA."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            nine_point_center_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00110(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X(110) = FOCUS OF KIEPERT PARABOLA on reference triangle ABC."""
            X_110 = eval_trilinear_point(model, A, B, C, 'csc(B - C)', ID="X(110)", classes=["selected", "target"])
            X_5 = nine_point_center_of(model, A, B, C, ID="X(5)", classes=["center", "incident"])
            X_49 = eval_trilinear_point(model, A, B, C, 'cos(3*A)', ID="X(49)", classes=["center", "incident"])
            central_line(model, X_5, X_49, 5, 49, incident=True)
            X_1 = incenter_of(model, A, B, C, ID="X(1)", classes=["center", "incident"])
            X_60 = eval_trilinear_point(model, A, B, C, '1/(1 + cos(B - C))', ID="X(60)", classes=["center", "incident"])
            central_line(model, X_1, X_60, 1, 60, incident=True)
            X_3 = circumcenter_of(model, A, B, C, ID="X(3)", classes=["center", "incident"])
            X_74 = eval_trilinear_point(model, A, B, C, '1/(cos(A) - 2*cos(B)*cos(C))', ID="X(74)", classes=["center", "incident"])
            central_line(model, X_3, X_74, 3, 74, incident=True)
            return promote_point_to_top(model, X_110)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X(110) on the given reference triangle."""
            model = create_model('X(110) = FOCUS OF KIEPERT PARABOLA')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00110(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00110.json"))
            write_center_svg(
                model_acute,
                here / "x-00110.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
