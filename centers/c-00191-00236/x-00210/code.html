X210¶

X10-CEVA CONJUGATE OF X37

X210 = X10-Ceva conjugate of X37
"""GEOMETOR construction model for X_210 = X_10-CEVA CONJUGATE OF X_37."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            mittenpunkt_of,
            promote_point_to_top,
            spieker_center_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00210(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_210 = X_10-CEVA CONJUGATE OF X_37 on reference triangle ABC."""
            X_210 = eval_trilinear_point(model, A, B, C, '(b + c)*(b + c - a)', ID="X_210", classes=["selected", "target"])
            X_10 = spieker_center_of(model, A, B, C, ID="X_10", classes=["center", "incident"])
            X_12 = eval_trilinear_point(model, A, B, C, '1 + cos(B - C)', ID="X_12", classes=["center", "incident"])
            central_line(model, X_10, X_12, 10, 12, incident=True)
            X_9 = mittenpunkt_of(model, A, B, C, ID="X_9", classes=["center", "incident"])
            X_55 = eval_trilinear_point(model, A, B, C, 'a*(b + c - a)', ID="X_55", classes=["center", "incident"])
            central_line(model, X_9, X_55, 9, 55, incident=True)
            X_31 = eval_trilinear_point(model, A, B, C, 'a**(2)', ID="X_31", classes=["center", "incident"])
            X_44 = eval_trilinear_point(model, A, B, C, 'b + c - 2*a', ID="X_44", classes=["center", "incident"])
            central_line(model, X_31, X_44, 31, 44, incident=True)
            return promote_point_to_top(model, X_210)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_210 on the given reference triangle."""
            model = create_model('X_210 = X_10-CEVA CONJUGATE OF X_37')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00210(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00210.json"))
            write_center_svg(
                model_acute,
                here / "x-00210.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
