X200¶

X8-CEVA CONJUGATE OF X9

X200 is the {X8,X78}-harmonic conjugate of X1. For a list of harmonic conjugates of X200, click Tables at the top of this page.
"""GEOMETOR construction model for X_200 = X_8-CEVA CONJUGATE OF X_9."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            mittenpunkt_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00200(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_200 = X_8-CEVA CONJUGATE OF X_9 on reference triangle ABC."""
            X_200 = eval_trilinear_point(model, A, B, C, '(cot(A/2))**(2)', ID="X_200", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            central_line(model, X_1, X_2, 1, 2, incident=True)
            X_9 = mittenpunkt_of(model, A, B, C, ID="X_9", classes=["center", "incident"])
            X_55 = eval_trilinear_point(model, A, B, C, 'a*(b + c - a)', ID="X_55", classes=["center", "incident"])
            central_line(model, X_9, X_55, 9, 55, incident=True)
            X_40 = eval_trilinear_point(model, A, B, C, 'cos(B) + cos(C) - cos(A) - 1', ID="X_40", classes=["center", "incident"])
            X_64 = eval_trilinear_point(model, A, B, C, '1/(cos(A) - cos(B)*cos(C))', ID="X_64", classes=["center", "incident"])
            central_line(model, X_40, X_64, 40, 64, incident=True)
            return promote_point_to_top(model, X_200)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_200 on the given reference triangle."""
            model = create_model('X_200 = X_8-CEVA CONJUGATE OF X_9')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00200(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00200.json"))
            write_center_svg(
                model_acute,
                here / "x-00200.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
