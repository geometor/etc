X10-CEVA CONJUGATE OF X65

X227 = X10-Ceva conjugate of X65
"""GEOMETOR construction model for X_227 = X_10-CEVA CONJUGATE OF X_65."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00227(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_227 = X_10-CEVA CONJUGATE OF X_65 on reference triangle ABC."""
            X_227 = eval_trilinear_point(model, A, B, C, '((b + c)*(cos(B) + cos(C) - cos(A) - 1)*tan(A/2))', ID="X_227", classes=["selected", "target"])
            X_12 = eval_trilinear_point(model, A, B, C, '1 + cos(B - C)', ID="X_12", classes=["center", "incident"])
            X_37 = eval_trilinear_point(model, A, B, C, 'b + c', ID="X_37", classes=["center", "incident"])
            central_line(model, X_12, X_37, 12, 37, incident=True)
            X_34 = eval_trilinear_point(model, A, B, C, '1 - sec(A)', ID="X_34", classes=["center", "incident"])
            X_55 = eval_trilinear_point(model, A, B, C, 'a*(b + c - a)', ID="X_55", classes=["center", "incident"])
            central_line(model, X_34, X_55, 34, 55, incident=True)
            X_42 = eval_trilinear_point(model, A, B, C, 'a*(b + c)', ID="X_42", classes=["center", "incident"])
            X_65 = eval_trilinear_point(model, A, B, C, 'cos(B) + cos(C)', ID="X_65", classes=["center", "incident"])
            central_line(model, X_42, X_65, 42, 65, incident=True)
            return promote_point_to_top(model, X_227)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_227 on the given reference triangle."""
            model = create_model('X_227 = X_10-CEVA CONJUGATE OF X_65')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00227(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00227.json"))
            write_center_svg(
                model_acute,
                here / "x-00227.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
