X945¶

ISOGONAL CONJUGATE OF X944

X945 is the isogonal conjugate of X944.
"""GEOMETOR construction model for X_945 = ISOGONAL CONJUGATE OF X_944."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            isogonal_conjugate,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00945(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_945 = ISOGONAL CONJUGATE OF X_944 on reference triangle ABC."""
            X_944 = eval_trilinear_point(model, A, B, C, 'b*c*(3*a**(4) - 2*a**(3)*(b + c) + (b - c)**(2)*(2*a*b + 2*a*c - 2*b*c - b**(2) - c**(2) - 2*a**(2)))', ID="X_944", classes=["center"])
            X_945 = isogonal_conjugate(model, A, B, C, X_944, ID="X_945", classes=["selected", "target"])
            X_78 = eval_trilinear_point(model, A, B, C, '1/(1 - sec(A))', ID="X_78", classes=["center", "incident"])
            X_517 = eval_trilinear_point(model, A, B, C, '-1 + cos(B) + cos(C)', ID="X_517", classes=["center", "incident"])
            central_line(model, X_78, X_517, 78, 517, incident=True)
            X_283 = eval_trilinear_point(model, A, B, C, '(cos(A))/(cos(B) + cos(C))', ID="X_283", classes=["center", "incident"])
            X_859 = eval_trilinear_point(model, A, B, C, '((sin(A) + sin(B))*sin(2*B)*sin(C - A) + (sin(A) + sin(C))*sin(2*C)*sin(B - A))', ID="X_859", classes=["center", "incident"])
            central_line(model, X_283, X_859, 283, 859, incident=True)
            return promote_point_to_top(model, X_945)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_945 on the given reference triangle."""
            model = create_model('X_945 = ISOGONAL CONJUGATE OF X_944')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00945(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00945.json"))
            write_center_svg(
                model_acute,
                here / "x-00945.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
