X90¶

X3-CROSS CONJUGATE OF X1

X90 = isogonal conjugate of X46
"""GEOMETOR construction model for X_90 = X_3-CROSS CONJUGATE OF X_1."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            isogonal_conjugate,
            mittenpunkt_of,
            orthocenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00090(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_90 = X_3-CROSS CONJUGATE OF X_1 on reference triangle ABC."""
            X_46 = eval_trilinear_point(model, A, B, C, 'cos(B) + cos(C) - cos(A)', ID="X_46", classes=["center"])
            X_90 = isogonal_conjugate(model, A, B, C, X_46, ID="X_90", classes=["selected", "target"])
            X_9 = mittenpunkt_of(model, A, B, C, ID="X_9", classes=["center", "incident"])
            X_35 = eval_trilinear_point(model, A, B, C, '1 + 2*cos(A)', ID="X_35", classes=["center", "incident"])
            central_line(model, X_9, X_35, 9, 35, incident=True)
            X_4 = orthocenter_of(model, A, B, C, ID="X_4", classes=["center", "incident"])
            central_line(model, X_4, X_46, 4, 46, incident=True)
            X_33 = eval_trilinear_point(model, A, B, C, '1 + sec(A)', ID="X_33", classes=["center", "incident"])
            X_47 = eval_trilinear_point(model, A, B, C, 'cos(2*A)', ID="X_47", classes=["center", "incident"])
            central_line(model, X_33, X_47, 33, 47, incident=True)
            return promote_point_to_top(model, X_90)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_90 on the given reference triangle."""
            model = create_model('X_90 = X_3-CROSS CONJUGATE OF X_1')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00090(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00090.json"))
            write_center_svg(
                model_acute,
                here / "x-00090.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
