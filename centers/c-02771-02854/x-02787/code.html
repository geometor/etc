X2787¶

ISOGONAL CONJUGATE OF X2703

X2787 is the isogonal conjugate of X2703.
"""GEOMETOR construction model for X_2787 = ISOGONAL CONJUGATE OF X_2703."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            feuerbach_point_of,
            isogonal_conjugate,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02787(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2787 = ISOGONAL CONJUGATE OF X_2703 on reference triangle ABC."""
            X_2703 = eval_trilinear_point(model, A, B, C, '(a/((b - c)*(a**(3) - b**(2)*c - b*c**(2) + a*b*c)))', ID="X_2703", classes=["center"])
            X_2787 = isogonal_conjugate(model, A, B, C, X_2703, ID="X_2787", classes=["selected", "target"])
            X_11 = feuerbach_point_of(model, A, B, C, ID="X_11", classes=["center", "incident"])
            X_115 = eval_trilinear_point(model, A, B, C, 'b*c*(b**(2) - c**(2))**(2)', ID="X_115", classes=["center", "incident"])
            central_line(model, X_11, X_115, 11, 115, incident=True)
            X_99 = eval_trilinear_point(model, A, B, C, 'b*c/(b**(2) - c**(2))', ID="X_99", classes=["center", "incident"])
            X_100 = eval_trilinear_point(model, A, B, C, '1/(b - c)', ID="X_100", classes=["center", "incident"])
            central_line(model, X_99, X_100, 99, 100, incident=True)
            X_98 = eval_trilinear_point(model, A, B, C, 'sec(A +  omega  )', ID="X_98", classes=["center", "incident"])
            X_104 = eval_trilinear_point(model, A, B, C, '1/(-1 + cos(B) + cos(C))', ID="X_104", classes=["center", "incident"])
            central_line(model, X_98, X_104, 98, 104, incident=True)
            return promote_point_to_top(model, X_2787)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2787 on the given reference triangle."""
            model = create_model('X_2787 = ISOGONAL CONJUGATE OF X_2703')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02787(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02787.json"))
            write_center_svg(
                model_acute,
                here / "x-02787.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
