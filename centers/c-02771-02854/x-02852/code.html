X2852¶

ISOGONAL CONJUGATE OF X2768

X2852 is the isogonal conjugate of X2768.
"""GEOMETOR construction model for X_2852 = ISOGONAL CONJUGATE OF X_2768."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            isogonal_conjugate,
            point_from_barycentrics,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02852(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2852 = ISOGONAL CONJUGATE OF X_2768 on reference triangle ABC."""
            X_2768 = point_from_barycentrics(model, A, B, C, 0.86924978, -0.20718284, 0.33793306, ID="X_2768", classes=["center"])
            X_2852 = isogonal_conjugate(model, A, B, C, X_2768, ID="X_2852", classes=["selected", "target"])
            X_109 = eval_trilinear_point(model, A, B, C, 'a/(cos(B) - cos(C))', ID="X_109", classes=["center", "incident"])
            X_111 = eval_trilinear_point(model, A, B, C, 'a/(2*a**(2) - b**(2) - c**(2))', ID="X_111", classes=["center", "incident"])
            central_line(model, X_109, X_111, 109, 111, incident=True)
            X_124 = eval_trilinear_point(model, A, B, C, '(b*c*(b + c - a)*(b - c)**(2)*((b + c)*(b**(2) + c**(2) - a**(2) - b*c) + a*b*c))', ID="X_124", classes=["center", "incident"])
            X_126 = eval_trilinear_point(model, A, B, C, '(b*c*(2*a**(2) - b**(2) - c**(2))*(b**(4) + c**(4) + a**(2)*(b**(2) + c**(2)) - 4*b**(2)*c**(2)))', ID="X_126", classes=["center", "incident"])
            central_line(model, X_124, X_126, 124, 126, incident=True)
            X_30 = eval_trilinear_point(model, A, B, C, 'cos(A) - 2*cos(B)*cos(C)', ID="X_30", classes=["center", "incident"])
            X_511 = eval_trilinear_point(model, A, B, C, 'cos(A +  omega  )', ID="X_511", classes=["center", "incident"])
            central_line(model, X_30, X_511, 30, 511, incident=True)
            return promote_point_to_top(model, X_2852)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2852 on the given reference triangle."""
            model = create_model('X_2852 = ISOGONAL CONJUGATE OF X_2768')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02852(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02852.json"))
            write_center_svg(
                model_acute,
                here / "x-02852.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
