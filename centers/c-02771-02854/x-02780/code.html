X2780¶

ISOGONAL CONJUGATE OF X2696

As the isogonal conjugate of a point on the circumcircle, X2780 lies on the line at infinity.
"""GEOMETOR construction model for X_2780 = ISOGONAL CONJUGATE OF X_2696."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            isogonal_conjugate,
            point_from_barycentrics,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02780(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2780 = ISOGONAL CONJUGATE OF X_2696 on reference triangle ABC."""
            X_2696 = point_from_barycentrics(model, A, B, C, 0.11096588, 1.0784607, -0.1894266, ID="X_2696", classes=["center"])
            X_2780 = isogonal_conjugate(model, A, B, C, X_2696, ID="X_2780", classes=["selected", "target"])
            X_74 = eval_trilinear_point(model, A, B, C, '1/(cos(A) - 2*cos(B)*cos(C))', ID="X_74", classes=["center", "incident"])
            X_111 = eval_trilinear_point(model, A, B, C, 'a/(2*a**(2) - b**(2) - c**(2))', ID="X_111", classes=["center", "incident"])
            central_line(model, X_74, X_111, 74, 111, incident=True)
            X_113 = eval_trilinear_point(model, A, B, C, 'sin(B)*sin(C)*((sin(C))/(cos(C) - 2*cos(A)*cos(B)) + (sin(B))/(cos(B) - 2*cos(A)*cos(C)))', ID="X_113", classes=["center", "incident"])
            X_126 = eval_trilinear_point(model, A, B, C, '(b*c*(2*a**(2) - b**(2) - c**(2))*(b**(4) + c**(4) + a**(2)*(b**(2) + c**(2)) - 4*b**(2)*c**(2)))', ID="X_126", classes=["center", "incident"])
            central_line(model, X_113, X_126, 113, 126, incident=True)
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center", "incident"])
            X_351 = eval_trilinear_point(model, A, B, C, '(a*(b**(2) - c**(2))*(b**(2) + c**(2) - 2*a**(2)))', ID="X_351", classes=["center", "incident"])
            central_line(model, X_3, X_351, 3, 351, incident=True)
            return promote_point_to_top(model, X_2780)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2780 on the given reference triangle."""
            model = create_model('X_2780 = ISOGONAL CONJUGATE OF X_2696')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02780(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02780.json"))
            write_center_svg(
                model_acute,
                here / "x-02780.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
