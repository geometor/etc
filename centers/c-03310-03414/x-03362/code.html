X3362¶

ISOGONAL CONJUGATE OF X1745

X3362 is the isogonal conjugate of X1745.
"""GEOMETOR construction model for X_3362 = ISOGONAL CONJUGATE OF X_1745."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            isogonal_conjugate,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03362(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3362 = ISOGONAL CONJUGATE OF X_1745 on reference triangle ABC."""
            X_1745 = eval_trilinear_point(model, A, B, C, '(sec(B) + sec(C) - sec(A))', ID="X_1745", classes=["center"])
            X_3362 = isogonal_conjugate(model, A, B, C, X_1745, ID="X_3362", classes=["selected", "target"])
            X_46 = eval_trilinear_point(model, A, B, C, 'cos(B) + cos(C) - cos(A)', ID="X_46", classes=["center", "incident"])
            X_296 = eval_trilinear_point(model, A, B, C, '(cos(A))/((cos(A))**(2) - cos(B)*cos(C))', ID="X_296", classes=["center", "incident"])
            central_line(model, X_46, X_296, 46, 296, incident=True)
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_1075 = eval_trilinear_point(model, A, B, C, '(sec(A))*((sec(B))**(2) + (sec(C))**(2) - (sec(A))**(2))', ID="X_1075", classes=["center", "incident"])
            central_line(model, X_1, X_1075, 1, 1075, incident=True)
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center", "incident"])
            central_line(model, X_3, X_1745, 3, 1745, incident=True)
            return promote_point_to_top(model, X_3362)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3362 on the given reference triangle."""
            model = create_model('X_3362 = ISOGONAL CONJUGATE OF X_1745')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03362(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03362.json"))
            write_center_svg(
                model_acute,
                here / "x-03362.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
