X3391¶

SEC(A - π/12) POINT

X3391 and related points occupy Bernard Gibert’s Table 38. See the note at X3364.
"""GEOMETOR construction model for X_3391 = SEC(A - π/12) POINT."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            isogonal_conjugate,
            nine_point_center_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03391(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3391 = SEC(A - π/12) POINT on reference triangle ABC."""
            X_3389 = eval_trilinear_point(model, A, B, C, 'cos(A -  pi  /12)', ID="X_3389", classes=["center"])
            X_3391 = isogonal_conjugate(model, A, B, C, X_3389, ID="X_3391", classes=["selected", "target"])
            X_5 = nine_point_center_of(model, A, B, C, ID="X_5", classes=["center", "incident"])
            X_16 = eval_trilinear_point(model, A, B, C, 'sin(A-  pi  /3)', ID="X_16", classes=["center", "incident"])
            central_line(model, X_5, X_16, 5, 16, incident=True)
            X_13 = eval_trilinear_point(model, A, B, C, 'csc(A +  pi  /3)', ID="X_13", classes=["center", "incident"])
            X_371 = eval_trilinear_point(model, A, B, C, 'cos(A -  pi  /4)', ID="X_371", classes=["center", "incident"])
            central_line(model, X_13, X_371, 13, 371, incident=True)
            X_18 = eval_trilinear_point(model, A, B, C, 'csc(A -  pi  /6)', ID="X_18", classes=["center", "incident"])
            X_372 = eval_trilinear_point(model, A, B, C, 'cos(A +  pi  /4)', ID="X_372", classes=["center", "incident"])
            central_line(model, X_18, X_372, 18, 372, incident=True)
            return promote_point_to_top(model, X_3391)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3391 on the given reference triangle."""
            model = create_model('X_3391 = SEC(A - π/12) POINT')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03391(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03391.json"))
            write_center_svg(
                model_acute,
                here / "x-03391.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
