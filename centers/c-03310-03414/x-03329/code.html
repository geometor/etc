X3329¶

INTERSECTION OF LINES X2X_6 AND X39X_83

X3229 = crossdifference of PU(148)
"""GEOMETOR construction model for X_3329 = INTERSECTION OF LINES X_2X_6 AND X_39X_83."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            nine_point_center_of,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03329(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3329 = INTERSECTION OF LINES X_2X_6 AND X_39X_83 on reference triangle ABC."""
            X_3329 = eval_trilinear_point(model, A, B, C, '(b*c*(a**(4) + 2*a**(2)*b**(2) + 2*a**(2)*c**(2) + b**(2)*c**(2)))', ID="X_3329", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            central_line(model, X_2, X_6, 2, 6, incident=True)
            X_39 = eval_trilinear_point(model, A, B, C, 'a*(b**(2) + c**(2))', ID="X_39", classes=["center", "incident"])
            X_83 = eval_trilinear_point(model, A, B, C, 'b*c/(b**(2) + c**(2))', ID="X_83", classes=["center", "incident"])
            central_line(model, X_39, X_83, 39, 83, incident=True)
            X_5 = nine_point_center_of(model, A, B, C, ID="X_5", classes=["center", "incident"])
            X_147 = eval_trilinear_point(model, A, B, C, 'b*c*(a**(8) + (b**(2) + c**(2))*a**(6) - (2*b**(4) + 3*b**(2)*c**(2) + 2*c**(4))*a**(4) + (b**(6) + b**(4)*c**(2) + b**(2)*c**(4) + c**(6))*a**(2) - b**(8) + b**(6)*c**(2) + b**(2)*c**(6) - c**(8))', ID="X_147", classes=["center", "incident"])
            central_line(model, X_5, X_147, 5, 147, incident=True)
            return promote_point_to_top(model, X_3329)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3329 on the given reference triangle."""
            model = create_model('X_3329 = INTERSECTION OF LINES X_2X_6 AND X_39X_83')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03329(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03329.json"))
            write_center_svg(
                model_acute,
                here / "x-03329.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
