X3402¶

TAN(A - ω) POINT

X3402 and related points occupy Bernard Gibert’s Table 38. See the note at X3364.
"""GEOMETOR construction model for X_3402 = TAN(A - ω) POINT."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03402(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3402 = TAN(A - ω) POINT on reference triangle ABC."""
            X_3402 = eval_trilinear_point(model, A, B, C, '(tan(A -  omega  ))', ID="X_3402", classes=["selected", "target"])
            X_42 = eval_trilinear_point(model, A, B, C, 'a*(b + c)', ID="X_42", classes=["center", "incident"])
            X_263 = eval_trilinear_point(model, A, B, C, 'a**(2)*sec(A -  omega  )', ID="X_263", classes=["center", "incident"])
            central_line(model, X_42, X_263, 42, 263, incident=True)
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_1755 = eval_trilinear_point(model, A, B, C, 'a*cos(A +  omega  )', ID="X_1755", classes=["center", "incident"])
            central_line(model, X_1, X_1755, 1, 1755, incident=True)
            X_1923 = eval_trilinear_point(model, A, B, C, 'a**(4)*(b**(2) + c**(2))', ID="X_1923", classes=["center", "incident"])
            X_1973 = eval_trilinear_point(model, A, B, C, 'a**(3)*cos(B)*cos(C)', ID="X_1973", classes=["center", "incident"])
            central_line(model, X_1923, X_1973, 1923, 1973, incident=True)
            return promote_point_to_top(model, X_3402)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3402 on the given reference triangle."""
            model = create_model('X_3402 = TAN(A - ω) POINT')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03402(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03402.json"))
            write_center_svg(
                model_acute,
                here / "x-03402.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
