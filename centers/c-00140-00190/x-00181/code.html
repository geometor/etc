X181¶

APOLLONIUS POINT

Let O(A),O(B),O(C) be the excircles. Apollonius’s Problem includes the construction of the circle O tangent to the three excircles and encompassing them. (The circle is called the Apollonius circle.) Let A' = O∩O(A), B'=O∩O(B), C'=O∩O(C). The lines AA',BB',CC' concur in X181. Yff derived trilinears in 1992.
"""GEOMETOR construction model for X_181 = APOLLONIUS POINT."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            promote_point_to_top,
            spieker_center_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00181(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_181 = APOLLONIUS POINT on reference triangle ABC."""
            X_181 = eval_trilinear_point(model, A, B, C, 'a*(b + c)**(2)/(b + c - a)', ID="X_181", classes=["selected", "target"])
            X_10 = spieker_center_of(model, A, B, C, ID="X_10", classes=["center", "incident"])
            X_12 = eval_trilinear_point(model, A, B, C, '1 + cos(B - C)', ID="X_12", classes=["center", "incident"])
            central_line(model, X_10, X_12, 10, 12, incident=True)
            X_31 = eval_trilinear_point(model, A, B, C, 'a**(2)', ID="X_31", classes=["center", "incident"])
            X_51 = eval_trilinear_point(model, A, B, C, 'a**(2)*cos(B - C)', ID="X_51", classes=["center", "incident"])
            central_line(model, X_31, X_51, 31, 51, incident=True)
            X_43 = eval_trilinear_point(model, A, B, C, 'a*b + a*c - b*c', ID="X_43", classes=["center", "incident"])
            X_57 = eval_trilinear_point(model, A, B, C, '1/(b + c - a)', ID="X_57", classes=["center", "incident"])
            central_line(model, X_43, X_57, 43, 57, incident=True)
            return promote_point_to_top(model, X_181)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_181 on the given reference triangle."""
            model = create_model('X_181 = APOLLONIUS POINT')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00181(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00181.json"))
            write_center_svg(
                model_acute,
                here / "x-00181.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
