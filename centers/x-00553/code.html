X(553) = POINT MAIA II — Code¶

X(553) = isogonal conjugate of X(33635)
"""GEOMETOR construction model for X(553) = POINT MAIA II."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            gergonne_point_of,
            incenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00553(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X(553) = POINT MAIA II on reference triangle ABC."""
            X_553 = eval_trilinear_point(model, A, B, C, 'b*c*(2*a + b + c)/(b + c - a)', ID="X(553)", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X(2)", classes=["center", "incident"])
            X_7 = gergonne_point_of(model, A, B, C, ID="X(7)", classes=["center", "incident"])
            central_line(model, X_2, X_7, 2, 7, incident=True)
            X_1 = incenter_of(model, A, B, C, ID="X(1)", classes=["center", "incident"])
            X_376 = eval_trilinear_point(model, A, B, C, '5*cos(A) - cos(B - C)', ID="X(376)", classes=["center", "incident"])
            central_line(model, X_1, X_376, 1, 376, incident=True)
            X_65 = eval_trilinear_point(model, A, B, C, 'cos(B) + cos(C)', ID="X(65)", classes=["center", "incident"])
            X_519 = eval_trilinear_point(model, A, B, C, '(2*a - b - c)/a', ID="X(519)", classes=["center", "incident"])
            central_line(model, X_65, X_519, 65, 519, incident=True)
            return promote_point_to_top(model, X_553)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X(553) on the given reference triangle."""
            model = create_model('X(553) = POINT MAIA II')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00553(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00553.json"))
            write_center_svg(
                model_acute,
                here / "x-00553.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
