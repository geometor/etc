X238¶

X1-LINE CONJUGATE OF X37

X238 = X1-line conjugate of X37
"""GEOMETOR construction model for X_238 = X_1-LINE CONJUGATE OF X_37."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
            spieker_center_of,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00238(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_238 = X_1-LINE CONJUGATE OF X_37 on reference triangle ABC."""
            X_238 = eval_trilinear_point(model, A, B, C, 'a**(2) - b*c', ID="X_238", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            central_line(model, X_1, X_6, 1, 6, incident=True)
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_31 = eval_trilinear_point(model, A, B, C, 'a**(2)', ID="X_31", classes=["center", "incident"])
            central_line(model, X_2, X_31, 2, 31, incident=True)
            X_10 = spieker_center_of(model, A, B, C, ID="X_10", classes=["center", "incident"])
            X_82 = eval_trilinear_point(model, A, B, C, '1/(b**(2) + c**(2))', ID="X_82", classes=["center", "incident"])
            central_line(model, X_10, X_82, 10, 82, incident=True)
            return promote_point_to_top(model, X_238)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_238 on the given reference triangle."""
            model = create_model('X_238 = X_1-LINE CONJUGATE OF X_37')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00238(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00238.json"))
            write_center_svg(
                model_acute,
                here / "x-00238.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
