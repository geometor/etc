X1813¶

INVERSE MIMOSA TRANSFORM OF X650

X1813 = X219-cross conjugate of X59
"""GEOMETOR construction model for X_1813 = INVERSE MIMOSA TRANSFORM OF X_650."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01813(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1813 = INVERSE MIMOSA TRANSFORM OF X_650 on reference triangle ABC."""
            X_1813 = eval_trilinear_point(model, A, B, C, '(cos(A))/(cos(B) - cos(C))', ID="X_1813", classes=["selected", "target"])
            X_48 = eval_trilinear_point(model, A, B, C, 'tan(B) + tan(C)', ID="X_48", classes=["center", "incident"])
            X_77 = eval_trilinear_point(model, A, B, C, '1/(1 + sec(A))', ID="X_77", classes=["center", "incident"])
            central_line(model, X_48, X_77, 48, 77, incident=True)
            X_109 = eval_trilinear_point(model, A, B, C, 'a/(cos(B) - cos(C))', ID="X_109", classes=["center", "incident"])
            X_110 = eval_trilinear_point(model, A, B, C, 'csc(B - C)', ID="X_110", classes=["center", "incident"])
            central_line(model, X_109, X_110, 109, 110, incident=True)
            X_283 = eval_trilinear_point(model, A, B, C, '(cos(A))/(cos(B) + cos(C))', ID="X_283", classes=["center", "incident"])
            X_296 = eval_trilinear_point(model, A, B, C, '(cos(A))/((cos(A))**(2) - cos(B)*cos(C))', ID="X_296", classes=["center", "incident"])
            central_line(model, X_283, X_296, 283, 296, incident=True)
            return promote_point_to_top(model, X_1813)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1813 on the given reference triangle."""
            model = create_model('X_1813 = INVERSE MIMOSA TRANSFORM OF X_650')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01813(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01813.json"))
            write_center_svg(
                model_acute,
                here / "x-01813.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
