X1801¶

INVERSE MIMOSA TRANSFORM OF X159

"""GEOMETOR construction model for X_1801 = INVERSE MIMOSA TRANSFORM OF X_159."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            intersect_lines,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01801(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1801 = INVERSE MIMOSA TRANSFORM OF X_159 on reference triangle ABC."""
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center"])
            X_49 = eval_trilinear_point(model, A, B, C, 'cos(3*A)', ID="X_49", classes=["center"])
            X_58 = eval_trilinear_point(model, A, B, C, 'a/(b + c)', ID="X_58", classes=["center"])
            X_78 = eval_trilinear_point(model, A, B, C, '1/(1 - sec(A))', ID="X_78", classes=["center"])
            central_line(model, X_3, X_49, 3, 49)
            central_line(model, X_58, X_78, 58, 78)
            X_1801 = intersect_lines(model, X_3, X_49, X_58, X_78, ID="X_1801", classes=["selected", "target"])
            X_171 = eval_trilinear_point(model, A, B, C, 'a**(2) + b*c', ID="X_171", classes=["center", "incident"])
            X_306 = eval_trilinear_point(model, A, B, C, '(b**(2)*c**(2))*(b + c)*cos(A)', ID="X_306", classes=["center", "incident"])
            central_line(model, X_171, X_306, 171, 306, incident=True)
            return promote_point_to_top(model, X_1801)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1801 on the given reference triangle."""
            model = create_model('X_1801 = INVERSE MIMOSA TRANSFORM OF X_159')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01801(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01801.json"))
            write_center_svg(
                model_acute,
                here / "x-01801.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
