X1800¶

INVERSE MIMOSA TRANSFORM OF X155

X1800 = X21-Ceva conjugate of X283
"""GEOMETOR construction model for X_1800 = INVERSE MIMOSA TRANSFORM OF X_155."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            ceva_conjugate,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01800(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1800 = INVERSE MIMOSA TRANSFORM OF X_155 on reference triangle ABC."""
            X_21 = eval_trilinear_point(model, A, B, C, '1/(cos(B) + cos(C))', ID="X_21", classes=["center"])
            X_283 = eval_trilinear_point(model, A, B, C, '(cos(A))/(cos(B) + cos(C))', ID="X_283", classes=["center"])
            X_1800 = ceva_conjugate(model, A, B, C, X_21, X_283, ID="X_1800", classes=["selected", "target"])
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center", "incident"])
            X_49 = eval_trilinear_point(model, A, B, C, 'cos(3*A)', ID="X_49", classes=["center", "incident"])
            central_line(model, X_3, X_49, 3, 49, incident=True)
            X_90 = eval_trilinear_point(model, A, B, C, '1/(cos(B) + cos(C) - cos(A))', ID="X_90", classes=["center", "incident"])
            central_line(model, X_21, X_90, 21, 90, incident=True)
            X_46 = eval_trilinear_point(model, A, B, C, 'cos(B) + cos(C) - cos(A)', ID="X_46", classes=["center", "incident"])
            X_453 = eval_trilinear_point(model, A, B, C, '((cos(B) + cos(C) - cos(A))**(2)/(cos(B) + cos(C)))', ID="X_453", classes=["center", "incident"])
            central_line(model, X_46, X_453, 46, 453, incident=True)
            return promote_point_to_top(model, X_1800)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1800 on the given reference triangle."""
            model = create_model('X_1800 = INVERSE MIMOSA TRANSFORM OF X_155')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01800(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01800.json"))
            write_center_svg(
                model_acute,
                here / "x-01800.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
