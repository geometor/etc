X1797¶

INVERSE MIMOSA TRANSFORM OF X44

X1797 = X903-Ceva conjugate of X106
"""GEOMETOR construction model for X_1797 = INVERSE MIMOSA TRANSFORM OF X_44."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            ceva_conjugate,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01797(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1797 = INVERSE MIMOSA TRANSFORM OF X_44 on reference triangle ABC."""
            X_903 = eval_trilinear_point(model, A, B, C, 'b*c/(b + c - 2*a)', ID="X_903", classes=["center"])
            X_106 = eval_trilinear_point(model, A, B, C, 'a/(2*a - b - c)', ID="X_106", classes=["center"])
            X_1797 = ceva_conjugate(model, A, B, C, X_903, X_106, ID="X_1797", classes=["selected", "target"])
            X_57 = eval_trilinear_point(model, A, B, C, '1/(b + c - a)', ID="X_57", classes=["center", "incident"])
            X_88 = eval_trilinear_point(model, A, B, C, '1/(b + c - 2*a)', ID="X_88", classes=["center", "incident"])
            central_line(model, X_57, X_88, 57, 88, incident=True)
            X_58 = eval_trilinear_point(model, A, B, C, 'a/(b + c)', ID="X_58", classes=["center", "incident"])
            central_line(model, X_58, X_106, 58, 106, incident=True)
            X_27 = eval_trilinear_point(model, A, B, C, '(sec(A))/(b + c)', ID="X_27", classes=["center", "incident"])
            X_648 = eval_trilinear_point(model, A, B, C, '1/(a*(b**(2) - c**(2))*(b**(2) + c**(2) - a**(2)))', ID="X_648", classes=["center", "incident"])
            central_line(model, X_27, X_648, 27, 648, incident=True)
            return promote_point_to_top(model, X_1797)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1797 on the given reference triangle."""
            model = create_model('X_1797 = INVERSE MIMOSA TRANSFORM OF X_44')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01797(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01797.json"))
            write_center_svg(
                model_acute,
                here / "x-01797.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
