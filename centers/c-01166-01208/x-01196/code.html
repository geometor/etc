X1196¶

3rd SARAGOSSA POINT OF X25

X1196 = complement of X305
"""GEOMETOR construction model for X_1196 = 3rd SARAGOSSA POINT OF X_25."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            complement,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01196(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1196 = 3rd SARAGOSSA POINT OF X_25 on reference triangle ABC."""
            X_305 = eval_trilinear_point(model, A, B, C, 'b**(4)*c**(4)*cos(A)', ID="X_305", classes=["center"])
            X_1196 = complement(model, A, B, C, X_305, ID="X_1196", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_39 = eval_trilinear_point(model, A, B, C, 'a*(b**(2) + c**(2))', ID="X_39", classes=["center", "incident"])
            central_line(model, X_2, X_39, 2, 39, incident=True)
            X_25 = eval_trilinear_point(model, A, B, C, 'sin(A)*tan(A)', ID="X_25", classes=["center", "incident"])
            X_32 = eval_trilinear_point(model, A, B, C, 'a**(3)', ID="X_32", classes=["center", "incident"])
            central_line(model, X_25, X_32, 25, 32, incident=True)
            X_22 = eval_trilinear_point(model, A, B, C, 'a*(b**(4) + c**(4) - a**(4))', ID="X_22", classes=["center", "incident"])
            X_187 = eval_trilinear_point(model, A, B, C, 'a*(2*a**(2) - b**(2) - c**(2))', ID="X_187", classes=["center", "incident"])
            central_line(model, X_22, X_187, 22, 187, incident=True)
            return promote_point_to_top(model, X_1196)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1196 on the given reference triangle."""
            model = create_model('X_1196 = 3rd SARAGOSSA POINT OF X_25')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01196(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01196.json"))
            write_center_svg(
                model_acute,
                here / "x-01196.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
