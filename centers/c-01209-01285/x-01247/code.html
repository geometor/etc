X1247¶

ISOGONAL CONJUGATE OF X1046

X1247 is the isogonal conjugate of X1046.
"""GEOMETOR construction model for X_1247 = ISOGONAL CONJUGATE OF X_1046."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            incenter_of,
            isogonal_conjugate,
            point_from_barycentrics,
            promote_point_to_top,
            spieker_center_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01247(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1247 = ISOGONAL CONJUGATE OF X_1046 on reference triangle ABC."""
            X_1046 = point_from_barycentrics(model, A, B, C, 0.13236391, 0.3064877, 0.5611484, ID="X_1046", classes=["center"])
            X_1247 = isogonal_conjugate(model, A, B, C, X_1046, ID="X_1247", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_409 = eval_barycentric_point(model, A, B, C, 'a*(a + b) * (a + c) * (a**(4) - a**(3)*b + a*b**(3) - b**(4) - a**(3)*c + a**(2)*b*c + a*b**(2)*c + b**(3)*c + a*b*c**(2) + 4*b**(2)*c**(2) + a*c**(3) + b*c**(3) - c**(4))', ID="X_409", classes=["center", "incident"])
            central_line(model, X_1, X_409, 1, 409, incident=True)
            X_158 = eval_trilinear_point(model, A, B, C, '(sec(A))**(2)', ID="X_158", classes=["center", "incident"])
            X_415 = eval_barycentric_point(model, A, B, C, '(a + b) * (a + c) * (a**(2) + b**(2) - c**(2)) * (a**(2) - b**(2) + c**(2)) * (a**(3) - 2*a**(2)*b + b**(3) - 2*a**(2)*c + a*b*c + c**(3))', ID="X_415", classes=["center", "incident"])
            central_line(model, X_158, X_415, 158, 415, incident=True)
            X_10 = spieker_center_of(model, A, B, C, ID="X_10", classes=["center", "incident"])
            X_846 = eval_trilinear_point(model, A, B, C, '(- a**(2) + b**(2) + c**(2) + b*c + c*a + a*b)', ID="X_846", classes=["center", "incident"])
            central_line(model, X_10, X_846, 10, 846, incident=True)
            return promote_point_to_top(model, X_1247)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1247 on the given reference triangle."""
            model = create_model('X_1247 = ISOGONAL CONJUGATE OF X_1046')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01247(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01247.json"))
            write_center_svg(
                model_acute,
                here / "x-01247.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
