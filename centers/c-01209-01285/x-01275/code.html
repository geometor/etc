X1275¶

ISOTOMIC CONJUGATE OF X1146

X1275 is the isotomic conjugate of X1146.
"""GEOMETOR construction model for X_1275 = ISOTOMIC CONJUGATE OF X_1146."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            gergonne_point_of,
            isotomic_conjugate,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01275(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1275 = ISOTOMIC CONJUGATE OF X_1146 on reference triangle ABC."""
            X_1146 = eval_trilinear_point(model, A, B, C, 'b*c*((b - c)*(b + c - a))**(2)', ID="X_1146", classes=["center"])
            X_1275 = isotomic_conjugate(model, A, B, C, X_1146, ID="X_1275", classes=["selected", "target"])
            X_7 = gergonne_point_of(model, A, B, C, ID="X_7", classes=["center", "incident"])
            X_59 = eval_trilinear_point(model, A, B, C, '1/(1 - cos(B - C))', ID="X_59", classes=["center", "incident"])
            central_line(model, X_7, X_59, 7, 59, incident=True)
            X_320 = eval_trilinear_point(model, A, B, C, '(1 - 2*cos(A))/a**(2)', ID="X_320", classes=["center", "incident"])
            X_765 = eval_trilinear_point(model, A, B, C, '(b - c) **(-2)', ID="X_765", classes=["center", "incident"])
            central_line(model, X_320, X_765, 320, 765, incident=True)
            X_522 = eval_trilinear_point(model, A, B, C, '(cos(B) - cos(C))*(csc(A))', ID="X_522", classes=["center", "incident"])
            X_664 = eval_trilinear_point(model, A, B, C, '1/(a*(b - c)*(b + c - a))', ID="X_664", classes=["center", "incident"])
            central_line(model, X_522, X_664, 522, 664, incident=True)
            return promote_point_to_top(model, X_1275)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1275 on the given reference triangle."""
            model = create_model('X_1275 = ISOTOMIC CONJUGATE OF X_1146')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01275(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01275.json"))
            write_center_svg(
                model_acute,
                here / "x-01275.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
