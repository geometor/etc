X1256¶

ISOGONAL CONJUGATE OF X1103

X1256 is the isogonal conjugate of X1103.
"""GEOMETOR construction model for X_1256 = ISOGONAL CONJUGATE OF X_1103."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            isogonal_conjugate,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01256(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1256 = ISOGONAL CONJUGATE OF X_1103 on reference triangle ABC."""
            X_1103 = eval_trilinear_point(model, A, B, C, '((cos(B) + cos(C) - cos(A) - 1)**(2))', ID="X_1103", classes=["center"])
            X_1256 = isogonal_conjugate(model, A, B, C, X_1103, ID="X_1256", classes=["selected", "target"])
            X_56 = eval_trilinear_point(model, A, B, C, 'a/(b + c - a)', ID="X_56", classes=["center", "incident"])
            X_84 = eval_trilinear_point(model, A, B, C, '1/(cos(B) + cos(C) - cos(A) - 1)', ID="X_84", classes=["center", "incident"])
            central_line(model, X_56, X_84, 56, 84, incident=True)
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_280 = eval_trilinear_point(model, A, B, C, '((csc(A/2))**(2))/(-1 - cos(A) + cos(B) + cos(C))', ID="X_280", classes=["center", "incident"])
            central_line(model, X_1, X_280, 1, 280, incident=True)
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_282 = eval_trilinear_point(model, A, B, C, '(cot(A/2))/(-1 - cos(A) + cos(B) + cos(C))', ID="X_282", classes=["center", "incident"])
            central_line(model, X_6, X_282, 6, 282, incident=True)
            return promote_point_to_top(model, X_1256)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1256 on the given reference triangle."""
            model = create_model('X_1256 = ISOGONAL CONJUGATE OF X_1103')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01256(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01256.json"))
            write_center_svg(
                model_acute,
                here / "x-01256.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
