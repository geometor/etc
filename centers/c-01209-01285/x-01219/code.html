X1219¶

ISOGONAL CONJUGATE OF X1191

X1219 is the isogonal conjugate of X1191.
"""GEOMETOR construction model for X_1219 = ISOGONAL CONJUGATE OF X_1191."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            isogonal_conjugate,
            nagel_point_of,
            point_from_barycentrics,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01219(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1219 = ISOGONAL CONJUGATE OF X_1191 on reference triangle ABC."""
            X_1191 = point_from_barycentrics(model, A, B, C, 0.19005848, 0.32280702, 0.4871345, ID="X_1191", classes=["center"])
            X_1219 = isogonal_conjugate(model, A, B, C, X_1191, ID="X_1219", classes=["selected", "target"])
            X_8 = nagel_point_of(model, A, B, C, ID="X_8", classes=["center", "incident"])
            X_57 = eval_trilinear_point(model, A, B, C, '1/(b + c - a)', ID="X_57", classes=["center", "incident"])
            central_line(model, X_8, X_57, 8, 57, incident=True)
            X_81 = eval_trilinear_point(model, A, B, C, '1/(b + c)', ID="X_81", classes=["center", "incident"])
            X_145 = eval_trilinear_point(model, A, B, C, 'b*c*(3*a - b - c)', ID="X_145", classes=["center", "incident"])
            central_line(model, X_81, X_145, 81, 145, incident=True)
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_341 = eval_trilinear_point(model, A, B, C, 'b**(2)*c**(2)*(b + c - a)**(2)', ID="X_341", classes=["center", "incident"])
            central_line(model, X_2, X_341, 2, 341, incident=True)
            return promote_point_to_top(model, X_1219)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1219 on the given reference triangle."""
            model = create_model('X_1219 = ISOGONAL CONJUGATE OF X_1191')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01219(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01219.json"))
            write_center_svg(
                model_acute,
                here / "x-01219.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
