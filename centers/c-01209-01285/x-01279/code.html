X1279¶

MIDPOINT OF X1 AND X238

X1279 is the midpoint of X1 and X238.
"""GEOMETOR construction model for X_1279 = MIDPOINT OF X_1 AND X_238."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            midpoint,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01279(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1279 = MIDPOINT OF X_1 AND X_238 on reference triangle ABC."""
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center"])
            X_238 = eval_trilinear_point(model, A, B, C, 'a**(2) - b*c', ID="X_238", classes=["center"])
            X_1279 = midpoint(model, X_1, X_238, ID="X_1279", classes=["selected", "target"], guide=False)
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            central_line(model, X_1, X_6, 1, 6, incident=True)
            X_31 = eval_trilinear_point(model, A, B, C, 'a**(2)', ID="X_31", classes=["center", "incident"])
            X_354 = eval_trilinear_point(model, A, B, C, '(b - c)**(2) - a*b - a*c', ID="X_354", classes=["center", "incident"])
            central_line(model, X_31, X_354, 31, 354, incident=True)
            X_145 = eval_trilinear_point(model, A, B, C, 'b*c*(3*a - b - c)', ID="X_145", classes=["center", "incident"])
            X_344 = eval_trilinear_point(model, A, B, C, '((csc(A/2))**(2))*((cos(B/2))**(4) + (cos(C/2))**(4) - (cos(A/2))**(4))', ID="X_344", classes=["center", "incident"])
            central_line(model, X_145, X_344, 145, 344, incident=True)
            return promote_point_to_top(model, X_1279)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1279 on the given reference triangle."""
            model = create_model('X_1279 = MIDPOINT OF X_1 AND X_238')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01279(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01279.json"))
            write_center_svg(
                model_acute,
                here / "x-01279.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
