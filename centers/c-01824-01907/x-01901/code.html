X1901¶

ZOSMA TRANSFORM OF X1172

X1901 = X653-Ceva conjugate of X523
"""GEOMETOR construction model for X_1901 = ZOSMA TRANSFORM OF X_1172."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            ceva_conjugate,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            mittenpunkt_of,
            orthocenter_of,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01901(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1901 = ZOSMA TRANSFORM OF X_1172 on reference triangle ABC."""
            X_653 = eval_trilinear_point(model, A, B, C, '1/(sec(B) - sec(C))', ID="X_653", classes=["center"])
            X_523 = eval_trilinear_point(model, A, B, C, 'sin(B -C)', ID="X_523", classes=["center"])
            X_1901 = ceva_conjugate(model, A, B, C, X_653, X_523, ID="X_1901", classes=["selected", "target"])
            X_4 = orthocenter_of(model, A, B, C, ID="X_4", classes=["center", "incident"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            central_line(model, X_4, X_6, 4, 6, incident=True)
            X_9 = mittenpunkt_of(model, A, B, C, ID="X_9", classes=["center", "incident"])
            X_46 = eval_trilinear_point(model, A, B, C, 'cos(B) + cos(C) - cos(A)', ID="X_46", classes=["center", "incident"])
            central_line(model, X_9, X_46, 9, 46, incident=True)
            X_12 = eval_trilinear_point(model, A, B, C, '1 + cos(B - C)', ID="X_12", classes=["center", "incident"])
            X_71 = eval_trilinear_point(model, A, B, C, '(b + c)*cos(A)', ID="X_71", classes=["center", "incident"])
            central_line(model, X_12, X_71, 12, 71, incident=True)
            return promote_point_to_top(model, X_1901)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1901 on the given reference triangle."""
            model = create_model('X_1901 = ZOSMA TRANSFORM OF X_1172')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01901(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01901.json"))
            write_center_svg(
                model_acute,
                here / "x-01901.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
