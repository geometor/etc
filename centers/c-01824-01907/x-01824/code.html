X1824¶

ZOSMA TRANSFORM OF X2

X1824 = intersection of tangents to hyperbola {{A, B, C,X4,X19}} at X4 and X19
"""GEOMETOR construction model for X_1824 = ZOSMA TRANSFORM OF X_2."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            isogonal_conjugate,
            nagel_point_of,
            orthocenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01824(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1824 = ZOSMA TRANSFORM OF X_2 on reference triangle ABC."""
            X_1444 = eval_trilinear_point(model, A, B, C, '(cot(A))/(b + c)', ID="X_1444", classes=["center"])
            X_1824 = isogonal_conjugate(model, A, B, C, X_1444, ID="X_1824", classes=["selected", "target"])
            X_4 = orthocenter_of(model, A, B, C, ID="X_4", classes=["center", "incident"])
            X_8 = nagel_point_of(model, A, B, C, ID="X_8", classes=["center", "incident"])
            central_line(model, X_4, X_8, 4, 8, incident=True)
            X_19 = eval_trilinear_point(model, A, B, C, 'tan(A)', ID="X_19", classes=["center", "incident"])
            X_25 = eval_trilinear_point(model, A, B, C, 'sin(A)*tan(A)', ID="X_25", classes=["center", "incident"])
            central_line(model, X_19, X_25, 19, 25, incident=True)
            X_65 = eval_trilinear_point(model, A, B, C, 'cos(B) + cos(C)', ID="X_65", classes=["center", "incident"])
            X_225 = eval_trilinear_point(model, A, B, C, '(sec(A))*(cos(B) + cos(C))', ID="X_225", classes=["center", "incident"])
            central_line(model, X_65, X_225, 65, 225, incident=True)
            return promote_point_to_top(model, X_1824)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1824 on the given reference triangle."""
            model = create_model('X_1824 = ZOSMA TRANSFORM OF X_2')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01824(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01824.json"))
            write_center_svg(
                model_acute,
                here / "x-01824.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
