X3525¶

EULER LINE INTERCEPT OF LINE X69X_575

X3525 = crosspoint of X3316 and X3317
"""GEOMETOR construction model for X_3525 = EULER LINE INTERCEPT OF LINE X_69X_575."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            crosspoint,
            eval_trilinear_point,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03525(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3525 = EULER LINE INTERCEPT OF LINE X_69X_575 on reference triangle ABC."""
            X_3316 = eval_trilinear_point(model, A, B, C, '1/(cos(A) + 2*sin(A))', ID="X_3316", classes=["center"])
            X_3317 = eval_trilinear_point(model, A, B, C, '1/(cos(A) - 2*sin(A))', ID="X_3317", classes=["center"])
            X_3525 = crosspoint(model, A, B, C, X_3316, X_3317, ID="X_3525", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center"])
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center"])
            central_line(model, X_2, X_3, 2, 3)
            X_69 = eval_trilinear_point(model, A, B, C, '(cos(A))/a**(2)', ID="X_69", classes=["center", "incident"])
            X_575 = eval_trilinear_point(model, A, B, C, '3*sin(A) + cos(A)*cot(omega) ', ID="X_575", classes=["center", "incident"])
            central_line(model, X_69, X_575, 69, 575, incident=True)
            X_498 = eval_trilinear_point(model, A, B, C, '1 + 2*sin(B)*sin(C)', ID="X_498", classes=["center", "incident"])
            X_1056 = eval_trilinear_point(model, A, B, C, '(2 + cos(B)*cos(C))', ID="X_1056", classes=["center", "incident"])
            central_line(model, X_498, X_1056, 498, 1056, incident=True)
            return promote_point_to_top(model, X_3525)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3525 on the given reference triangle."""
            model = create_model('X_3525 = EULER LINE INTERCEPT OF LINE X_69X_575')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03525(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03525.json"))
            write_center_svg(
                model_acute,
                here / "x-03525.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
