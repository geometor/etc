X3584¶

INTERSECTION OF LINES X1X_2 AND X12X_30

"""GEOMETOR construction model for X_3584 = INTERSECTION OF LINES X_1X_2 AND X_12X_30."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03584(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3584 = INTERSECTION OF LINES X_1X_2 AND X_12X_30 on reference triangle ABC."""
            X_3584 = eval_trilinear_point(model, A, B, C, '(b*c*(a**(4) + b**(4) + c**(4) - a**(2)*(3*b*c + 2*b**(2) + 2*c**(2)) - 2*b**(2)*c**(2)))', ID="X_3584", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            central_line(model, X_1, X_2, 1, 2, incident=True)
            X_12 = eval_trilinear_point(model, A, B, C, '1 + cos(B - C)', ID="X_12", classes=["center", "incident"])
            X_30 = eval_trilinear_point(model, A, B, C, 'cos(A) - 2*cos(B)*cos(C)', ID="X_30", classes=["center", "incident"])
            central_line(model, X_12, X_30, 12, 30, incident=True)
            X_55 = eval_trilinear_point(model, A, B, C, 'a*(b + c - a)', ID="X_55", classes=["center", "incident"])
            X_381 = eval_trilinear_point(model, A, B, C, '2*cos(B - C) - cos(A)', ID="X_381", classes=["center", "incident"])
            central_line(model, X_55, X_381, 55, 381, incident=True)
            return promote_point_to_top(model, X_3584)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3584 on the given reference triangle."""
            model = create_model('X_3584 = INTERSECTION OF LINES X_1X_2 AND X_12X_30')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03584(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03584.json"))
            write_center_svg(
                model_acute,
                here / "x-03584.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
