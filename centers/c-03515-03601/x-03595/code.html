X3595¶

ISOTOMIC CONJUGATE OF 2nd DIXIT POINT

X3595 is the isotomic conjugate of 2nd dixit point.
"""GEOMETOR construction model for X_3595 = ISOTOMIC CONJUGATE OF 2nd DIXIT POINT."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            orthocenter_of,
            point_from_barycentrics,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03595(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3595 = ISOTOMIC CONJUGATE OF 2nd DIXIT POINT on reference triangle ABC."""
            X_3595 = eval_trilinear_point(model, A, B, C, '(b*c*(b**(2) + c**(2) - a**(2) - 6*area ))', ID="X_3595", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            central_line(model, X_2, X_6, 2, 6, incident=True)
            X_4 = orthocenter_of(model, A, B, C, ID="X_4", classes=["center", "incident"])
            X_1164 = eval_trilinear_point(model, A, B, C, '(( area + a**(2))*(2*area - b**(2) - c**(2))/(a*(b**(2) + c**(2) - a**(2))))', ID="X_1164", classes=["center", "incident"])
            central_line(model, X_4, X_1164, 4, 1164, incident=True)
            X_642 = point_from_barycentrics(model, A, B, C, -0.045985566, 0.53183937, 0.5141462, ID="X_642", classes=["center", "incident"])
            X_1587 = eval_trilinear_point(model, A, B, C, '(sin(A) + cos(B)*cos(C))', ID="X_1587", classes=["center", "incident"])
            central_line(model, X_642, X_1587, 642, 1587, incident=True)
            return promote_point_to_top(model, X_3595)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3595 on the given reference triangle."""
            model = create_model('X_3595 = ISOTOMIC CONJUGATE OF 2nd DIXIT POINT')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03595(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03595.json"))
            write_center_svg(
                model_acute,
                here / "x-03595.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
