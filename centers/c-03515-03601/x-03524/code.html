X3524¶

EULER LINE INTERCEPT OF LINE X35X_1058

X3524 = trisector nearest X3 of segment X2X3
"""GEOMETOR construction model for X_3524 = EULER LINE INTERCEPT OF LINE X_35X_1058."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03524(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3524 = EULER LINE INTERCEPT OF LINE X_35X_1058 on reference triangle ABC."""
            X_3524 = eval_trilinear_point(model, A, B, C, '3*cos(A) + sin(B)*sin(C)', ID="X_3524", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center"])
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center"])
            central_line(model, X_2, X_3, 2, 3)
            X_40 = eval_trilinear_point(model, A, B, C, 'cos(B) + cos(C) - cos(A) - 1', ID="X_40", classes=["center", "incident"])
            X_551 = eval_trilinear_point(model, A, B, C, '(4*a + b + c)/a', ID="X_551", classes=["center", "incident"])
            central_line(model, X_40, X_551, 40, 551, incident=True)
            X_36 = eval_trilinear_point(model, A, B, C, '1 - 2*cos(A)', ID="X_36", classes=["center", "incident"])
            X_1056 = eval_trilinear_point(model, A, B, C, '(2 + cos(B)*cos(C))', ID="X_1056", classes=["center", "incident"])
            central_line(model, X_36, X_1056, 36, 1056, incident=True)
            return promote_point_to_top(model, X_3524)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3524 on the given reference triangle."""
            model = create_model('X_3524 = EULER LINE INTERCEPT OF LINE X_35X_1058')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03524(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03524.json"))
            write_center_svg(
                model_acute,
                here / "x-03524.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
