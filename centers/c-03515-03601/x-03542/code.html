X3542¶

EULER LINE INTERCEPT OF LINE X33X_498

X3542 = inverse-in-orthocentroidal-circle of X3541
"""GEOMETOR construction model for X_3542 = EULER LINE INTERCEPT OF LINE X_33X_498."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03542(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3542 = EULER LINE INTERCEPT OF LINE X_33X_498 on reference triangle ABC."""
            X_3542 = eval_trilinear_point(model, A, B, C, '(sec(A) - 2*sin(B)*sin(C))', ID="X_3542", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center"])
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center"])
            central_line(model, X_2, X_3, 2, 3)
            X_33 = eval_trilinear_point(model, A, B, C, '1 + sec(A)', ID="X_33", classes=["center", "incident"])
            X_498 = eval_trilinear_point(model, A, B, C, '1 + 2*sin(B)*sin(C)', ID="X_498", classes=["center", "incident"])
            central_line(model, X_33, X_498, 33, 498, incident=True)
            X_34 = eval_trilinear_point(model, A, B, C, '1 - sec(A)', ID="X_34", classes=["center", "incident"])
            X_499 = eval_trilinear_point(model, A, B, C, '1 - 2*sin(B)*sin(C)', ID="X_499", classes=["center", "incident"])
            central_line(model, X_34, X_499, 34, 499, incident=True)
            return promote_point_to_top(model, X_3542)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3542 on the given reference triangle."""
            model = create_model('X_3542 = EULER LINE INTERCEPT OF LINE X_33X_498')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03542(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03542.json"))
            write_center_svg(
                model_acute,
                here / "x-03542.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
