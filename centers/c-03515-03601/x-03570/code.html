X3570¶

INTERSECTION OF LINES P(6)U(8) AND P(8)U(6)

The points given by trilinears P(6) =\(b : c : a\) and U(6) =\(c : a : b\) are indexed at Bicentric Pairs (accessible using the Tables button at the top of ETC), and likewise for their isogonal conjugates P(8) =\(1/b : 1/c : 1/a\) and U(8) =\(1/c : 1/a : 1/b\). The bicentric pair of lines P(6)U(8) and P(8)U(6) concur in X3570. Contributed by Peter Moses, July 7, 2009..
"""GEOMETOR construction model for X_3570 = INTERSECTION OF LINES P(6)U(8) AND P(8)U(6)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03570(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3570 = INTERSECTION OF LINES P(6)U(8) AND P(8)U(6) on reference triangle ABC."""
            X_3570 = eval_trilinear_point(model, A, B, C, 'b*c*(a**(2) - b*c)/(b - c)', ID="X_3570", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            central_line(model, X_2, X_6, 2, 6, incident=True)
            X_100 = eval_trilinear_point(model, A, B, C, '1/(b - c)', ID="X_100", classes=["center", "incident"])
            X_190 = eval_trilinear_point(model, A, B, C, 'b*c/(b - c)', ID="X_190", classes=["center", "incident"])
            central_line(model, X_100, X_190, 100, 190, incident=True)
            X_101 = eval_trilinear_point(model, A, B, C, 'a/(b - c)', ID="X_101", classes=["center", "incident"])
            X_668 = eval_trilinear_point(model, A, B, C, '1/(a**(2)*(b - c))', ID="X_668", classes=["center", "incident"])
            central_line(model, X_101, X_668, 101, 668, incident=True)
            return promote_point_to_top(model, X_3570)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3570 on the given reference triangle."""
            model = create_model('X_3570 = INTERSECTION OF LINES P(6)U(8) AND P(8)U(6)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03570(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03570.json"))
            write_center_svg(
                model_acute,
                here / "x-03570.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
