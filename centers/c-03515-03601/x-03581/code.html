X3581¶

INTERSECTION OF LINES X3X_6 AND X30X_74

Let A1B1C1 and A2B2C2 be the 1st and 2nd Ehrmann circumscribing triangles. Let A* be the crosssum of A1 and A2, and define B* and C* cyclically. The lines AA*, BB*, CC* concur in X3581.
"""GEOMETOR construction model for X_3581 = INTERSECTION OF LINES X_3X_6 AND X_30X_74."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03581(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3581 = INTERSECTION OF LINES X_3X_6 AND X_30X_74 on reference triangle ABC."""
            X_3581 = eval_trilinear_point(model, A, B, C, '(a*(2*b**(4) + 2*c**(4) - a**(4) - a**(2)*b**(2) - a**(2)*c**(2) - 4*b**(2)*c**(2))*((b**(2) + c**(2) - a**(2))**(2) - b**(2)*c**(2)))', ID="X_3581", classes=["selected", "target"])
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center", "incident"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            central_line(model, X_3, X_6, 3, 6, incident=True)
            X_30 = eval_trilinear_point(model, A, B, C, 'cos(A) - 2*cos(B)*cos(C)', ID="X_30", classes=["center", "incident"])
            X_74 = eval_trilinear_point(model, A, B, C, '1/(cos(A) - 2*cos(B)*cos(C))', ID="X_74", classes=["center", "incident"])
            central_line(model, X_30, X_74, 30, 74, incident=True)
            X_186 = eval_trilinear_point(model, A, B, C, '4*cos(A) - sec(A)', ID="X_186", classes=["center", "incident"])
            X_323 = eval_trilinear_point(model, A, B, C, 'sin(3*A)*(csc(A))**(2)', ID="X_323", classes=["center", "incident"])
            central_line(model, X_186, X_323, 186, 323, incident=True)
            return promote_point_to_top(model, X_3581)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3581 on the given reference triangle."""
            model = create_model('X_3581 = INTERSECTION OF LINES X_3X_6 AND X_30X_74')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03581(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03581.json"))
            write_center_svg(
                model_acute,
                here / "x-03581.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
