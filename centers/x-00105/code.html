X(105) = Λ(INCENTER, SYMMEDIAN POINT) — Code¶

X(105) is the perspector of ABC and the (degenerate) side-triangle of the circumcevian triangles of X(3513) and X(3514). (Randy Hutson, June 7, 2019)
"""GEOMETOR construction model for X(105) = Λ(INCENTER, SYMMEDIAN POINT)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            feuerbach_point_of,
            incenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00105(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X(105) = Λ(INCENTER, SYMMEDIAN POINT) on reference triangle ABC."""
            X_105 = eval_trilinear_point(model, A, B, C, '1/(b**(2) + c**(2) - a*(b + c))', ID="X(105)", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X(2)", classes=["center", "incident"])
            X_11 = feuerbach_point_of(model, A, B, C, ID="X(11)", classes=["center", "incident"])
            central_line(model, X_2, X_11, 2, 11, incident=True)
            X_1 = incenter_of(model, A, B, C, ID="X(1)", classes=["center", "incident"])
            X_41 = eval_trilinear_point(model, A, B, C, 'a**(2)*(b + c - a)', ID="X(41)", classes=["center", "incident"])
            central_line(model, X_1, X_41, 1, 41, incident=True)
            X_31 = eval_trilinear_point(model, A, B, C, 'a**(2)', ID="X(31)", classes=["center", "incident"])
            X_57 = eval_trilinear_point(model, A, B, C, '1/(b + c - a)', ID="X(57)", classes=["center", "incident"])
            central_line(model, X_31, X_57, 31, 57, incident=True)
            return promote_point_to_top(model, X_105)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X(105) on the given reference triangle."""
            model = create_model('X(105) = Λ(INCENTER, SYMMEDIAN POINT)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00105(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00105.json"))
            write_center_svg(
                model_acute,
                here / "x-00105.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
