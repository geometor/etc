X(537) = DIRECTION OF VECTOR AX + BX + CX, where X = X(38) — Code¶

Barycentrics af(a,b,c) : bf(b,c,a) : cf(c,a,b)
"""GEOMETOR construction model for X(537) = DIRECTION OF VECTOR AX + BX + CX, where X = X(38)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            intersect_lines,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00537(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X(537) = DIRECTION OF VECTOR AX + BX + CX, where X = X(38) on reference triangle ABC."""
            X_2 = centroid_of(model, A, B, C, ID="X(2)", classes=["center"])
            X_38 = eval_trilinear_point(model, A, B, C, 'b**(2) + c**(2)', ID="X(38)", classes=["center"])
            X_1 = incenter_of(model, A, B, C, ID="X(1)", classes=["center"])
            X_190 = eval_trilinear_point(model, A, B, C, 'b*c/(b - c)', ID="X(190)", classes=["center"])
            central_line(model, X_2, X_38, 2, 38)
            central_line(model, X_1, X_190, 1, 190)
            X_537 = intersect_lines(model, X_2, X_38, X_1, X_190, ID="X(537)", classes=["selected", "target"])
            X_30 = eval_trilinear_point(model, A, B, C, 'cos(A) - 2*cos(B)*cos(C)', ID="X(30)", classes=["center", "incident"])
            X_511 = eval_trilinear_point(model, A, B, C, 'cos(A + omega )', ID="X(511)", classes=["center", "incident"])
            central_line(model, X_30, X_511, 30, 511, incident=True)
            return promote_point_to_top(model, X_537)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X(537) on the given reference triangle."""
            model = create_model('X(537) = DIRECTION OF VECTOR AX + BX + CX, where X = X(38)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00537(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00537.json"))
            write_center_svg(
                model_acute,
                here / "x-00537.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
