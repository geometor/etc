X37¶

CROSSPOINT OF INCENTER AND CENTROID

X37 is the crosspoint of incenter and centroid.
"""GEOMETOR construction model for X_37 = CROSSPOINT OF INCENTER AND CENTROID."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00037(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_37 = CROSSPOINT OF INCENTER AND CENTROID on reference triangle ABC."""
            X_37 = eval_trilinear_point(model, A, B, C, 'b + c', ID="X_37", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            central_line(model, X_1, X_6, 1, 6, incident=True)
            X_19 = eval_trilinear_point(model, A, B, C, 'tan(A)', ID="X_19", classes=["center", "incident"])
            X_25 = eval_trilinear_point(model, A, B, C, 'sin(A)*tan(A)', ID="X_25", classes=["center", "incident"])
            central_line(model, X_19, X_25, 19, 25, incident=True)
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_75 = eval_trilinear_point(model, A, B, C, '1/a**(2)', ID="X_75", classes=["center", "incident"])
            central_line(model, X_2, X_75, 2, 75, incident=True)
            return promote_point_to_top(model, X_37)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_37 on the given reference triangle."""
            model = create_model('X_37 = CROSSPOINT OF INCENTER AND CENTROID')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00037(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00037.json"))
            write_center_svg(
                model_acute,
                here / "x-00037.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
