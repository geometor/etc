X44¶

X6-LINE CONJUGATE OF X1

X44 = crossdifference of PU(55)
"""GEOMETOR construction model for X_44 = X_6-LINE CONJUGATE OF X_1."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00044(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_44 = X_6-LINE CONJUGATE OF X_1 on reference triangle ABC."""
            X_44 = eval_trilinear_point(model, A, B, C, 'b + c - 2*a', ID="X_44", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            central_line(model, X_1, X_6, 1, 6, incident=True)
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_89 = eval_trilinear_point(model, A, B, C, '1/(2*b + 2*c - a)', ID="X_89", classes=["center", "incident"])
            central_line(model, X_2, X_89, 2, 89, incident=True)
            X_31 = eval_trilinear_point(model, A, B, C, 'a**(2)', ID="X_31", classes=["center", "incident"])
            X_210 = eval_trilinear_point(model, A, B, C, '(b + c)*(b + c - a)', ID="X_210", classes=["center", "incident"])
            central_line(model, X_31, X_210, 31, 210, incident=True)
            return promote_point_to_top(model, X_44)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_44 on the given reference triangle."""
            model = create_model('X_44 = X_6-LINE CONJUGATE OF X_1')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00044(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00044.json"))
            write_center_svg(
                model_acute,
                here / "x-00044.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
