X45¶

X9-BETH CONJUGATE OF X1

X45 is the {X1,X9}-harmonic conjugate of X44.
"""GEOMETOR construction model for X_45 = X_9-BETH CONJUGATE OF X_1."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00045(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_45 = X_9-BETH CONJUGATE OF X_1 on reference triangle ABC."""
            X_45 = eval_trilinear_point(model, A, B, C, '2*b + 2*c - a', ID="X_45", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            central_line(model, X_1, X_6, 1, 6, incident=True)
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_88 = eval_trilinear_point(model, A, B, C, '1/(b + c - 2*a)', ID="X_88", classes=["center", "incident"])
            central_line(model, X_2, X_88, 2, 88, incident=True)
            X_53 = eval_trilinear_point(model, A, B, C, 'tan(A)*cos(B - C)', ID="X_53", classes=["center", "incident"])
            X_281 = eval_trilinear_point(model, A, B, C, 'sec(A)*cot(A/2)', ID="X_281", classes=["center", "incident"])
            central_line(model, X_53, X_281, 53, 281, incident=True)
            return promote_point_to_top(model, X_45)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_45 on the given reference triangle."""
            model = create_model('X_45 = X_9-BETH CONJUGATE OF X_1')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00045(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00045.json"))
            write_center_svg(
                model_acute,
                here / "x-00045.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
