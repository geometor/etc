X33¶

PERSPECTOR OF THE ORTHIC AND INTANGENTS TRIANGLES

X33 is the {X1,X4}-harmonic conjugate of X34. For a list of other harmonic conjugates, click Tables at the top of this page.
"""GEOMETOR construction model for X_33 = PERSPECTOR OF THE ORTHIC AND INTANGENTS TRIANGLES."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            orthocenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00033(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_33 = PERSPECTOR OF THE ORTHIC AND INTANGENTS TRIANGLES on reference triangle ABC."""
            X_33 = eval_trilinear_point(model, A, B, C, '1 + sec(A)', ID="X_33", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_4 = orthocenter_of(model, A, B, C, ID="X_4", classes=["center", "incident"])
            central_line(model, X_1, X_4, 1, 4, incident=True)
            X_19 = eval_trilinear_point(model, A, B, C, 'tan(A)', ID="X_19", classes=["center", "incident"])
            X_25 = eval_trilinear_point(model, A, B, C, 'sin(A)*tan(A)', ID="X_25", classes=["center", "incident"])
            central_line(model, X_19, X_25, 19, 25, incident=True)
            X_24 = eval_trilinear_point(model, A, B, C, 'sec(A)*cos(2*A)', ID="X_24", classes=["center", "incident"])
            X_35 = eval_trilinear_point(model, A, B, C, '1 + 2*cos(A)', ID="X_35", classes=["center", "incident"])
            central_line(model, X_24, X_35, 24, 35, incident=True)
            return promote_point_to_top(model, X_33)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_33 on the given reference triangle."""
            model = create_model('X_33 = PERSPECTOR OF THE ORTHIC AND INTANGENTS TRIANGLES')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00033(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00033.json"))
            write_center_svg(
                model_acute,
                here / "x-00033.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
