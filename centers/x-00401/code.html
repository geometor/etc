X(401) = BAILEY POINT — Code¶

As a point on the Euler line, X(401) has Shinagawa coefficients (EF + F2 - S2, 2S2).
"""GEOMETOR construction model for X(401) = BAILEY POINT."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            point_from_barycentrics,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00401(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X(401) = BAILEY POINT on reference triangle ABC."""
            X_401 = eval_trilinear_point(model, A, B, C, '(sin(2*B)*sin(2*C) - (sin(2*A))**(2))*(csc(A))', ID="X(401)", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X(2)", classes=["center"])
            X_3 = circumcenter_of(model, A, B, C, ID="X(3)", classes=["center"])
            central_line(model, X_2, X_3, 2, 3)
            X_97 = eval_trilinear_point(model, A, B, C, 'cot(A)*sec(B - C)', ID="X(97)", classes=["center", "incident"])
            X_276 = point_from_barycentrics(model, A, B, C, 0.26546393, 0.27267734, 0.46185873, ID="X(276)", classes=["center", "incident"])
            central_line(model, X_97, X_276, 97, 276, incident=True)
            X_50 = eval_trilinear_point(model, A, B, C, 'sin(3*A)', ID="X(50)", classes=["center", "incident"])
            X_338 = eval_trilinear_point(model, A, B, C, '(b**(2) - c**(2))**(2)/a**(3)', ID="X(338)", classes=["center", "incident"])
            central_line(model, X_50, X_338, 50, 338, incident=True)
            return promote_point_to_top(model, X_401)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X(401) on the given reference triangle."""
            model = create_model('X(401) = BAILEY POINT')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00401(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00401.json"))
            write_center_svg(
                model_acute,
                here / "x-00401.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
