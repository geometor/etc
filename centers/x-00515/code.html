X(515) = ISOGONAL CONJUGATE OF X(102) — Code¶

As the isogonal conjugate of a point on the circumcircle, X(515) lies on the line at infinity.
"""GEOMETOR construction model for X(515) = ISOGONAL CONJUGATE OF X(102)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            isogonal_conjugate,
            nagel_point_of,
            orthocenter_of,
            promote_point_to_top,
            spieker_center_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00515(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X(515) = ISOGONAL CONJUGATE OF X(102) on reference triangle ABC."""
            X_102 = eval_trilinear_point(model, A, B, C, '1/(sin(B)*(sec(A) - sec(B)) + sin(C)*(sec(A) - sec(C)))', ID="X(102)", classes=["center"])
            X_515 = isogonal_conjugate(model, A, B, C, X_102, ID="X(515)", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X(1)", classes=["center", "incident"])
            X_4 = orthocenter_of(model, A, B, C, ID="X(4)", classes=["center", "incident"])
            central_line(model, X_1, X_4, 1, 4, incident=True)
            X_3 = circumcenter_of(model, A, B, C, ID="X(3)", classes=["center", "incident"])
            X_10 = spieker_center_of(model, A, B, C, ID="X(10)", classes=["center", "incident"])
            central_line(model, X_3, X_10, 3, 10, incident=True)
            X_8 = nagel_point_of(model, A, B, C, ID="X(8)", classes=["center", "incident"])
            X_20 = eval_trilinear_point(model, A, B, C, 'cos(A) - cos(B)*cos(C)', ID="X(20)", classes=["center", "incident"])
            central_line(model, X_8, X_20, 8, 20, incident=True)
            return promote_point_to_top(model, X_515)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X(515) on the given reference triangle."""
            model = create_model('X(515) = ISOGONAL CONJUGATE OF X(102)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00515(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00515.json"))
            write_center_svg(
                model_acute,
                here / "x-00515.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
