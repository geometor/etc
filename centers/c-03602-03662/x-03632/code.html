X3632¶

H⁻¹(X8; M, 1/2)

X3632 = homothetic center of outer Garcia triangle and mid-triangle of medial and anticomplementary triangles
"""GEOMETOR construction model for X_3632 = H⁻¹(X_8; M, 1/2)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03632(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3632 = H⁻¹(X_8; M, 1/2) on reference triangle ABC."""
            X_3632 = eval_trilinear_point(model, A, B, C, 'b*c*(3*a - 2*b - 2*c)', ID="X_3632", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            central_line(model, X_1, X_2, 1, 2, incident=True)
            X_40 = eval_trilinear_point(model, A, B, C, 'cos(B) + cos(C) - cos(A) - 1', ID="X_40", classes=["center", "incident"])
            X_550 = eval_trilinear_point(model, A, B, C, '- cos(B - C) + 4*cos(A)', ID="X_550", classes=["center", "incident"])
            central_line(model, X_40, X_550, 40, 550, incident=True)
            X_35 = eval_trilinear_point(model, A, B, C, '1 + 2*cos(A)', ID="X_35", classes=["center", "incident"])
            X_956 = eval_trilinear_point(model, A, B, C, '(a**(3) - a*(b - c)**(2) - 2*b*c*(b + c))', ID="X_956", classes=["center", "incident"])
            central_line(model, X_35, X_956, 35, 956, incident=True)
            return promote_point_to_top(model, X_3632)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3632 on the given reference triangle."""
            model = create_model('X_3632 = H⁻¹(X_8; M, 1/2)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03632(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03632.json"))
            write_center_svg(
                model_acute,
                here / "x-03632.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
