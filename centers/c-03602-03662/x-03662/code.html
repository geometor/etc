X3662¶

2nd KIRIKAMI-MOSES POINT

X3662 = 3(\(a^{2}\) + \(b^{2}\) + \(c^{2})\)*X2 + (\(a^{2}\) + \(b^{2}\) + \(c^{2}\) -2bc - 2ca - 2ab)X7
"""GEOMETOR construction model for X_3662 = 2nd KIRIKAMI-MOSES POINT."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            gergonne_point_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03662(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3662 = 2nd KIRIKAMI-MOSES POINT on reference triangle ABC."""
            X_3662 = eval_trilinear_point(model, A, B, C, '(b*c*(b**(2) + c**(2) - b*c))', ID="X_3662", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_7 = gergonne_point_of(model, A, B, C, ID="X_7", classes=["center", "incident"])
            central_line(model, X_2, X_7, 2, 7, incident=True)
            X_75 = eval_trilinear_point(model, A, B, C, '1/a**(2)', ID="X_75", classes=["center", "incident"])
            X_141 = eval_trilinear_point(model, A, B, C, 'b*c*(b**(2) + c**(2))', ID="X_141", classes=["center", "incident"])
            central_line(model, X_75, X_141, 75, 141, incident=True)
            X_69 = eval_trilinear_point(model, A, B, C, '(cos(A))/a**(2)', ID="X_69", classes=["center", "incident"])
            X_239 = eval_trilinear_point(model, A, B, C, 'b*c*(a**(2) - b*c)', ID="X_239", classes=["center", "incident"])
            central_line(model, X_69, X_239, 69, 239, incident=True)
            return promote_point_to_top(model, X_3662)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3662 on the given reference triangle."""
            model = create_model('X_3662 = 2nd KIRIKAMI-MOSES POINT')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03662(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03662.json"))
            write_center_svg(
                model_acute,
                here / "x-03662.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
