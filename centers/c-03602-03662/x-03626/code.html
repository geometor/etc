X3626¶

H⁻¹(X1; M, -2)

X3626 = midpoint of X8 and X10
"""GEOMETOR construction model for X_3626 = H⁻¹(X_1; M, -2)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            midpoint,
            nagel_point_of,
            promote_point_to_top,
            spieker_center_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03626(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3626 = H⁻¹(X_1; M, -2) on reference triangle ABC."""
            X_8 = nagel_point_of(model, A, B, C, ID="X_8", classes=["center"])
            X_10 = spieker_center_of(model, A, B, C, ID="X_10", classes=["center"])
            X_3626 = midpoint(model, X_8, X_10, ID="X_3626", classes=["selected", "target"], guide=False)
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            central_line(model, X_1, X_2, 1, 2, incident=True)
            X_44 = eval_trilinear_point(model, A, B, C, 'b + c - 2*a', ID="X_44", classes=["center", "incident"])
            X_594 = eval_trilinear_point(model, A, B, C, '(b + c)**(2)/a', ID="X_594", classes=["center", "incident"])
            central_line(model, X_44, X_594, 44, 594, incident=True)
            X_355 = eval_trilinear_point(model, A, B, C, 'a*cos(A) - (b + c)*cos(B - C)', ID="X_355", classes=["center", "incident"])
            X_382 = eval_trilinear_point(model, A, B, C, 'cos(A) - 4*cos(B)*cos(C)', ID="X_382", classes=["center", "incident"])
            central_line(model, X_355, X_382, 355, 382, incident=True)
            return promote_point_to_top(model, X_3626)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3626 on the given reference triangle."""
            model = create_model('X_3626 = H⁻¹(X_1; M, -2)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03626(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03626.json"))
            write_center_svg(
                model_acute,
                here / "x-03626.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
