X3610¶

1st AYME-MOSES PERSPECTOR

Moses found that the locus of X such that the cevian triangle of X is perspective to the Ayme triangle is a cubic which passes through the points Xi for i = 1, 2, 19, 75, 279, 304, 346, 2184. A barycentric equation for this Ayme-Moses cubic follows:

(Cyclic sum of ayz[\(by(a^{2}\) + \(b^{2}\) - \(c^{2})\) - \(cz(a^{2}\) - \(b^{2}\) + \(c^{2}\)] ) = 0.

"""GEOMETOR construction model for X_3610 = 1st AYME-MOSES PERSPECTOR."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            promote_point_to_top,
            spieker_center_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03610(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3610 = 1st AYME-MOSES PERSPECTOR on reference triangle ABC."""
            X_3610 = eval_trilinear_point(model, A, B, C, 'b*c*(b + c)*(a**(2) - b**(2) - c**(2))*(a**(2) + b**(2) + c**(2) + 2*b*c)', ID="X_3610", classes=["selected", "target"])
            X_10 = spieker_center_of(model, A, B, C, ID="X_10", classes=["center", "incident"])
            X_37 = eval_trilinear_point(model, A, B, C, 'b + c', ID="X_37", classes=["center", "incident"])
            central_line(model, X_10, X_37, 10, 37, incident=True)
            X_19 = eval_trilinear_point(model, A, B, C, 'tan(A)', ID="X_19", classes=["center", "incident"])
            X_346 = eval_trilinear_point(model, A, B, C, 'b*c*(b + c - a)**(2)', ID="X_346", classes=["center", "incident"])
            central_line(model, X_19, X_346, 19, 346, incident=True)
            X_612 = eval_trilinear_point(model, A, B, C, 'a**(2) + b**(2) + c**(2) + 2*b*c', ID="X_612", classes=["center", "incident"])
            X_2345 = eval_barycentric_point(model, A, B, C, 'b*c + SW', ID="X_2345", classes=["center", "incident"])
            central_line(model, X_612, X_2345, 612, 2345, incident=True)
            return promote_point_to_top(model, X_3610)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3610 on the given reference triangle."""
            model = create_model('X_3610 = 1st AYME-MOSES PERSPECTOR')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03610(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03610.json"))
            write_center_svg(
                model_acute,
                here / "x-03610.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
