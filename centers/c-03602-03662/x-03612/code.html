X3612¶

INTERSECTION OF LINES X1X_3 AND X21X_90

Let Ja, Jb, Jc be the excenters and I the incenter. Let A' be the centroid of JbJcI, and define B' and C' cyclically. A'B'C' is also the cross-triangle of the excentral and 2nd circumperp triangles. A'B'C' is homothetic to the reflection triangle of X1 at X3612. (Randy Hutson, July 31 2018)
"""GEOMETOR construction model for X_3612 = INTERSECTION OF LINES X_1X_3 AND X_21X_90."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03612(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3612 = INTERSECTION OF LINES X_1X_3 AND X_21X_90 on reference triangle ABC."""
            X_3612 = eval_trilinear_point(model, A, B, C, '3*cos(A) + cos(B) + cos(C)', ID="X_3612", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center", "incident"])
            central_line(model, X_1, X_3, 1, 3, incident=True)
            X_21 = eval_trilinear_point(model, A, B, C, '1/(cos(B) + cos(C))', ID="X_21", classes=["center", "incident"])
            X_90 = eval_trilinear_point(model, A, B, C, '1/(cos(B) + cos(C) - cos(A))', ID="X_90", classes=["center", "incident"])
            central_line(model, X_21, X_90, 21, 90, incident=True)
            X_498 = eval_trilinear_point(model, A, B, C, '1 + 2*sin(B)*sin(C)', ID="X_498", classes=["center", "incident"])
            X_515 = eval_trilinear_point(model, A, B, C, '(b + c)*sec(A) - b*sec(B) - c*sec(C)', ID="X_515", classes=["center", "incident"])
            central_line(model, X_498, X_515, 498, 515, incident=True)
            return promote_point_to_top(model, X_3612)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3612 on the given reference triangle."""
            model = create_model('X_3612 = INTERSECTION OF LINES X_1X_3 AND X_21X_90')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03612(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03612.json"))
            write_center_svg(
                model_acute,
                here / "x-03612.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
