X3616¶

H(X1; M, 1/2)

Let OA be the circle tangent to side BC at its midpoint and to the circumcircle on the side of BC opposite A. Define OB and OC cyclically. Let LA be the external tangent to circles OB and OC that is the nearer of the two to OA. Define LB and LC cyclically. Let A' = LB∩LC, and define B' and C' cyclically. Then A'B'C' is homothetic to ABC, and the center of homothety is X3616. See the reference at X1001.
"""GEOMETOR construction model for X_3616 = H(X_1; M, 1/2)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            gergonne_point_of,
            incenter_of,
            isogonal_conjugate,
            point_from_barycentrics,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03616(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3616 = H(X_1; M, 1/2) on reference triangle ABC."""
            X_2334 = point_from_barycentrics(model, A, B, C, 0.2463661, 0.33111604, 0.42251786, ID="X_2334", classes=["center"])
            X_3616 = isogonal_conjugate(model, A, B, C, X_2334, ID="X_3616", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            central_line(model, X_1, X_2, 1, 2, incident=True)
            X_7 = gergonne_point_of(model, A, B, C, ID="X_7", classes=["center", "incident"])
            X_21 = eval_trilinear_point(model, A, B, C, '1/(cos(B) + cos(C))', ID="X_21", classes=["center", "incident"])
            central_line(model, X_7, X_21, 7, 21, incident=True)
            X_29 = eval_trilinear_point(model, A, B, C, '(sec(A))/(cos(B) + cos(C))', ID="X_29", classes=["center", "incident"])
            X_278 = eval_trilinear_point(model, A, B, C, 'sec(A)*tan(A/2)', ID="X_278", classes=["center", "incident"])
            central_line(model, X_29, X_278, 29, 278, incident=True)
            return promote_point_to_top(model, X_3616)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3616 on the given reference triangle."""
            model = create_model('X_3616 = H(X_1; M, 1/2)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03616(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03616.json"))
            write_center_svg(
                model_acute,
                here / "x-03616.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
