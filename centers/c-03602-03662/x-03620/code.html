X3620¶

H(X6; M, -2)

X3620 = midpoint of X69 and X3618
"""GEOMETOR construction model for X_3620 = H(X_6; M, -2)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            midpoint,
            nagel_point_of,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03620(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3620 = H(X_6; M, -2) on reference triangle ABC."""
            X_69 = eval_trilinear_point(model, A, B, C, '(cos(A))/a**(2)', ID="X_69", classes=["center"])
            X_3618 = eval_trilinear_point(model, A, B, C, '(b*c*(3*a**(2) + b**(2) + c**(2)))', ID="X_3618", classes=["center"])
            X_3620 = midpoint(model, X_69, X_3618, ID="X_3620", classes=["selected", "target"], guide=False)
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            central_line(model, X_2, X_6, 2, 6, incident=True)
            X_20 = eval_trilinear_point(model, A, B, C, 'cos(A) - cos(B)*cos(C)', ID="X_20", classes=["center", "incident"])
            X_1352 = eval_trilinear_point(model, A, B, C, 'a - 2*R*cos(B-C)*cot(omega) ', ID="X_1352", classes=["center", "incident"])
            central_line(model, X_20, X_1352, 20, 1352, incident=True)
            X_8 = nagel_point_of(model, A, B, C, ID="X_8", classes=["center", "incident"])
            X_1738 = eval_barycentric_point(model, A, B, C, '(2*a*b*c - (b + c)*(a**(2) - (b - c)**(2)))', ID="X_1738", classes=["center", "incident"])
            central_line(model, X_8, X_1738, 8, 1738, incident=True)
            return promote_point_to_top(model, X_3620)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3620 on the given reference triangle."""
            model = create_model('X_3620 = H(X_6; M, -2)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03620(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03620.json"))
            write_center_svg(
                model_acute,
                here / "x-03620.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
