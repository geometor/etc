X1627¶

TCC-PERSPECTOR OF X251

X1627 is the tcc-perspector of X251.
"""GEOMETOR construction model for X_1627 = TCC-PERSPECTOR OF X_251."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01627(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1627 = TCC-PERSPECTOR OF X_251 on reference triangle ABC."""
            X_1627 = eval_barycentric_point(model, A, B, C, 'a**(2)*(a**(4) + a**(2)*b**(2) + a**(2)*c**(2) - b**(2)*c**(2))', ID="X_1627", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_32 = eval_trilinear_point(model, A, B, C, 'a**(3)', ID="X_32", classes=["center", "incident"])
            central_line(model, X_2, X_32, 2, 32, incident=True)
            X_25 = eval_trilinear_point(model, A, B, C, 'sin(A)*tan(A)', ID="X_25", classes=["center", "incident"])
            X_111 = eval_trilinear_point(model, A, B, C, 'a/(2*a**(2) - b**(2) - c**(2))', ID="X_111", classes=["center", "incident"])
            central_line(model, X_25, X_111, 25, 111, incident=True)
            X_110 = eval_trilinear_point(model, A, B, C, 'csc(B - C)', ID="X_110", classes=["center", "incident"])
            X_699 = eval_trilinear_point(model, A, B, C, '(1/(a **(-5)*(b **(-2) + c **(-2)) - a **(-3)*(b **(-4) + c **(-4))))', ID="X_699", classes=["center", "incident"])
            central_line(model, X_110, X_699, 110, 699, incident=True)
            return promote_point_to_top(model, X_1627)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1627 on the given reference triangle."""
            model = create_model('X_1627 = TCC-PERSPECTOR OF X_251')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01627(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01627.json"))
            write_center_svg(
                model_acute,
                here / "x-01627.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
