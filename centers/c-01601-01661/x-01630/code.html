X1630¶

TCC-PERSPECTOR OF X284

X1630 is the tcc-perspector of X284.
"""GEOMETOR construction model for X_1630 = TCC-PERSPECTOR OF X_284."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            intersect_lines,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01630(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1630 = TCC-PERSPECTOR OF X_284 on reference triangle ABC."""
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center"])
            X_19 = eval_trilinear_point(model, A, B, C, 'tan(A)', ID="X_19", classes=["center"])
            X_101 = eval_trilinear_point(model, A, B, C, 'a/(b - c)', ID="X_101", classes=["center"])
            X_102 = eval_trilinear_point(model, A, B, C, '1/(sin(B)*(sec(A) - sec(B)) + sin(C)*(sec(A) - sec(C)))', ID="X_102", classes=["center"])
            central_line(model, X_1, X_19, 1, 19)
            central_line(model, X_101, X_102, 101, 102)
            X_1630 = intersect_lines(model, X_1, X_19, X_101, X_102, ID="X_1630", classes=["selected", "target"])
            X_109 = eval_trilinear_point(model, A, B, C, 'a/(cos(B) - cos(C))', ID="X_109", classes=["center", "incident"])
            X_577 = eval_trilinear_point(model, A, B, C, 'sin(A)*(cos(A))**(2)', ID="X_577", classes=["center", "incident"])
            central_line(model, X_109, X_577, 109, 577, incident=True)
            return promote_point_to_top(model, X_1630)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1630 on the given reference triangle."""
            model = create_model('X_1630 = TCC-PERSPECTOR OF X_284')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01630(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01630.json"))
            write_center_svg(
                model_acute,
                here / "x-01630.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
