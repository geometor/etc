X1624¶

TCC-PERSPECTOR OF X162

X1624 is the tcc-perspector of X162.
"""GEOMETOR construction model for X_1624 = TCC-PERSPECTOR OF X_162."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            intersect_lines,
            point_from_barycentrics,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01624(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1624 = TCC-PERSPECTOR OF X_162 on reference triangle ABC."""
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center"])
            X_113 = eval_trilinear_point(model, A, B, C, 'sin(B)*sin(C)*((sin(C))/(cos(C) - 2*cos(A)*cos(B)) + (sin(B))/(cos(B) - 2*cos(A)*cos(C)))', ID="X_113", classes=["center"])
            X_25 = eval_trilinear_point(model, A, B, C, 'sin(A)*tan(A)', ID="X_25", classes=["center"])
            X_132 = point_from_barycentrics(model, A, B, C, 0.11002416, 0.049913231, 0.84006261, ID="X_132", classes=["center"])
            central_line(model, X_3, X_113, 3, 113)
            central_line(model, X_25, X_132, 25, 132)
            X_1624 = intersect_lines(model, X_3, X_113, X_25, X_132, ID="X_1624", classes=["selected", "target"])
            X_110 = eval_trilinear_point(model, A, B, C, 'csc(B - C)', ID="X_110", classes=["center", "incident"])
            X_351 = eval_trilinear_point(model, A, B, C, '(a*(b**(2) - c**(2))*(b**(2) + c**(2) - 2*a**(2)))', ID="X_351", classes=["center", "incident"])
            central_line(model, X_110, X_351, 110, 351, incident=True)
            return promote_point_to_top(model, X_1624)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1624 on the given reference triangle."""
            model = create_model('X_1624 = TCC-PERSPECTOR OF X_162')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01624(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01624.json"))
            write_center_svg(
                model_acute,
                here / "x-01624.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
