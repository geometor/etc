X1635¶

TRIPOLAR CENTROID OF X1

X1635 is the tripolar centroid of X1.
"""GEOMETOR construction model for X_1635 = TRIPOLAR CENTROID OF X_1."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01635(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1635 = TRIPOLAR CENTROID OF X_1 on reference triangle ABC."""
            X_1635 = eval_trilinear_point(model, A, B, C, '(b - c)*(b + c - 2*a)', ID="X_1635", classes=["selected", "target"])
            X_100 = eval_trilinear_point(model, A, B, C, '1/(b - c)', ID="X_100", classes=["center", "incident"])
            X_101 = eval_trilinear_point(model, A, B, C, 'a/(b - c)', ID="X_101", classes=["center", "incident"])
            central_line(model, X_100, X_101, 100, 101, incident=True)
            X_44 = eval_trilinear_point(model, A, B, C, 'b + c - 2*a', ID="X_44", classes=["center", "incident"])
            X_513 = eval_trilinear_point(model, A, B, C, 'b - c', ID="X_513", classes=["center", "incident"])
            central_line(model, X_44, X_513, 44, 513, incident=True)
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_812 = eval_trilinear_point(model, A, B, C, 'a **(-2)*(b**(1) - c**(1)) + a**(0)*(b **(-1) - c **(-1))', ID="X_812", classes=["center", "incident"])
            central_line(model, X_2, X_812, 2, 812, incident=True)
            return promote_point_to_top(model, X_1635)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1635 on the given reference triangle."""
            model = create_model('X_1635 = TRIPOLAR CENTROID OF X_1')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01635(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01635.json"))
            write_center_svg(
                model_acute,
                here / "x-01635.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
