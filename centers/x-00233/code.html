X(233) = X(2)-CEVA CONJUGATE OF X(140) — Code¶

X(233) = isogonal conjugate of X(288)
"""GEOMETOR construction model for X(233) = X(2)-CEVA CONJUGATE OF X(140)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            nine_point_center_of,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00233(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X(233) = X(2)-CEVA CONJUGATE OF X(140) on reference triangle ABC."""
            X_233 = eval_trilinear_point(model, A, B, C, '(b*cos(C - A) + c*cos(B - A))*cos(B - C)', ID="X(233)", classes=["selected", "target"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X(6)", classes=["center", "incident"])
            X_17 = eval_trilinear_point(model, A, B, C, 'csc(A + pi /6)', ID="X(17)", classes=["center", "incident"])
            central_line(model, X_6, X_17, 6, 17, incident=True)
            X_5 = nine_point_center_of(model, A, B, C, ID="X(5)", classes=["center", "incident"])
            X_53 = eval_trilinear_point(model, A, B, C, 'tan(A)*cos(B - C)', ID="X(53)", classes=["center", "incident"])
            central_line(model, X_5, X_53, 5, 53, incident=True)
            X_2 = centroid_of(model, A, B, C, ID="X(2)", classes=["center", "incident"])
            X_95 = eval_trilinear_point(model, A, B, C, 'b**(2)*c**(2)*sec(B - C)', ID="X(95)", classes=["center", "incident"])
            central_line(model, X_2, X_95, 2, 95, incident=True)
            return promote_point_to_top(model, X_233)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X(233) on the given reference triangle."""
            model = create_model('X(233) = X(2)-CEVA CONJUGATE OF X(140)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00233(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00233.json"))
            write_center_svg(
                model_acute,
                here / "x-00233.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
