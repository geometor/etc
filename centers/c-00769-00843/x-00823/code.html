X823¶

ISOGONAL CONJUGATE OF X822

X823 is the isogonal conjugate of X822.
"""GEOMETOR construction model for X_823 = ISOGONAL CONJUGATE OF X_822."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            isogonal_conjugate,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00823(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_823 = ISOGONAL CONJUGATE OF X_822 on reference triangle ABC."""
            X_822 = eval_trilinear_point(model, A, B, C, '(sec(B))**(2) - (sec(C))**(2)', ID="X_822", classes=["center"])
            X_823 = isogonal_conjugate(model, A, B, C, X_822, ID="X_823", classes=["selected", "target"])
            X_100 = eval_trilinear_point(model, A, B, C, '1/(b - c)', ID="X_100", classes=["center", "incident"])
            X_107 = eval_trilinear_point(model, A, B, C, '1/(cos(A)*(sin(2*B) - sin(2*C)))', ID="X_107", classes=["center", "incident"])
            central_line(model, X_100, X_107, 100, 107, incident=True)
            X_264 = eval_trilinear_point(model, A, B, C, 'csc(A)*csc(2*A)', ID="X_264", classes=["center", "incident"])
            X_379 = eval_trilinear_point(model, A, B, C, '(b*c*(a**(5) + (b + c)*(b*c*a**(2) - (b*c + c*a + a*b)*(b - c)**(2))))', ID="X_379", classes=["center", "incident"])
            central_line(model, X_264, X_379, 264, 379, incident=True)
            X_110 = eval_trilinear_point(model, A, B, C, 'csc(B - C)', ID="X_110", classes=["center", "incident"])
            X_681 = eval_trilinear_point(model, A, B, C, '(((sec(A))**(2))/(sin(B)*(cos(B))**(2) - sin(C)*(cos(C))**(2)))', ID="X_681", classes=["center", "incident"])
            central_line(model, X_110, X_681, 110, 681, incident=True)
            return promote_point_to_top(model, X_823)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_823 on the given reference triangle."""
            model = create_model('X_823 = ISOGONAL CONJUGATE OF X_822')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00823(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00823.json"))
            write_center_svg(
                model_acute,
                here / "x-00823.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
