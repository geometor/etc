X754¶

EVEN (0, 4) INFINITY POINT

As the isogonal conjugate of a point on the circumcircle, X754 lies on the line at infinity.
"""GEOMETOR construction model for X_754 = EVEN (0, 4) INFINITY POINT."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00754(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_754 = EVEN (0, 4) INFINITY POINT on reference triangle ABC."""
            X_754 = eval_trilinear_point(model, A, B, C, '(a **(-1)*(b**(4) + c**(4)) - a**(3)*(b**(0) + c**(0)))', ID="X_754", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_32 = eval_trilinear_point(model, A, B, C, 'a**(3)', ID="X_32", classes=["center", "incident"])
            central_line(model, X_2, X_32, 2, 32, incident=True)
            X_115 = eval_trilinear_point(model, A, B, C, 'b*c*(b**(2) - c**(2))**(2)', ID="X_115", classes=["center", "incident"])
            X_316 = eval_trilinear_point(model, A, B, C, '(b*c*(b**(4) + c**(4) - a**(4) - b**(2)*c**(2)))', ID="X_316", classes=["center", "incident"])
            central_line(model, X_115, X_316, 115, 316, incident=True)
            X_187 = eval_trilinear_point(model, A, B, C, 'a*(2*a**(2) - b**(2) - c**(2))', ID="X_187", classes=["center", "incident"])
            X_325 = eval_trilinear_point(model, A, B, C, '(csc(A))**(2)', ID="X_325", classes=["center", "incident"])
            central_line(model, X_187, X_325, 187, 325, incident=True)
            return promote_point_to_top(model, X_754)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_754 on the given reference triangle."""
            model = create_model('X_754 = EVEN (0, 4) INFINITY POINT')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00754(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00754.json"))
            write_center_svg(
                model_acute,
                here / "x-00754.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
