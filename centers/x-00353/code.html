X(353) = INVERSE-IN-BROCARD-CIRCLE OF X(111) — Code¶

Let AaBaCa, AbBbCb, AcBcCc be the A-, B- and C-anti-altimedial triangles. Let A’ be the trilinear product Aa*Ab*Ac, and define B’ and C’ cyclically. Triangle A’B’C’ is the anticomplementary triangle of the 1st Brocard triangle, and is similar to the anti-orthocentroidal triangle, with similitude center X(353). (Randy Hutson, November 2, 2017)
"""GEOMETOR construction model for X(353) = INVERSE-IN-BROCARD-CIRCLE OF X(111)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00353(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X(353) = INVERSE-IN-BROCARD-CIRCLE OF X(111) on reference triangle ABC."""
            X_353 = eval_trilinear_point(model, A, B, C, 'a*(4*a**(4)- 2*b**(4)- 2*c**(4)- b**(2)*c**(2)- 4*a**(2)*b**(2)-4*a**(2)*c**(2))', ID="X(353)", classes=["selected", "target"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X(6)", classes=["center", "incident"])
            X_23 = eval_trilinear_point(model, A, B, C, 'a*(b**(4) + c**(4) - a**(4) - b**(2)*c**(2))', ID="X(23)", classes=["center", "incident"])
            central_line(model, X_6, X_23, 6, 23, incident=True)
            X_111 = eval_trilinear_point(model, A, B, C, 'a/(2*a**(2) - b**(2) - c**(2))', ID="X(111)", classes=["center", "incident"])
            X_182 = eval_trilinear_point(model, A, B, C, 'cos(A- omega )', ID="X(182)", classes=["center", "incident"])
            central_line(model, X_111, X_182, 111, 182, incident=True)
            X_3 = circumcenter_of(model, A, B, C, ID="X(3)", classes=["center", "incident"])
            X_352 = eval_trilinear_point(model, A, B, C, 'a*(-a**(4)- b**(4)- c**(4)- 5*b**(2)*c**(2)+ 4*a**(2)*b**(2)+ 4*a**(2)*c**(2))', ID="X(352)", classes=["center", "incident"])
            central_line(model, X_3, X_352, 3, 352, incident=True)
            return promote_point_to_top(model, X_353)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X(353) on the given reference triangle."""
            model = create_model('X(353) = INVERSE-IN-BROCARD-CIRCLE OF X(111)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00353(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00353.json"))
            write_center_svg(
                model_acute,
                here / "x-00353.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
