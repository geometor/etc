X(696) = EVEN (- 4, - 3) INFINITY POINT — Code¶

If m and n are distinct integers, this form fits the definition of even polynomial center as in Clark Kimberling, “Functional equations associated with triangle geometry,” Aequationes Mathematicae 45 (1993) 127-152. This form, perhaps appearing initially here (July 7, 2001) defines a triangle center for arbitrary distinct real numbers m and n. Selected even infinity and circumcircle points begin at X(696); odd ones begin at X(768).
"""GEOMETOR construction model for X(696) = EVEN (- 4, - 3) INFINITY POINT."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00696(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X(696) = EVEN (- 4, - 3) INFINITY POINT on reference triangle ABC."""
            X_696 = eval_trilinear_point(model, A, B, C, 'a**(-5)*(b**(-3)+ c**(-3)) - a**(-4)*(b**(-4)+ c**(-4))', ID="X(696)", classes=["selected", "target"])
            X_30 = eval_trilinear_point(model, A, B, C, 'cos(A) - 2*cos(B)*cos(C)', ID="X(30)", classes=["center", "incident"])
            X_511 = eval_trilinear_point(model, A, B, C, 'cos(A + omega )', ID="X(511)", classes=["center", "incident"])
            central_line(model, X_30, X_511, 30, 511, incident=True)
            X_313 = eval_trilinear_point(model, A, B, C, '(b + c)*b**(3)*c**(3)', ID="X(313)", classes=["center", "incident"])
            X_561 = eval_trilinear_point(model, A, B, C, 'a **(- 4)', ID="X(561)", classes=["center", "incident"])
            central_line(model, X_313, X_561, 313, 561, incident=True)
            return promote_point_to_top(model, X_696)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X(696) on the given reference triangle."""
            model = create_model('X(696) = EVEN (- 4, - 3) INFINITY POINT')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00696(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00696.json"))
            write_center_svg(
                model_acute,
                here / "x-00696.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
