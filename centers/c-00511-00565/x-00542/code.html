X542¶

DIRECTION OF VECTOR AX + BX + CX, where X = X98

X542 = crossdifference of every pair of points on line X6X526
"""GEOMETOR construction model for X_542 = DIRECTION OF VECTOR AX + BX + CX, where X = X_98."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00542(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_542 = DIRECTION OF VECTOR AX + BX + CX, where X = X_98 on reference triangle ABC."""
            X_542 = eval_barycentric_point(model, A, B, C, '2*a**(6) - 2*a**(4)*(b**(2) + c**(2)) + a**(2)*(b**(4) + c**(4)) - (b**(2) - c**(2))**(2)*(b**(2) + c**(2))', ID="X_542", classes=["selected", "target"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_13 = eval_trilinear_point(model, A, B, C, 'csc(A +  pi  /3)', ID="X_13", classes=["center", "incident"])
            central_line(model, X_6, X_13, 6, 13, incident=True)
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center", "incident"])
            X_67 = eval_trilinear_point(model, A, B, C, 'b*c/(b**(4) + c**(4) - a**(4) - b**(2)*c**(2))', ID="X_67", classes=["center", "incident"])
            central_line(model, X_3, X_67, 3, 67, incident=True)
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_98 = eval_trilinear_point(model, A, B, C, 'sec(A +  omega  )', ID="X_98", classes=["center", "incident"])
            central_line(model, X_2, X_98, 2, 98, incident=True)
            return promote_point_to_top(model, X_542)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_542 on the given reference triangle."""
            model = create_model('X_542 = DIRECTION OF VECTOR AX + BX + CX, where X = X_98')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00542(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00542.json"))
            write_center_svg(
                model_acute,
                here / "x-00542.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
