X533¶

DIRECTION OF VECTOR AX + BX + CX, where X = X18

X533 = isogonal conjugate of X2381
"""GEOMETOR construction model for X_533 = DIRECTION OF VECTOR AX + BX + CX, where X = X_18."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            intersect_lines,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00533(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_533 = DIRECTION OF VECTOR AX + BX + CX, where X = X_18 on reference triangle ABC."""
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center"])
            X_18 = eval_trilinear_point(model, A, B, C, 'csc(A -  pi  /6)', ID="X_18", classes=["center"])
            X_14 = eval_trilinear_point(model, A, B, C, 'csc(A -  pi  /3)', ID="X_14", classes=["center"])
            X_299 = eval_trilinear_point(model, A, B, C, '(csc(A))**(2)*sin(A -  pi  /3)', ID="X_299", classes=["center"])
            central_line(model, X_2, X_18, 2, 18)
            central_line(model, X_14, X_299, 14, 299)
            X_533 = intersect_lines(model, X_2, X_18, X_14, X_299, ID="X_533", classes=["selected", "target"])
            X_15 = eval_trilinear_point(model, A, B, C, 'sin(A +  pi  /3)', ID="X_15", classes=["center", "incident"])
            X_298 = eval_trilinear_point(model, A, B, C, '(csc(A))**(2)*sin(A +  pi  /3)', ID="X_298", classes=["center", "incident"])
            central_line(model, X_15, X_298, 15, 298, incident=True)
            return promote_point_to_top(model, X_533)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_533 on the given reference triangle."""
            model = create_model('X_533 = DIRECTION OF VECTOR AX + BX + CX, where X = X_18')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00533(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00533.json"))
            write_center_svg(
                model_acute,
                here / "x-00533.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
