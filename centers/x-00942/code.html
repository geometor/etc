X(942) = INCIRCLE-INVERSE OF X(36) — Code¶

Let A’B’C’ be the incentral triangle of triangle ABC. Let LA be the line of reflection of line BC in line B’C’, and define LB and LC cyclically. Let A’’ = LB∩LC, and define B’’ and C’’ cyclically. The lines AA’’, BB’’, CC’’ concur in X(942). (Randy Hutson, 9/23/2011)
"""GEOMETOR construction model for X(942) = INCIRCLE-INVERSE OF X(36)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            gergonne_point_of,
            incenter_of,
            midpoint,
            orthocenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00942(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X(942) = INCIRCLE-INVERSE OF X(36) on reference triangle ABC."""
            X_1 = incenter_of(model, A, B, C, ID="X(1)", classes=["center"])
            X_65 = eval_trilinear_point(model, A, B, C, 'cos(B) + cos(C)', ID="X(65)", classes=["center"])
            X_942 = midpoint(model, X_1, X_65, ID="X(942)", classes=["selected", "target"], guide=False)
            X_3 = circumcenter_of(model, A, B, C, ID="X(3)", classes=["center", "incident"])
            central_line(model, X_1, X_3, 1, 3, incident=True)
            X_4 = orthocenter_of(model, A, B, C, ID="X(4)", classes=["center", "incident"])
            X_7 = gergonne_point_of(model, A, B, C, ID="X(7)", classes=["center", "incident"])
            central_line(model, X_4, X_7, 4, 7, incident=True)
            X_2 = centroid_of(model, A, B, C, ID="X(2)", classes=["center", "incident"])
            X_72 = eval_trilinear_point(model, A, B, C, '(b + c)*cot(A)', ID="X(72)", classes=["center", "incident"])
            central_line(model, X_2, X_72, 2, 72, incident=True)
            return promote_point_to_top(model, X_942)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X(942) on the given reference triangle."""
            model = create_model('X(942) = INCIRCLE-INVERSE OF X(36)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00942(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00942.json"))
            write_center_svg(
                model_acute,
                here / "x-00942.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
