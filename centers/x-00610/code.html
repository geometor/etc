X(610) = X(63)-CEVA CONJUGATE OF X(1) — Code¶

Barycentrics a(area - a2 cot A) : b(area - b2 cot B) : c(area - c2 cot C)
"""GEOMETOR construction model for X(610) = X(63)-CEVA CONJUGATE OF X(1)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            ceva_conjugate,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            mittenpunkt_of,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00610(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X(610) = X(63)-CEVA CONJUGATE OF X(1) on reference triangle ABC."""
            X_63 = eval_trilinear_point(model, A, B, C, 'cot(A)', ID="X(63)", classes=["center"])
            X_1 = incenter_of(model, A, B, C, ID="X(1)", classes=["center"])
            X_610 = ceva_conjugate(model, A, B, C, X_63, X_1, ID="X(610)", classes=["selected", "target"])
            X_3 = circumcenter_of(model, A, B, C, ID="X(3)", classes=["center", "incident"])
            X_9 = mittenpunkt_of(model, A, B, C, ID="X(9)", classes=["center", "incident"])
            central_line(model, X_3, X_9, 3, 9, incident=True)
            X_19 = eval_trilinear_point(model, A, B, C, 'tan(A)', ID="X(19)", classes=["center", "incident"])
            central_line(model, X_1, X_19, 1, 19, incident=True)
            X_6 = symmedian_point_of(model, A, B, C, ID="X(6)", classes=["center", "incident"])
            X_57 = eval_trilinear_point(model, A, B, C, '1/(b + c - a)', ID="X(57)", classes=["center", "incident"])
            central_line(model, X_6, X_57, 6, 57, incident=True)
            return promote_point_to_top(model, X_610)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X(610) on the given reference triangle."""
            model = create_model('X(610) = X(63)-CEVA CONJUGATE OF X(1)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00610(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00610.json"))
            write_center_svg(
                model_acute,
                here / "x-00610.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
