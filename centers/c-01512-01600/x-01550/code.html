ORTHOJOIN OF X110

X1550 = orthopole of line X115X125
"""GEOMETOR construction model for X_1550 = ORTHOJOIN OF X_110."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            intersect_lines,
            orthocenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01550(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1550 = ORTHOJOIN OF X_110 on reference triangle ABC."""
            X_30 = eval_trilinear_point(model, A, B, C, 'cos(A) - 2*cos(B)*cos(C)', ID="X_30", classes=["center"])
            X_74 = eval_trilinear_point(model, A, B, C, '1/(cos(A) - 2*cos(B)*cos(C))', ID="X_74", classes=["center"])
            X_98 = eval_trilinear_point(model, A, B, C, 'sec(A +  omega  )', ID="X_98", classes=["center"])
            X_230 = eval_trilinear_point(model, A, B, C, 'b*c*(a**(2)*(2*a**(2) - b**(2) - c**(2)) + (b**(2) - c**(2))**(2))', ID="X_230", classes=["center"])
            central_line(model, X_30, X_74, 30, 74)
            central_line(model, X_98, X_230, 98, 230)
            X_1550 = intersect_lines(model, X_30, X_74, X_98, X_230, ID="X_1550", classes=["selected", "target"])
            X_4 = orthocenter_of(model, A, B, C, ID="X_4", classes=["center", "incident"])
            X_1499 = eval_trilinear_point(model, A, B, C, 'b*c*(b**(2) - c**(2))*(b**(2) + c**(2) - 5*a**(2))', ID="X_1499", classes=["center", "incident"])
            central_line(model, X_4, X_1499, 4, 1499, incident=True)
            return promote_point_to_top(model, X_1550)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1550 on the given reference triangle."""
            model = create_model('X_1550 = ORTHOJOIN OF X_110')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01550(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01550.json"))
            write_center_svg(
                model_acute,
                here / "x-01550.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
