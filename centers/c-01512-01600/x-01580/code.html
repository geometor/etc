X1580¶

POINT ALUDRA

X1580 = crossdifference of every pair of points on line X38X661
"""GEOMETOR construction model for X_1580 = POINT ALUDRA."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01580(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1580 = POINT ALUDRA on reference triangle ABC."""
            X_1580 = eval_trilinear_point(model, A, B, C, 'a**(4) - b**(2)*c**(2)', ID="X_1580", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_21 = eval_trilinear_point(model, A, B, C, '1/(cos(B) + cos(C))', ID="X_21", classes=["center", "incident"])
            central_line(model, X_1, X_21, 1, 21, incident=True)
            X_41 = eval_trilinear_point(model, A, B, C, 'a**(2)*(b + c - a)', ID="X_41", classes=["center", "incident"])
            X_43 = eval_trilinear_point(model, A, B, C, 'a*b + a*c - b*c', ID="X_43", classes=["center", "incident"])
            central_line(model, X_41, X_43, 41, 43, incident=True)
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_256 = eval_trilinear_point(model, A, B, C, '1/(a**(2) + b*c)', ID="X_256", classes=["center", "incident"])
            central_line(model, X_6, X_256, 6, 256, incident=True)
            return promote_point_to_top(model, X_1580)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1580 on the given reference triangle."""
            model = create_model('X_1580 = POINT ALUDRA')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01580(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01580.json"))
            write_center_svg(
                model_acute,
                here / "x-01580.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
