X1513¶

ORTHOJOIN OF X2

X1513 = orthopole of Lemoine axis
"""GEOMETOR construction model for X_1513 = ORTHOJOIN OF X_2."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            point_from_barycentrics,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01513(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1513 = ORTHOJOIN OF X_2 on reference triangle ABC."""
            X_1513 = eval_trilinear_point(model, A, B, C, '(b*c*(b**(4) + c**(4) - a**(2)*b**(2) - a**(2)*c**(2))*(3*a**(4) + (b**(2) - c**(2))**(2)))', ID="X_1513", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center"])
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center"])
            central_line(model, X_2, X_3, 2, 3)
            X_98 = eval_trilinear_point(model, A, B, C, 'sec(A +  omega  )', ID="X_98", classes=["center", "incident"])
            X_230 = eval_trilinear_point(model, A, B, C, 'b*c*(a**(2)*(2*a**(2) - b**(2) - c**(2)) + (b**(2) - c**(2))**(2))', ID="X_230", classes=["center", "incident"])
            central_line(model, X_98, X_230, 98, 230, incident=True)
            X_132 = point_from_barycentrics(model, A, B, C, 0.11002416, 0.049913231, 0.84006261, ID="X_132", classes=["center", "incident"])
            X_232 = eval_trilinear_point(model, A, B, C, 'tan(A)*cos(A +  omega  )', ID="X_232", classes=["center", "incident"])
            central_line(model, X_132, X_232, 132, 232, incident=True)
            return promote_point_to_top(model, X_1513)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1513 on the given reference triangle."""
            model = create_model('X_1513 = ORTHOJOIN OF X_2')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01513(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01513.json"))
            write_center_svg(
                model_acute,
                here / "x-01513.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
