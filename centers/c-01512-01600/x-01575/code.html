X1575¶

EXSIMILICENTER OF SPIEKER AND (1/2)-MOSES CIRCLES

The Moses circle, M, is introduced at X1015; the (1/2)-Moses circle is concentric to M with half the radius of M. The insimilicenter of the Spieker and (1/2)-Moses circles is X1107.
"""GEOMETOR construction model for X_1575 = EXSIMILICENTER OF SPIEKER AND (1/2)-MOSES CIRCLES."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            complement,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            promote_point_to_top,
            spieker_center_of,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01575(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1575 = EXSIMILICENTER OF SPIEKER AND (1/2)-MOSES CIRCLES on reference triangle ABC."""
            X_350 = eval_trilinear_point(model, A, B, C, '(a**(2) - b*c)*b**(2)*c**(2)', ID="X_350", classes=["center"])
            X_1575 = complement(model, A, B, C, X_350, ID="X_1575", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_37 = eval_trilinear_point(model, A, B, C, 'b + c', ID="X_37", classes=["center", "incident"])
            central_line(model, X_2, X_37, 2, 37, incident=True)
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_43 = eval_trilinear_point(model, A, B, C, 'a*b + a*c - b*c', ID="X_43", classes=["center", "incident"])
            central_line(model, X_6, X_43, 6, 43, incident=True)
            X_10 = spieker_center_of(model, A, B, C, ID="X_10", classes=["center", "incident"])
            X_39 = eval_trilinear_point(model, A, B, C, 'a*(b**(2) + c**(2))', ID="X_39", classes=["center", "incident"])
            central_line(model, X_10, X_39, 10, 39, incident=True)
            return promote_point_to_top(model, X_1575)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1575 on the given reference triangle."""
            model = create_model('X_1575 = EXSIMILICENTER OF SPIEKER AND (1/2)-MOSES CIRCLES')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01575(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01575.json"))
            write_center_svg(
                model_acute,
                here / "x-01575.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
