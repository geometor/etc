X1543¶

ORTHOJOIN OF X57

"""GEOMETOR construction model for X_1543 = ORTHOJOIN OF X_57."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            incenter_of,
            intersect_lines,
            orthocenter_of,
            point_from_barycentrics,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01543(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1543 = ORTHOJOIN OF X_57 on reference triangle ABC."""
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center"])
            X_4 = orthocenter_of(model, A, B, C, ID="X_4", classes=["center"])
            X_516 = eval_trilinear_point(model, A, B, C, 'b*c*((a - b)*cot(C) + (a - c)*cot(B))', ID="X_516", classes=["center"])
            X_972 = eval_trilinear_point(model, A, B, C, '1/(a**(4)*(b + c) - 2*a**(3)*(b**(2) + c**(2) - b*c) + 2*a*(b - c)**(2)*(b**(2) + c**(2) + b*c) - (b - c)**(2)*(b + c)**(3))', ID="X_972", classes=["center"])
            central_line(model, X_1, X_4, 1, 4)
            central_line(model, X_516, X_972, 516, 972)
            X_1543 = intersect_lines(model, X_1, X_4, X_516, X_972, ID="X_1543", classes=["selected", "target"])
            X_908 = eval_barycentric_point(model, A, B, C, 'cos(B) + cos(C) - 1', ID="X_908", classes=["center", "incident"])
            X_1530 = point_from_barycentrics(model, A, B, C, 2.5426587, 1.172619, -2.7152778, ID="X_1530", classes=["center", "incident"])
            central_line(model, X_908, X_1530, 908, 1530, incident=True)
            return promote_point_to_top(model, X_1543)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1543 on the given reference triangle."""
            model = create_model('X_1543 = ORTHOJOIN OF X_57')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01543(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01543.json"))
            write_center_svg(
                model_acute,
                here / "x-01543.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
