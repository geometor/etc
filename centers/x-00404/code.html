X(404) = {X(2),X(3)}-HARMONIC CONJUGATE OF X(21) — Code¶

X(404) is the {X(2),X(3)}-harmonic conjugate of X(21). For a list of harmonic conjugates of X(404), click Tables at the top of this page.
"""GEOMETOR construction model for X(404) = {X(2),X(3)}-HARMONIC CONJUGATE OF X(21)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            circumcenter_of,
            complement,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            nagel_point_of,
            promote_point_to_top,
            spieker_center_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00404(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X(404) = {X(2),X(3)}-HARMONIC CONJUGATE OF X(21) on reference triangle ABC."""
            X_320 = eval_trilinear_point(model, A, B, C, '(1 - 2*cos(A))/a**(2)', ID="X(320)", classes=["center"])
            X_404 = complement(model, A, B, C, X_320, ID="X(404)", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X(2)", classes=["center"])
            X_3 = circumcenter_of(model, A, B, C, ID="X(3)", classes=["center"])
            central_line(model, X_2, X_3, 2, 3)
            X_10 = spieker_center_of(model, A, B, C, ID="X(10)", classes=["center", "incident"])
            X_36 = eval_trilinear_point(model, A, B, C, '1 - 2*cos(A)', ID="X(36)", classes=["center", "incident"])
            central_line(model, X_10, X_36, 10, 36, incident=True)
            X_8 = nagel_point_of(model, A, B, C, ID="X(8)", classes=["center", "incident"])
            X_56 = eval_trilinear_point(model, A, B, C, 'a/(b + c - a)', ID="X(56)", classes=["center", "incident"])
            central_line(model, X_8, X_56, 8, 56, incident=True)
            return promote_point_to_top(model, X_404)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X(404) on the given reference triangle."""
            model = create_model('X(404) = {X(2),X(3)}-HARMONIC CONJUGATE OF X(21)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00404(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00404.json"))
            write_center_svg(
                model_acute,
                here / "x-00404.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
