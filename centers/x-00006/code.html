X(6) = SYMMEDIAN POINT (LEMOINE POINT, GREBE POINT) — Code¶

LEMOINE POINT • GREBE POINT

X(6) is the point of concurrence of the symmedians (i.e., reflections of medians in corresponding angle bisectors). X(6) is the point which, when given by actual trilinear distances x,y,z, minimizes x2 + y2 + z2.
"""GEOMETOR construction model for X(6) = SYMMEDIAN POINT."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            mittenpunkt_of,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00006(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X(6) = SYMMEDIAN POINT on reference triangle ABC."""
            X_6 = symmedian_point_of(model, A, B, C, ID="X(6)", classes=["selected", "target"], draw_symmedians=True)
            X_1 = incenter_of(model, A, B, C, ID="X(1)", classes=["center", "incident"])
            X_9 = mittenpunkt_of(model, A, B, C, ID="X(9)", classes=["center", "incident"])
            central_line(model, X_1, X_9, 1, 9, incident=True)
            X_3 = circumcenter_of(model, A, B, C, ID="X(3)", classes=["center", "incident"])
            X_15 = eval_trilinear_point(model, A, B, C, 'sin(A + pi /3)', ID="X(15)", classes=["center", "incident"])
            central_line(model, X_3, X_15, 3, 15, incident=True)
            X_13 = eval_trilinear_point(model, A, B, C, 'csc(A + pi /3)', ID="X(13)", classes=["center", "incident"])
            X_14 = eval_trilinear_point(model, A, B, C, 'csc(A - pi /3)', ID="X(14)", classes=["center", "incident"])
            central_line(model, X_13, X_14, 13, 14, incident=True)
            return promote_point_to_top(model, X_6)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X(6) on the given reference triangle."""
            model = create_model('X(6) = SYMMEDIAN POINT')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00006(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00006.json"))
            write_center_svg(
                model_acute,
                here / "x-00006.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
