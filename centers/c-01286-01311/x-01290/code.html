X1290¶

COLLINGS TRANSFORM OF X36

X1290 = cevapoint of X36 and X513
"""GEOMETOR construction model for X_1290 = COLLINGS TRANSFORM OF X_36."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            intersect_lines,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01290(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1290 = COLLINGS TRANSFORM OF X_36 on reference triangle ABC."""
            X_23 = eval_trilinear_point(model, A, B, C, 'a*(b**(4) + c**(4) - a**(4) - b**(2)*c**(2))', ID="X_23", classes=["center"])
            X_105 = eval_trilinear_point(model, A, B, C, '1/(b**(2) + c**(2) - a*(b + c))', ID="X_105", classes=["center"])
            X_30 = eval_trilinear_point(model, A, B, C, 'cos(A) - 2*cos(B)*cos(C)', ID="X_30", classes=["center"])
            X_104 = eval_trilinear_point(model, A, B, C, '1/(-1 + cos(B) + cos(C))', ID="X_104", classes=["center"])
            central_line(model, X_23, X_105, 23, 105)
            central_line(model, X_30, X_104, 30, 104)
            X_1290 = intersect_lines(model, X_23, X_105, X_30, X_104, ID="X_1290", classes=["selected", "target"])
            X_74 = eval_trilinear_point(model, A, B, C, '1/(cos(A) - 2*cos(B)*cos(C))', ID="X_74", classes=["center", "incident"])
            X_517 = eval_trilinear_point(model, A, B, C, '-1 + cos(B) + cos(C)', ID="X_517", classes=["center", "incident"])
            central_line(model, X_74, X_517, 74, 517, incident=True)
            return promote_point_to_top(model, X_1290)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1290 on the given reference triangle."""
            model = create_model('X_1290 = COLLINGS TRANSFORM OF X_36')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01290(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01290.json"))
            write_center_svg(
                model_acute,
                here / "x-01290.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
