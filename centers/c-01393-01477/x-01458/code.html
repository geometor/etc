X1458¶

X518-BETH CONJUGATE OF X518

X1458 = crosssum of X1 and X516
"""GEOMETOR construction model for X_1458 = X_518-BETH CONJUGATE OF X_518."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            gergonne_point_of,
            incenter_of,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01458(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1458 = X_518-BETH CONJUGATE OF X_518 on reference triangle ABC."""
            X_1458 = eval_barycentric_point(model, A, B, C, 'a**(2)*(b**(2) + c**(2) - a*b - a*c)/(a - b - c)', ID="X_1458", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_7 = gergonne_point_of(model, A, B, C, ID="X_7", classes=["center", "incident"])
            central_line(model, X_1, X_7, 1, 7, incident=True)
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_41 = eval_trilinear_point(model, A, B, C, 'a**(2)*(b + c - a)', ID="X_41", classes=["center", "incident"])
            central_line(model, X_6, X_41, 6, 41, incident=True)
            X_36 = eval_trilinear_point(model, A, B, C, '1 - 2*cos(A)', ID="X_36", classes=["center", "incident"])
            X_59 = eval_trilinear_point(model, A, B, C, '1/(1 - cos(B - C))', ID="X_59", classes=["center", "incident"])
            central_line(model, X_36, X_59, 36, 59, incident=True)
            return promote_point_to_top(model, X_1458)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1458 on the given reference triangle."""
            model = create_model('X_1458 = X_518-BETH CONJUGATE OF X_518')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01458(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01458.json"))
            write_center_svg(
                model_acute,
                here / "x-01458.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
