X984-BETH CONJUGATE OF X984

X1469 = X1350-of-Mandart-incircle-triangle
"""GEOMETOR construction model for X_1469 = X_984-BETH CONJUGATE OF X_984."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            gergonne_point_of,
            nagel_point_of,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01469(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1469 = X_984-BETH CONJUGATE OF X_984 on reference triangle ABC."""
            X_1469 = eval_trilinear_point(model, A, B, C, 'a*(b**(2) + b*c + c**(2))/(a - b - c)', ID="X_1469", classes=["selected", "target"])
            X_7 = gergonne_point_of(model, A, B, C, ID="X_7", classes=["center", "incident"])
            X_8 = nagel_point_of(model, A, B, C, ID="X_8", classes=["center", "incident"])
            central_line(model, X_7, X_8, 7, 8, incident=True)
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_41 = eval_trilinear_point(model, A, B, C, 'a**(2)*(b + c - a)', ID="X_41", classes=["center", "incident"])
            central_line(model, X_6, X_41, 6, 41, incident=True)
            X_43 = eval_trilinear_point(model, A, B, C, 'a*b + a*c - b*c', ID="X_43", classes=["center", "incident"])
            X_57 = eval_trilinear_point(model, A, B, C, '1/(b + c - a)', ID="X_57", classes=["center", "incident"])
            central_line(model, X_43, X_57, 43, 57, incident=True)
            return promote_point_to_top(model, X_1469)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1469 on the given reference triangle."""
            model = create_model('X_1469 = X_984-BETH CONJUGATE OF X_984')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01469(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01469.json"))
            write_center_svg(
                model_acute,
                here / "x-01469.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
