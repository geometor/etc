X1397¶

X31-BETH CONJUGATE OF X31

X1397 = X560-cross conjugate of X32
"""GEOMETOR construction model for X_1397 = X_31-BETH CONJUGATE OF X_31."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01397(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1397 = X_31-BETH CONJUGATE OF X_31 on reference triangle ABC."""
            X_1397 = eval_trilinear_point(model, A, B, C, '((sin(A)*sin(A/2))**(2))', ID="X_1397", classes=["selected", "target"])
            X_56 = eval_trilinear_point(model, A, B, C, 'a/(b + c - a)', ID="X_56", classes=["center", "incident"])
            X_58 = eval_trilinear_point(model, A, B, C, 'a/(b + c)', ID="X_58", classes=["center", "incident"])
            central_line(model, X_56, X_58, 56, 58, incident=True)
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_181 = eval_trilinear_point(model, A, B, C, 'a*(b + c)**(2)/(b + c - a)', ID="X_181", classes=["center", "incident"])
            central_line(model, X_6, X_181, 6, 181, incident=True)
            X_31 = eval_trilinear_point(model, A, B, C, 'a**(2)', ID="X_31", classes=["center", "incident"])
            X_184 = eval_trilinear_point(model, A, B, C, 'a**(2)*cos(A)', ID="X_184", classes=["center", "incident"])
            central_line(model, X_31, X_184, 31, 184, incident=True)
            return promote_point_to_top(model, X_1397)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1397 on the given reference triangle."""
            model = create_model('X_1397 = X_31-BETH CONJUGATE OF X_31')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01397(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01397.json"))
            write_center_svg(
                model_acute,
                here / "x-01397.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
