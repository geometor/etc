X452-BETH CONJUGATE OF X452

"""GEOMETOR construction model for X_1453 = X_452-BETH CONJUGATE OF X_452."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01453(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1453 = X_452-BETH CONJUGATE OF X_452 on reference triangle ABC."""
            X_1453 = eval_trilinear_point(model, A, B, C, '(3*a**(3) + b**(3) + c**(3) + 3*a**(2)*b + 3*a**(2)*c + a*b**(2) + a*c**(2) - b**(2)*c - b*c**(2) + 2*a*b*c)', ID="X_1453", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            central_line(model, X_1, X_6, 1, 6, incident=True)
            X_28 = eval_trilinear_point(model, A, B, C, '(tan(A))/(b + c)', ID="X_28", classes=["center", "incident"])
            X_34 = eval_trilinear_point(model, A, B, C, '1 - sec(A)', ID="X_34", classes=["center", "incident"])
            central_line(model, X_28, X_34, 28, 34, incident=True)
            X_31 = eval_trilinear_point(model, A, B, C, 'a**(2)', ID="X_31", classes=["center", "incident"])
            X_40 = eval_trilinear_point(model, A, B, C, 'cos(B) + cos(C) - cos(A) - 1', ID="X_40", classes=["center", "incident"])
            central_line(model, X_31, X_40, 31, 40, incident=True)
            return promote_point_to_top(model, X_1453)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1453 on the given reference triangle."""
            model = create_model('X_1453 = X_452-BETH CONJUGATE OF X_452')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01453(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01453.json"))
            write_center_svg(
                model_acute,
                here / "x-01453.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
