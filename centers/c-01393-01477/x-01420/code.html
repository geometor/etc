X1420¶

X145-BETH CONJUGATE OF X145

X1420 = homothetic center of 1st Johnson-Yff triangle and cross-triangle of Aquila and anti-Aquila triangles
"""GEOMETOR construction model for X_1420 = X_145-BETH CONJUGATE OF X_145."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01420(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1420 = X_145-BETH CONJUGATE OF X_145 on reference triangle ABC."""
            X_1420 = eval_trilinear_point(model, A, B, C, '3*cos(A) + cos(B) + cos(C) - 3', ID="X_1420", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center", "incident"])
            central_line(model, X_1, X_3, 1, 3, incident=True)
            X_34 = eval_trilinear_point(model, A, B, C, '1 - sec(A)', ID="X_34", classes=["center", "incident"])
            X_106 = eval_trilinear_point(model, A, B, C, 'a/(2*a - b - c)', ID="X_106", classes=["center", "incident"])
            central_line(model, X_34, X_106, 34, 106, incident=True)
            X_84 = eval_trilinear_point(model, A, B, C, '1/(cos(B) + cos(C) - cos(A) - 1)', ID="X_84", classes=["center", "incident"])
            X_104 = eval_trilinear_point(model, A, B, C, '1/(-1 + cos(B) + cos(C))', ID="X_104", classes=["center", "incident"])
            central_line(model, X_84, X_104, 84, 104, incident=True)
            return promote_point_to_top(model, X_1420)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1420 on the given reference triangle."""
            model = create_model('X_1420 = X_145-BETH CONJUGATE OF X_145')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01420(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01420.json"))
            write_center_svg(
                model_acute,
                here / "x-01420.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
