X1451¶

X405-BETH CONJUGATE OF X405

"""GEOMETOR construction model for X_1451 = X_405-BETH CONJUGATE OF X_405."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            intersect_lines,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01451(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1451 = X_405-BETH CONJUGATE OF X_405 on reference triangle ABC."""
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center"])
            X_41 = eval_trilinear_point(model, A, B, C, 'a**(2)*(b + c - a)', ID="X_41", classes=["center"])
            X_28 = eval_trilinear_point(model, A, B, C, '(tan(A))/(b + c)', ID="X_28", classes=["center"])
            X_34 = eval_trilinear_point(model, A, B, C, '1 - sec(A)', ID="X_34", classes=["center"])
            central_line(model, X_6, X_41, 6, 41)
            central_line(model, X_28, X_34, 28, 34)
            X_1451 = intersect_lines(model, X_6, X_41, X_28, X_34, ID="X_1451", classes=["selected", "target"])
            X_31 = eval_trilinear_point(model, A, B, C, 'a**(2)', ID="X_31", classes=["center", "incident"])
            X_65 = eval_trilinear_point(model, A, B, C, 'cos(B) + cos(C)', ID="X_65", classes=["center", "incident"])
            central_line(model, X_31, X_65, 31, 65, incident=True)
            return promote_point_to_top(model, X_1451)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1451 on the given reference triangle."""
            model = create_model('X_1451 = X_405-BETH CONJUGATE OF X_405')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01451(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01451.json"))
            write_center_svg(
                model_acute,
                here / "x-01451.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
