X225-BETH CONJUGATE OF X225

X1426 = X34-Ceva conjugate of X1042
"""GEOMETOR construction model for X_1426 = X_225-BETH CONJUGATE OF X_225."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            gergonne_point_of,
            intersect_lines,
            orthocenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01426(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1426 = X_225-BETH CONJUGATE OF X_225 on reference triangle ABC."""
            X_4 = orthocenter_of(model, A, B, C, ID="X_4", classes=["center"])
            X_7 = gergonne_point_of(model, A, B, C, ID="X_7", classes=["center"])
            X_25 = eval_trilinear_point(model, A, B, C, 'sin(A)*tan(A)', ID="X_25", classes=["center"])
            X_34 = eval_trilinear_point(model, A, B, C, '1 - sec(A)', ID="X_34", classes=["center"])
            central_line(model, X_4, X_7, 4, 7)
            central_line(model, X_25, X_34, 25, 34)
            X_1426 = intersect_lines(model, X_4, X_7, X_25, X_34, ID="X_1426", classes=["selected", "target"])
            X_65 = eval_trilinear_point(model, A, B, C, 'cos(B) + cos(C)', ID="X_65", classes=["center", "incident"])
            X_225 = eval_trilinear_point(model, A, B, C, '(sec(A))*(cos(B) + cos(C))', ID="X_225", classes=["center", "incident"])
            central_line(model, X_65, X_225, 65, 225, incident=True)
            return promote_point_to_top(model, X_1426)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1426 on the given reference triangle."""
            model = create_model('X_1426 = X_225-BETH CONJUGATE OF X_225')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01426(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01426.json"))
            write_center_svg(
                model_acute,
                here / "x-01426.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
