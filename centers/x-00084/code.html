X(84) = ISOGONAL CONJUGATE OF X(40) — Code¶

Let A’,B’,C’ be the excenters. The perpendiculars from B’ to AB and from C’ to AC meet in a point A″. Points B″ and C″ are determined cyclically. The hexyl triangle, A″B″C″, is perspective to ABC, and X(84) is the perspector.
"""GEOMETOR construction model for X(84) = ISOGONAL CONJUGATE OF X(40)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            isogonal_conjugate,
            mittenpunkt_of,
            nagel_point_of,
            orthocenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00084(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X(84) = ISOGONAL CONJUGATE OF X(40) on reference triangle ABC."""
            X_40 = eval_trilinear_point(model, A, B, C, 'cos(B) + cos(C) - cos(A) - 1', ID="X(40)", classes=["center"])
            X_84 = isogonal_conjugate(model, A, B, C, X_40, ID="X(84)", classes=["selected", "target"])
            X_3 = circumcenter_of(model, A, B, C, ID="X(3)", classes=["center", "incident"])
            X_9 = mittenpunkt_of(model, A, B, C, ID="X(9)", classes=["center", "incident"])
            central_line(model, X_3, X_9, 3, 9, incident=True)
            X_8 = nagel_point_of(model, A, B, C, ID="X(8)", classes=["center", "incident"])
            X_20 = eval_trilinear_point(model, A, B, C, 'cos(A) - cos(B)*cos(C)', ID="X(20)", classes=["center", "incident"])
            central_line(model, X_8, X_20, 8, 20, incident=True)
            X_4 = orthocenter_of(model, A, B, C, ID="X(4)", classes=["center", "incident"])
            X_57 = eval_trilinear_point(model, A, B, C, '1/(b + c - a)', ID="X(57)", classes=["center", "incident"])
            central_line(model, X_4, X_57, 4, 57, incident=True)
            return promote_point_to_top(model, X_84)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X(84) on the given reference triangle."""
            model = create_model('X(84) = ISOGONAL CONJUGATE OF X(40)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00084(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00084.json"))
            write_center_svg(
                model_acute,
                here / "x-00084.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
