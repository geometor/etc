X900¶

CROSSDIFFERENCE OF X6 AND X101

As the isogonal conjugate of a point that lies on the circumcircle, X900 lies on the line at infinity.
"""GEOMETOR construction model for X_900 = CROSSDIFFERENCE OF X_6 AND X_101."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            feuerbach_point_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00900(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_900 = CROSSDIFFERENCE OF X_6 AND X_101 on reference triangle ABC."""
            X_900 = eval_trilinear_point(model, A, B, C, '(b - c)*(b + c - 2*a)/a', ID="X_900", classes=["selected", "target"])
            X_11 = feuerbach_point_of(model, A, B, C, ID="X_11", classes=["center", "incident"])
            X_244 = eval_trilinear_point(model, A, B, C, '(b - c)**(2)', ID="X_244", classes=["center", "incident"])
            central_line(model, X_11, X_244, 11, 244, incident=True)
            X_100 = eval_trilinear_point(model, A, B, C, '1/(b - c)', ID="X_100", classes=["center", "incident"])
            X_190 = eval_trilinear_point(model, A, B, C, 'b*c/(b - c)', ID="X_190", classes=["center", "incident"])
            central_line(model, X_100, X_190, 100, 190, incident=True)
            X_30 = eval_trilinear_point(model, A, B, C, 'cos(A) - 2*cos(B)*cos(C)', ID="X_30", classes=["center", "incident"])
            X_511 = eval_trilinear_point(model, A, B, C, 'cos(A +  omega  )', ID="X_511", classes=["center", "incident"])
            central_line(model, X_30, X_511, 30, 511, incident=True)
            return promote_point_to_top(model, X_900)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_900 on the given reference triangle."""
            model = create_model('X_900 = CROSSDIFFERENCE OF X_6 AND X_101')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00900(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00900.json"))
            write_center_svg(
                model_acute,
                here / "x-00900.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
