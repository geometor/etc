X869¶

INTERSECTION OF LINES X1X_2 AND X31X_32

X869 = crossdifference of every pair of points on line X649X693
"""GEOMETOR construction model for X_869 = INTERSECTION OF LINES X_1X_2 AND X_31X_32."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00869(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_869 = INTERSECTION OF LINES X_1X_2 AND X_31X_32 on reference triangle ABC."""
            X_869 = eval_trilinear_point(model, A, B, C, '(a**(2)*(b**(2) + c**(2) + b*c))', ID="X_869", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            central_line(model, X_1, X_2, 1, 2, incident=True)
            X_31 = eval_trilinear_point(model, A, B, C, 'a**(2)', ID="X_31", classes=["center", "incident"])
            X_32 = eval_trilinear_point(model, A, B, C, 'a**(3)', ID="X_32", classes=["center", "incident"])
            central_line(model, X_31, X_32, 31, 32, incident=True)
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_292 = eval_trilinear_point(model, A, B, C, 'a/(a**(2) - b*c)', ID="X_292", classes=["center", "incident"])
            central_line(model, X_6, X_292, 6, 292, incident=True)
            return promote_point_to_top(model, X_869)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_869 on the given reference triangle."""
            model = create_model('X_869 = INTERSECTION OF LINES X_1X_2 AND X_31X_32')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00869(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00869.json"))
            write_center_svg(
                model_acute,
                here / "x-00869.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
