X1767¶

MIMOSA TRANSFORM OF X347

X1767 = X522-Ceva conjugate of X1
"""GEOMETOR construction model for X_1767 = MIMOSA TRANSFORM OF X_347."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            ceva_conjugate,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01767(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1767 = MIMOSA TRANSFORM OF X_347 on reference triangle ABC."""
            X_522 = eval_trilinear_point(model, A, B, C, '(cos(B) - cos(C))*(csc(A))', ID="X_522", classes=["center"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center"])
            X_1767 = ceva_conjugate(model, A, B, C, X_522, X_1, ID="X_1767", classes=["selected", "target"])
            X_19 = eval_trilinear_point(model, A, B, C, 'tan(A)', ID="X_19", classes=["center", "incident"])
            X_57 = eval_trilinear_point(model, A, B, C, '1/(b + c - a)', ID="X_57", classes=["center", "incident"])
            central_line(model, X_19, X_57, 19, 57, incident=True)
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center", "incident"])
            X_207 = eval_trilinear_point(model, A, B, C, '((1 - sec(A))*(sec(B) + sec(C) - sec(A) - 1))', ID="X_207", classes=["center", "incident"])
            central_line(model, X_3, X_207, 3, 207, incident=True)
            X_46 = eval_trilinear_point(model, A, B, C, 'cos(B) + cos(C) - cos(A)', ID="X_46", classes=["center", "incident"])
            X_208 = eval_trilinear_point(model, A, B, C, '((1 - sec(A))*(cos(B) + cos(C) - cos(A) - 1))', ID="X_208", classes=["center", "incident"])
            central_line(model, X_46, X_208, 46, 208, incident=True)
            return promote_point_to_top(model, X_1767)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1767 on the given reference triangle."""
            model = create_model('X_1767 = MIMOSA TRANSFORM OF X_347')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01767(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01767.json"))
            write_center_svg(
                model_acute,
                here / "x-01767.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
