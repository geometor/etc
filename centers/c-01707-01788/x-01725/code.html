X1725¶

MIMOSA TRANSFORM OF X74

Let L denote the line having trilinears of X1725 as coefficients. Then L is the line passing through X3 perpendicular to the Euler line.
"""GEOMETOR construction model for X_1725 = MIMOSA TRANSFORM OF X_74."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01725(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1725 = MIMOSA TRANSFORM OF X_74 on reference triangle ABC."""
            X_1725 = eval_trilinear_point(model, A, B, C, '1 + cos(2*B) + cos(2*C)', ID="X_1725", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_21 = eval_trilinear_point(model, A, B, C, '1/(cos(B) + cos(C))', ID="X_21", classes=["center", "incident"])
            central_line(model, X_1, X_21, 1, 21, incident=True)
            X_33 = eval_trilinear_point(model, A, B, C, '1 + sec(A)', ID="X_33", classes=["center", "incident"])
            X_46 = eval_trilinear_point(model, A, B, C, 'cos(B) + cos(C) - cos(A)', ID="X_46", classes=["center", "incident"])
            central_line(model, X_33, X_46, 33, 46, incident=True)
            X_34 = eval_trilinear_point(model, A, B, C, '1 - sec(A)', ID="X_34", classes=["center", "incident"])
            X_90 = eval_trilinear_point(model, A, B, C, '1/(cos(B) + cos(C) - cos(A))', ID="X_90", classes=["center", "incident"])
            central_line(model, X_34, X_90, 34, 90, incident=True)
            return promote_point_to_top(model, X_1725)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1725 on the given reference triangle."""
            model = create_model('X_1725 = MIMOSA TRANSFORM OF X_74')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01725(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01725.json"))
            write_center_svg(
                model_acute,
                here / "x-01725.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
