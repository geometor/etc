X1744¶

MIMOSA TRANSFORM OF X226

"""GEOMETOR construction model for X_1744 = MIMOSA TRANSFORM OF X_226."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            intersect_lines,
            mittenpunkt_of,
            point_from_barycentrics,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01744(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1744 = MIMOSA TRANSFORM OF X_226 on reference triangle ABC."""
            X_9 = mittenpunkt_of(model, A, B, C, ID="X_9", classes=["center"])
            X_46 = eval_trilinear_point(model, A, B, C, 'cos(B) + cos(C) - cos(A)', ID="X_46", classes=["center"])
            X_19 = eval_trilinear_point(model, A, B, C, 'tan(A)', ID="X_19", classes=["center"])
            X_91 = eval_trilinear_point(model, A, B, C, 'sec(2*A)', ID="X_91", classes=["center"])
            central_line(model, X_9, X_46, 9, 46)
            central_line(model, X_19, X_91, 19, 91)
            X_1744 = intersect_lines(model, X_9, X_46, X_19, X_91, ID="X_1744", classes=["selected", "target"])
            X_1046 = point_from_barycentrics(model, A, B, C, 0.13236391, 0.3064877, 0.5611484, ID="X_1046", classes=["center", "incident"])
            X_1409 = eval_trilinear_point(model, A, B, C, 'a*(sec(B) + sec(C))', ID="X_1409", classes=["center", "incident"])
            central_line(model, X_1046, X_1409, 1046, 1409, incident=True)
            return promote_point_to_top(model, X_1744)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1744 on the given reference triangle."""
            model = create_model('X_1744 = MIMOSA TRANSFORM OF X_226')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01744(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01744.json"))
            write_center_svg(
                model_acute,
                here / "x-01744.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
