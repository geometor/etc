X1746¶

MIMOSA TRANSFORM OF X261

"""GEOMETOR construction model for X_1746 = MIMOSA TRANSFORM OF X_261."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            intersect_lines,
            orthocenter_of,
            point_from_barycentrics,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01746(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1746 = MIMOSA TRANSFORM OF X_261 on reference triangle ABC."""
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center"])
            X_572 = eval_trilinear_point(model, A, B, C, '(cos(A - U))', ID="X_572", classes=["center"])
            X_4 = orthocenter_of(model, A, B, C, ID="X_4", classes=["center"])
            X_580 = point_from_barycentrics(model, A, B, C, 0.002197609, 0.25263713, 0.74516526, ID="X_580", classes=["center"])
            central_line(model, X_2, X_572, 2, 572)
            central_line(model, X_4, X_580, 4, 580)
            X_1746 = intersect_lines(model, X_2, X_572, X_4, X_580, ID="X_1746", classes=["selected", "target"])
            X_57 = eval_trilinear_point(model, A, B, C, '1/(b + c - a)', ID="X_57", classes=["center", "incident"])
            X_1111 = eval_trilinear_point(model, A, B, C, '(((b - c)/a)**(2))', ID="X_1111", classes=["center", "incident"])
            central_line(model, X_57, X_1111, 57, 1111, incident=True)
            return promote_point_to_top(model, X_1746)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1746 on the given reference triangle."""
            model = create_model('X_1746 = MIMOSA TRANSFORM OF X_261')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01746(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01746.json"))
            write_center_svg(
                model_acute,
                here / "x-01746.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
