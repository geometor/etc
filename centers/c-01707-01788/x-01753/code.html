X1753¶

MIMOSA TRANSFORM OF X280

X1753 = X92-aleph conjugate of X1767
"""GEOMETOR construction model for X_1753 = MIMOSA TRANSFORM OF X_280."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            intersect_lines,
            mittenpunkt_of,
            orthocenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01753(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1753 = MIMOSA TRANSFORM OF X_280 on reference triangle ABC."""
            X_4 = orthocenter_of(model, A, B, C, ID="X_4", classes=["center"])
            X_9 = mittenpunkt_of(model, A, B, C, ID="X_9", classes=["center"])
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center"])
            X_33 = eval_trilinear_point(model, A, B, C, '1 + sec(A)', ID="X_33", classes=["center"])
            central_line(model, X_4, X_9, 4, 9)
            central_line(model, X_3, X_33, 3, 33)
            X_1753 = intersect_lines(model, X_4, X_9, X_3, X_33, ID="X_1753", classes=["selected", "target"])
            X_46 = eval_trilinear_point(model, A, B, C, 'cos(B) + cos(C) - cos(A)', ID="X_46", classes=["center", "incident"])
            X_208 = eval_trilinear_point(model, A, B, C, '((1 - sec(A))*(cos(B) + cos(C) - cos(A) - 1))', ID="X_208", classes=["center", "incident"])
            central_line(model, X_46, X_208, 46, 208, incident=True)
            return promote_point_to_top(model, X_1753)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1753 on the given reference triangle."""
            model = create_model('X_1753 = MIMOSA TRANSFORM OF X_280')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01753(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01753.json"))
            write_center_svg(
                model_acute,
                here / "x-01753.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
