X1788¶

MIMOSA TRANSFORM OF X1476

X1788 = cevapoint of X46 and X1722
"""GEOMETOR construction model for X_1788 = MIMOSA TRANSFORM OF X_1476."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            cevapoint,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            gergonne_point_of,
            nagel_point_of,
            orthocenter_of,
            point_from_barycentrics,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01788(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1788 = MIMOSA TRANSFORM OF X_1476 on reference triangle ABC."""
            X_46 = eval_trilinear_point(model, A, B, C, 'cos(B) + cos(C) - cos(A)', ID="X_46", classes=["center"])
            X_1722 = point_from_barycentrics(model, A, B, C, -0.18888889, 0.33333333, 0.85555556, ID="X_1722", classes=["center"])
            X_1788 = cevapoint(model, A, B, C, X_46, X_1722, ID="X_1788", classes=["selected", "target"])
            X_7 = gergonne_point_of(model, A, B, C, ID="X_7", classes=["center", "incident"])
            X_12 = eval_trilinear_point(model, A, B, C, '1 + cos(B - C)', ID="X_12", classes=["center", "incident"])
            central_line(model, X_7, X_12, 7, 12, incident=True)
            X_4 = orthocenter_of(model, A, B, C, ID="X_4", classes=["center", "incident"])
            central_line(model, X_4, X_46, 4, 46, incident=True)
            X_8 = nagel_point_of(model, A, B, C, ID="X_8", classes=["center", "incident"])
            X_56 = eval_trilinear_point(model, A, B, C, 'a/(b + c - a)', ID="X_56", classes=["center", "incident"])
            central_line(model, X_8, X_56, 8, 56, incident=True)
            return promote_point_to_top(model, X_1788)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1788 on the given reference triangle."""
            model = create_model('X_1788 = MIMOSA TRANSFORM OF X_1476')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01788(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01788.json"))
            write_center_svg(
                model_acute,
                here / "x-01788.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
