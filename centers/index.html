Encyclopedia of Triangle Centers¶

X(1) = INCENTER

\(1 : 1 : 1\)

X(2) = CENTROID

\(1/a : 1/b : 1/c\)

X(3) = CIRCUMCENTER

\(\cos A : \cos B : \cos C\)

X(4) = ORTHOCENTER

\(\sec A : \sec B : \sec C\)

X(5) = NINE-POINT CENTER

\(\cos(B - C) : \cos(C - A) : \cos(A - B)\)

X(6) = SYMMEDIAN POINT (LEMOINE POINT, GREBE POINT)

\(a : b : c\)

X(7) = GERGONNE POINT

\(bc/(b + c - a) : ca/(c + a - b) : ab/(a + b - c)\)

X(8) = NAGEL POINT

\((b + c - a)/a : (c + a - b)/b : (a + b - c)/c\)

X(9) = MITTENPUNKT

\(b + c - a : c + a - b : a + b - c\)

X(10) = SPIEKER CENTER

\(bc(b + c) : ca(c + a) : ab(a + b)\)

X(11) = FEUERBACH POINT

\(1 - \cos(B - C) : 1 - \cos(C - A) : 1 - \cos(A - B)\)

X(12) = {X(1),X(5)}-HARMONIC CONJUGATE OF X(11)

\(1 + \cos(B - C) : 1 + \cos(C - A) : 1 + \cos(A -B) : :\)

X(13) = 1st ISOGONIC CENTER (FERMAT POINT, TORRICELLI POINT)

\(\csc(A + \pi /3) : \csc(B + \pi /3) : \csc(C + \pi /3)\)

X(14) = 2nd ISOGONIC CENTER

\(\csc(A - \pi /3) : \csc(B - \pi /3) : \csc(C - \pi /3), = \sec(A + \pi /6) : \sec(B + \pi /6) : \sec(C + \pi /6)\)

X(15) = 1st ISODYNAMIC POINT

\(\sin(A + \pi /3) : \sin(B + \pi /3) : \sin(C + \pi /3)\)

X(16) = 2nd ISODYNAMIC POINT

\(\sin(A- \pi /3) : \sin(B - \pi /3) : \sin(C - \pi /3)\)

X(17) = 1st NAPOLEON POINT

\(\csc(A + \pi /6) : \csc(B + \pi /6) : \csc(C + \pi /6)\)

X(18) = 2nd NAPOLEON POINT

\(\csc(A - \pi /6) : \csc(B - \pi /6) : \csc(C - \pi /6)\)

X(19) = CLAWSON POINT

\(\tan A : \tan B : \tan C\)

X(20) = DE LONGCHAMPS POINT

\(\cos A - \cos B \cos C : \cos B - \cos C \cos A : \cos C - \cos A \cos B\)

X(21) = SCHIFFLER POINT

\(1/(\cos B + \cos C) : 1/(\cos C + \cos A) : 1/(\cos A + \cos B), = f(a,b,c) : f(b,c,a) : f(c,a,b)\), where f(a,b,c) = (b + c - a)/(b + c)

X(22) = EXETER POINT

\(a(b^{4} + c^{4} - a^{4}) : b(c^{4} + a^{4} - b^{4}) : c(a^{4} + b^{4} - c^{4})\)

X(23) = FAR-OUT POINT

\(a[b^{4} + c^{4} - a^{4} - b^{2}c^{2}] : :\)

X(24) = PERSPECTOR OF ABC AND ORTHIC-OF-ORTHIC TRIANGLE

\(\sec A \cos 2A : \sec B \cos 2B : \sec C \cos 2C, Tllrilinears \sec A - 2 \cos A : \sec B - 2 \cos B : \sec C - 2 \cos C\)

X(25) = HOMOTHETIC CENTER OF ORTHIC AND TANGENTIAL TRIANGLES

\(\sin A \tan A : :\)

X(26) = CIRCUMCENTER OF THE TANGENTIAL TRIANGLE

\(a[b^{2}\cos 2B + c^{2}\cos 2C - a^{2}\cos 2A] : :\)

X(27) = CEVAPOINT OF ORTHOCENTER AND CLAWSON CENTER

\((\sec A)/(b + c) : (\sec B)/(c + a) : (\sec C)/(a + b)\)

X(28) = CEVAPOINT OF X(19) AND X(25)

\((\tan A)/(b + c) : (\tan B)/(c + a) : (\tan C)/(a + b)\)

X(29) = CEVAPOINT OF INCENTER AND ORTHOCENTER

\((\sec A)/(\cos B + \cos C) : (\sec B)/(\cos C + \cos A) : (\sec C)/(\cos A + \cos B)\)

X(30) = EULER INFINITY POINT

\(\cos A - 2 \cos B \cos C : \cos B - 2 \cos C \cos A : \cos C - 2 \cos A cosB\)

X(31) = 2nd POWER POINT

\(a^{2} : b^{2} : c^{2}\)

X(32) = 3rd POWER POINT

\(a^{3} : b^{3} : c^{3}\)

X(33) = PERSPECTOR OF THE ORTHIC AND INTANGENTS TRIANGLES

\(1 + \sec A : 1 + \sec B : 1 + \sec C\)

X(34) = -BETH CONJUGATE OF X(4)

\(1 - \sec A : 1 - \sec B : 1 - \sec C\)

X(35) = {X(1),X(3)}-HARMONIC CONJUGATE OF X(36)

\(1 + 2 \cos A : 1 + 2 \cos B : 1 + 2 \cos C\)

X(36) = INVERSE-IN-CIRCUMCIRCLE OF INCENTER

\(1 - 2 \cos A : 1 - 2 \cos B : 1 - 2 \cos C\)

X(37) = CROSSPOINT OF INCENTER AND CENTROID

\(b + c : c + a : a + b\)

X(38) = CROSSPOINT OF X(1) AND X(75)

\(b^{2} + c^{2} : :\)

X(39) = BROCARD MIDPOINT

\(a(b^{2} + c^{2}) : b(c^{2} + a^{2}) : c(a^{2} + b^{2})\)

X(40) = BEVAN POINT

\(\cos B + \cos C - \cos A - 1 : :\)

X(41) = -CEVA CONJUGATE OF X(31)

\(a^{2}(b + c - a) : b^{2}(c + a - b) : c^{2}(a + b - c)\)

X(42) = CROSSPOINT OF INCENTER AND SYMMEDIAN POINT

\(a(b + c) : b(c + a) : c(a + b)\)

X(43) = -CEVA CONJUGATE OF X(1)

\(ab + ac - bc : bc + ba - ca : ca + cb - ab\)

X(44) = -LINE CONJUGATE OF X(1)

\(b + c - 2a : c + a - 2b : a + b - 2c\)

X(45) = -BETH CONJUGATE OF X(1)

\(2b + 2c - a : 2c + 2a - b : 2a + 2b - c\)

X(46) = -CEVA CONJUGATE OF X(1)

\(\cos B + \cos C - \cos A : :\)

X(47) = -BETH CONJUGATE OF X(34)

\(\cos 2A : \cos 2B : \cos 2C = f(a,b,c) : f(b,c,a) : f(c,a,b)\), where, f(a,b,c) = a2(a4 + b4 + c4 - 2a2b2 - 2a2c2)

X(48) = CROSSPOINT OF X(1) AND X(63)

\(\tan B + \tan C : :\)

X(49) = CENTER OF SINE-TRIPLE-ANGLE CIRCLE

\(\cos 3A : \cos 3B : \cos 3C\)

X(50) = -CEVA CONJUGATE OF X(184)

\(\sin 3A : :\)

X(51) = CENTROID OF ORTHIC TRIANGLE

\(a^{2}\cos(B - C) : :\)

X(52) = ORTHOCENTER OF ORTHIC TRIANGLE

\(\cos 2A \cos(B - C) : \cos 2B \cos(C - A) : \cos 2C \cos(A - B)\)

X(53) = SYMMEDIAN POINT OF ORTHIC TRIANGLE

\(\tan A \cos(B - C) : \tan B \cos(C - A) : \tan C \cos(A - B)\)

X(54) = KOSNITA POINT

\(\sec(B - C) : \sec(C - A) : \sec(A - B)\)

X(55) = INSIMILICENTER(CIRCUMCIRCLE, INCIRCLE)

\(a(b + c - a) : b(c + a - b) : c(a + b - c)\)

X(56) = EXSIMILICENTER(CIRCUMCIRCLE, INCIRCLE)

\(a/(b + c - a) : b/(c + a - b) : c/(a + b - c)\)

X(57) = ISOGONAL CONJUGATE OF X(9)

\(1/(b + c - a) : 1/(c + a - b) : 1/(a + b - c)\)

X(58) = ISOGONAL CONJUGATE OF X(10)

\(a/(b + c) : b/(c + a) : c/(a + b)\)

X(59) = ISOGONAL CONJUGATE OF X(11)

\(1/[1 - \cos(B - C)] : 1/[1 - \cos(C - A)] : 1/[1 - \cos(A - B)]\)

X(60) = ISOGONAL CONJUGATE OF X(12)

\(1/[1 + \cos(B - C)] : 1/[1 + \cos(C - A)] : 1/[1 + \cos(A - B)]\)

X(61) = ISOGONAL CONJUGATE OF X(17)

\(\sin(A + \pi /6) : \sin(B + \pi /6) : \sin(C + \pi /6)\)

X(62) = ISOGONAL CONJUGATE OF X(18)

\(\sin(A - \pi /6) : \sin(B - \pi /6) : \sin(C - \pi /6)\)

X(63) = ISOGONAL CONJUGATE OF X(19)

\(\cot A : \cot B : \cot C\)

X(64) = ISOGONAL CONJUGATE OF X(20)

\(1/(\cos A - \cos B \cos C) : 1/(\cos B - \cos C \cos A) : 1/(\cos C - \cos A \cos B)\)

X(65) = ORTHOCENTER OF THE INTOUCH TRIANGLE

\(\cos B + \cos C : \cos C + \cos A : \cos A + \cos B\)

X(66) = ISOGONAL CONJUGATE OF X(22)

\(bc/(b^{4} + c^{4} - a^{4}) : :\)

X(67) = ISOGONAL CONJUGATE OF X(23)

\(bc/(b^{4} + c^{4} - a^{4} - b^{2}c^{2}) : :\)

X(68) = PRASOLOV POINT

\(\cos A \sec 2A : \cos B \sec 2B : \cos C \sec 2C\)

X(69) = SYMMEDIAN POINT OF THE ANTICOMPLEMENTARY TRIANGLE

\((\cos A)/a^{2} : (\cos B)/b^{2} : (\cos C)/c^{2}\)

X(70) = ISOGONAL CONJUGATE OF X(26)

\(bc/[b^{2} \cos 2B + c^{2} \cos 2C - a^{2} \cos 2A] : :\)

X(71) = ISOGONAL CONJUGATE OF X(27)

\((b + c) \cos A : (c + a) \cos B : (a + b) \cos C\)

X(72) = ISOGONAL CONJUGATE OF X(28)

\((b + c) \cot A : (c + a) \cot B : (a + b) \cot C\)

X(73) = CROSSPOINT OF INCENTER AND CIRCUMCENTER

\(\sec B + \sec C : \sec C + \sec A : \sec A + \sec B\)

X(74) = ISOGONAL CONJUGATE OF EULER INFINITY POINT

\(1/(\cos A - 2 \cos B \cos C) : 1/(\cos B - 2 \cos C \cos A) : 1/(\cos C - 2 \cos A \cos B)\)

X(75) = ISOTOMIC CONJUGATE OF INCENTER

\(1/a^{2} : 1/b^{2} : 1/c^{2}\)

X(76) = 3rd BROCARD POINT

\(1/a^{3} : 1/b^{3} : 1/c^{3}\)

X(77) = ISOGONAL CONJUGATE OF X(33)

\(1/(1 + \sec A) : 1/(1 + \sec B) : 1/(1 + \sec C)\)

X(78) = ISOGONAL CONJUGATE OF X(34)

\(1/(1 - \sec A) : 1/(1 - \sec B) : 1/(1 - \sec C)\)

X(79) = ISOGONAL CONJUGATE OF X(35)

\(1/(1 + 2 \cos A) : :\)

X(80) = REFLECTION OF INCENTER IN FEUERBACH POINT

\(1/(1 - 2 \cos A) : 1/(1 - 2 \cos B) : 1/(1 - 2 \cos C)\)

X(81) = CEVAPOINT OF INCENTER AND SYMMEDIAN POINT

\(1/(b + c) : 1/(c + a) : 1/(a + b)\)

X(82) = ISOGONAL CONJUGATE OF X(38)

\(1/(b^{2} + c^{2}) : 1/(c^{2} + a^{2}) : 1/(a^{2} + b^{2})\)

X(83) = CEVAPOINT OF CENTROID AND SYMMEDIAN POINT

\(bc/(b^{2} + c^{2}) : ca/(c^{2} + a^{2}) : ab/(a^{2} + b^{2}), = \csc(A + \omega ) : \csc(B + \omega ) : \csc(C + \omega )\)

X(84) = ISOGONAL CONJUGATE OF X(40)

\(1/(\cos B + \cos C - \cos A - 1) : :\)

X(85) = ISOTOMIC CONJUGATE OF X(9)

\(b^{2}c^{2}/(b + c - a) : c^{2}a^{2}/(c + a - b) : a^{2}b^{2}/(a + b - c), = \tan(A/2) \csc^{2}A : \tan(B/2) \csc^{2}B : \tan(C/2) \csc^{2}C\)

X(86) = CEVAPOINT OF INCENTER AND CENTROID

\(bc/(b + c) : ca/(c + a) : ab/(a + b)\)

X(87) = -CROSS CONJUGATE OF X(1)

\(1/(ab + ac - bc) : 1/(bc + ba - ca) : 1/(ca + cb - ab)\)

X(88) = ISOGONAL CONJUGATE OF X(44)

\(1/(b + c - 2a) : 1/(c + a - 2b) : 1/(a + b - 2c)\)

X(89) = ISOGONAL CONJUGATE OF X(45)

\(1/(2b + 2c - a) : 1/(2c + 2a - b) : 1/(2a + 2b - c)\)

X(90) = -CROSS CONJUGATE OF X(1)

\(1/(\cos B + \cos C - \cos A) : 1/(\cos C + \cos A - \cos B) : 1/(\cos A + \cos B - \cos C)\)

X(91) = ISOGONAL CONJUGATE OF X(47)

\(\sec 2A : \sec 2B : \sec 2C\)

X(92) = CEVAPOINT OF INCENTER AND CLAWSON POINT

\(\csc 2A : \csc 2B : \csc 2C\)

X(93) = ISOGONAL CONJUGATE OF X(49)

\(\sec 3A : \sec 3B : \sec 3C\)

X(94) = ISOGONAL CONJUGATE OF X(50)

X(95) = CEVAPOINT OF CENTROID AND CIRCUMCENTER

\(b^{2}c^{2}\sec(B - C) : :\)

X(96) = ISOGONAL CONJUGATE OF X(52)

\(\sec 2A \sec(B - C) : \sec 2B \sec(C - A) : \sec 2C \sec(A - B)\)

X(97) = ISOGONAL CONJUGATE OF X(53)

\(\cot A \sec(B - C) : \cot B \sec(C - A) : \cot C \sec(A - B)\)
