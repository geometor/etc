X2-ISOCONJUGATE OF X79

X2174 = isogonal conjugate of X30690
"""GEOMETOR construction model for X_2174 = X_2-ISOCONJUGATE OF X_79."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            crosspoint,
            eval_trilinear_point,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02174(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2174 = X_2-ISOCONJUGATE OF X_79 on reference triangle ABC."""
            X_35 = eval_trilinear_point(model, A, B, C, '1 + 2*cos(A)', ID="X_35", classes=["center"])
            X_2003 = eval_trilinear_point(model, A, B, C, 'sin(3*A/2)*sec(A/2)', ID="X_2003", classes=["center"])
            X_2174 = crosspoint(model, A, B, C, X_35, X_2003, ID="X_2174", classes=["selected", "target"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_41 = eval_trilinear_point(model, A, B, C, 'a**(2)*(b + c - a)', ID="X_41", classes=["center", "incident"])
            central_line(model, X_6, X_41, 6, 41, incident=True)
            X_37 = eval_trilinear_point(model, A, B, C, 'b + c', ID="X_37", classes=["center", "incident"])
            X_101 = eval_trilinear_point(model, A, B, C, 'a/(b - c)', ID="X_101", classes=["center", "incident"])
            central_line(model, X_37, X_101, 37, 101, incident=True)
            X_199 = eval_trilinear_point(model, A, B, C, '(a*(b**(4) + c**(4) - a**(4) + (b**(2) + c**(2) - a**(2))*(b*c + c*a + a*b)))', ID="X_199", classes=["center", "incident"])
            X_209 = eval_trilinear_point(model, A, B, C, '((sin(B) + sin(C))*(sin(A) + sin(A - B) + sin(A - C)))', ID="X_209", classes=["center", "incident"])
            central_line(model, X_199, X_209, 199, 209, incident=True)
            return promote_point_to_top(model, X_2174)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2174 on the given reference triangle."""
            model = create_model('X_2174 = X_2-ISOCONJUGATE OF X_79')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02174(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02174.json"))
            write_center_svg(
                model_acute,
                here / "x-02174.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
