X2250¶

X2-ISOCONJUGATE OF X859

X2250 = crosspoint of X37 and X2245
"""GEOMETOR construction model for X_2250 = X_2-ISOCONJUGATE OF X_859."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            crosspoint,
            eval_trilinear_point,
            mittenpunkt_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02250(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2250 = X_2-ISOCONJUGATE OF X_859 on reference triangle ABC."""
            X_37 = eval_trilinear_point(model, A, B, C, 'b + c', ID="X_37", classes=["center"])
            X_2245 = eval_trilinear_point(model, A, B, C, '(sin(A) + sin(B) + sin(C))*cos(A) - (cos(A) + cos(B) + cos(C))*sin(A)', ID="X_2245", classes=["center"])
            X_2250 = crosspoint(model, A, B, C, X_37, X_2245, ID="X_2250", classes=["selected", "target"])
            X_9 = mittenpunkt_of(model, A, B, C, ID="X_9", classes=["center", "incident"])
            X_48 = eval_trilinear_point(model, A, B, C, 'tan(B) + tan(C)', ID="X_48", classes=["center", "incident"])
            central_line(model, X_9, X_48, 9, 48, incident=True)
            X_31 = eval_trilinear_point(model, A, B, C, 'a**(2)', ID="X_31", classes=["center", "incident"])
            X_33 = eval_trilinear_point(model, A, B, C, '1 + sec(A)', ID="X_33", classes=["center", "incident"])
            central_line(model, X_31, X_33, 31, 33, incident=True)
            X_63 = eval_trilinear_point(model, A, B, C, 'cot(A)', ID="X_63", classes=["center", "incident"])
            X_190 = eval_trilinear_point(model, A, B, C, 'b*c/(b - c)', ID="X_190", classes=["center", "incident"])
            central_line(model, X_63, X_190, 63, 190, incident=True)
            return promote_point_to_top(model, X_2250)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2250 on the given reference triangle."""
            model = create_model('X_2250 = X_2-ISOCONJUGATE OF X_859')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02250(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02250.json"))
            write_center_svg(
                model_acute,
                here / "x-02250.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
