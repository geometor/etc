X2192¶

X2-ISOCONJUGATE OF X223

X2192 = isogonal conjugate of X347
"""GEOMETOR construction model for X_2192 = X_2-ISOCONJUGATE OF X_223."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            isogonal_conjugate,
            point_from_barycentrics,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02192(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2192 = X_2-ISOCONJUGATE OF X_223 on reference triangle ABC."""
            X_347 = point_from_barycentrics(model, A, B, C, 0.57303371, 0.49438202, -0.06741573, ID="X_347", classes=["center"])
            X_2192 = isogonal_conjugate(model, A, B, C, X_347, ID="X_2192", classes=["selected", "target"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_33 = eval_trilinear_point(model, A, B, C, '1 + sec(A)', ID="X_33", classes=["center", "incident"])
            central_line(model, X_6, X_33, 6, 33, incident=True)
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_84 = eval_trilinear_point(model, A, B, C, '1/(cos(B) + cos(C) - cos(A) - 1)', ID="X_84", classes=["center", "incident"])
            central_line(model, X_1, X_84, 1, 84, incident=True)
            X_48 = eval_trilinear_point(model, A, B, C, 'tan(B) + tan(C)', ID="X_48", classes=["center", "incident"])
            X_55 = eval_trilinear_point(model, A, B, C, 'a*(b + c - a)', ID="X_55", classes=["center", "incident"])
            central_line(model, X_48, X_55, 48, 55, incident=True)
            return promote_point_to_top(model, X_2192)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2192 on the given reference triangle."""
            model = create_model('X_2192 = X_2-ISOCONJUGATE OF X_223')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02192(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02192.json"))
            write_center_svg(
                model_acute,
                here / "x-02192.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
