X2163¶

X2-ISOCONJUGATE OF X45

Let IaIbIc be the excentral triangle of a triangle ABC. Let A2 and A3 be the points where the perpendicular bisector of BC meets the internal angle bisector of angle ABC and the internal angle bisector of angle BCA, respectively. Let A'2 and A'3 be the points where the perpendicular bisector of BC meets the external angle bisector of angle ABC and the external angle bisector of angle BCA, respectively. Let Oab, Oac be the circumcenters of triangles IbA2A'3, IcA3A'2, respectively. Define Obc, Oba, Oca, Ocb cyclically. Then the triangle having sidelines OabOac, ObcOba, OcaOcb is perspective to ABC, and the perspector is X2163. See Angel Montesdeoca, HG090218: El centro del triángulo X2163.
"""GEOMETOR construction model for X_2163 = X_2-ISOCONJUGATE OF X_45."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            intersect_lines,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02163(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2163 = X_2-ISOCONJUGATE OF X_45 on reference triangle ABC."""
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center"])
            X_36 = eval_trilinear_point(model, A, B, C, '1 - 2*cos(A)', ID="X_36", classes=["center"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center"])
            X_89 = eval_trilinear_point(model, A, B, C, '1/(2*b + 2*c - a)', ID="X_89", classes=["center"])
            central_line(model, X_6, X_36, 6, 36)
            central_line(model, X_1, X_89, 1, 89)
            X_2163 = intersect_lines(model, X_6, X_36, X_1, X_89, ID="X_2163", classes=["selected", "target"])
            X_31 = eval_trilinear_point(model, A, B, C, 'a**(2)', ID="X_31", classes=["center", "incident"])
            X_106 = eval_trilinear_point(model, A, B, C, 'a/(2*a - b - c)', ID="X_106", classes=["center", "incident"])
            central_line(model, X_31, X_106, 31, 106, incident=True)
            return promote_point_to_top(model, X_2163)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2163 on the given reference triangle."""
            model = create_model('X_2163 = X_2-ISOCONJUGATE OF X_45')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02163(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02163.json"))
            write_center_svg(
                model_acute,
                here / "x-02163.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
