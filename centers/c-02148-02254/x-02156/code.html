X2156¶

X2-ISOCONJUGATE OF X22

X2156 = isogonal conjugate of X1760
"""GEOMETOR construction model for X_2156 = X_2-ISOCONJUGATE OF X_22."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            isogonal_conjugate,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02156(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2156 = X_2-ISOCONJUGATE OF X_22 on reference triangle ABC."""
            X_1760 = eval_trilinear_point(model, A, B, C, '(a**(4) - b**(4) - c**(4))', ID="X_1760", classes=["center"])
            X_2156 = isogonal_conjugate(model, A, B, C, X_1760, ID="X_2156", classes=["selected", "target"])
            X_38 = eval_trilinear_point(model, A, B, C, 'b**(2) + c**(2)', ID="X_38", classes=["center", "incident"])
            X_48 = eval_trilinear_point(model, A, B, C, 'tan(B) + tan(C)', ID="X_48", classes=["center", "incident"])
            central_line(model, X_38, X_48, 38, 48, incident=True)
            X_66 = eval_trilinear_point(model, A, B, C, 'b*c/(b**(4) + c**(4) - a**(4))', ID="X_66", classes=["center", "incident"])
            X_71 = eval_trilinear_point(model, A, B, C, '(b + c)*cos(A)', ID="X_71", classes=["center", "incident"])
            central_line(model, X_66, X_71, 66, 71, incident=True)
            X_63 = eval_trilinear_point(model, A, B, C, 'cot(A)', ID="X_63", classes=["center", "incident"])
            X_1930 = eval_trilinear_point(model, A, B, C, 'b**(2)*c**(2)*(b**(2) + c**(2))', ID="X_1930", classes=["center", "incident"])
            central_line(model, X_63, X_1930, 63, 1930, incident=True)
            return promote_point_to_top(model, X_2156)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2156 on the given reference triangle."""
            model = create_model('X_2156 = X_2-ISOCONJUGATE OF X_22')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02156(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02156.json"))
            write_center_svg(
                model_acute,
                here / "x-02156.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
