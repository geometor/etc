X2-ISOCONJUGATE OF X743

X2239 = crosssum of X1 and X2239
"""GEOMETOR construction model for X_2239 = X_2-ISOCONJUGATE OF X_743."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            intersect_lines,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02239(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2239 = X_2-ISOCONJUGATE OF X_743 on reference triangle ABC."""
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center"])
            X_31 = eval_trilinear_point(model, A, B, C, 'a**(2)', ID="X_31", classes=["center"])
            X_38 = eval_trilinear_point(model, A, B, C, 'b**(2) + c**(2)', ID="X_38", classes=["center"])
            X_42 = eval_trilinear_point(model, A, B, C, 'a*(b + c)', ID="X_42", classes=["center"])
            central_line(model, X_2, X_31, 2, 31)
            central_line(model, X_38, X_42, 38, 42)
            X_2239 = intersect_lines(model, X_2, X_31, X_38, X_42, ID="X_2239", classes=["selected", "target"])
            X_43 = eval_trilinear_point(model, A, B, C, 'a*b + a*c - b*c', ID="X_43", classes=["center", "incident"])
            X_63 = eval_trilinear_point(model, A, B, C, 'cot(A)', ID="X_63", classes=["center", "incident"])
            central_line(model, X_43, X_63, 43, 63, incident=True)
            return promote_point_to_top(model, X_2239)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2239 on the given reference triangle."""
            model = create_model('X_2239 = X_2-ISOCONJUGATE OF X_743')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02239(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02239.json"))
            write_center_svg(
                model_acute,
                here / "x-02239.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
