X2150¶

X2-ISOCONJUGATE OF X12

X2150 = X593-Ceva conjugate of X849
"""GEOMETOR construction model for X_2150 = X_2-ISOCONJUGATE OF X_12."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            ceva_conjugate,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            mittenpunkt_of,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02150(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2150 = X_2-ISOCONJUGATE OF X_12 on reference triangle ABC."""
            X_593 = eval_trilinear_point(model, A, B, C, 'a/(b + c)**(2)', ID="X_593", classes=["center"])
            X_849 = eval_trilinear_point(model, A, B, C, '(a/(b + c))**(2)', ID="X_849", classes=["center"])
            X_2150 = ceva_conjugate(model, A, B, C, X_593, X_849, ID="X_2150", classes=["selected", "target"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_163 = eval_trilinear_point(model, A, B, C, '(sin(2*A))/(sin(2*B) - sin(2*C))', ID="X_163", classes=["center", "incident"])
            central_line(model, X_6, X_163, 6, 163, incident=True)
            X_60 = eval_trilinear_point(model, A, B, C, '1/(1 + cos(B - C))', ID="X_60", classes=["center", "incident"])
            X_283 = eval_trilinear_point(model, A, B, C, '(cos(A))/(cos(B) + cos(C))', ID="X_283", classes=["center", "incident"])
            central_line(model, X_60, X_283, 60, 283, incident=True)
            X_9 = mittenpunkt_of(model, A, B, C, ID="X_9", classes=["center", "incident"])
            X_1098 = eval_trilinear_point(model, A, B, C, '(1/(cos(B) + cos(C))**(2))', ID="X_1098", classes=["center", "incident"])
            central_line(model, X_9, X_1098, 9, 1098, incident=True)
            return promote_point_to_top(model, X_2150)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2150 on the given reference triangle."""
            model = create_model('X_2150 = X_2-ISOCONJUGATE OF X_12')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02150(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02150.json"))
            write_center_svg(
                model_acute,
                here / "x-02150.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
