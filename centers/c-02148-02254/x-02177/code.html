X2177¶

X2-ISOCONJUGATE OF X89

X2177 = X2099-Ceva conjugate of X1405
"""GEOMETOR construction model for X_2177 = X_2-ISOCONJUGATE OF X_89."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            ceva_conjugate,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            point_from_barycentrics,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02177(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2177 = X_2-ISOCONJUGATE OF X_89 on reference triangle ABC."""
            X_2099 = eval_trilinear_point(model, A, B, C, '((a + b - c)*(a - b + c)*(a - 2*b - 2*c))', ID="X_2099", classes=["center"])
            X_1405 = point_from_barycentrics(model, A, B, C, 0.18363064, 0.30220357, 0.51416579, ID="X_1405", classes=["center"])
            X_2177 = ceva_conjugate(model, A, B, C, X_2099, X_1405, ID="X_2177", classes=["selected", "target"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_31 = eval_trilinear_point(model, A, B, C, 'a**(2)', ID="X_31", classes=["center", "incident"])
            central_line(model, X_6, X_31, 6, 31, incident=True)
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_88 = eval_trilinear_point(model, A, B, C, '1/(b + c - 2*a)', ID="X_88", classes=["center", "incident"])
            central_line(model, X_1, X_88, 1, 88, incident=True)
            X_43 = eval_trilinear_point(model, A, B, C, 'a*b + a*c - b*c', ID="X_43", classes=["center", "incident"])
            X_748 = eval_trilinear_point(model, A, B, C, 'a**(2) - 2*b*c', ID="X_748", classes=["center", "incident"])
            central_line(model, X_43, X_748, 43, 748, incident=True)
            return promote_point_to_top(model, X_2177)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2177 on the given reference triangle."""
            model = create_model('X_2177 = X_2-ISOCONJUGATE OF X_89')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02177(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02177.json"))
            write_center_svg(
                model_acute,
                here / "x-02177.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
