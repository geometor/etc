X2194¶

X2-ISOCONJUGATE OF X226

X2194 = isogonal conjugate of X1441
"""GEOMETOR construction model for X_2194 = X_2-ISOCONJUGATE OF X_226."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            isogonal_conjugate,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02194(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2194 = X_2-ISOCONJUGATE OF X_226 on reference triangle ABC."""
            X_1441 = eval_barycentric_point(model, A, B, C, '(csc(A))*(cos(B) + cos(C))', ID="X_1441", classes=["center"])
            X_2194 = isogonal_conjugate(model, A, B, C, X_1441, ID="X_2194", classes=["selected", "target"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_25 = eval_trilinear_point(model, A, B, C, 'sin(A)*tan(A)', ID="X_25", classes=["center", "incident"])
            central_line(model, X_6, X_25, 6, 25, incident=True)
            X_31 = eval_trilinear_point(model, A, B, C, 'a**(2)', ID="X_31", classes=["center", "incident"])
            X_48 = eval_trilinear_point(model, A, B, C, 'tan(B) + tan(C)', ID="X_48", classes=["center", "incident"])
            central_line(model, X_31, X_48, 31, 48, incident=True)
            X_21 = eval_trilinear_point(model, A, B, C, '1/(cos(B) + cos(C))', ID="X_21", classes=["center", "incident"])
            X_60 = eval_trilinear_point(model, A, B, C, '1/(1 + cos(B - C))', ID="X_60", classes=["center", "incident"])
            central_line(model, X_21, X_60, 21, 60, incident=True)
            return promote_point_to_top(model, X_2194)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2194 on the given reference triangle."""
            model = create_model('X_2194 = X_2-ISOCONJUGATE OF X_226')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02194(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02194.json"))
            write_center_svg(
                model_acute,
                here / "x-02194.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
