X2173¶

X2-ISOCONJUGATE OF X74

X2173 = crossdifference of every pair of points on line X1X656
"""GEOMETOR construction model for X_2173 = X_2-ISOCONJUGATE OF X_74."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02173(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2173 = X_2-ISOCONJUGATE OF X_74 on reference triangle ABC."""
            X_2173 = eval_trilinear_point(model, A, B, C, 'a*(cos(A) - 2*cos(B)*cos(C))', ID="X_2173", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_19 = eval_trilinear_point(model, A, B, C, 'tan(A)', ID="X_19", classes=["center", "incident"])
            central_line(model, X_1, X_19, 1, 19, incident=True)
            X_45 = eval_trilinear_point(model, A, B, C, '2*b + 2*c - a', ID="X_45", classes=["center", "incident"])
            X_198 = eval_trilinear_point(model, A, B, C, '(a*(cos(B) + cos(C) - cos(A) - 1))', ID="X_198", classes=["center", "incident"])
            central_line(model, X_45, X_198, 45, 198, incident=True)
            X_44 = eval_trilinear_point(model, A, B, C, 'b + c - 2*a', ID="X_44", classes=["center", "incident"])
            X_513 = eval_trilinear_point(model, A, B, C, 'b - c', ID="X_513", classes=["center", "incident"])
            central_line(model, X_44, X_513, 44, 513, incident=True)
            return promote_point_to_top(model, X_2173)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2173 on the given reference triangle."""
            model = create_model('X_2173 = X_2-ISOCONJUGATE OF X_74')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02173(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02173.json"))
            write_center_svg(
                model_acute,
                here / "x-02173.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
