X2-ISOCONJUGATE OF X903

X2251 = crossdifference of every pair of points on line X75X693
"""GEOMETOR construction model for X_2251 = X_2-ISOCONJUGATE OF X_903."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            intersect_lines,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02251(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2251 = X_2-ISOCONJUGATE OF X_903 on reference triangle ABC."""
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center"])
            X_36 = eval_trilinear_point(model, A, B, C, '1 - 2*cos(A)', ID="X_36", classes=["center"])
            X_31 = eval_trilinear_point(model, A, B, C, 'a**(2)', ID="X_31", classes=["center"])
            X_32 = eval_trilinear_point(model, A, B, C, 'a**(3)', ID="X_32", classes=["center"])
            central_line(model, X_6, X_36, 6, 36)
            central_line(model, X_31, X_32, 31, 32)
            X_2251 = intersect_lines(model, X_6, X_36, X_31, X_32, ID="X_2251", classes=["selected", "target"])
            X_187 = eval_trilinear_point(model, A, B, C, 'a*(2*a**(2) - b**(2) - c**(2))', ID="X_187", classes=["center", "incident"])
            X_672 = eval_trilinear_point(model, A, B, C, 'a*(b**(2) + c**(2) - a*(b + c))', ID="X_672", classes=["center", "incident"])
            central_line(model, X_187, X_672, 187, 672, incident=True)
            return promote_point_to_top(model, X_2251)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2251 on the given reference triangle."""
            model = create_model('X_2251 = X_2-ISOCONJUGATE OF X_903')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02251(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02251.json"))
            write_center_svg(
                model_acute,
                here / "x-02251.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
