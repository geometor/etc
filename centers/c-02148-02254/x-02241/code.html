X2-ISOCONJUGATE OF X749

"""GEOMETOR construction model for X_2241 = X_2-ISOCONJUGATE OF X_749."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            intersect_lines,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02241(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2241 = X_2-ISOCONJUGATE OF X_749 on reference triangle ABC."""
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center"])
            X_32 = eval_trilinear_point(model, A, B, C, 'a**(3)', ID="X_32", classes=["center"])
            X_39 = eval_trilinear_point(model, A, B, C, 'a*(b**(2) + c**(2))', ID="X_39", classes=["center"])
            X_55 = eval_trilinear_point(model, A, B, C, 'a*(b + c - a)', ID="X_55", classes=["center"])
            central_line(model, X_1, X_32, 1, 32)
            central_line(model, X_39, X_55, 39, 55)
            X_2241 = intersect_lines(model, X_1, X_32, X_39, X_55, ID="X_2241", classes=["selected", "target"])
            X_56 = eval_trilinear_point(model, A, B, C, 'a/(b + c - a)', ID="X_56", classes=["center", "incident"])
            X_187 = eval_trilinear_point(model, A, B, C, 'a*(2*a**(2) - b**(2) - c**(2))', ID="X_187", classes=["center", "incident"])
            central_line(model, X_56, X_187, 56, 187, incident=True)
            return promote_point_to_top(model, X_2241)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2241 on the given reference triangle."""
            model = create_model('X_2241 = X_2-ISOCONJUGATE OF X_749')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02241(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02241.json"))
            write_center_svg(
                model_acute,
                here / "x-02241.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
