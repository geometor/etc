X2364¶

X2-ISOCONJUGATE OF X2099

X2364 = trilinear pole of line X654X663
"""GEOMETOR construction model for X_2364 = X_2-ISOCONJUGATE OF X_2099."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            ceva_conjugate,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            point_from_barycentrics,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02364(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2364 = X_2-ISOCONJUGATE OF X_2099 on reference triangle ABC."""
            X_89 = eval_trilinear_point(model, A, B, C, '1/(2*b + 2*c - a)', ID="X_89", classes=["center"])
            X_2163 = point_from_barycentrics(model, A, B, C, 0.18436578, 0.30973451, 0.50589971, ID="X_2163", classes=["center"])
            X_2364 = ceva_conjugate(model, A, B, C, X_89, X_2163, ID="X_2364", classes=["selected", "target"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_36 = eval_trilinear_point(model, A, B, C, '1 - 2*cos(A)', ID="X_36", classes=["center", "incident"])
            central_line(model, X_6, X_36, 6, 36, incident=True)
            X_57 = eval_trilinear_point(model, A, B, C, '1/(b + c - a)', ID="X_57", classes=["center", "incident"])
            central_line(model, X_57, X_89, 57, 89, incident=True)
            X_19 = eval_trilinear_point(model, A, B, C, 'tan(A)', ID="X_19", classes=["center", "incident"])
            X_1449 = eval_trilinear_point(model, A, B, C, '3*a + b + c', ID="X_1449", classes=["center", "incident"])
            central_line(model, X_19, X_1449, 19, 1449, incident=True)
            return promote_point_to_top(model, X_2364)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2364 on the given reference triangle."""
            model = create_model('X_2364 = X_2-ISOCONJUGATE OF X_2099')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02364(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02364.json"))
            write_center_svg(
                model_acute,
                here / "x-02364.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
