X2362¶

X2-ISOCONJUGATE OF X2066

X2362 = crosspoint of X4 and X1123
"""GEOMETOR construction model for X_2362 = X_2-ISOCONJUGATE OF X_2066."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            crosspoint,
            eval_trilinear_point,
            incenter_of,
            orthocenter_of,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02362(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2362 = X_2-ISOCONJUGATE OF X_2066 on reference triangle ABC."""
            X_4 = orthocenter_of(model, A, B, C, ID="X_4", classes=["center"])
            X_1123 = eval_trilinear_point(model, A, B, C, '1/(1 +sin(A))', ID="X_1123", classes=["center"])
            X_2362 = crosspoint(model, A, B, C, X_4, X_1123, ID="X_2362", classes=["selected", "target"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_19 = eval_trilinear_point(model, A, B, C, 'tan(A)', ID="X_19", classes=["center", "incident"])
            central_line(model, X_6, X_19, 6, 19, incident=True)
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_176 = eval_trilinear_point(model, A, B, C, '1 + sec(A/2)*cos(B/2)*cos(C/2)', ID="X_176", classes=["center", "incident"])
            central_line(model, X_2, X_176, 2, 176, incident=True)
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_372 = eval_trilinear_point(model, A, B, C, 'cos(A +  pi  /4)', ID="X_372", classes=["center", "incident"])
            central_line(model, X_1, X_372, 1, 372, incident=True)
            return promote_point_to_top(model, X_2362)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2362 on the given reference triangle."""
            model = create_model('X_2362 = X_2-ISOCONJUGATE OF X_2066')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02362(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02362.json"))
            write_center_svg(
                model_acute,
                here / "x-02362.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
