X2308¶

X2-ISOCONJUGATE OF X1255

X2308 = isogonal conjugate of X1268
"""GEOMETOR construction model for X_2308 = X_2-ISOCONJUGATE OF X_1255."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            isogonal_conjugate,
            point_from_barycentrics,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02308(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2308 = X_2-ISOCONJUGATE OF X_1255 on reference triangle ABC."""
            X_1268 = point_from_barycentrics(model, A, B, C, 0.34742328, 0.33294731, 0.31962942, ID="X_1268", classes=["center"])
            X_2308 = isogonal_conjugate(model, A, B, C, X_1268, ID="X_2308", classes=["selected", "target"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_31 = eval_trilinear_point(model, A, B, C, 'a**(2)', ID="X_31", classes=["center", "incident"])
            central_line(model, X_6, X_31, 6, 31, incident=True)
            X_36 = eval_trilinear_point(model, A, B, C, '1 - 2*cos(A)', ID="X_36", classes=["center", "incident"])
            X_58 = eval_trilinear_point(model, A, B, C, 'a/(b + c)', ID="X_58", classes=["center", "incident"])
            central_line(model, X_36, X_58, 36, 58, incident=True)
            X_81 = eval_trilinear_point(model, A, B, C, '1/(b + c)', ID="X_81", classes=["center", "incident"])
            X_238 = eval_trilinear_point(model, A, B, C, 'a**(2) - b*c', ID="X_238", classes=["center", "incident"])
            central_line(model, X_81, X_238, 81, 238, incident=True)
            return promote_point_to_top(model, X_2308)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2308 on the given reference triangle."""
            model = create_model('X_2308 = X_2-ISOCONJUGATE OF X_1255')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02308(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02308.json"))
            write_center_svg(
                model_acute,
                here / "x-02308.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
