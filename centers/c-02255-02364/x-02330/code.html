X2330¶

X2-ISOCONJUGATE OF X1432

X2330 = X171-Ceva conjugate of X172
"""GEOMETOR construction model for X_2330 = X_2-ISOCONJUGATE OF X_1432."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            ceva_conjugate,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02330(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2330 = X_2-ISOCONJUGATE OF X_1432 on reference triangle ABC."""
            X_171 = eval_trilinear_point(model, A, B, C, 'a**(2) + b*c', ID="X_171", classes=["center"])
            X_172 = eval_trilinear_point(model, A, B, C, 'a**(3) + a*b*c', ID="X_172", classes=["center"])
            X_2330 = ceva_conjugate(model, A, B, C, X_171, X_172, ID="X_2330", classes=["selected", "target"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_31 = eval_trilinear_point(model, A, B, C, 'a**(2)', ID="X_31", classes=["center", "incident"])
            central_line(model, X_6, X_31, 6, 31, incident=True)
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_182 = eval_trilinear_point(model, A, B, C, 'cos(A-  omega  )', ID="X_182", classes=["center", "incident"])
            central_line(model, X_1, X_182, 1, 182, incident=True)
            X_35 = eval_trilinear_point(model, A, B, C, '1 + 2*cos(A)', ID="X_35", classes=["center", "incident"])
            X_511 = eval_trilinear_point(model, A, B, C, 'cos(A +  omega  )', ID="X_511", classes=["center", "incident"])
            central_line(model, X_35, X_511, 35, 511, incident=True)
            return promote_point_to_top(model, X_2330)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2330 on the given reference triangle."""
            model = create_model('X_2330 = X_2-ISOCONJUGATE OF X_1432')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02330(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02330.json"))
            write_center_svg(
                model_acute,
                here / "x-02330.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
