X2-ISOCONJUGATE OF X969

"""GEOMETOR construction model for X_2271 = X_2-ISOCONJUGATE OF X_969."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02271(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2271 = X_2-ISOCONJUGATE OF X_969 on reference triangle ABC."""
            X_2271 = eval_trilinear_point(model, A, B, C, 'cos(A) + sin(A)*(csc(A) + csc(B) + csc(C))', ID="X_2271", classes=["selected", "target"])
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center", "incident"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            central_line(model, X_3, X_6, 3, 6, incident=True)
            X_25 = eval_trilinear_point(model, A, B, C, 'sin(A)*tan(A)', ID="X_25", classes=["center", "incident"])
            X_41 = eval_trilinear_point(model, A, B, C, 'a**(2)*(b + c - a)', ID="X_41", classes=["center", "incident"])
            central_line(model, X_25, X_41, 25, 41, incident=True)
            X_55 = eval_trilinear_point(model, A, B, C, 'a*(b + c - a)', ID="X_55", classes=["center", "incident"])
            X_213 = eval_trilinear_point(model, A, B, C, '(b + c)*a**(2)', ID="X_213", classes=["center", "incident"])
            central_line(model, X_55, X_213, 55, 213, incident=True)
            return promote_point_to_top(model, X_2271)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2271 on the given reference triangle."""
            model = create_model('X_2271 = X_2-ISOCONJUGATE OF X_969')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02271(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02271.json"))
            write_center_svg(
                model_acute,
                here / "x-02271.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
