X2357¶

X2-ISOCONJUGATE OF X1817

X2357 = X84-Ceva conjugate of X1903
"""GEOMETOR construction model for X_2357 = X_2-ISOCONJUGATE OF X_1817."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            ceva_conjugate,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            nagel_point_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02357(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2357 = X_2-ISOCONJUGATE OF X_1817 on reference triangle ABC."""
            X_84 = eval_trilinear_point(model, A, B, C, '1/(cos(B) + cos(C) - cos(A) - 1)', ID="X_84", classes=["center"])
            X_1903 = eval_barycentric_point(model, A, B, C, 'a*(b + c)/(a**(3) + a**(2)*(b + c) - a*(b + c)**(2) - (b - c)**(2)*(b + c))', ID="X_1903", classes=["center"])
            X_2357 = ceva_conjugate(model, A, B, C, X_84, X_1903, ID="X_2357", classes=["selected", "target"])
            X_8 = nagel_point_of(model, A, B, C, ID="X_8", classes=["center", "incident"])
            X_20 = eval_trilinear_point(model, A, B, C, 'cos(A) - cos(B)*cos(C)', ID="X_20", classes=["center", "incident"])
            central_line(model, X_8, X_20, 8, 20, incident=True)
            X_48 = eval_trilinear_point(model, A, B, C, 'tan(B) + tan(C)', ID="X_48", classes=["center", "incident"])
            X_55 = eval_trilinear_point(model, A, B, C, 'a*(b + c - a)', ID="X_55", classes=["center", "incident"])
            central_line(model, X_48, X_55, 48, 55, incident=True)
            X_71 = eval_trilinear_point(model, A, B, C, '(b + c)*cos(A)', ID="X_71", classes=["center", "incident"])
            X_210 = eval_trilinear_point(model, A, B, C, '(b + c)*(b + c - a)', ID="X_210", classes=["center", "incident"])
            central_line(model, X_71, X_210, 71, 210, incident=True)
            return promote_point_to_top(model, X_2357)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2357 on the given reference triangle."""
            model = create_model('X_2357 = X_2-ISOCONJUGATE OF X_1817')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02357(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02357.json"))
            write_center_svg(
                model_acute,
                here / "x-02357.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
