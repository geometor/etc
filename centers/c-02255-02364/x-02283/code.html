X2283¶

X2-ISOCONJUGATE OF X1024

X2283 = isogonal conjugate of X885
"""GEOMETOR construction model for X_2283 = X_2-ISOCONJUGATE OF X_1024."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            incenter_of,
            isogonal_conjugate,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02283(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2283 = X_2-ISOCONJUGATE OF X_1024 on reference triangle ABC."""
            X_885 = eval_barycentric_point(model, A, B, C, '(b - c)*(a - b - c)/(b**(2) + c**(2) - a*b - a*c)', ID="X_885", classes=["center"])
            X_2283 = isogonal_conjugate(model, A, B, C, X_885, ID="X_2283", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center", "incident"])
            central_line(model, X_1, X_3, 1, 3, incident=True)
            X_101 = eval_trilinear_point(model, A, B, C, 'a/(b - c)', ID="X_101", classes=["center", "incident"])
            X_109 = eval_trilinear_point(model, A, B, C, 'a/(cos(B) - cos(C))', ID="X_109", classes=["center", "incident"])
            central_line(model, X_101, X_109, 101, 109, incident=True)
            X_100 = eval_trilinear_point(model, A, B, C, '1/(b - c)', ID="X_100", classes=["center", "incident"])
            X_658 = eval_trilinear_point(model, A, B, C, '1/((1 + cos(A))*(cos(B) - cos(C)))', ID="X_658", classes=["center", "incident"])
            central_line(model, X_100, X_658, 100, 658, incident=True)
            return promote_point_to_top(model, X_2283)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2283 on the given reference triangle."""
            model = create_model('X_2283 = X_2-ISOCONJUGATE OF X_1024')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02283(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02283.json"))
            write_center_svg(
                model_acute,
                here / "x-02283.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
