X2320¶

X2-ISOCONJUGATE OF X1405

X2320 = isogonal conjugate of X2099
"""GEOMETOR construction model for X_2320 = X_2-ISOCONJUGATE OF X_1405."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            gergonne_point_of,
            incenter_of,
            isogonal_conjugate,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02320(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2320 = X_2-ISOCONJUGATE OF X_1405 on reference triangle ABC."""
            X_2099 = eval_trilinear_point(model, A, B, C, '((a + b - c)*(a - b + c)*(a - 2*b - 2*c))', ID="X_2099", classes=["center"])
            X_2320 = isogonal_conjugate(model, A, B, C, X_2099, ID="X_2320", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_80 = eval_trilinear_point(model, A, B, C, '1/(1 - 2*cos(A))', ID="X_80", classes=["center", "incident"])
            central_line(model, X_2, X_80, 2, 80, incident=True)
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_89 = eval_trilinear_point(model, A, B, C, '1/(2*b + 2*c - a)', ID="X_89", classes=["center", "incident"])
            central_line(model, X_1, X_89, 1, 89, incident=True)
            X_7 = gergonne_point_of(model, A, B, C, ID="X_7", classes=["center", "incident"])
            X_1319 = eval_trilinear_point(model, A, B, C, '(b + c - 2*a)/ (b + c - a)', ID="X_1319", classes=["center", "incident"])
            central_line(model, X_7, X_1319, 7, 1319, incident=True)
            return promote_point_to_top(model, X_2320)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2320 on the given reference triangle."""
            model = create_model('X_2320 = X_2-ISOCONJUGATE OF X_1405')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02320(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02320.json"))
            write_center_svg(
                model_acute,
                here / "x-02320.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
