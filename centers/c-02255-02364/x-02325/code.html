X2325¶

X2-ISOCONJUGATE OF X1417

X2325 = complement of X1266
"""GEOMETOR construction model for X_2325 = X_2-ISOCONJUGATE OF X_1417."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            complement,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            mittenpunkt_of,
            nagel_point_of,
            point_from_barycentrics,
            promote_point_to_top,
            spieker_center_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02325(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2325 = X_2-ISOCONJUGATE OF X_1417 on reference triangle ABC."""
            X_1266 = point_from_barycentrics(model, A, B, C, -3, 1, 3, ID="X_1266", classes=["center"])
            X_2325 = complement(model, A, B, C, X_1266, ID="X_2325", classes=["selected", "target"])
            X_8 = nagel_point_of(model, A, B, C, ID="X_8", classes=["center", "incident"])
            X_9 = mittenpunkt_of(model, A, B, C, ID="X_9", classes=["center", "incident"])
            central_line(model, X_8, X_9, 8, 9, incident=True)
            X_10 = spieker_center_of(model, A, B, C, ID="X_10", classes=["center", "incident"])
            X_45 = eval_trilinear_point(model, A, B, C, '2*b + 2*c - a', ID="X_45", classes=["center", "incident"])
            central_line(model, X_10, X_45, 10, 45, incident=True)
            X_37 = eval_trilinear_point(model, A, B, C, 'b + c', ID="X_37", classes=["center", "incident"])
            X_39 = eval_trilinear_point(model, A, B, C, 'a*(b**(2) + c**(2))', ID="X_39", classes=["center", "incident"])
            central_line(model, X_37, X_39, 37, 39, incident=True)
            return promote_point_to_top(model, X_2325)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2325 on the given reference triangle."""
            model = create_model('X_2325 = X_2-ISOCONJUGATE OF X_1417')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02325(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02325.json"))
            write_center_svg(
                model_acute,
                here / "x-02325.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
