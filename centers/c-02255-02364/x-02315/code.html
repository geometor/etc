X2315¶

X2-ISOCONJUGATE OF X1300

X2315 = X2159-Ceva conjugate of X48
"""GEOMETOR construction model for X_2315 = X_2-ISOCONJUGATE OF X_1300."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            ceva_conjugate,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            point_from_barycentrics,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02315(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2315 = X_2-ISOCONJUGATE OF X_1300 on reference triangle ABC."""
            X_2159 = point_from_barycentrics(model, A, B, C, 0.47826441, 1.3314881, -0.80975252, ID="X_2159", classes=["center"])
            X_48 = eval_trilinear_point(model, A, B, C, 'tan(B) + tan(C)', ID="X_48", classes=["center"])
            X_2315 = ceva_conjugate(model, A, B, C, X_2159, X_48, ID="X_2315", classes=["selected", "target"])
            X_19 = eval_trilinear_point(model, A, B, C, 'tan(A)', ID="X_19", classes=["center", "incident"])
            X_91 = eval_trilinear_point(model, A, B, C, 'sec(2*A)', ID="X_91", classes=["center", "incident"])
            central_line(model, X_19, X_91, 19, 91, incident=True)
            X_255 = eval_trilinear_point(model, A, B, C, '(cos(A))**(2)', ID="X_255", classes=["center", "incident"])
            central_line(model, X_48, X_255, 48, 255, incident=True)
            X_44 = eval_trilinear_point(model, A, B, C, 'b + c - 2*a', ID="X_44", classes=["center", "incident"])
            X_513 = eval_trilinear_point(model, A, B, C, 'b - c', ID="X_513", classes=["center", "incident"])
            central_line(model, X_44, X_513, 44, 513, incident=True)
            return promote_point_to_top(model, X_2315)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2315 on the given reference triangle."""
            model = create_model('X_2315 = X_2-ISOCONJUGATE OF X_1300')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02315(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02315.json"))
            write_center_svg(
                model_acute,
                here / "x-02315.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
