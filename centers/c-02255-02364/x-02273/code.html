X2273¶

X2-ISOCONJUGATE OF X977

"""GEOMETOR construction model for X_2273 = X_2-ISOCONJUGATE OF X_977."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            intersect_lines,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02273(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2273 = X_2-ISOCONJUGATE OF X_977 on reference triangle ABC."""
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center"])
            X_39 = eval_trilinear_point(model, A, B, C, 'a*(b**(2) + c**(2))', ID="X_39", classes=["center"])
            X_48 = eval_trilinear_point(model, A, B, C, 'tan(B) + tan(C)', ID="X_48", classes=["center"])
            central_line(model, X_1, X_6, 1, 6)
            central_line(model, X_39, X_48, 39, 48)
            X_2273 = intersect_lines(model, X_1, X_6, X_39, X_48, ID="X_2273", classes=["selected", "target"])
            X_32 = eval_trilinear_point(model, A, B, C, 'a**(3)', ID="X_32", classes=["center", "incident"])
            X_71 = eval_trilinear_point(model, A, B, C, '(b + c)*cos(A)', ID="X_71", classes=["center", "incident"])
            central_line(model, X_32, X_71, 32, 71, incident=True)
            return promote_point_to_top(model, X_2273)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2273 on the given reference triangle."""
            model = create_model('X_2273 = X_2-ISOCONJUGATE OF X_977')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02273(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02273.json"))
            write_center_svg(
                model_acute,
                here / "x-02273.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
