X2269¶

X2-ISOCONJUGATE OF X961

"""GEOMETOR construction model for X_2269 = X_2-ISOCONJUGATE OF X_961."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            intersect_lines,
            mittenpunkt_of,
            nagel_point_of,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02269(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2269 = X_2-ISOCONJUGATE OF X_961 on reference triangle ABC."""
            X_8 = nagel_point_of(model, A, B, C, ID="X_8", classes=["center"])
            X_9 = mittenpunkt_of(model, A, B, C, ID="X_9", classes=["center"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center"])
            X_31 = eval_trilinear_point(model, A, B, C, 'a**(2)', ID="X_31", classes=["center"])
            central_line(model, X_8, X_9, 8, 9)
            central_line(model, X_6, X_31, 6, 31)
            X_2269 = intersect_lines(model, X_8, X_9, X_6, X_31, ID="X_2269", classes=["selected", "target"])
            X_63 = eval_trilinear_point(model, A, B, C, 'cot(A)', ID="X_63", classes=["center", "incident"])
            X_193 = eval_trilinear_point(model, A, B, C, '(csc(A))*(cot(B) + cot(C) - cot(A))', ID="X_193", classes=["center", "incident"])
            central_line(model, X_63, X_193, 63, 193, incident=True)
            return promote_point_to_top(model, X_2269)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2269 on the given reference triangle."""
            model = create_model('X_2269 = X_2-ISOCONJUGATE OF X_961')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02269(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02269.json"))
            write_center_svg(
                model_acute,
                here / "x-02269.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
