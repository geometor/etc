X2319¶

X2-ISOCONJUGATE OF X1403

X2319 = isogonal conjugate of X1423
"""GEOMETOR construction model for X_2319 = X_2-ISOCONJUGATE OF X_1403."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            isogonal_conjugate,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02319(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2319 = X_2-ISOCONJUGATE OF X_1403 on reference triangle ABC."""
            X_1423 = eval_trilinear_point(model, A, B, C, '((a*b + a*c - b*c)/(b + c - a))', ID="X_1423", classes=["center"])
            X_2319 = isogonal_conjugate(model, A, B, C, X_1423, ID="X_2319", classes=["selected", "target"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_43 = eval_trilinear_point(model, A, B, C, 'a*b + a*c - b*c', ID="X_43", classes=["center", "incident"])
            central_line(model, X_6, X_43, 6, 43, incident=True)
            X_57 = eval_trilinear_point(model, A, B, C, '1/(b + c - a)', ID="X_57", classes=["center", "incident"])
            X_239 = eval_trilinear_point(model, A, B, C, 'b*c*(a**(2) - b*c)', ID="X_239", classes=["center", "incident"])
            central_line(model, X_57, X_239, 57, 239, incident=True)
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_893 = eval_trilinear_point(model, A, B, C, 'a/(a**(2) + b*c)', ID="X_893", classes=["center", "incident"])
            central_line(model, X_1, X_893, 1, 893, incident=True)
            return promote_point_to_top(model, X_2319)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2319 on the given reference triangle."""
            model = create_model('X_2319 = X_2-ISOCONJUGATE OF X_1403')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02319(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02319.json"))
            write_center_svg(
                model_acute,
                here / "x-02319.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
