X2116¶

POINT KEPLER VI

"""GEOMETOR construction model for X_2116 = POINT KEPLER VI."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            intersect_lines,
            point_from_barycentrics,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02116(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2116 = POINT KEPLER VI on reference triangle ABC."""
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center"])
            X_105 = eval_trilinear_point(model, A, B, C, '1/(b**(2) + c**(2) - a*(b + c))', ID="X_105", classes=["center"])
            X_1423 = eval_trilinear_point(model, A, B, C, '((a*b + a*c - b*c)/(b + c - a))', ID="X_1423", classes=["center"])
            X_2107 = eval_trilinear_point(model, A, B, C, '(a/((a + b)*(a + c)*(b**(2)*c**(2) - a**(2)*b**(2) - a**(2)*c**(2) + a*b*c*(b + c - a))))', ID="X_2107", classes=["center"])
            central_line(model, X_6, X_105, 6, 105)
            central_line(model, X_1423, X_2107, 1423, 2107)
            X_2116 = intersect_lines(model, X_6, X_105, X_1423, X_2107, ID="X_2116", classes=["selected", "target"])
            X_1931 = eval_trilinear_point(model, A, B, C, '(((b + c)**(2) - (a + b)*(a + c))/(b + c))', ID="X_1931", classes=["center", "incident"])
            X_2115 = point_from_barycentrics(model, A, B, C, 1.2446873, -1.0163934, 0.77170613, ID="X_2115", classes=["center", "incident"])
            central_line(model, X_1931, X_2115, 1931, 2115, incident=True)
            return promote_point_to_top(model, X_2116)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2116 on the given reference triangle."""
            model = create_model('X_2116 = POINT KEPLER VI')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02116(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02116.json"))
            write_center_svg(
                model_acute,
                here / "x-02116.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
