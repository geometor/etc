X2100¶

REFLECTION OF X1 IN X1113

X2100 is the reflection of X1 in X1113.
"""GEOMETOR construction model for X_2100 = REFLECTION OF X_1 IN X_1113."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            intersect_lines,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02100(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2100 = REFLECTION OF X_1 IN X_1113 on reference triangle ABC."""
            X_30 = eval_trilinear_point(model, A, B, C, 'cos(A) - 2*cos(B)*cos(C)', ID="X_30", classes=["center"])
            X_40 = eval_trilinear_point(model, A, B, C, 'cos(B) + cos(C) - cos(A) - 1', ID="X_40", classes=["center"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center"])
            X_1113 = eval_trilinear_point(model, A, B, C, '(R - d)*cos(A) - 2*R*cos(B)*cos(C)', ID="X_1113", classes=["center"])
            central_line(model, X_30, X_40, 30, 40)
            central_line(model, X_1, X_1113, 1, 1113)
            X_2100 = intersect_lines(model, X_30, X_40, X_1, X_1113, ID="X_2100", classes=["selected", "target"])
            X_165 = eval_trilinear_point(model, A, B, C, 'tan(B/2) + tan(C/2) - tan(A/2)', ID="X_165", classes=["center", "incident"])
            X_1114 = eval_trilinear_point(model, A, B, C, '(R + d)*cos(A) - 2*R*cos(B)*cos(C)', ID="X_1114", classes=["center", "incident"])
            central_line(model, X_165, X_1114, 165, 1114, incident=True)
            return promote_point_to_top(model, X_2100)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2100 on the given reference triangle."""
            model = create_model('X_2100 = REFLECTION OF X_1 IN X_1113')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02100(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02100.json"))
            write_center_svg(
                model_acute,
                here / "x-02100.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
