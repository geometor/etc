X2083¶

NK-TRANSFORM OF X19

X2083 = X811-Ceva conjugate of X810
"""GEOMETOR construction model for X_2083 = NK-TRANSFORM OF X_19."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            ceva_conjugate,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02083(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2083 = NK-TRANSFORM OF X_19 on reference triangle ABC."""
            X_811 = eval_trilinear_point(model, A, B, C, '(csc(2*A))/((cos(B))**(2) - (cos(C))**(2))', ID="X_811", classes=["center"])
            X_810 = eval_trilinear_point(model, A, B, C, 'sin(2*A)*((cos(B))**(2) - (cos(C))**(2))', ID="X_810", classes=["center"])
            X_2083 = ceva_conjugate(model, A, B, C, X_811, X_810, ID="X_2083", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_163 = eval_trilinear_point(model, A, B, C, '(sin(2*A))/(sin(2*B) - sin(2*C))', ID="X_163", classes=["center", "incident"])
            central_line(model, X_1, X_163, 1, 163, incident=True)
            X_19 = eval_trilinear_point(model, A, B, C, 'tan(A)', ID="X_19", classes=["center", "incident"])
            X_158 = eval_trilinear_point(model, A, B, C, '(sec(A))**(2)', ID="X_158", classes=["center", "incident"])
            central_line(model, X_19, X_158, 19, 158, incident=True)
            X_63 = eval_trilinear_point(model, A, B, C, 'cot(A)', ID="X_63", classes=["center", "incident"])
            X_304 = eval_trilinear_point(model, A, B, C, '(cot(A))*(csc(A))**(2)', ID="X_304", classes=["center", "incident"])
            central_line(model, X_63, X_304, 63, 304, incident=True)
            return promote_point_to_top(model, X_2083)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2083 on the given reference triangle."""
            model = create_model('X_2083 = NK-TRANSFORM OF X_19')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02083(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02083.json"))
            write_center_svg(
                model_acute,
                here / "x-02083.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
