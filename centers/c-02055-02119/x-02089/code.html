X2089¶

3rd MID-ARC POINT

X2089 = X7-Ceva conjugate of X174
"""GEOMETOR construction model for X_2089 = 3rd MID-ARC POINT."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            ceva_conjugate,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            gergonne_point_of,
            incenter_of,
            point_from_barycentrics,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02089(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2089 = 3rd MID-ARC POINT on reference triangle ABC."""
            X_7 = gergonne_point_of(model, A, B, C, ID="X_7", classes=["center"])
            X_174 = eval_trilinear_point(model, A, B, C, 'sec(A/2)', ID="X_174", classes=["center"])
            X_2089 = ceva_conjugate(model, A, B, C, X_7, X_174, ID="X_2089", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_167 = point_from_barycentrics(model, A, B, C, 0.58849841, 0.49152493, -0.080023337, ID="X_167", classes=["center", "incident"])
            central_line(model, X_1, X_167, 1, 167, incident=True)
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_178 = eval_trilinear_point(model, A, B, C, '(cos(B/2) + cos(C/2))*csc(A)', ID="X_178", classes=["center", "incident"])
            central_line(model, X_2, X_178, 2, 178, incident=True)
            X_1488 = eval_trilinear_point(model, A, B, C, '1/(1 + sin(A/2))', ID="X_1488", classes=["center", "incident"])
            central_line(model, X_7, X_1488, 7, 1488, incident=True)
            return promote_point_to_top(model, X_2089)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2089 on the given reference triangle."""
            model = create_model('X_2089 = 3rd MID-ARC POINT')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02089(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02089.json"))
            write_center_svg(
                model_acute,
                here / "x-02089.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
