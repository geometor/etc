X2107¶

X2-ISOCONJUGATE OF X2106

X2107 = isogonal conjugate of X2669
"""GEOMETOR construction model for X_2107 = X_2-ISOCONJUGATE OF X_2106."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02107(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2107 = X_2-ISOCONJUGATE OF X_2106 on reference triangle ABC."""
            X_2107 = eval_trilinear_point(model, A, B, C, '(a/((a + b)*(a + c)*(b**(2)*c**(2) - a**(2)*b**(2) - a**(2)*c**(2) + a*b*c*(b + c - a))))', ID="X_2107", classes=["selected", "target"])
            X_238 = eval_trilinear_point(model, A, B, C, 'a**(2) - b*c', ID="X_238", classes=["center", "incident"])
            X_741 = eval_trilinear_point(model, A, B, C, '(1/(a **(-2)*(b**(1) + c**(1)) - a**(0)*(b **(-1) + c **(-1))))', ID="X_741", classes=["center", "incident"])
            central_line(model, X_238, X_741, 238, 741, incident=True)
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_1655 = eval_trilinear_point(model, A, B, C, '(b*c*(-1/a**(2) + 1/b**(2) + 1/c**(2) + 1/(b*c) + 1/(c*a) + 1/(a*b)))', ID="X_1655", classes=["center", "incident"])
            central_line(model, X_1, X_1655, 1, 1655, incident=True)
            X_31 = eval_trilinear_point(model, A, B, C, 'a**(2)', ID="X_31", classes=["center", "incident"])
            X_1979 = eval_trilinear_point(model, A, B, C, '(a**(3)*(b - c)**(2) - a*b*c*(a - b)*(a - c))', ID="X_1979", classes=["center", "incident"])
            central_line(model, X_31, X_1979, 31, 1979, incident=True)
            return promote_point_to_top(model, X_2107)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2107 on the given reference triangle."""
            model = create_model('X_2107 = X_2-ISOCONJUGATE OF X_2106')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02107(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02107.json"))
            write_center_svg(
                model_acute,
                here / "x-02107.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
