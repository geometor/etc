X2078¶

INVERSE-IN-CIRCUMCIRCLE OF X57

X2078 = inverse-in-circumcircle of X57
"""GEOMETOR construction model for X_2078 = INVERSE-IN-CIRCUMCIRCLE OF X_57."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02078(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2078 = INVERSE-IN-CIRCUMCIRCLE OF X_57 on reference triangle ABC."""
            X_2078 = eval_trilinear_point(model, A, B, C, '(r - 2*R + 2*(r + R)*cos(A))', ID="X_2078", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center", "incident"])
            central_line(model, X_1, X_3, 1, 3, incident=True)
            X_73 = eval_trilinear_point(model, A, B, C, 'sec(B) + sec(C)', ID="X_73", classes=["center", "incident"])
            X_595 = eval_trilinear_point(model, A, B, C, 'r + R*(cos(2*A) - 1)', ID="X_595", classes=["center", "incident"])
            central_line(model, X_73, X_595, 73, 595, incident=True)
            X_59 = eval_trilinear_point(model, A, B, C, '1/(1 - cos(B - C))', ID="X_59", classes=["center", "incident"])
            X_672 = eval_trilinear_point(model, A, B, C, 'a*(b**(2) + c**(2) - a*(b + c))', ID="X_672", classes=["center", "incident"])
            central_line(model, X_59, X_672, 59, 672, incident=True)
            return promote_point_to_top(model, X_2078)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2078 on the given reference triangle."""
            model = create_model('X_2078 = INVERSE-IN-CIRCUMCIRCLE OF X_57')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02078(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02078.json"))
            write_center_svg(
                model_acute,
                here / "x-02078.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
