X2082¶

NK-TRANSFORM OF X9

X2082 = X664-Ceva conjugate of X663
"""GEOMETOR construction model for X_2082 = NK-TRANSFORM OF X_9."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            ceva_conjugate,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            mittenpunkt_of,
            nagel_point_of,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02082(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2082 = NK-TRANSFORM OF X_9 on reference triangle ABC."""
            X_664 = eval_trilinear_point(model, A, B, C, '1/(a*(b - c)*(b + c - a))', ID="X_664", classes=["center"])
            X_663 = eval_trilinear_point(model, A, B, C, 'a*(b - c)*(b + c - a)', ID="X_663", classes=["center"])
            X_2082 = ceva_conjugate(model, A, B, C, X_664, X_663, ID="X_2082", classes=["selected", "target"])
            X_8 = nagel_point_of(model, A, B, C, ID="X_8", classes=["center", "incident"])
            X_9 = mittenpunkt_of(model, A, B, C, ID="X_9", classes=["center", "incident"])
            central_line(model, X_8, X_9, 8, 9, incident=True)
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_19 = eval_trilinear_point(model, A, B, C, 'tan(A)', ID="X_19", classes=["center", "incident"])
            central_line(model, X_6, X_19, 6, 19, incident=True)
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_41 = eval_trilinear_point(model, A, B, C, 'a**(2)*(b + c - a)', ID="X_41", classes=["center", "incident"])
            central_line(model, X_1, X_41, 1, 41, incident=True)
            return promote_point_to_top(model, X_2082)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2082 on the given reference triangle."""
            model = create_model('X_2082 = NK-TRANSFORM OF X_9')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02082(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02082.json"))
            write_center_svg(
                model_acute,
                here / "x-02082.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
