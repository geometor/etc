X(717) = EVEN (- 3, 0) CIRCUMCIRCLE POINT — Code¶

X(717) = isogonal conjugate of X(716)
"""GEOMETOR construction model for X(717) = EVEN (- 3, 0) CIRCUMCIRCLE POINT."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            isogonal_conjugate,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00717(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X(717) = EVEN (- 3, 0) CIRCUMCIRCLE POINT on reference triangle ABC."""
            X_716 = eval_trilinear_point(model, A, B, C, 'a**(-4)*(b**(0)+ c**(0)) - a**(-1)*(b**(-3)+ c**(-3))', ID="X(716)", classes=["center"])
            X_717 = isogonal_conjugate(model, A, B, C, X_716, ID="X(717)", classes=["selected", "target"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X(6)", classes=["center", "incident"])
            X_789 = eval_trilinear_point(model, A, B, C, 'a **(- 2)/(b**(3) - c**(3))', ID="X(789)", classes=["center", "incident"])
            central_line(model, X_6, X_789, 6, 789, incident=True)
            X_560 = eval_trilinear_point(model, A, B, C, 'a**(4)', ID="X(560)", classes=["center", "incident"])
            X_825 = eval_trilinear_point(model, A, B, C, '1/(a**(-1)*(b**(3) - c**(3)) + a**(2)*(b**(0) - c**(0)))', ID="X(825)", classes=["center", "incident"])
            central_line(model, X_560, X_825, 560, 825, incident=True)
            return promote_point_to_top(model, X_717)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X(717) on the given reference triangle."""
            model = create_model('X(717) = EVEN (- 3, 0) CIRCUMCIRCLE POINT')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00717(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00717.json"))
            write_center_svg(
                model_acute,
                here / "x-00717.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
