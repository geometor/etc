X1937¶

TRILINEAR POLE OF PU(15)

X1937 = trilinear pole of line X65X650
"""GEOMETOR construction model for X_1937 = TRILINEAR POLE OF PU(15)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            isogonal_conjugate,
            nagel_point_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01937(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1937 = TRILINEAR POLE OF PU(15) on reference triangle ABC."""
            X_1936 = eval_trilinear_point(model, A, B, C, '(cos(A))**(2) - cos(B)*cos(C)', ID="X_1936", classes=["center"])
            X_1937 = isogonal_conjugate(model, A, B, C, X_1936, ID="X_1937", classes=["selected", "target"])
            X_21 = eval_trilinear_point(model, A, B, C, '1/(cos(B) + cos(C))', ID="X_21", classes=["center", "incident"])
            X_73 = eval_trilinear_point(model, A, B, C, 'sec(B) + sec(C)', ID="X_73", classes=["center", "incident"])
            central_line(model, X_21, X_73, 21, 73, incident=True)
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_185 = eval_trilinear_point(model, A, B, C, '(cos(A))*(1 - cos(A)*cos(B - C))', ID="X_185", classes=["center", "incident"])
            central_line(model, X_1, X_185, 1, 185, incident=True)
            X_8 = nagel_point_of(model, A, B, C, ID="X_8", classes=["center", "incident"])
            X_201 = eval_trilinear_point(model, A, B, C, '(cos(A))*(1 + cos(B - C))', ID="X_201", classes=["center", "incident"])
            central_line(model, X_8, X_201, 8, 201, incident=True)
            return promote_point_to_top(model, X_1937)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1937 on the given reference triangle."""
            model = create_model('X_1937 = TRILINEAR POLE OF PU(15)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01937(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01937.json"))
            write_center_svg(
                model_acute,
                here / "x-01937.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
