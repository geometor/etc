X1931¶

CROSSDIFFERENCE OF PU(32)

X1931 = perspector of conic {{A, B, C,PU(31)}}
"""GEOMETOR construction model for X_1931 = CROSSDIFFERENCE OF PU(32)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01931(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1931 = CROSSDIFFERENCE OF PU(32) on reference triangle ABC."""
            X_1931 = eval_trilinear_point(model, A, B, C, '(((b + c)**(2) - (a + b)*(a + c))/(b + c))', ID="X_1931", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_21 = eval_trilinear_point(model, A, B, C, '1/(cos(B) + cos(C))', ID="X_21", classes=["center", "incident"])
            central_line(model, X_1, X_21, 1, 21, incident=True)
            X_99 = eval_trilinear_point(model, A, B, C, 'b*c/(b**(2) - c**(2))', ID="X_99", classes=["center", "incident"])
            X_239 = eval_trilinear_point(model, A, B, C, 'b*c*(a**(2) - b*c)', ID="X_239", classes=["center", "incident"])
            central_line(model, X_99, X_239, 99, 239, incident=True)
            X_44 = eval_trilinear_point(model, A, B, C, 'b + c - 2*a', ID="X_44", classes=["center", "incident"])
            X_662 = eval_trilinear_point(model, A, B, C, '1/(cot(B) - cot(C))', ID="X_662", classes=["center", "incident"])
            central_line(model, X_44, X_662, 44, 662, incident=True)
            return promote_point_to_top(model, X_1931)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1931 on the given reference triangle."""
            model = create_model('X_1931 = CROSSDIFFERENCE OF PU(32)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01931(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01931.json"))
            write_center_svg(
                model_acute,
                here / "x-01931.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
