X(125) = CENTER OF JERABEK HYPERBOLA — Code¶

X(125) is the pole of the Fermat axis with respect to the Dao-Moses-Telv circle. (Randy Hutson, December 14, 2014)
"""GEOMETOR construction model for X(125) = CENTER OF JERABEK HYPERBOLA."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            orthocenter_of,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00125(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X(125) = CENTER OF JERABEK HYPERBOLA on reference triangle ABC."""
            X_125 = eval_trilinear_point(model, A, B, C, 'cos(A)*(sin(B - C))**(2)', ID="X(125)", classes=["selected", "target"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X(6)", classes=["center", "incident"])
            X_67 = eval_trilinear_point(model, A, B, C, 'b*c/(b**(4) + c**(4) - a**(4) - b**(2)*c**(2))', ID="X(67)", classes=["center", "incident"])
            central_line(model, X_6, X_67, 6, 67, incident=True)
            X_4 = orthocenter_of(model, A, B, C, ID="X(4)", classes=["center", "incident"])
            X_74 = eval_trilinear_point(model, A, B, C, '1/(cos(A) - 2*cos(B)*cos(C))', ID="X(74)", classes=["center", "incident"])
            central_line(model, X_4, X_74, 4, 74, incident=True)
            X_2 = centroid_of(model, A, B, C, ID="X(2)", classes=["center", "incident"])
            X_98 = eval_trilinear_point(model, A, B, C, 'sec(A + omega )', ID="X(98)", classes=["center", "incident"])
            central_line(model, X_2, X_98, 2, 98, incident=True)
            return promote_point_to_top(model, X_125)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X(125) on the given reference triangle."""
            model = create_model('X(125) = CENTER OF JERABEK HYPERBOLA')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00125(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00125.json"))
            write_center_svg(
                model_acute,
                here / "x-00125.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
