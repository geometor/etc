X(63) = ISOGONAL CONJUGATE OF X(19) — Code¶

Let Oa be the A-extraversion of the Conway circle (the circle centered at the A-excenter and passing through A, with radius sqrt(r_a^2 + s^2), where r_a is the A-exradius). Let Pa be the perspector of Oa, and La the polar of Pa wrt Oa. Define Lb and Lc cyclically. Let A’ = Lb∩Lc, B’ = Lc∩La, C’ = La∩Lb. Triangle A’B’C’ is perspective to the excentral triangle at X(63). (Randy Hutson, February 10, 2016)
"""GEOMETOR construction model for X(63) = ISOGONAL CONJUGATE OF X(19)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            gergonne_point_of,
            incenter_of,
            isogonal_conjugate,
            nagel_point_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00063(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X(63) = ISOGONAL CONJUGATE OF X(19) on reference triangle ABC."""
            X_19 = eval_trilinear_point(model, A, B, C, 'tan(A)', ID="X(19)", classes=["center"])
            X_63 = isogonal_conjugate(model, A, B, C, X_19, ID="X(63)", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X(2)", classes=["center", "incident"])
            X_7 = gergonne_point_of(model, A, B, C, ID="X(7)", classes=["center", "incident"])
            central_line(model, X_2, X_7, 2, 7, incident=True)
            X_1 = incenter_of(model, A, B, C, ID="X(1)", classes=["center", "incident"])
            X_21 = eval_trilinear_point(model, A, B, C, '1/(cos(B) + cos(C))', ID="X(21)", classes=["center", "incident"])
            central_line(model, X_1, X_21, 1, 21, incident=True)
            X_8 = nagel_point_of(model, A, B, C, ID="X(8)", classes=["center", "incident"])
            X_20 = eval_trilinear_point(model, A, B, C, 'cos(A) - cos(B)*cos(C)', ID="X(20)", classes=["center", "incident"])
            central_line(model, X_8, X_20, 8, 20, incident=True)
            return promote_point_to_top(model, X_63)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X(63) on the given reference triangle."""
            model = create_model('X(63) = ISOGONAL CONJUGATE OF X(19)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00063(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00063.json"))
            write_center_svg(
                model_acute,
                here / "x-00063.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
