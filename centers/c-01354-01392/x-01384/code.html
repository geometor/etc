X1384¶

2nd GRINBERG HOMOTHETIC CENTER

Let GA be the circumcenter of triangle BCG, where G - centroid(ABC). Define GB and GC cyclically. Triangle GAGBGC is homothetic to the pedal triangle of X6, and X1384 is the center of the homothety.
"""GEOMETOR construction model for X_1384 = 2nd GRINBERG HOMOTHETIC CENTER."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            ceva_conjugate,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01384(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1384 = 2nd GRINBERG HOMOTHETIC CENTER on reference triangle ABC."""
            X_1383 = eval_trilinear_point(model, A, B, C, 'a/(2*b**(2) + 2*c**(2) - a**(2))', ID="X_1383", classes=["center"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center"])
            X_1384 = ceva_conjugate(model, A, B, C, X_1383, X_6, ID="X_1384", classes=["selected", "target"])
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center", "incident"])
            central_line(model, X_3, X_6, 3, 6, incident=True)
            X_25 = eval_trilinear_point(model, A, B, C, 'sin(A)*tan(A)', ID="X_25", classes=["center", "incident"])
            X_111 = eval_trilinear_point(model, A, B, C, 'a/(2*a**(2) - b**(2) - c**(2))', ID="X_111", classes=["center", "incident"])
            central_line(model, X_25, X_111, 25, 111, incident=True)
            X_230 = eval_trilinear_point(model, A, B, C, 'b*c*(a**(2)*(2*a**(2) - b**(2) - c**(2)) + (b**(2) - c**(2))**(2))', ID="X_230", classes=["center", "incident"])
            X_381 = eval_trilinear_point(model, A, B, C, '2*cos(B - C) - cos(A)', ID="X_381", classes=["center", "incident"])
            central_line(model, X_230, X_381, 230, 381, incident=True)
            return promote_point_to_top(model, X_1384)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1384 on the given reference triangle."""
            model = create_model('X_1384 = 2nd GRINBERG HOMOTHETIC CENTER')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01384(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01384.json"))
            write_center_svg(
                model_acute,
                here / "x-01384.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
