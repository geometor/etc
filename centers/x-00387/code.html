X(387) = INTERSECTION OF LINES X(1,2) AND X(4,6) — Code¶

X(387) = crossdifference of every pair of points on line X(520)X(649)
"""GEOMETOR construction model for X(387) = INTERSECTION OF LINES X(1,2) AND X(4,6)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            orthocenter_of,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00387(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X(387) = INTERSECTION OF LINES X(1,2) AND X(4,6) on reference triangle ABC."""
            X_387 = eval_trilinear_point(model, A, B, C, 'b*c*(-a**(4) + 2*a**(2)*(a + b + c)**(2) + (b**(2) - c**(2))**(2))', ID="X(387)", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X(1)", classes=["center", "incident"])
            X_2 = centroid_of(model, A, B, C, ID="X(2)", classes=["center", "incident"])
            central_line(model, X_1, X_2, 1, 2, incident=True)
            X_4 = orthocenter_of(model, A, B, C, ID="X(4)", classes=["center", "incident"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X(6)", classes=["center", "incident"])
            central_line(model, X_4, X_6, 4, 6, incident=True)
            X_20 = eval_trilinear_point(model, A, B, C, 'cos(A) - cos(B)*cos(C)', ID="X(20)", classes=["center", "incident"])
            X_58 = eval_trilinear_point(model, A, B, C, 'a/(b + c)', ID="X(58)", classes=["center", "incident"])
            central_line(model, X_20, X_58, 20, 58, incident=True)
            return promote_point_to_top(model, X_387)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X(387) on the given reference triangle."""
            model = create_model('X(387) = INTERSECTION OF LINES X(1,2) AND X(4,6)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00387(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00387.json"))
            write_center_svg(
                model_acute,
                here / "x-00387.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
