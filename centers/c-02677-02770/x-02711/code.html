X2711¶

SR(X99, X105)

X2711 = reflection of X2704 in X3
"""GEOMETOR construction model for X_2711 = SR(X_99, X_105)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            point_from_barycentrics,
            promote_point_to_top,
            reflection_in_point,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02711(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2711 = SR(X_99, X_105) on reference triangle ABC."""
            X_2704 = point_from_barycentrics(model, A, B, C, 0.25390548, 1.1213867, -0.37529215, ID="X_2704", classes=["center"])
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center"])
            X_2711 = reflection_in_point(model, X_2704, X_3, ID="X_2711", classes=["selected", "target"], guide=False)
            X_99 = eval_trilinear_point(model, A, B, C, 'b*c/(b**(2) - c**(2))', ID="X_99", classes=["center", "incident"])
            X_518 = eval_trilinear_point(model, A, B, C, '(a*b + a*c - b**(2) - c**(2))', ID="X_518", classes=["center", "incident"])
            central_line(model, X_99, X_518, 99, 518, incident=True)
            X_105 = eval_trilinear_point(model, A, B, C, '1/(b**(2) + c**(2) - a*(b + c))', ID="X_105", classes=["center", "incident"])
            X_512 = eval_trilinear_point(model, A, B, C, 'a*(b**(2) - c**(2))', ID="X_512", classes=["center", "incident"])
            central_line(model, X_105, X_512, 105, 512, incident=True)
            X_35 = eval_trilinear_point(model, A, B, C, '1 + 2*cos(A)', ID="X_35", classes=["center", "incident"])
            X_813 = eval_trilinear_point(model, A, B, C, '1/(a**(-2)*(b**(1) - c**(1)) + a**(0)*(b**(-1) - c **(-1)))', ID="X_813", classes=["center", "incident"])
            central_line(model, X_35, X_813, 35, 813, incident=True)
            return promote_point_to_top(model, X_2711)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2711 on the given reference triangle."""
            model = create_model('X_2711 = SR(X_99, X_105)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02711(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02711.json"))
            write_center_svg(
                model_acute,
                here / "x-02711.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
