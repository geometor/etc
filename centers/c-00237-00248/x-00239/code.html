X239¶

X1-LINE CONJUGATE OF X42

X239 = X1-line conjugate of X42
"""GEOMETOR construction model for X_239 = X_1-LINE CONJUGATE OF X_42."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            gergonne_point_of,
            incenter_of,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00239(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_239 = X_1-LINE CONJUGATE OF X_42 on reference triangle ABC."""
            X_239 = eval_trilinear_point(model, A, B, C, 'b*c*(a**(2) - b*c)', ID="X_239", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            central_line(model, X_1, X_2, 1, 2, incident=True)
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_75 = eval_trilinear_point(model, A, B, C, '1/a**(2)', ID="X_75", classes=["center", "incident"])
            central_line(model, X_6, X_75, 6, 75, incident=True)
            X_7 = gergonne_point_of(model, A, B, C, ID="X_7", classes=["center", "incident"])
            X_193 = eval_trilinear_point(model, A, B, C, '(csc(A))*(cot(B) + cot(C) - cot(A))', ID="X_193", classes=["center", "incident"])
            central_line(model, X_7, X_193, 7, 193, incident=True)
            return promote_point_to_top(model, X_239)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_239 on the given reference triangle."""
            model = create_model('X_239 = X_1-LINE CONJUGATE OF X_42')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00239(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00239.json"))
            write_center_svg(
                model_acute,
                here / "x-00239.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
