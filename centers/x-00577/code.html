"""GEOMETOR construction model for X(577) = HARMONIC OF X(216)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            circumcenter_of,
            complement,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00577(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X(577) = HARMONIC OF X(216) on reference triangle ABC."""
            X_317 = eval_trilinear_point(model, A, B, C, 'b*c*cot(2*A)', ID="X(317)", classes=["center"])
            X_577 = complement(model, A, B, C, X_317, ID="X(577)", classes=["selected", "target"])
            X_3 = circumcenter_of(model, A, B, C, ID="X(3)", classes=["center", "incident"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X(6)", classes=["center", "incident"])
            central_line(model, X_3, X_6, 3, 6, incident=True)
            X_30 = eval_trilinear_point(model, A, B, C, 'cos(A) - 2*cos(B)*cos(C)', ID="X(30)", classes=["center", "incident"])
            X_53 = eval_trilinear_point(model, A, B, C, 'tan(A)*cos(B - C)', ID="X(53)", classes=["center", "incident"])
            central_line(model, X_30, X_53, 30, 53, incident=True)
            X_2 = centroid_of(model, A, B, C, ID="X(2)", classes=["center", "incident"])
            X_95 = eval_trilinear_point(model, A, B, C, 'b**(2)*c**(2)*sec(B - C)', ID="X(95)", classes=["center", "incident"])
            central_line(model, X_2, X_95, 2, 95, incident=True)
            return promote_point_to_top(model, X_577)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X(577) on the given reference triangle."""
            model = create_model('X(577) = HARMONIC OF X(216)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00577(model, A, B, C)
            return model


        if __name__ == "__main__":
