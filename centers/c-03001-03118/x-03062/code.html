X3062¶

ISOGONAL CONJUGATE OF X165

X3062 is the isogonal conjugate of X165.
"""GEOMETOR construction model for X_3062 = ISOGONAL CONJUGATE OF X_165."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            isogonal_conjugate,
            mittenpunkt_of,
            nagel_point_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03062(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3062 = ISOGONAL CONJUGATE OF X_165 on reference triangle ABC."""
            X_165 = eval_trilinear_point(model, A, B, C, 'tan(B/2) + tan(C/2) - tan(A/2)', ID="X_165", classes=["center"])
            X_3062 = isogonal_conjugate(model, A, B, C, X_165, ID="X_3062", classes=["selected", "target"])
            X_8 = nagel_point_of(model, A, B, C, ID="X_8", classes=["center", "incident"])
            X_144 = eval_trilinear_point(model, A, B, C, '(csc(A))*(tan(B/2) + tan(C/2) - tan(A/2))', ID="X_144", classes=["center", "incident"])
            central_line(model, X_8, X_144, 8, 144, incident=True)
            X_9 = mittenpunkt_of(model, A, B, C, ID="X_9", classes=["center", "incident"])
            central_line(model, X_9, X_165, 9, 165, incident=True)
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_971 = eval_trilinear_point(model, A, B, C, 'a**(4)*(b + c) - 2*a**(3)*(b**(2) + c**(2) - b*c) + 2*a*(b - c)**(2)*(b**(2) + c**(2) + b*c) - (b - c)**(2)*(b + c)**(3)', ID="X_971", classes=["center", "incident"])
            central_line(model, X_1, X_971, 1, 971, incident=True)
            return promote_point_to_top(model, X_3062)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3062 on the given reference triangle."""
            model = create_model('X_3062 = ISOGONAL CONJUGATE OF X_165')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03062(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03062.json"))
            write_center_svg(
                model_acute,
                here / "x-03062.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
