X3049¶

INTERSECTION X6X_523∩X520X_647

X3049 = reflection of X2451 in X6
"""GEOMETOR construction model for X_3049 = INTERSECTION X_6X_523∩X_520X_647."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            promote_point_to_top,
            reflection_in_point,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03049(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3049 = INTERSECTION X_6X_523∩X_520X_647 on reference triangle ABC."""
            X_2451 = eval_trilinear_point(model, A, B, C, '(a*(b**(2) - c**(2))*(a**(4) + 2*b**(2)*c**(2) - a**(2)*b**(2)- a**(2)*c**(2)))', ID="X_2451", classes=["center"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center"])
            X_3049 = reflection_in_point(model, X_2451, X_6, ID="X_3049", classes=["selected", "target"], guide=False)
            X_523 = eval_trilinear_point(model, A, B, C, 'sin(B -C)', ID="X_523", classes=["center", "incident"])
            central_line(model, X_6, X_523, 6, 523, incident=True)
            X_520 = eval_trilinear_point(model, A, B, C, '(cos(A))*(sin(2*B) - sin(2*C))', ID="X_520", classes=["center", "incident"])
            X_647 = eval_trilinear_point(model, A, B, C, 'a*(b**(2) - c**(2))*(b**(2) + c**(2) - a**(2))', ID="X_647", classes=["center", "incident"])
            central_line(model, X_520, X_647, 520, 647, incident=True)
            X_669 = eval_trilinear_point(model, A, B, C, 'a**(3)*(b**(2) - c**(2))', ID="X_669", classes=["center", "incident"])
            X_688 = eval_trilinear_point(model, A, B, C, '(a**(3)*(b**(4) - c**(4)))', ID="X_688", classes=["center", "incident"])
            central_line(model, X_669, X_688, 669, 688, incident=True)
            return promote_point_to_top(model, X_3049)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3049 on the given reference triangle."""
            model = create_model('X_3049 = INTERSECTION X_6X_523∩X_520X_647')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03049(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03049.json"))
            write_center_svg(
                model_acute,
                here / "x-03049.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
