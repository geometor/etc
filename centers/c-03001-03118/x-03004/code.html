X3004¶

(DE LONGCHAMPS LINE)∩(GERGONNE LINE)

X3004 = crossdifference of every pair of points on line X32X55
"""GEOMETOR construction model for X_3004 = (DE LONGCHAMPS LINE)∩(GERGONNE LINE)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03004(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3004 = (DE LONGCHAMPS LINE)∩(GERGONNE LINE) on reference triangle ABC."""
            X_3004 = eval_trilinear_point(model, A, B, C, '(b*c*(b - c)*(b**(2) + c**(2) + a*b + a*c))', ID="X_3004", classes=["selected", "target"])
            X_241 = eval_trilinear_point(model, A, B, C, '(cos(B/2))**(4) - ((cos(A/2))**(2))*((cos(B/2))**(2) +(cos(C/2))**(2)) + (cos(C/2))**(4)', ID="X_241", classes=["center", "incident"])
            X_514 = eval_trilinear_point(model, A, B, C, '(b - c)/a', ID="X_514", classes=["center", "incident"])
            central_line(model, X_241, X_514, 241, 514, incident=True)
            X_325 = eval_trilinear_point(model, A, B, C, '(csc(A))**(2)', ID="X_325", classes=["center", "incident"])
            X_523 = eval_trilinear_point(model, A, B, C, 'sin(B -C)', ID="X_523", classes=["center", "incident"])
            central_line(model, X_325, X_523, 325, 523, incident=True)
            X_661 = eval_trilinear_point(model, A, B, C, 'cot(B) - cot(C)', ID="X_661", classes=["center", "incident"])
            X_918 = eval_barycentric_point(model, A, B, C, '((b - c)*(b**(2) + c**(2) -a*b - a*c))', ID="X_918", classes=["center", "incident"])
            central_line(model, X_661, X_918, 661, 918, incident=True)
            return promote_point_to_top(model, X_3004)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3004 on the given reference triangle."""
            model = create_model('X_3004 = (DE LONGCHAMPS LINE)∩(GERGONNE LINE)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03004(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03004.json"))
            write_center_svg(
                model_acute,
                here / "x-03004.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
