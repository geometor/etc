X3061¶

INTERSECTION OF LINES X1X_6 AND X2X_257

X3061 = X3056-cross conjugate of X982
"""GEOMETOR construction model for X_3061 = INTERSECTION OF LINES X_1X_6 AND X_2X_257."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
            spieker_center_of,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03061(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3061 = INTERSECTION OF LINES X_1X_6 AND X_2X_257 on reference triangle ABC."""
            X_3061 = eval_trilinear_point(model, A, B, C, '((b + c - a)*(b**(2) + c**(2) - b*c))', ID="X_3061", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            central_line(model, X_1, X_6, 1, 6, incident=True)
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_257 = eval_trilinear_point(model, A, B, C, '1/(a**(3) + a*b*c)', ID="X_257", classes=["center", "incident"])
            central_line(model, X_2, X_257, 2, 257, incident=True)
            X_10 = spieker_center_of(model, A, B, C, ID="X_10", classes=["center", "incident"])
            X_262 = eval_trilinear_point(model, A, B, C, 'sec(A -  omega  )', ID="X_262", classes=["center", "incident"])
            central_line(model, X_10, X_262, 10, 262, incident=True)
            return promote_point_to_top(model, X_3061)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3061 on the given reference triangle."""
            model = create_model('X_3061 = INTERSECTION OF LINES X_1X_6 AND X_2X_257')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03061(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03061.json"))
            write_center_svg(
                model_acute,
                here / "x-03061.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
