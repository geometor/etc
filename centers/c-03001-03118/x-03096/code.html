X3096¶

INTERSECTION OF LINES X2X_32 AND X76X_141

X3096 = homothetic center of medial triangle and 5th Brocard triangle
"""GEOMETOR construction model for X_3096 = INTERSECTION OF LINES X_2X_32 AND X_76X_141."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03096(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3096 = INTERSECTION OF LINES X_2X_32 AND X_76X_141 on reference triangle ABC."""
            X_3096 = eval_trilinear_point(model, A, B, C, 'b*c*(b**(4) + c**(4) + b**(2)*c**(2) + c**(2)*a**(2) + a**(2)*b**(2))', ID="X_3096", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_32 = eval_trilinear_point(model, A, B, C, 'a**(3)', ID="X_32", classes=["center", "incident"])
            central_line(model, X_2, X_32, 2, 32, incident=True)
            X_76 = eval_trilinear_point(model, A, B, C, '1/a**(3)', ID="X_76", classes=["center", "incident"])
            X_141 = eval_trilinear_point(model, A, B, C, 'b*c*(b**(2) + c**(2))', ID="X_141", classes=["center", "incident"])
            central_line(model, X_76, X_141, 76, 141, incident=True)
            X_114 = eval_trilinear_point(model, A, B, C, 'b*c*(b*sec(B +  omega  ) + c*sec(C +  omega  ))', ID="X_114", classes=["center", "incident"])
            X_631 = eval_trilinear_point(model, A, B, C, '2*cos(A) + cos(B)*cos(C)', ID="X_631", classes=["center", "incident"])
            central_line(model, X_114, X_631, 114, 631, incident=True)
            return promote_point_to_top(model, X_3096)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3096 on the given reference triangle."""
            model = create_model('X_3096 = INTERSECTION OF LINES X_2X_32 AND X_76X_141')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03096(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03096.json"))
            write_center_svg(
                model_acute,
                here / "x-03096.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
