X3107¶

CENTER OF 2nd BROCARD EQUILATERAL TRIANGLE

X3107 is the center of the equilateral triangle introduced at X3105.
"""GEOMETOR construction model for X_3107 = CENTER OF 2nd BROCARD EQUILATERAL TRIANGLE."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            midpoint,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03107(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3107 = CENTER OF 2nd BROCARD EQUILATERAL TRIANGLE on reference triangle ABC."""
            X_3105 = eval_trilinear_point(model, A, B, C, 'cos(A +  omega  +  pi  /6)', ID="X_3105", classes=["center"])
            X_3106 = eval_trilinear_point(model, A, B, C, 'cos(A +  omega  -  pi  /3)', ID="X_3106", classes=["center"])
            X_3107 = midpoint(model, X_3105, X_3106, ID="X_3107", classes=["selected", "target"], guide=False)
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center", "incident"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            central_line(model, X_3, X_6, 3, 6, incident=True)
            X_17 = eval_trilinear_point(model, A, B, C, 'csc(A +  pi  /6)', ID="X_17", classes=["center", "incident"])
            X_76 = eval_trilinear_point(model, A, B, C, '1/a**(3)', ID="X_76", classes=["center", "incident"])
            central_line(model, X_17, X_76, 17, 76, incident=True)
            X_14 = eval_trilinear_point(model, A, B, C, 'csc(A -  pi  /3)', ID="X_14", classes=["center", "incident"])
            X_262 = eval_trilinear_point(model, A, B, C, 'sec(A -  omega  )', ID="X_262", classes=["center", "incident"])
            central_line(model, X_14, X_262, 14, 262, incident=True)
            return promote_point_to_top(model, X_3107)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3107 on the given reference triangle."""
            model = create_model('X_3107 = CENTER OF 2nd BROCARD EQUILATERAL TRIANGLE')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03107(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03107.json"))
            write_center_svg(
                model_acute,
                here / "x-03107.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
