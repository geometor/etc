X3110¶

ORTHOGONAL PROJECTION OF X1 ON BROCARD AXIS

X3110 = crossdifference of every pair of points on the line X523X2238
"""GEOMETOR construction model for X_3110 = ORTHOGONAL PROJECTION OF X_1 ON BROCARD AXIS."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03110(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3110 = ORTHOGONAL PROJECTION OF X_1 ON BROCARD AXIS on reference triangle ABC."""
            X_3110 = eval_trilinear_point(model, A, B, C, '(a*(b**(4) + c**(4) + 2*a**(2)*b*c - a*(b + c)*(b**(2) + c**(2)))/(b + c))', ID="X_3110", classes=["selected", "target"])
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center", "incident"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            central_line(model, X_3, X_6, 3, 6, incident=True)
            X_60 = eval_trilinear_point(model, A, B, C, '1/(1 + cos(B - C))', ID="X_60", classes=["center", "incident"])
            X_249 = eval_trilinear_point(model, A, B, C, '(csc(A))*(csc(B - C))**(2)', ID="X_249", classes=["center", "incident"])
            central_line(model, X_60, X_249, 60, 249, incident=True)
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_512 = eval_trilinear_point(model, A, B, C, 'a*(b**(2) - c**(2))', ID="X_512", classes=["center", "incident"])
            central_line(model, X_1, X_512, 1, 512, incident=True)
            return promote_point_to_top(model, X_3110)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3110 on the given reference triangle."""
            model = create_model('X_3110 = ORTHOGONAL PROJECTION OF X_1 ON BROCARD AXIS')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03110(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03110.json"))
            write_center_svg(
                model_acute,
                here / "x-03110.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
