X3008¶

(GERGONNE LINE)∩(NAGEL LINE)

X3008 = cevapoint of X1279 and X2348
"""GEOMETOR construction model for X_3008 = (GERGONNE LINE)∩(NAGEL LINE)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            cevapoint,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            point_from_barycentrics,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03008(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3008 = (GERGONNE LINE)∩(NAGEL LINE) on reference triangle ABC."""
            X_1279 = eval_trilinear_point(model, A, B, C, '((b - c)**(2) - a*(b + c - 2*a))', ID="X_1279", classes=["center"])
            X_2348 = point_from_barycentrics(model, A, B, C, -2.8, 0.72, 3.08, ID="X_2348", classes=["center"])
            X_3008 = cevapoint(model, A, B, C, X_1279, X_2348, ID="X_3008", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            central_line(model, X_1, X_2, 1, 2, incident=True)
            X_57 = eval_trilinear_point(model, A, B, C, '1/(b + c - a)', ID="X_57", classes=["center", "incident"])
            X_169 = eval_trilinear_point(model, A, B, C, '- (sin(A))*(cos(A/2))**(2) + (sin(B))*(cos(B/2))**(2) + (sin(C))*(cos(C/2))**(2)', ID="X_169", classes=["center", "incident"])
            central_line(model, X_57, X_169, 57, 169, incident=True)
            X_218 = eval_trilinear_point(model, A, B, C, '((cos(A/2))**(2)*((cos(B/2))**(4) + (cos(C/2))**(4) - (cos(A/2))**(4)))', ID="X_218", classes=["center", "incident"])
            X_226 = eval_trilinear_point(model, A, B, C, '(csc(A))*(cos(B) + cos(C))', ID="X_226", classes=["center", "incident"])
            central_line(model, X_218, X_226, 218, 226, incident=True)
            return promote_point_to_top(model, X_3008)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3008 on the given reference triangle."""
            model = create_model('X_3008 = (GERGONNE LINE)∩(NAGEL LINE)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03008(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03008.json"))
            write_center_svg(
                model_acute,
                here / "x-03008.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
