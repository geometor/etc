X3014¶

(DE LONGCHAMPS LINE)∩(FERMAT AXIS)

X3014 = isotomic conjugate of X35568
"""GEOMETOR construction model for X_3014 = (DE LONGCHAMPS LINE)∩(FERMAT AXIS)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03014(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3014 = (DE LONGCHAMPS LINE)∩(FERMAT AXIS) on reference triangle ABC."""
            X_3014 = eval_trilinear_point(model, A, B, C, '((sin(B))**(3)*sin(A - B)*(1 + 2*cos(2*C)) + (sin(C))**(3)*sin(A - C)*(1 + 2*cos(2*B)))', ID="X_3014", classes=["selected", "target"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_13 = eval_trilinear_point(model, A, B, C, 'csc(A +  pi  /3)', ID="X_13", classes=["center", "incident"])
            central_line(model, X_6, X_13, 6, 13, incident=True)
            X_325 = eval_trilinear_point(model, A, B, C, '(csc(A))**(2)', ID="X_325", classes=["center", "incident"])
            X_523 = eval_trilinear_point(model, A, B, C, 'sin(B -C)', ID="X_523", classes=["center", "incident"])
            central_line(model, X_325, X_523, 325, 523, incident=True)
            X_868 = eval_trilinear_point(model, A, B, C, 'sin(2*B)*sin(A - C)*csc(A - B) - sin(2*C)*sin(A - B)*csc(A - C)', ID="X_868", classes=["center", "incident"])
            X_2854 = eval_barycentric_point(model, A, B, C, 'a**(2)*(b**(6) + c**(6) - a**(4)*b**(2) - a**(4)*c**(2) + 4*a**(2)*b**(2)*c**(2) - 2*b**(4)*c**(2) - 2*b**(2)*c**(4))', ID="X_2854", classes=["center", "incident"])
            central_line(model, X_868, X_2854, 868, 2854, incident=True)
            return promote_point_to_top(model, X_3014)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3014 on the given reference triangle."""
            model = create_model('X_3014 = (DE LONGCHAMPS LINE)∩(FERMAT AXIS)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03014(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03014.json"))
            write_center_svg(
                model_acute,
                here / "x-03014.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
