X3016¶

(LEMOINE AXIS)∩(FERMAT AXIS)

X3016 = reflection of X2088 in X2493
"""GEOMETOR construction model for X_3016 = (LEMOINE AXIS)∩(FERMAT AXIS)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            promote_point_to_top,
            reflection_in_point,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03016(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3016 = (LEMOINE AXIS)∩(FERMAT AXIS) on reference triangle ABC."""
            X_2088 = eval_trilinear_point(model, A, B, C, '(a*(b**(2) - c**(2))**(2)*((b**(2) + c**(2) - a**(2))**(2) - b**(2)*c**(2)))', ID="X_2088", classes=["center"])
            X_2493 = eval_trilinear_point(model, A, B, C, 'a*(a**(6)*(b**(2) + c**(2)) - a**(4)*(b**(2) + c**(2))**(2) - a**(2)*(b**(6) + c**(6) - 2*b**(4)*c**(2) - 2*b**(2)*c**(4)) + (b**(4) + c**(4) - b**(2)*c**(2))*(b**(2) - c**(2))**(2))', ID="X_2493", classes=["center"])
            X_3016 = reflection_in_point(model, X_2088, X_2493, ID="X_3016", classes=["selected", "target"], guide=False)
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_13 = eval_trilinear_point(model, A, B, C, 'csc(A +  pi  /3)', ID="X_13", classes=["center", "incident"])
            central_line(model, X_6, X_13, 6, 13, incident=True)
            X_187 = eval_trilinear_point(model, A, B, C, 'a*(2*a**(2) - b**(2) - c**(2))', ID="X_187", classes=["center", "incident"])
            X_237 = eval_trilinear_point(model, A, B, C, 'a**(2)*cos(A +  omega  )', ID="X_237", classes=["center", "incident"])
            central_line(model, X_187, X_237, 187, 237, incident=True)
            X_316 = eval_trilinear_point(model, A, B, C, '(b*c*(b**(4) + c**(4) - a**(4) - b**(2)*c**(2)))', ID="X_316", classes=["center", "incident"])
            X_323 = eval_trilinear_point(model, A, B, C, 'sin(3*A)*(csc(A))**(2)', ID="X_323", classes=["center", "incident"])
            central_line(model, X_316, X_323, 316, 323, incident=True)
            return promote_point_to_top(model, X_3016)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3016 on the given reference triangle."""
            model = create_model('X_3016 = (LEMOINE AXIS)∩(FERMAT AXIS)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03016(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03016.json"))
            write_center_svg(
                model_acute,
                here / "x-03016.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
