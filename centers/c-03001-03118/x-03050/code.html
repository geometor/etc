INTERSECTION X6X_523∩X50X_647

X3050 = reflection of X6 in X3049
"""GEOMETOR construction model for X_3050 = INTERSECTION X_6X_523∩X_50X_647."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03050(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3050 = INTERSECTION X_6X_523∩X_50X_647 on reference triangle ABC."""
            X_3050 = eval_trilinear_point(model, A, B, C, '(a*(b**(2) - c**(2))*(a**(4) - a**(2)*b**(2) - a**(2)*c**(2) - b**(2)*c**(2)))', ID="X_3050", classes=["selected", "target"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_523 = eval_trilinear_point(model, A, B, C, 'sin(B -C)', ID="X_523", classes=["center", "incident"])
            central_line(model, X_6, X_523, 6, 523, incident=True)
            X_50 = eval_trilinear_point(model, A, B, C, 'sin(3*A)', ID="X_50", classes=["center", "incident"])
            X_647 = eval_trilinear_point(model, A, B, C, 'a*(b**(2) - c**(2))*(b**(2) + c**(2) - a**(2))', ID="X_647", classes=["center", "incident"])
            central_line(model, X_50, X_647, 50, 647, incident=True)
            X_512 = eval_trilinear_point(model, A, B, C, 'a*(b**(2) - c**(2))', ID="X_512", classes=["center", "incident"])
            X_1691 = eval_trilinear_point(model, A, B, C, 'sin(A - 2*omega  )', ID="X_1691", classes=["center", "incident"])
            central_line(model, X_512, X_1691, 512, 1691, incident=True)
            return promote_point_to_top(model, X_3050)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3050 on the given reference triangle."""
            model = create_model('X_3050 = INTERSECTION X_6X_523∩X_50X_647')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03050(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03050.json"))
            write_center_svg(
                model_acute,
                here / "x-03050.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
