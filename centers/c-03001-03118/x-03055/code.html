X3055¶

{X2,X6}-HARMONIC CONJUGATE OF X3054

X3055 is the {X2,X6}-harmonic conjugate of X3054.
"""GEOMETOR construction model for X_3055 = {X_2,X_6}-HARMONIC CONJUGATE OF X_3054."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            nine_point_center_of,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03055(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3055 = {X_2,X_6}-HARMONIC CONJUGATE OF X_3054 on reference triangle ABC."""
            X_3055 = eval_trilinear_point(model, A, B, C, '(b*c*(2*a**(4) + 3*b**(4) + 3*c**(4) - 7*a**(2)*b**(2) - 7*a**(2)*c**(2) - 6*b**(2)*c**(2)))', ID="X_3055", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            central_line(model, X_2, X_6, 2, 6, incident=True)
            X_140 = eval_trilinear_point(model, A, B, C, '2*cos(A) + cos(B - C)', ID="X_140", classes=["center", "incident"])
            X_187 = eval_trilinear_point(model, A, B, C, 'a*(2*a**(2) - b**(2) - c**(2))', ID="X_187", classes=["center", "incident"])
            central_line(model, X_140, X_187, 140, 187, incident=True)
            X_5 = nine_point_center_of(model, A, B, C, ID="X_5", classes=["center", "incident"])
            X_574 = eval_trilinear_point(model, A, B, C, 'a*(a**(2) - 2*b**(2) - 2*c**(2))', ID="X_574", classes=["center", "incident"])
            central_line(model, X_5, X_574, 5, 574, incident=True)
            return promote_point_to_top(model, X_3055)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3055 on the given reference triangle."""
            model = create_model('X_3055 = {X_2,X_6}-HARMONIC CONJUGATE OF X_3054')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03055(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03055.json"))
            write_center_svg(
                model_acute,
                here / "x-03055.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
