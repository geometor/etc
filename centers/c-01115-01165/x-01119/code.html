X1119¶

2nd HATZIPOLAKIS PERSPECTOR

Let triangle A0B0C0 be as defined for X1118. Let A1 be the orthogonal projection of A0 onto line BC, and define B1 and C1 cyclically. Then triangle A1B1C1 is perspective to ABC, and the perspector is X1119. (Antreas Hatzipolakis, #5321, 4/30/02)
"""GEOMETOR construction model for X_1119 = 2nd HATZIPOLAKIS PERSPECTOR."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            gergonne_point_of,
            orthocenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01119(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1119 = 2nd HATZIPOLAKIS PERSPECTOR on reference triangle ABC."""
            X_1119 = eval_trilinear_point(model, A, B, C, '((1 - sec(A))/(1 + cos(A)))', ID="X_1119", classes=["selected", "target"])
            X_4 = orthocenter_of(model, A, B, C, ID="X_4", classes=["center", "incident"])
            X_7 = gergonne_point_of(model, A, B, C, ID="X_7", classes=["center", "incident"])
            central_line(model, X_4, X_7, 4, 7, incident=True)
            X_19 = eval_trilinear_point(model, A, B, C, 'tan(A)', ID="X_19", classes=["center", "incident"])
            X_57 = eval_trilinear_point(model, A, B, C, '1/(b + c - a)', ID="X_57", classes=["center", "incident"])
            central_line(model, X_19, X_57, 19, 57, incident=True)
            X_34 = eval_trilinear_point(model, A, B, C, '1 - sec(A)', ID="X_34", classes=["center", "incident"])
            X_269 = eval_trilinear_point(model, A, B, C, '(tan(A/2))**(2)', ID="X_269", classes=["center", "incident"])
            central_line(model, X_34, X_269, 34, 269, incident=True)
            return promote_point_to_top(model, X_1119)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1119 on the given reference triangle."""
            model = create_model('X_1119 = 2nd HATZIPOLAKIS PERSPECTOR')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01119(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01119.json"))
            write_center_svg(
                model_acute,
                here / "x-01119.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
