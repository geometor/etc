X1116¶

CENTER OF THE LESTER CIRCLE

The appearance of i in the following list means that Xi lies on the Lester circle: 3, 5, 13, 14, 1117, 5671, 14854, 15475, 15535, 15536, 15537, 15538, 15539, 15540, 15541, 15542, 15543, 15544, 15545, 15546, 15547, 15548, 15549, 15550, 15551, 15552, 15553, 15554, 15555, 34365
"""GEOMETOR construction model for X_1116 = CENTER OF THE LESTER CIRCLE."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01116(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1116 = CENTER OF THE LESTER CIRCLE on reference triangle ABC."""
            X_1116 = eval_trilinear_point(model, A, B, C, 'b*c*(b**(2)-c**(2))*(2*(a**(2)-b**(2))*(c**(2)-a**(2)) + 3*R**(2)*(2*a**(2)-b**(2)-c**(2)) - a**(2)*(a**(2)+b**(2)+c**(2)) + a**(4)+b**(4)+c**(4))', ID="X_1116", classes=["selected", "target"])
            X_115 = eval_trilinear_point(model, A, B, C, 'b*c*(b**(2) - c**(2))**(2)', ID="X_115", classes=["center", "incident"])
            X_125 = eval_trilinear_point(model, A, B, C, 'cos(A)*(sin(B - C))**(2)', ID="X_125", classes=["center", "incident"])
            central_line(model, X_115, X_125, 115, 125, incident=True)
            X_140 = eval_trilinear_point(model, A, B, C, '2*cos(A) + cos(B - C)', ID="X_140", classes=["center", "incident"])
            X_523 = eval_trilinear_point(model, A, B, C, 'sin(B -C)', ID="X_523", classes=["center", "incident"])
            central_line(model, X_140, X_523, 140, 523, incident=True)
            return promote_point_to_top(model, X_1116)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1116 on the given reference triangle."""
            model = create_model('X_1116 = CENTER OF THE LESTER CIRCLE')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01116(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01116.json"))
            write_center_svg(
                model_acute,
                here / "x-01116.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
