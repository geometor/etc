X1149¶

ISOGONAL CONJUGATE OF X1120

X1149 is the isogonal conjugate of X1120.
"""GEOMETOR construction model for X_1149 = ISOGONAL CONJUGATE OF X_1120."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            isogonal_conjugate,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01149(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1149 = ISOGONAL CONJUGATE OF X_1120 on reference triangle ABC."""
            X_1120 = eval_trilinear_point(model, A, B, C, '(b*c/(b**(2) + c**(2) + b*a + c*a - 4*b*c))', ID="X_1120", classes=["center"])
            X_1149 = isogonal_conjugate(model, A, B, C, X_1120, ID="X_1149", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            central_line(model, X_1, X_2, 1, 2, incident=True)
            X_36 = eval_trilinear_point(model, A, B, C, '1 - 2*cos(A)', ID="X_36", classes=["center", "incident"])
            X_106 = eval_trilinear_point(model, A, B, C, 'a/(2*a - b - c)', ID="X_106", classes=["center", "incident"])
            central_line(model, X_36, X_106, 36, 106, incident=True)
            X_38 = eval_trilinear_point(model, A, B, C, 'b**(2) + c**(2)', ID="X_38", classes=["center", "incident"])
            X_392 = eval_trilinear_point(model, A, B, C, '(b + c)*(b**(2) + c**(2) - a**(2)) + 4*a*b*c', ID="X_392", classes=["center", "incident"])
            central_line(model, X_38, X_392, 38, 392, incident=True)
            return promote_point_to_top(model, X_1149)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1149 on the given reference triangle."""
            model = create_model('X_1149 = ISOGONAL CONJUGATE OF X_1120')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01149(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01149.json"))
            write_center_svg(
                model_acute,
                here / "x-01149.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
