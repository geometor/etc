X2998¶

CD(X32)

X2998 = trilinear pole of line X512X625 (the complement of the Lemoine axis)
"""GEOMETOR construction model for X_2998 = CD(X_32)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            isogonal_conjugate,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02998(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2998 = CD(X_32) on reference triangle ABC."""
            X_1613 = eval_trilinear_point(model, A, B, C, '(a*(b**(2)*c**(2) - c**(2)*a**(2) - a**(2)*b**(2)))', ID="X_1613", classes=["center"])
            X_2998 = isogonal_conjugate(model, A, B, C, X_1613, ID="X_2998", classes=["selected", "target"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_194 = eval_trilinear_point(model, A, B, C, 'b*c*(a**(2)*b**(2) + a**(2)*c**(2) - b**(2)*c**(2))', ID="X_194", classes=["center", "incident"])
            central_line(model, X_6, X_194, 6, 194, incident=True)
            X_42 = eval_trilinear_point(model, A, B, C, 'a*(b + c)', ID="X_42", classes=["center", "incident"])
            X_192 = eval_trilinear_point(model, A, B, C, 'b*c*(c*a + a*b - b*c)', ID="X_192", classes=["center", "incident"])
            central_line(model, X_42, X_192, 42, 192, incident=True)
            X_25 = eval_trilinear_point(model, A, B, C, 'sin(A)*tan(A)', ID="X_25", classes=["center", "incident"])
            X_385 = eval_trilinear_point(model, A, B, C, 'b*c*(a**(4) - b**(2)*c**(2))', ID="X_385", classes=["center", "incident"])
            central_line(model, X_25, X_385, 25, 385, incident=True)
            return promote_point_to_top(model, X_2998)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2998 on the given reference triangle."""
            model = create_model('X_2998 = CD(X_32)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02998(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02998.json"))
            write_center_svg(
                model_acute,
                here / "x-02998.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
