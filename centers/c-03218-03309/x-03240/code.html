X3240¶

INTERSECTION OF LINES X1X_2 AND X6X_100

X3240 = crossdifference of every pair of points on the line X649X891
"""GEOMETOR construction model for X_3240 = INTERSECTION OF LINES X_1X_2 AND X_6X_100."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03240(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3240 = INTERSECTION OF LINES X_1X_2 AND X_6X_100 on reference triangle ABC."""
            X_3240 = eval_trilinear_point(model, A, B, C, 'b*c - 2*a*(b + c)', ID="X_3240", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            central_line(model, X_1, X_2, 1, 2, incident=True)
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_100 = eval_trilinear_point(model, A, B, C, '1/(b - c)', ID="X_100", classes=["center", "incident"])
            central_line(model, X_6, X_100, 6, 100, incident=True)
            X_89 = eval_trilinear_point(model, A, B, C, '1/(2*b + 2*c - a)', ID="X_89", classes=["center", "incident"])
            X_291 = eval_trilinear_point(model, A, B, C, '1/(a**(2) - b*c)', ID="X_291", classes=["center", "incident"])
            central_line(model, X_89, X_291, 89, 291, incident=True)
            return promote_point_to_top(model, X_3240)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3240 on the given reference triangle."""
            model = create_model('X_3240 = INTERSECTION OF LINES X_1X_2 AND X_6X_100')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03240(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03240.json"))
            write_center_svg(
                model_acute,
                here / "x-03240.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
