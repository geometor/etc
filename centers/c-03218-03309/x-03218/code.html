X3218¶

INTERSECTION OF LINES X2X_7 AND X8X_46

X3218 = anticomplement of X908
"""GEOMETOR construction model for X_3218 = INTERSECTION OF LINES X_2X_7 AND X_8X_46."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            gergonne_point_of,
            incenter_of,
            isogonal_conjugate,
            nagel_point_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03218(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3218 = INTERSECTION OF LINES X_2X_7 AND X_8X_46 on reference triangle ABC."""
            X_2161 = eval_trilinear_point(model, A, B, C, '((sin(A))/(1 - 2*cos(A)))', ID="X_2161", classes=["center"])
            X_3218 = isogonal_conjugate(model, A, B, C, X_2161, ID="X_3218", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_7 = gergonne_point_of(model, A, B, C, ID="X_7", classes=["center", "incident"])
            central_line(model, X_2, X_7, 2, 7, incident=True)
            X_8 = nagel_point_of(model, A, B, C, ID="X_8", classes=["center", "incident"])
            X_46 = eval_trilinear_point(model, A, B, C, 'cos(B) + cos(C) - cos(A)', ID="X_46", classes=["center", "incident"])
            central_line(model, X_8, X_46, 8, 46, incident=True)
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_89 = eval_trilinear_point(model, A, B, C, '1/(2*b + 2*c - a)', ID="X_89", classes=["center", "incident"])
            central_line(model, X_1, X_89, 1, 89, incident=True)
            return promote_point_to_top(model, X_3218)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3218 on the given reference triangle."""
            model = create_model('X_3218 = INTERSECTION OF LINES X_2X_7 AND X_8X_46')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03218(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03218.json"))
            write_center_svg(
                model_acute,
                here / "x-03218.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
