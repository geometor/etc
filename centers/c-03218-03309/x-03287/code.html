X3287¶

INTERSECTION OF LINES X44X_513 AND X523X_879

Let L denote the line through X1 parallel to the Brocard axis, X3X6. Coefficients for a trilinear equation for L are the trilinears for X3287.
"""GEOMETOR construction model for X_3287 = INTERSECTION OF LINES X_44X_513 AND X_523X_879."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03287(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3287 = INTERSECTION OF LINES X_44X_513 AND X_523X_879 on reference triangle ABC."""
            X_3287 = eval_trilinear_point(model, A, B, C, '(b - c)*(b + c - a)*(a**(2) + b*c)', ID="X_3287", classes=["selected", "target"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_523 = eval_trilinear_point(model, A, B, C, 'sin(B -C)', ID="X_523", classes=["center", "incident"])
            central_line(model, X_6, X_523, 6, 523, incident=True)
            X_44 = eval_trilinear_point(model, A, B, C, 'b + c - 2*a', ID="X_44", classes=["center", "incident"])
            X_513 = eval_trilinear_point(model, A, B, C, 'b - c', ID="X_513", classes=["center", "incident"])
            central_line(model, X_44, X_513, 44, 513, incident=True)
            X_37 = eval_trilinear_point(model, A, B, C, 'b + c', ID="X_37", classes=["center", "incident"])
            X_2605 = eval_trilinear_point(model, A, B, C, '(sin(C- A) - sin(B - A))', ID="X_2605", classes=["center", "incident"])
            central_line(model, X_37, X_2605, 37, 2605, incident=True)
            return promote_point_to_top(model, X_3287)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3287 on the given reference triangle."""
            model = create_model('X_3287 = INTERSECTION OF LINES X_44X_513 AND X_523X_879')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03287(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03287.json"))
            write_center_svg(
                model_acute,
                here / "x-03287.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
