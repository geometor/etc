X3222¶

ISOGONAL CONJUGATE OF X3221

X3222 is the isogonal conjugate of X3221.
"""GEOMETOR construction model for X_3222 = ISOGONAL CONJUGATE OF X_3221."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            isogonal_conjugate,
            point_from_barycentrics,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03222(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3222 = ISOGONAL CONJUGATE OF X_3221 on reference triangle ABC."""
            X_3221 = eval_trilinear_point(model, A, B, C, '(a*(b**(2) - c**(2))*(c**(2)*a**(2) + a**(2)*b**(2) - b**(2)*c**(2)))', ID="X_3221", classes=["center"])
            X_3222 = isogonal_conjugate(model, A, B, C, X_3221, ID="X_3222", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_699 = eval_trilinear_point(model, A, B, C, '(1/(a **(-5)*(b **(-2) + c **(-2)) - a **(-3)*(b **(-4) + c **(-4))))', ID="X_699", classes=["center", "incident"])
            central_line(model, X_2, X_699, 2, 699, incident=True)
            X_645 = eval_trilinear_point(model, A, B, C, '((b + c - a)/(a*(b**(2) - c**(2))))', ID="X_645", classes=["center", "incident"])
            X_803 = eval_trilinear_point(model, A, B, C, '(1/(a **(-3)*(b**(1) - c**(1)) + a**(0)*(b **(-2) - c **(-2))))', ID="X_803", classes=["center", "incident"])
            central_line(model, X_645, X_803, 645, 803, incident=True)
            X_799 = eval_trilinear_point(model, A, B, C, '(csc(A))**(2)/((cos(B))**(2) - (cos(C))**(2))', ID="X_799", classes=["center", "incident"])
            X_932 = point_from_barycentrics(model, A, B, C, 0.76211763, -0.28424928, 0.52213165, ID="X_932", classes=["center", "incident"])
            central_line(model, X_799, X_932, 799, 932, incident=True)
            return promote_point_to_top(model, X_3222)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3222 on the given reference triangle."""
            model = create_model('X_3222 = ISOGONAL CONJUGATE OF X_3221')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03222(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03222.json"))
            write_center_svg(
                model_acute,
                here / "x-03222.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
