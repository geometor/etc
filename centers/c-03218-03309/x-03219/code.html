X3219¶

INTERSECTION OF LINES X2X_7 AND X8X_90

This point occurs in a forthcoming paper by Sadi Abu-Saymeh, Mowaffaq Hajja, and Hellmuth Stachel, “Another cubic associated with the triangle.”
"""GEOMETOR construction model for X_3219 = INTERSECTION OF LINES X_2X_7 AND X_8X_90."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            gergonne_point_of,
            isogonal_conjugate,
            nagel_point_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03219(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3219 = INTERSECTION OF LINES X_2X_7 AND X_8X_90 on reference triangle ABC."""
            X_2160 = eval_trilinear_point(model, A, B, C, '(sin(A))/(1 + 2*cos(A))', ID="X_2160", classes=["center"])
            X_3219 = isogonal_conjugate(model, A, B, C, X_2160, ID="X_3219", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_7 = gergonne_point_of(model, A, B, C, ID="X_7", classes=["center", "incident"])
            central_line(model, X_2, X_7, 2, 7, incident=True)
            X_21 = eval_trilinear_point(model, A, B, C, '1/(cos(B) + cos(C))', ID="X_21", classes=["center", "incident"])
            X_72 = eval_trilinear_point(model, A, B, C, '(b + c)*cot(A)', ID="X_72", classes=["center", "incident"])
            central_line(model, X_21, X_72, 21, 72, incident=True)
            X_8 = nagel_point_of(model, A, B, C, ID="X_8", classes=["center", "incident"])
            X_90 = eval_trilinear_point(model, A, B, C, '1/(cos(B) + cos(C) - cos(A))', ID="X_90", classes=["center", "incident"])
            central_line(model, X_8, X_90, 8, 90, incident=True)
            return promote_point_to_top(model, X_3219)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3219 on the given reference triangle."""
            model = create_model('X_3219 = INTERSECTION OF LINES X_2X_7 AND X_8X_90')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03219(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03219.json"))
            write_center_svg(
                model_acute,
                here / "x-03219.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
