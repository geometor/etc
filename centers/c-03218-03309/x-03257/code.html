X3257¶

ISOGONAL CONJUGATE OF X1635

X3257 is the isogonal conjugate of X1635.
"""GEOMETOR construction model for X_3257 = ISOGONAL CONJUGATE OF X_1635."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            isogonal_conjugate,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03257(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3257 = ISOGONAL CONJUGATE OF X_1635 on reference triangle ABC."""
            X_1635 = eval_trilinear_point(model, A, B, C, '(b - c)*(b + c - 2*a)', ID="X_1635", classes=["center"])
            X_3257 = isogonal_conjugate(model, A, B, C, X_1635, ID="X_3257", classes=["selected", "target"])
            X_44 = eval_trilinear_point(model, A, B, C, 'b + c - 2*a', ID="X_44", classes=["center", "incident"])
            X_88 = eval_trilinear_point(model, A, B, C, '1/(b + c - 2*a)', ID="X_88", classes=["center", "incident"])
            central_line(model, X_44, X_88, 44, 88, incident=True)
            X_106 = eval_trilinear_point(model, A, B, C, 'a/(2*a - b - c)', ID="X_106", classes=["center", "incident"])
            X_238 = eval_trilinear_point(model, A, B, C, 'a**(2) - b*c', ID="X_238", classes=["center", "incident"])
            central_line(model, X_106, X_238, 106, 238, incident=True)
            X_100 = eval_trilinear_point(model, A, B, C, '1/(b - c)', ID="X_100", classes=["center", "incident"])
            X_513 = eval_trilinear_point(model, A, B, C, 'b - c', ID="X_513", classes=["center", "incident"])
            central_line(model, X_100, X_513, 100, 513, incident=True)
            return promote_point_to_top(model, X_3257)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3257 on the given reference triangle."""
            model = create_model('X_3257 = ISOGONAL CONJUGATE OF X_1635')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03257(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03257.json"))
            write_center_svg(
                model_acute,
                here / "x-03257.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
