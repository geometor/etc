X2574¶

ISOGONAL CONJUGATE OF X1113

As the isogonal conjugate of a point on the circumcircle, X2574 lies on the line at infinity.
"""GEOMETOR construction model for X_2574 = ISOGONAL CONJUGATE OF X_1113."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            isogonal_conjugate,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02574(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2574 = ISOGONAL CONJUGATE OF X_1113 on reference triangle ABC."""
            X_1113 = eval_trilinear_point(model, A, B, C, '(R - d)*cos(A) - 2*R*cos(B)*cos(C)', ID="X_1113", classes=["center"])
            X_2574 = isogonal_conjugate(model, A, B, C, X_1113, ID="X_2574", classes=["selected", "target"])
            X_30 = eval_trilinear_point(model, A, B, C, 'cos(A) - 2*cos(B)*cos(C)', ID="X_30", classes=["center", "incident"])
            X_511 = eval_trilinear_point(model, A, B, C, 'cos(A +  omega  )', ID="X_511", classes=["center", "incident"])
            central_line(model, X_30, X_511, 30, 511, incident=True)
            X_74 = eval_trilinear_point(model, A, B, C, '1/(cos(A) - 2*cos(B)*cos(C))', ID="X_74", classes=["center", "incident"])
            X_1114 = eval_trilinear_point(model, A, B, C, '(R + d)*cos(A) - 2*R*cos(B)*cos(C)', ID="X_1114", classes=["center", "incident"])
            central_line(model, X_74, X_1114, 74, 1114, incident=True)
            X_110 = eval_trilinear_point(model, A, B, C, 'csc(B - C)', ID="X_110", classes=["center", "incident"])
            central_line(model, X_110, X_1113, 110, 1113, incident=True)
            return promote_point_to_top(model, X_2574)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2574 on the given reference triangle."""
            model = create_model('X_2574 = ISOGONAL CONJUGATE OF X_1113')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02574(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02574.json"))
            write_center_svg(
                model_acute,
                here / "x-02574.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
