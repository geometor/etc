X2567¶

EXSIMILICENTER(NINE-POINT CIRCLE, 2nd BROCARD CIRCLE)

X2567 is the exsimilicenter(nine-point circle, 2nd Brocard circle).
"""GEOMETOR construction model for X_2567 = EXSIMILICENTER(NINE-POINT CIRCLE, 2nd BROCARD CIRCLE)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            feuerbach_point_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02567(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2567 = EXSIMILICENTER(NINE-POINT CIRCLE, 2nd BROCARD CIRCLE) on reference triangle ABC."""
            X_2567 = eval_trilinear_point(model, A, B, C, '(cos(A) - e*cos(B - C))', ID="X_2567", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center"])
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center"])
            central_line(model, X_2, X_3, 2, 3)
            X_11 = feuerbach_point_of(model, A, B, C, ID="X_11", classes=["center", "incident"])
            X_2565 = eval_trilinear_point(model, A, B, C, '(e - cos(A))', ID="X_2565", classes=["center", "incident"])
            central_line(model, X_11, X_2565, 11, 2565, incident=True)
            X_12 = eval_trilinear_point(model, A, B, C, '1 + cos(B - C)', ID="X_12", classes=["center", "incident"])
            X_2566 = eval_trilinear_point(model, A, B, C, '(cos(A) + e*cos(B - C))', ID="X_2566", classes=["center", "incident"])
            central_line(model, X_12, X_2566, 12, 2566, incident=True)
            return promote_point_to_top(model, X_2567)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2567 on the given reference triangle."""
            model = create_model('X_2567 = EXSIMILICENTER(NINE-POINT CIRCLE, 2nd BROCARD CIRCLE)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02567(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02567.json"))
            write_center_svg(
                model_acute,
                here / "x-02567.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
