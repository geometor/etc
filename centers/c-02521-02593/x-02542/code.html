"""GEOMETOR construction model for X_2542 = INSIMILICENTER(BROCARD CIRCLE, (X_4,2R))."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,

            create_model,
            eval_trilinear_point,
            orthocenter_of,
            point_from_barycentrics,
            promote_point_to_top,

        from geometor.etc.render import write_center_svg


        def construct_x_02542(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point

            """Construct X_2542 = INSIMILICENTER(BROCARD CIRCLE, (X_4,2R)) on reference triangle ABC."""
            X_2542 = eval_trilinear_point(model, A, B, C, 'e*cos(B)*cos(C) + cos(A -  omega  )', ID="X_2542", classes=["selected", "target"])
            X_4 = orthocenter_of(model, A, B, C, ID="X_4", classes=["center", "incident"])
            X_83 = eval_trilinear_point(model, A, B, C, 'b*c/(b**(2) + c**(2))', ID="X_83", classes=["center", "incident"])
            central_line(model, X_4, X_83, 4, 83, incident=True)

        def build_model(
