X2564¶

INSIMILICENTER(INCIRCLE, 2nd BROCARD CIRCLE)

X2564 is the insimilicenter(incircle, 2nd Brocard circle).
"""GEOMETOR construction model for X_2564 = INSIMILICENTER(INCIRCLE, 2nd BROCARD CIRCLE)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02564(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2564 = INSIMILICENTER(INCIRCLE, 2nd BROCARD CIRCLE) on reference triangle ABC."""
            X_2564 = eval_trilinear_point(model, A, B, C, '(e + cos(A))', ID="X_2564", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center", "incident"])
            central_line(model, X_1, X_3, 1, 3, incident=True)
            X_1342 = eval_trilinear_point(model, A, B, C, 'sin(A) + cos(A)*cot( omega  /2)', ID="X_1342", classes=["center", "incident"])
            X_1675 = eval_trilinear_point(model, A, B, C, 'e - cos(A -  omega  )', ID="X_1675", classes=["center", "incident"])
            central_line(model, X_1342, X_1675, 1342, 1675, incident=True)
            X_1343 = eval_trilinear_point(model, A, B, C, 'sin(A) - cos(A)*tan( omega  /2)', ID="X_1343", classes=["center", "incident"])
            X_1674 = eval_trilinear_point(model, A, B, C, 'e + cos(A -  omega  )', ID="X_1674", classes=["center", "incident"])
            central_line(model, X_1343, X_1674, 1343, 1674, incident=True)
            return promote_point_to_top(model, X_2564)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2564 on the given reference triangle."""
            model = create_model('X_2564 = INSIMILICENTER(INCIRCLE, 2nd BROCARD CIRCLE)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02564(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02564.json"))
            write_center_svg(
                model_acute,
                here / "x-02564.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
