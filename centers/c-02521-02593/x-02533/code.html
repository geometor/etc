X2533¶

RADICAL CENTER OF {BEVAN, FUHRMANN, (X4,2R)} CIRCLES

X2533 = crosspoint of X10 and X668
"""GEOMETOR construction model for X_2533 = RADICAL CENTER OF {BEVAN, FUHRMANN, (X_4,2R)} CIRCLES."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            crosspoint,
            eval_barycentric_point,
            eval_trilinear_point,
            promote_point_to_top,
            spieker_center_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02533(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2533 = RADICAL CENTER OF {BEVAN, FUHRMANN, (X_4,2R)} CIRCLES on reference triangle ABC."""
            X_10 = spieker_center_of(model, A, B, C, ID="X_10", classes=["center"])
            X_668 = eval_trilinear_point(model, A, B, C, '1/(a**(2)*(b - c))', ID="X_668", classes=["center"])
            X_2533 = crosspoint(model, A, B, C, X_10, X_668, ID="X_2533", classes=["selected", "target"])
            X_514 = eval_trilinear_point(model, A, B, C, '(b - c)/a', ID="X_514", classes=["center", "incident"])
            central_line(model, X_10, X_514, 10, 514, incident=True)
            X_65 = eval_trilinear_point(model, A, B, C, 'cos(B) + cos(C)', ID="X_65", classes=["center", "incident"])
            X_879 = eval_barycentric_point(model, A, B, C, '(b**(2) - c**(2))*(b**(2) + c**(2) - a**(2))/(b**(4) + c**(4) - a**(2)*b**(2) - a**(2)*c**(2))', ID="X_879", classes=["center", "incident"])
            central_line(model, X_65, X_879, 65, 879, incident=True)
            X_523 = eval_trilinear_point(model, A, B, C, 'sin(B -C)', ID="X_523", classes=["center", "incident"])
            X_656 = eval_trilinear_point(model, A, B, C, 'tan(B) - tan(C)', ID="X_656", classes=["center", "incident"])
            central_line(model, X_523, X_656, 523, 656, incident=True)
            return promote_point_to_top(model, X_2533)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2533 on the given reference triangle."""
            model = create_model('X_2533 = RADICAL CENTER OF {BEVAN, FUHRMANN, (X_4,2R)} CIRCLES')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02533(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02533.json"))
            write_center_svg(
                model_acute,
                here / "x-02533.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
