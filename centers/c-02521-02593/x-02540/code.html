X2540¶

INSIMILICENTER(NINE-POINT CIRCLE, SPIEKER RADICAL CIRCLE)

X2540 is the insimilicenter(nine-point circle, Spieker radical circle).
"""GEOMETOR construction model for X_2540 = INSIMILICENTER(NINE-POINT CIRCLE, SPIEKER RADICAL CIRCLE)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            intersect_lines,
            nine_point_center_of,
            orthocenter_of,
            point_from_barycentrics,
            promote_point_to_top,
            spieker_center_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02540(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2540 = INSIMILICENTER(NINE-POINT CIRCLE, SPIEKER RADICAL CIRCLE) on reference triangle ABC."""
            X_5 = nine_point_center_of(model, A, B, C, ID="X_5", classes=["center"])
            X_10 = spieker_center_of(model, A, B, C, ID="X_10", classes=["center"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center"])
            X_2537 = point_from_barycentrics(model, A, B, C, 0.69416008, 0.61880388, -0.31296395, ID="X_2537", classes=["center"])
            central_line(model, X_5, X_10, 5, 10)
            central_line(model, X_2, X_2537, 2, 2537)
            X_2540 = intersect_lines(model, X_5, X_10, X_2, X_2537, ID="X_2540", classes=["selected", "target"])
            X_4 = orthocenter_of(model, A, B, C, ID="X_4", classes=["center", "incident"])
            X_2536 = point_from_barycentrics(model, A, B, C, 0.40205108, 0.36842473, 0.22952419, ID="X_2536", classes=["center", "incident"])
            central_line(model, X_4, X_2536, 4, 2536, incident=True)
            return promote_point_to_top(model, X_2540)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2540 on the given reference triangle."""
            model = create_model('X_2540 = INSIMILICENTER(NINE-POINT CIRCLE, SPIEKER RADICAL CIRCLE)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02540(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02540.json"))
            write_center_svg(
                model_acute,
                here / "x-02540.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
