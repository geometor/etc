X2548¶

INSIMILICENTER(MOSES CIRCLE, (X4,2R))

The circle (X4,2R) is identified at X2474.
"""GEOMETOR construction model for X_2548 = INSIMILICENTER(MOSES CIRCLE, (X_4,2R))."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            nine_point_center_of,
            orthocenter_of,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02548(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2548 = INSIMILICENTER(MOSES CIRCLE, (X_4,2R)) on reference triangle ABC."""
            X_2548 = eval_trilinear_point(model, A, B, C, '(2*cos(B)*cos(C) + sin(A +  omega  )*csc(omega) )', ID="X_2548", classes=["selected", "target"])
            X_5 = nine_point_center_of(model, A, B, C, ID="X_5", classes=["center", "incident"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            central_line(model, X_5, X_6, 5, 6, incident=True)
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_32 = eval_trilinear_point(model, A, B, C, 'a**(3)', ID="X_32", classes=["center", "incident"])
            central_line(model, X_2, X_32, 2, 32, incident=True)
            X_4 = orthocenter_of(model, A, B, C, ID="X_4", classes=["center", "incident"])
            X_39 = eval_trilinear_point(model, A, B, C, 'a*(b**(2) + c**(2))', ID="X_39", classes=["center", "incident"])
            central_line(model, X_4, X_39, 4, 39, incident=True)
            return promote_point_to_top(model, X_2548)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2548 on the given reference triangle."""
            model = create_model('X_2548 = INSIMILICENTER(MOSES CIRCLE, (X_4,2R))')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02548(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02548.json"))
            write_center_svg(
                model_acute,
                here / "x-02548.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
