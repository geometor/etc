X(458) = EULER LINE INTERCEPT OF LINE X(76)X(275) — Code¶

As a point on the Euler line, X(458) has Shinagawa coefficients ((E + F)F, S2).
"""GEOMETOR construction model for X(458) = EULER LINE INTERCEPT OF LINE X(76)X(275)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00458(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X(458) = EULER LINE INTERCEPT OF LINE X(76)X(275) on reference triangle ABC."""
            X_458 = eval_barycentric_point(model, A, B, C, 'tan(A) + cot(omega)', ID="X(458)", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X(2)", classes=["center"])
            X_3 = circumcenter_of(model, A, B, C, ID="X(3)", classes=["center"])
            central_line(model, X_2, X_3, 2, 3)
            X_6 = symmedian_point_of(model, A, B, C, ID="X(6)", classes=["center", "incident"])
            X_264 = eval_trilinear_point(model, A, B, C, 'csc(A)*csc(2*A)', ID="X(264)", classes=["center", "incident"])
            central_line(model, X_6, X_264, 6, 264, incident=True)
            X_76 = eval_trilinear_point(model, A, B, C, '1/a**(3)', ID="X(76)", classes=["center", "incident"])
            X_275 = eval_trilinear_point(model, A, B, C, 'csc(2*A)*sec(B - C)', ID="X(275)", classes=["center", "incident"])
            central_line(model, X_76, X_275, 76, 275, incident=True)
            return promote_point_to_top(model, X_458)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X(458) on the given reference triangle."""
            model = create_model('X(458) = EULER LINE INTERCEPT OF LINE X(76)X(275)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00458(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00458.json"))
            write_center_svg(
                model_acute,
                here / "x-00458.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
