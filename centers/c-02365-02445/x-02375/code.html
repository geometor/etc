X2375¶

CIR(X81)

X2375 = isogonal conjugate of X8682
"""GEOMETOR construction model for X_2375 = CIR(X_81)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            intersect_lines,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02375(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2375 = CIR(X_81) on reference triangle ABC."""
            X_37 = eval_trilinear_point(model, A, B, C, 'b + c', ID="X_37", classes=["center"])
            X_99 = eval_trilinear_point(model, A, B, C, 'b*c/(b**(2) - c**(2))', ID="X_99", classes=["center"])
            X_110 = eval_trilinear_point(model, A, B, C, 'csc(B - C)', ID="X_110", classes=["center"])
            X_213 = eval_trilinear_point(model, A, B, C, '(b + c)*a**(2)', ID="X_213", classes=["center"])
            central_line(model, X_37, X_99, 37, 99)
            central_line(model, X_110, X_213, 110, 213)
            X_2375 = intersect_lines(model, X_37, X_99, X_110, X_213, ID="X_2375", classes=["selected", "target"])
            X_101 = eval_trilinear_point(model, A, B, C, 'a/(b - c)', ID="X_101", classes=["center", "incident"])
            X_872 = eval_trilinear_point(model, A, B, C, '(a*(b + c))**(2)', ID="X_872", classes=["center", "incident"])
            central_line(model, X_101, X_872, 101, 872, incident=True)
            return promote_point_to_top(model, X_2375)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2375 on the given reference triangle."""
            model = create_model('X_2375 = CIR(X_81)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02375(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02375.json"))
            write_center_svg(
                model_acute,
                here / "x-02375.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
