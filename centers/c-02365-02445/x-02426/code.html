X2426¶

ISOGONAL CONJUGATE OF X2400

X2426 is the isogonal conjugate of X2400.
"""GEOMETOR construction model for X_2426 = ISOGONAL CONJUGATE OF X_2400."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            isogonal_conjugate,
            point_from_barycentrics,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02426(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2426 = ISOGONAL CONJUGATE OF X_2400 on reference triangle ABC."""
            X_2400 = point_from_barycentrics(model, A, B, C, -0.30177515, 1.1065089, 0.19526627, ID="X_2400", classes=["center"])
            X_2426 = isogonal_conjugate(model, A, B, C, X_2400, ID="X_2426", classes=["selected", "target"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_31 = eval_trilinear_point(model, A, B, C, 'a**(2)', ID="X_31", classes=["center", "incident"])
            central_line(model, X_6, X_31, 6, 31, incident=True)
            X_109 = eval_trilinear_point(model, A, B, C, 'a/(cos(B) - cos(C))', ID="X_109", classes=["center", "incident"])
            X_692 = eval_trilinear_point(model, A, B, C, 'a**(2)/(b - c)', ID="X_692", classes=["center", "incident"])
            central_line(model, X_109, X_692, 109, 692, incident=True)
            X_884 = point_from_barycentrics(model, A, B, C, -0.045819015, 0.98969072, 0.056128293, ID="X_884", classes=["center", "incident"])
            X_919 = point_from_barycentrics(model, A, B, C, -0.11560694, 0.83236994, 0.28323699, ID="X_919", classes=["center", "incident"])
            central_line(model, X_884, X_919, 884, 919, incident=True)
            return promote_point_to_top(model, X_2426)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2426 on the given reference triangle."""
            model = create_model('X_2426 = ISOGONAL CONJUGATE OF X_2400')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02426(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02426.json"))
            write_center_svg(
                model_acute,
                here / "x-02426.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
