X2401¶

GIBERT-SIMSON TRANSFORM OF X104

X2401 = cevapoint of X650 and X900
"""GEOMETOR construction model for X_2401 = GIBERT-SIMSON TRANSFORM OF X_104."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            intersect_lines,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02401(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2401 = GIBERT-SIMSON TRANSFORM OF X_104 on reference triangle ABC."""
            X_104 = eval_trilinear_point(model, A, B, C, '1/(-1 + cos(B) + cos(C))', ID="X_104", classes=["center"])
            X_105 = eval_trilinear_point(model, A, B, C, '1/(b**(2) + c**(2) - a*(b + c))', ID="X_105", classes=["center"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center"])
            X_522 = eval_trilinear_point(model, A, B, C, '(cos(B) - cos(C))*(csc(A))', ID="X_522", classes=["center"])
            central_line(model, X_104, X_105, 104, 105)
            central_line(model, X_1, X_522, 1, 522)
            X_2401 = intersect_lines(model, X_104, X_105, X_1, X_522, ID="X_2401", classes=["selected", "target"])
            X_57 = eval_trilinear_point(model, A, B, C, '1/(b + c - a)', ID="X_57", classes=["center", "incident"])
            X_514 = eval_trilinear_point(model, A, B, C, '(b - c)/a', ID="X_514", classes=["center", "incident"])
            central_line(model, X_57, X_514, 57, 514, incident=True)
            return promote_point_to_top(model, X_2401)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2401 on the given reference triangle."""
            model = create_model('X_2401 = GIBERT-SIMSON TRANSFORM OF X_104')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02401(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02401.json"))
            write_center_svg(
                model_acute,
                here / "x-02401.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
