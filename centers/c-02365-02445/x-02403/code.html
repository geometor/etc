X2403¶

GIBERT-SIMSON TRANSFORM OF X106

X2403 = isogonal conjugate of X2429
"""GEOMETOR construction model for X_2403 = GIBERT-SIMSON TRANSFORM OF X_106."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            point_from_barycentrics,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02403(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2403 = GIBERT-SIMSON TRANSFORM OF X_106 on reference triangle ABC."""
            X_2403 = eval_trilinear_point(model, A, B, C, '(b*c*(b - c)*(3*a - b - c)/(2*a - b - c))', ID="X_2403", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_514 = eval_trilinear_point(model, A, B, C, '(b - c)/a', ID="X_514", classes=["center", "incident"])
            central_line(model, X_2, X_514, 2, 514, incident=True)
            X_523 = eval_trilinear_point(model, A, B, C, 'sin(B -C)', ID="X_523", classes=["center", "incident"])
            X_1222 = point_from_barycentrics(model, A, B, C, 0.3683818, 0.31991051, 0.31170768, ID="X_1222", classes=["center", "incident"])
            central_line(model, X_523, X_1222, 523, 1222, incident=True)
            X_901 = eval_trilinear_point(model, A, B, C, 'a/((b - c)*(b + c - 2*a))', ID="X_901", classes=["center", "incident"])
            X_2398 = point_from_barycentrics(model, A, B, C, -1.2222222, 0.33333333, 1.8888889, ID="X_2398", classes=["center", "incident"])
            central_line(model, X_901, X_2398, 901, 2398, incident=True)
            return promote_point_to_top(model, X_2403)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2403 on the given reference triangle."""
            model = create_model('X_2403 = GIBERT-SIMSON TRANSFORM OF X_106')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02403(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02403.json"))
            write_center_svg(
                model_acute,
                here / "x-02403.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
