X2424¶

ISOGONAL CONJUGATE OF X2398

X2424 is the isogonal conjugate of X2398.
"""GEOMETOR construction model for X_2424 = ISOGONAL CONJUGATE OF X_2398."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            isogonal_conjugate,
            point_from_barycentrics,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02424(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2424 = ISOGONAL CONJUGATE OF X_2398 on reference triangle ABC."""
            X_2398 = point_from_barycentrics(model, A, B, C, -1.2222222, 0.33333333, 1.8888889, ID="X_2398", classes=["center"])
            X_2424 = isogonal_conjugate(model, A, B, C, X_2398, ID="X_2424", classes=["selected", "target"])
            X_103 = eval_trilinear_point(model, A, B, C, 'a/((a - b)*cot(C) + (a - c)*cot(B))', ID="X_103", classes=["center", "incident"])
            X_106 = eval_trilinear_point(model, A, B, C, 'a/(2*a - b - c)', ID="X_106", classes=["center", "incident"])
            central_line(model, X_103, X_106, 103, 106, incident=True)
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_657 = eval_trilinear_point(model, A, B, C, '(1 + cos(A))*(cos(B) - cos(C))', ID="X_657", classes=["center", "incident"])
            central_line(model, X_6, X_657, 6, 657, incident=True)
            X_56 = eval_trilinear_point(model, A, B, C, 'a/(b + c - a)', ID="X_56", classes=["center", "incident"])
            X_663 = eval_trilinear_point(model, A, B, C, 'a*(b - c)*(b + c - a)', ID="X_663", classes=["center", "incident"])
            central_line(model, X_56, X_663, 56, 663, incident=True)
            return promote_point_to_top(model, X_2424)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2424 on the given reference triangle."""
            model = create_model('X_2424 = ISOGONAL CONJUGATE OF X_2398')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02424(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02424.json"))
            write_center_svg(
                model_acute,
                here / "x-02424.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
