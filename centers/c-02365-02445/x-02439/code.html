X2439¶

ISOGONAL CONJUGATE OF X2413

X2439 is the isogonal conjugate of X2413.
"""GEOMETOR construction model for X_2439 = ISOGONAL CONJUGATE OF X_2413."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            isogonal_conjugate,
            point_from_barycentrics,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02439(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2439 = ISOGONAL CONJUGATE OF X_2413 on reference triangle ABC."""
            X_2413 = point_from_barycentrics(model, A, B, C, -0.049419878, 0.94479088, 0.104629, ID="X_2413", classes=["center"])
            X_2439 = isogonal_conjugate(model, A, B, C, X_2413, ID="X_2439", classes=["selected", "target"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_17 = eval_trilinear_point(model, A, B, C, 'csc(A +  pi  /6)', ID="X_17", classes=["center", "incident"])
            central_line(model, X_6, X_17, 6, 17, incident=True)
            return promote_point_to_top(model, X_2439)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2439 on the given reference triangle."""
            model = create_model('X_2439 = ISOGONAL CONJUGATE OF X_2413')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02439(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02439.json"))
            write_center_svg(
                model_acute,
                here / "x-02439.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
