X2392¶

ISOGONAL CONJUGATE OF X2372

As the isogonal conjugate of a point on the circumcircle, X2392 lies on the line at infinity.
"""GEOMETOR construction model for X_2392 = ISOGONAL CONJUGATE OF X_2372."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            isogonal_conjugate,
            point_from_barycentrics,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02392(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2392 = ISOGONAL CONJUGATE OF X_2372 on reference triangle ABC."""
            X_2372 = point_from_barycentrics(model, A, B, C, 0.32015831, 1.1249398, -0.44509814, ID="X_2372", classes=["center"])
            X_2392 = isogonal_conjugate(model, A, B, C, X_2372, ID="X_2392", classes=["selected", "target"])
            X_36 = eval_trilinear_point(model, A, B, C, '1 - 2*cos(A)', ID="X_36", classes=["center", "incident"])
            X_58 = eval_trilinear_point(model, A, B, C, 'a/(b + c)', ID="X_58", classes=["center", "incident"])
            central_line(model, X_36, X_58, 36, 58, incident=True)
            X_30 = eval_trilinear_point(model, A, B, C, 'cos(A) - 2*cos(B)*cos(C)', ID="X_30", classes=["center", "incident"])
            X_511 = eval_trilinear_point(model, A, B, C, 'cos(A +  omega  )', ID="X_511", classes=["center", "incident"])
            central_line(model, X_30, X_511, 30, 511, incident=True)
            X_484 = eval_trilinear_point(model, A, B, C, '1 + 2*(cos(A) - cos(B) - cos(C))', ID="X_484", classes=["center", "incident"])
            X_1046 = point_from_barycentrics(model, A, B, C, 0.13236391, 0.3064877, 0.5611484, ID="X_1046", classes=["center", "incident"])
            central_line(model, X_484, X_1046, 484, 1046, incident=True)
            return promote_point_to_top(model, X_2392)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2392 on the given reference triangle."""
            model = create_model('X_2392 = ISOGONAL CONJUGATE OF X_2372')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02392(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02392.json"))
            write_center_svg(
                model_acute,
                here / "x-02392.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
