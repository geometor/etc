X2427¶

ISOGONAL CONJUGATE OF X2401

X2427 is the isogonal conjugate of X2401.
"""GEOMETOR construction model for X_2427 = ISOGONAL CONJUGATE OF X_2401."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            isogonal_conjugate,
            point_from_barycentrics,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02427(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2427 = ISOGONAL CONJUGATE OF X_2401 on reference triangle ABC."""
            X_2401 = point_from_barycentrics(model, A, B, C, -0.16666667, 1, 0.16666667, ID="X_2401", classes=["center"])
            X_2427 = isogonal_conjugate(model, A, B, C, X_2401, ID="X_2427", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            central_line(model, X_1, X_6, 1, 6, incident=True)
            X_101 = eval_trilinear_point(model, A, B, C, 'a/(b - c)', ID="X_101", classes=["center", "incident"])
            X_109 = eval_trilinear_point(model, A, B, C, 'a/(cos(B) - cos(C))', ID="X_109", classes=["center", "incident"])
            central_line(model, X_101, X_109, 101, 109, incident=True)
            X_905 = eval_trilinear_point(model, A, B, C, '(b - c)*cot(A)', ID="X_905", classes=["center", "incident"])
            X_1025 = eval_trilinear_point(model, A, B, C, '((b**(2) + c**(2) - a*(b + c))/((b - c)*(b + c - a)))', ID="X_1025", classes=["center", "incident"])
            central_line(model, X_905, X_1025, 905, 1025, incident=True)
            return promote_point_to_top(model, X_2427)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2427 on the given reference triangle."""
            model = create_model('X_2427 = ISOGONAL CONJUGATE OF X_2401')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02427(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02427.json"))
            write_center_svg(
                model_acute,
                here / "x-02427.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
