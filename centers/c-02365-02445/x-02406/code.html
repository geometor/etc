X2406¶

GIBERT-SIMSON TRANSFORM OF X109

X2406 = trilinear pole of line X117X515 (the Simson line of X109)
"""GEOMETOR construction model for X_2406 = GIBERT-SIMSON TRANSFORM OF X_109."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            gergonne_point_of,
            intersect_lines,
            point_from_barycentrics,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02406(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2406 = GIBERT-SIMSON TRANSFORM OF X_109 on reference triangle ABC."""
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center"])
            X_7 = gergonne_point_of(model, A, B, C, ID="X_7", classes=["center"])
            X_651 = eval_trilinear_point(model, A, B, C, '1/(cos(B) - cos(C))', ID="X_651", classes=["center"])
            X_653 = eval_trilinear_point(model, A, B, C, '1/(sec(B) - sec(C))', ID="X_653", classes=["center"])
            central_line(model, X_2, X_7, 2, 7)
            central_line(model, X_651, X_653, 651, 653)
            X_2406 = intersect_lines(model, X_2, X_7, X_651, X_653, ID="X_2406", classes=["selected", "target"])
            X_59 = eval_trilinear_point(model, A, B, C, '1/(1 - cos(B - C))', ID="X_59", classes=["center", "incident"])
            X_2398 = point_from_barycentrics(model, A, B, C, -1.2222222, 0.33333333, 1.8888889, ID="X_2398", classes=["center", "incident"])
            central_line(model, X_59, X_2398, 59, 2398, incident=True)
            return promote_point_to_top(model, X_2406)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2406 on the given reference triangle."""
            model = create_model('X_2406 = GIBERT-SIMSON TRANSFORM OF X_109')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02406(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02406.json"))
            write_center_svg(
                model_acute,
                here / "x-02406.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
