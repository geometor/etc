X2665¶

TRILINEAR POLE OF PU(84)

X2665 = isogonal conjugate of X2664
"""GEOMETOR construction model for X_2665 = TRILINEAR POLE OF PU(84)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            isogonal_conjugate,
            point_from_barycentrics,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02665(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2665 = TRILINEAR POLE OF PU(84) on reference triangle ABC."""
            X_2664 = eval_trilinear_point(model, A, B, C, '((1/a + 1/b)*(1/a + 1/c) - (1/b + 1/c)**(2))', ID="X_2664", classes=["center"])
            X_2665 = isogonal_conjugate(model, A, B, C, X_2664, ID="X_2665", classes=["selected", "target"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_1045 = point_from_barycentrics(model, A, B, C, 0.2223905, 0.33634079, 0.44126872, ID="X_1045", classes=["center", "incident"])
            central_line(model, X_6, X_1045, 6, 1045, incident=True)
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_1655 = eval_trilinear_point(model, A, B, C, '(b*c*(-1/a**(2) + 1/b**(2) + 1/c**(2) + 1/(b*c) + 1/(c*a) + 1/(a*b)))', ID="X_1655", classes=["center", "incident"])
            central_line(model, X_1, X_1655, 1, 1655, incident=True)
            X_43 = eval_trilinear_point(model, A, B, C, 'a*b + a*c - b*c', ID="X_43", classes=["center", "incident"])
            X_2279 = point_from_barycentrics(model, A, B, C, 0.18360932, 0.3092951, 0.50709557, ID="X_2279", classes=["center", "incident"])
            central_line(model, X_43, X_2279, 43, 2279, incident=True)
            return promote_point_to_top(model, X_2665)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2665 on the given reference triangle."""
            model = create_model('X_2665 = TRILINEAR POLE OF PU(84)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02665(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02665.json"))
            write_center_svg(
                model_acute,
                here / "x-02665.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
