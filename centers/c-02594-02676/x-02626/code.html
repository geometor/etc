X2626¶

CROSSDIFFERENCE OF PU(73)

X2626 = X2621-Ceva conjugate of X2625
"""GEOMETOR construction model for X_2626 = CROSSDIFFERENCE OF PU(73)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02626(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2626 = CROSSDIFFERENCE OF PU(73) on reference triangle ABC."""
            X_2626 = eval_trilinear_point(model, A, B, C, 'cot(A - B)*cot(A - C) + (cot(B - C))**(2)', ID="X_2626", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_564 = eval_trilinear_point(model, A, B, C, 'cos(2*B - 2*C)', ID="X_564", classes=["center", "incident"])
            central_line(model, X_1, X_564, 1, 564, incident=True)
            X_163 = eval_trilinear_point(model, A, B, C, '(sin(2*A))/(sin(2*B) - sin(2*C))', ID="X_163", classes=["center", "incident"])
            X_1983 = eval_trilinear_point(model, A, B, C, '(sin(C - A) - sin(A - B))/sin(B - C)', ID="X_1983", classes=["center", "incident"])
            central_line(model, X_163, X_1983, 163, 1983, incident=True)
            X_2616 = eval_trilinear_point(model, A, B, C, 'tan(B - C)', ID="X_2616", classes=["center", "incident"])
            X_2625 = eval_trilinear_point(model, A, B, C, 'cot(A - B)*cot(A - C) - (cot(B - C))**(2)', ID="X_2625", classes=["center", "incident"])
            central_line(model, X_2616, X_2625, 2616, 2625, incident=True)
            return promote_point_to_top(model, X_2626)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2626 on the given reference triangle."""
            model = create_model('X_2626 = CROSSDIFFERENCE OF PU(73)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02626(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02626.json"))
            write_center_svg(
                model_acute,
                here / "x-02626.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
