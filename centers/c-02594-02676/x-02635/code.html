SUM OF PU(76)

X2635 = bicentric sum of PU(76)
"""GEOMETOR construction model for X_2635 = SUM OF PU(76)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            feuerbach_point_of,
            incenter_of,
            orthocenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02635(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2635 = SUM OF PU(76) on reference triangle ABC."""
            X_2635 = eval_trilinear_point(model, A, B, C, 'sec(B) + sec(C) - 2*sec(A)', ID="X_2635", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_4 = orthocenter_of(model, A, B, C, ID="X_4", classes=["center", "incident"])
            central_line(model, X_1, X_4, 1, 4, incident=True)
            X_44 = eval_trilinear_point(model, A, B, C, 'b + c - 2*a', ID="X_44", classes=["center", "incident"])
            X_513 = eval_trilinear_point(model, A, B, C, 'b - c', ID="X_513", classes=["center", "incident"])
            central_line(model, X_44, X_513, 44, 513, incident=True)
            X_11 = feuerbach_point_of(model, A, B, C, ID="X_11", classes=["center", "incident"])
            X_1458 = eval_barycentric_point(model, A, B, C, 'a**(2)*(b**(2) + c**(2) - a*b - a*c)/(a - b - c)', ID="X_1458", classes=["center", "incident"])
            central_line(model, X_11, X_1458, 11, 1458, incident=True)
            return promote_point_to_top(model, X_2635)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2635 on the given reference triangle."""
            model = create_model('X_2635 = SUM OF PU(76)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02635(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02635.json"))
            write_center_svg(
                model_acute,
                here / "x-02635.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
