X2672¶

2nd GOLDEN ARBELOS POINT

In the construction of X2671, if A' is the center of the s-Archimedean circle of the inward arbelos on segment BC, and B' and C' are defined cyclically, then A'A, B'B, C'C concur in X2672.
"""GEOMETOR construction model for X_2672 = 2nd GOLDEN ARBELOS POINT."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02672(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2672 = 2nd GOLDEN ARBELOS POINT on reference triangle ABC."""
            X_2672 = eval_trilinear_point(model, A, B, C, 'sec(A + a*r*c*cot(3 - 5**(1/2)))', ID="X_2672", classes=["selected", "target"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_2671 = eval_trilinear_point(model, A, B, C, 'sec(A - a*r*c*cot(3 - 5**(1/2)))', ID="X_2671", classes=["center", "incident"])
            central_line(model, X_6, X_2671, 6, 2671, incident=True)
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_2673 = eval_trilinear_point(model, A, B, C, 'cos(A - a*r*c*cot(3 - 5**(1/2)))', ID="X_2673", classes=["center", "incident"])
            central_line(model, X_2, X_2673, 2, 2673, incident=True)
            return promote_point_to_top(model, X_2672)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2672 on the given reference triangle."""
            model = create_model('X_2672 = 2nd GOLDEN ARBELOS POINT')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02672(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02672.json"))
            write_center_svg(
                model_acute,
                here / "x-02672.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
