X2627¶

TRILINEAR POLE OF PU(73)

X2627 = isogonal conjugate of X2626
"""GEOMETOR construction model for X_2627 = TRILINEAR POLE OF PU(73)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            incenter_of,
            isogonal_conjugate,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02627(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2627 = TRILINEAR POLE OF PU(73) on reference triangle ABC."""
            X_2626 = eval_trilinear_point(model, A, B, C, 'cot(A - B)*cot(A - C) + (cot(B - C))**(2)', ID="X_2626", classes=["center"])
            X_2627 = isogonal_conjugate(model, A, B, C, X_2626, ID="X_2627", classes=["selected", "target"])
            X_110 = eval_trilinear_point(model, A, B, C, 'csc(B - C)', ID="X_110", classes=["center", "incident"])
            X_1296 = eval_barycentric_point(model, A, B, C, 'a**(2)/((b**(2) - c**(2))*(5*a**(2) - b**(2) - c**(2)))', ID="X_1296", classes=["center", "incident"])
            central_line(model, X_110, X_1296, 110, 1296, incident=True)
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_2621 = eval_trilinear_point(model, A, B, C, '(1/(tan(A - B)*tan(A - C) + (tan(B - C))**(2)))', ID="X_2621", classes=["center", "incident"])
            central_line(model, X_1, X_2621, 1, 2621, incident=True)
            X_1101 = eval_trilinear_point(model, A, B, C, '((csc(B - C))**(2))', ID="X_1101", classes=["center", "incident"])
            X_2618 = eval_trilinear_point(model, A, B, C, 'tan(C - A) - tan(B - A)', ID="X_2618", classes=["center", "incident"])
            central_line(model, X_1101, X_2618, 1101, 2618, incident=True)
            return promote_point_to_top(model, X_2627)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2627 on the given reference triangle."""
            model = create_model('X_2627 = TRILINEAR POLE OF PU(73)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02627(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02627.json"))
            write_center_svg(
                model_acute,
                here / "x-02627.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
