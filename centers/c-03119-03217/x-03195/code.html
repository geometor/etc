X3195¶

X33-CEVA CONJUGATE OF X25

"""GEOMETOR construction model for X_3195 = X_33-CEVA CONJUGATE OF X_25."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            intersect_lines,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03195(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3195 = X_33-CEVA CONJUGATE OF X_25 on reference triangle ABC."""
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center"])
            X_33 = eval_trilinear_point(model, A, B, C, '1 + sec(A)', ID="X_33", classes=["center"])
            X_25 = eval_trilinear_point(model, A, B, C, 'sin(A)*tan(A)', ID="X_25", classes=["center"])
            X_31 = eval_trilinear_point(model, A, B, C, 'a**(2)', ID="X_31", classes=["center"])
            central_line(model, X_6, X_33, 6, 33)
            central_line(model, X_25, X_31, 25, 31)
            X_3195 = intersect_lines(model, X_6, X_33, X_25, X_31, ID="X_3195", classes=["selected", "target"])
            X_108 = eval_trilinear_point(model, A, B, C, 'a/(sec(B) - sec(C))', ID="X_108", classes=["center", "incident"])
            X_222 = eval_trilinear_point(model, A, B, C, 'cos(A)*tan(A/2)', ID="X_222", classes=["center", "incident"])
            central_line(model, X_108, X_222, 108, 222, incident=True)
            return promote_point_to_top(model, X_3195)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3195 on the given reference triangle."""
            model = create_model('X_3195 = X_33-CEVA CONJUGATE OF X_25')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03195(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03195.json"))
            write_center_svg(
                model_acute,
                here / "x-03195.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
