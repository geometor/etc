X3148¶

X262-CEVA CONJUGATE OF X6

X3148 = inverse-in-orthocentroidal circle of X2450
"""GEOMETOR construction model for X_3148 = X_262-CEVA CONJUGATE OF X_6."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            ceva_conjugate,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03148(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3148 = X_262-CEVA CONJUGATE OF X_6 on reference triangle ABC."""
            X_262 = eval_trilinear_point(model, A, B, C, 'sec(A -  omega  )', ID="X_262", classes=["center"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center"])
            X_3148 = ceva_conjugate(model, A, B, C, X_262, X_6, ID="X_3148", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center"])
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center"])
            central_line(model, X_2, X_3, 2, 3)
            X_32 = eval_trilinear_point(model, A, B, C, 'a**(3)', ID="X_32", classes=["center", "incident"])
            X_51 = eval_trilinear_point(model, A, B, C, 'a**(2)*cos(B - C)', ID="X_51", classes=["center", "incident"])
            central_line(model, X_32, X_51, 32, 51, incident=True)
            X_157 = eval_trilinear_point(model, A, B, C, '(a*(b**(3)*cos(B) + c**(3)*cos(C) - a**(3)*cos(A)))', ID="X_157", classes=["center", "incident"])
            central_line(model, X_6, X_157, 6, 157, incident=True)
            return promote_point_to_top(model, X_3148)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3148 on the given reference triangle."""
            model = create_model('X_3148 = X_262-CEVA CONJUGATE OF X_6')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03148(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03148.json"))
            write_center_svg(
                model_acute,
                here / "x-03148.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
