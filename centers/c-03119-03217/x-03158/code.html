X3158¶

X1-CEVA CONJUGATE OF X9

X3158 = crosssum of X513 and X3020
"""GEOMETOR construction model for X_3158 = X_1-CEVA CONJUGATE OF X_9."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            mittenpunkt_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03158(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3158 = X_1-CEVA CONJUGATE OF X_9 on reference triangle ABC."""
            X_3158 = eval_trilinear_point(model, A, B, C, '((b + c - a)*(b + c - 3*a))', ID="X_3158", classes=["selected", "target"])
            X_9 = mittenpunkt_of(model, A, B, C, ID="X_9", classes=["center", "incident"])
            X_55 = eval_trilinear_point(model, A, B, C, 'a*(b + c - a)', ID="X_55", classes=["center", "incident"])
            central_line(model, X_9, X_55, 9, 55, incident=True)
            X_57 = eval_trilinear_point(model, A, B, C, '1/(b + c - a)', ID="X_57", classes=["center", "incident"])
            X_100 = eval_trilinear_point(model, A, B, C, '1/(b - c)', ID="X_100", classes=["center", "incident"])
            central_line(model, X_57, X_100, 57, 100, incident=True)
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_474 = eval_trilinear_point(model, A, B, C, 'cos(A) - (a + b + c)/a', ID="X_474", classes=["center", "incident"])
            central_line(model, X_1, X_474, 1, 474, incident=True)
            return promote_point_to_top(model, X_3158)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3158 on the given reference triangle."""
            model = create_model('X_3158 = X_1-CEVA CONJUGATE OF X_9')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03158(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03158.json"))
            write_center_svg(
                model_acute,
                here / "x-03158.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
