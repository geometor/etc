X3207¶

X55-CEVA CONJUGATE OF X6

X3207 = X55-Ceva conjugate of X6
"""GEOMETOR construction model for X_3207 = X_55-CEVA CONJUGATE OF X_6."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            ceva_conjugate,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03207(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3207 = X_55-CEVA CONJUGATE OF X_6 on reference triangle ABC."""
            X_55 = eval_trilinear_point(model, A, B, C, 'a*(b + c - a)', ID="X_55", classes=["center"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center"])
            X_3207 = ceva_conjugate(model, A, B, C, X_55, X_6, ID="X_3207", classes=["selected", "target"])
            X_41 = eval_trilinear_point(model, A, B, C, 'a**(2)*(b + c - a)', ID="X_41", classes=["center", "incident"])
            central_line(model, X_6, X_41, 6, 41, incident=True)
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center", "incident"])
            X_101 = eval_trilinear_point(model, A, B, C, 'a/(b - c)', ID="X_101", classes=["center", "incident"])
            central_line(model, X_3, X_101, 3, 101, incident=True)
            X_36 = eval_trilinear_point(model, A, B, C, '1 - 2*cos(A)', ID="X_36", classes=["center", "incident"])
            X_218 = eval_trilinear_point(model, A, B, C, '((cos(A/2))**(2)*((cos(B/2))**(4) + (cos(C/2))**(4) - (cos(A/2))**(4)))', ID="X_218", classes=["center", "incident"])
            central_line(model, X_36, X_218, 36, 218, incident=True)
            return promote_point_to_top(model, X_3207)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3207 on the given reference triangle."""
            model = create_model('X_3207 = X_55-CEVA CONJUGATE OF X_6')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03207(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03207.json"))
            write_center_svg(
                model_acute,
                here / "x-03207.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
