X3179¶

X13-CEVA CONJUGATE OF X1

X3179 = X13-Ceva conjugate of X1
"""GEOMETOR construction model for X_3179 = X_13-CEVA CONJUGATE OF X_1."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            ceva_conjugate,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            mittenpunkt_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03179(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3179 = X_13-CEVA CONJUGATE OF X_1 on reference triangle ABC."""
            X_13 = eval_trilinear_point(model, A, B, C, 'csc(A +  pi  /3)', ID="X_13", classes=["center"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center"])
            X_3179 = ceva_conjugate(model, A, B, C, X_13, X_1, ID="X_3179", classes=["selected", "target"])
            X_15 = eval_trilinear_point(model, A, B, C, 'sin(A +  pi  /3)', ID="X_15", classes=["center", "incident"])
            central_line(model, X_1, X_15, 1, 15, incident=True)
            X_9 = mittenpunkt_of(model, A, B, C, ID="X_9", classes=["center", "incident"])
            X_46 = eval_trilinear_point(model, A, B, C, 'cos(B) + cos(C) - cos(A)', ID="X_46", classes=["center", "incident"])
            central_line(model, X_9, X_46, 9, 46, incident=True)
            X_484 = eval_trilinear_point(model, A, B, C, '1 + 2*(cos(A) - cos(B) - cos(C))', ID="X_484", classes=["center", "incident"])
            central_line(model, X_13, X_484, 13, 484, incident=True)
            return promote_point_to_top(model, X_3179)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3179 on the given reference triangle."""
            model = create_model('X_3179 = X_13-CEVA CONJUGATE OF X_1')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03179(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03179.json"))
            write_center_svg(
                model_acute,
                here / "x-03179.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
