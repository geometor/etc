X3156¶

X486-CEVA CONJUGATE OF X6

"""GEOMETOR construction model for X_3156 = X_486-CEVA CONJUGATE OF X_6."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            intersect_lines,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03156(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3156 = X_486-CEVA CONJUGATE OF X_6 on reference triangle ABC."""
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center"])
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center"])
            X_51 = eval_trilinear_point(model, A, B, C, 'a**(2)*cos(B - C)', ID="X_51", classes=["center"])
            X_371 = eval_trilinear_point(model, A, B, C, 'cos(A -  pi  /4)', ID="X_371", classes=["center"])
            central_line(model, X_2, X_3, 2, 3)
            central_line(model, X_51, X_371, 51, 371)
            X_3156 = intersect_lines(model, X_2, X_3, X_51, X_371, ID="X_3156", classes=["selected", "target"])
            X_184 = eval_trilinear_point(model, A, B, C, 'a**(2)*cos(A)', ID="X_184", classes=["center", "incident"])
            X_372 = eval_trilinear_point(model, A, B, C, 'cos(A +  pi  /4)', ID="X_372", classes=["center", "incident"])
            central_line(model, X_184, X_372, 184, 372, incident=True)
            return promote_point_to_top(model, X_3156)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3156 on the given reference triangle."""
            model = create_model('X_3156 = X_486-CEVA CONJUGATE OF X_6')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03156(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03156.json"))
            write_center_svg(
                model_acute,
                here / "x-03156.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
