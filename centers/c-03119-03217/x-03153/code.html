X3153¶

X328-CEVA CONJUGATE OF X2

X3153 = anticomplement of X186
"""GEOMETOR construction model for X_3153 = X_328-CEVA CONJUGATE OF X_2."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            anticomplement,
            central_line,
            centroid_of,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03153(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3153 = X_328-CEVA CONJUGATE OF X_2 on reference triangle ABC."""
            X_186 = eval_trilinear_point(model, A, B, C, '4*cos(A) - sec(A)', ID="X_186", classes=["center"])
            X_3153 = anticomplement(model, A, B, C, X_186, ID="X_3153", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center"])
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center"])
            central_line(model, X_2, X_3, 2, 3)
            X_265 = eval_trilinear_point(model, A, B, C, 'sin(2*A)*csc(3*A)', ID="X_265", classes=["center", "incident"])
            X_1154 = eval_trilinear_point(model, A, B, C, '(1 - 4*(cos(A))**(2))*cos(B - C)', ID="X_1154", classes=["center", "incident"])
            central_line(model, X_265, X_1154, 265, 1154, incident=True)
            X_146 = eval_barycentric_point(model, A, B, C, 'a**(10) + a**(8)*(b**(2) + c**(2)) - a**(6)*(8*b**(4) - 9*b**(2)*c**(2) + 8*c**(4)) + 2*a**(4)*(b**(2) + c**(2))*(4*b**(4) - 7*b**(2)*c**(2) + 4*c**(4)) - a**(2)*(b**(2) - c**(2))**(2)*(b**(4) + 9*b**(2)*c**(2) + c**(4)) - (b**(2) - c**(2))**(4)*(b**(2) + c**(2))', ID="X_146", classes=["center", "incident"])
            X_1531 = eval_trilinear_point(model, A, B, C, '(2*a**(4) - a**(2)*(b**(2) + c**(2)) - b**(4) - c**(4) + 2*b**(2)*c**(2))*(a**(5) - a*(b**(2) - c**(2))**(2) - 4*b*c*(b**(3)*cos(B) + c**(3)*cos(C)))', ID="X_1531", classes=["center", "incident"])
            central_line(model, X_146, X_1531, 146, 1531, incident=True)
            return promote_point_to_top(model, X_3153)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3153 on the given reference triangle."""
            model = create_model('X_3153 = X_328-CEVA CONJUGATE OF X_2')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03153(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03153.json"))
            write_center_svg(
                model_acute,
                here / "x-03153.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
