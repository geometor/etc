X2514¶

RADICAL CENTER OF {CIRCUMCIRCLE, 2nd LEMOINE CIRCLE, (X4,2R)}

The circle (X4,2R) is identified at X2474.
"""GEOMETOR construction model for X_2514 = RADICAL CENTER OF {CIRCUMCIRCLE, 2nd LEMOINE CIRCLE, (X_4,2R)}."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            promote_point_to_top,
            reflection_in_point,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02514(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2514 = RADICAL CENTER OF {CIRCUMCIRCLE, 2nd LEMOINE CIRCLE, (X_4,2R)} on reference triangle ABC."""
            X_669 = eval_trilinear_point(model, A, B, C, 'a**(3)*(b**(2) - c**(2))', ID="X_669", classes=["center"])
            X_2485 = eval_trilinear_point(model, A, B, C, '(a*(b**(2) - c**(2))*(b**(4) + c**(4) - a**(4)))', ID="X_2485", classes=["center"])
            X_2514 = reflection_in_point(model, X_669, X_2485, ID="X_2514", classes=["selected", "target"], guide=False)
            X_325 = eval_trilinear_point(model, A, B, C, '(csc(A))**(2)', ID="X_325", classes=["center", "incident"])
            X_523 = eval_trilinear_point(model, A, B, C, 'sin(B -C)', ID="X_523", classes=["center", "incident"])
            central_line(model, X_325, X_523, 325, 523, incident=True)
            X_512 = eval_trilinear_point(model, A, B, C, 'a*(b**(2) - c**(2))', ID="X_512", classes=["center", "incident"])
            X_1692 = eval_trilinear_point(model, A, B, C, '2*sin(A - 2*omega  ) + sin(A + 2*omega  ) - sin(A)', ID="X_1692", classes=["center", "incident"])
            central_line(model, X_512, X_1692, 512, 1692, incident=True)
            X_525 = eval_trilinear_point(model, A, B, C, '(b*cos(B) - c*cos(C))/a', ID="X_525", classes=["center", "incident"])
            X_2474 = eval_trilinear_point(model, A, B, C, '(a*(b**(4) - c**(4))*(a**(4) + b**(4) + c**(4) + 2*a**(2)*b**(2) + 2*a**(2)*c**(2) - 2*b**(2)*c**(2)))', ID="X_2474", classes=["center", "incident"])
            central_line(model, X_525, X_2474, 525, 2474, incident=True)
            return promote_point_to_top(model, X_2514)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2514 on the given reference triangle."""
            model = create_model('X_2514 = RADICAL CENTER OF {CIRCUMCIRCLE, 2nd LEMOINE CIRCLE, (X_4,2R)}')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02514(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02514.json"))
            write_center_svg(
                model_acute,
                here / "x-02514.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
