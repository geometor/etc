X2501¶

RADICAL CENTER OF {CIRCUMCIRCLE, NINE-POINT CIRCLE, TAYLOR CIRCLE}

X2501 = perspector of circumconic centered at X136
"""GEOMETOR construction model for X_2501 = RADICAL CENTER OF {CIRCUMCIRCLE, NINE-POINT CIRCLE, TAYLOR CIRCLE}."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            ceva_conjugate,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02501(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2501 = RADICAL CENTER OF {CIRCUMCIRCLE, NINE-POINT CIRCLE, TAYLOR CIRCLE} on reference triangle ABC."""
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center"])
            X_136 = eval_barycentric_point(model, A, B, C, '(b**(2) - c**(2))**(2)*(a**(4) + b**(4) + c**(4) - 2*a**(2)*b**(2) - 2*a**(2)*c**(2))/(a**(2) - b**(2) - c**(2))', ID="X_136", classes=["center"])
            X_2501 = ceva_conjugate(model, A, B, C, X_2, X_136, ID="X_2501", classes=["selected", "target"])
            X_230 = eval_trilinear_point(model, A, B, C, 'b*c*(a**(2)*(2*a**(2) - b**(2) - c**(2)) + (b**(2) - c**(2))**(2))', ID="X_230", classes=["center", "incident"])
            X_231 = eval_trilinear_point(model, A, B, C, '(cos(B))*(4*(sin(C))**(2) - 1)*(cos(A)*sin(B) - sin(A)*cos(B)) - (cos(C))*(4*(sin(B))**(2) - 1)*(cos(C)*sin(A) - sin(C)*cos(A))', ID="X_231", classes=["center", "incident"])
            central_line(model, X_230, X_231, 230, 231, incident=True)
            X_112 = eval_trilinear_point(model, A, B, C, 'a/(sin(2*B) - sin(2*C))', ID="X_112", classes=["center", "incident"])
            X_476 = eval_trilinear_point(model, A, B, C, '1/((1 + 2*cos(2*A))*sin(B - C))', ID="X_476", classes=["center", "incident"])
            central_line(model, X_112, X_476, 112, 476, incident=True)
            X_25 = eval_trilinear_point(model, A, B, C, 'sin(A)*tan(A)', ID="X_25", classes=["center", "incident"])
            X_669 = eval_trilinear_point(model, A, B, C, 'a**(3)*(b**(2) - c**(2))', ID="X_669", classes=["center", "incident"])
            central_line(model, X_25, X_669, 25, 669, incident=True)
            return promote_point_to_top(model, X_2501)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2501 on the given reference triangle."""
            model = create_model('X_2501 = RADICAL CENTER OF {CIRCUMCIRCLE, NINE-POINT CIRCLE, TAYLOR CIRCLE}')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02501(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02501.json"))
            write_center_svg(
                model_acute,
                here / "x-02501.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
