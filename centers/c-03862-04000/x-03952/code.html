X244com[INVERSE(n(INCENTRAL TRIANGLE))]

X3952 = anticomplement of X244
"""GEOMETOR construction model for X_3952 = X_244com[INVERSE(n(INCENTRAL TRIANGLE))]."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            isogonal_conjugate,
            nagel_point_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03952(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3952 = X_244com[INVERSE(n(INCENTRAL TRIANGLE))] on reference triangle ABC."""
            X_3733 = eval_barycentric_point(model, A, B, C, 'a**(2)*(b - c)/(b + c)', ID="X_3733", classes=["center"])
            X_3952 = isogonal_conjugate(model, A, B, C, X_3733, ID="X_3952", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_38 = eval_trilinear_point(model, A, B, C, 'b**(2) + c**(2)', ID="X_38", classes=["center", "incident"])
            central_line(model, X_2, X_38, 2, 38, incident=True)
            X_8 = nagel_point_of(model, A, B, C, ID="X_8", classes=["center", "incident"])
            X_80 = eval_trilinear_point(model, A, B, C, '1/(1 - 2*cos(A))', ID="X_80", classes=["center", "incident"])
            central_line(model, X_8, X_80, 8, 80, incident=True)
            X_100 = eval_trilinear_point(model, A, B, C, '1/(b - c)', ID="X_100", classes=["center", "incident"])
            X_190 = eval_trilinear_point(model, A, B, C, 'b*c/(b - c)', ID="X_190", classes=["center", "incident"])
            central_line(model, X_100, X_190, 100, 190, incident=True)
            return promote_point_to_top(model, X_3952)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3952 on the given reference triangle."""
            model = create_model('X_3952 = X_244com[INVERSE(n(INCENTRAL TRIANGLE))]')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03952(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03952.json"))
            write_center_svg(
                model_acute,
                here / "x-03952.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
