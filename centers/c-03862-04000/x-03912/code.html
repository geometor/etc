"""GEOMETOR construction model for X_3912 = X_3747com(INTOUCH TRIANGLE)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            isogonal_conjugate,
            mittenpunkt_of,
            point_from_barycentrics,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03912(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3912 = X_3747com(INTOUCH TRIANGLE) on reference triangle ABC."""
            X_1438 = point_from_barycentrics(model, A, B, C, 0.077220077, 1.1119691, -0.18918919, ID="X_1438", classes=["center"])
            X_3912 = isogonal_conjugate(model, A, B, C, X_1438, ID="X_3912", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            central_line(model, X_1, X_2, 1, 2, incident=True)
            X_9 = mittenpunkt_of(model, A, B, C, ID="X_9", classes=["center", "incident"])
            X_69 = eval_trilinear_point(model, A, B, C, '(cos(A))/a**(2)', ID="X_69", classes=["center", "incident"])
            central_line(model, X_9, X_69, 9, 69, incident=True)
            X_76 = eval_trilinear_point(model, A, B, C, '1/a**(3)', ID="X_76", classes=["center", "incident"])
            X_85 = eval_trilinear_point(model, A, B, C, 'b**(2)*c**(2)/(b + c - a)', ID="X_85", classes=["center", "incident"])
            central_line(model, X_76, X_85, 76, 85, incident=True)
            return promote_point_to_top(model, X_3912)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3912 on the given reference triangle."""
            model = create_model('X_3912 = X_3747com(INTOUCH TRIANGLE)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03912(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03912.json"))
            write_center_svg(
                model_acute,
                here / "x-03912.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
