X1212com[INVERSE(n(INCENTRAL TRIANGLE))]

"""GEOMETOR construction model for X_3991 = X_1212com[INVERSE(n(INCENTRAL TRIANGLE))]."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
            spieker_center_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03991(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3991 = X_1212com[INVERSE(n(INCENTRAL TRIANGLE))] on reference triangle ABC."""
            X_3991 = eval_barycentric_point(model, A, B, C, '(a*(b + c)*(a**(2) + b**(2) + c**(2) - 2*a*b - 2*a*c))', ID="X_3991", classes=["selected", "target"])
            X_10 = spieker_center_of(model, A, B, C, ID="X_10", classes=["center", "incident"])
            X_37 = eval_trilinear_point(model, A, B, C, 'b + c', ID="X_37", classes=["center", "incident"])
            central_line(model, X_10, X_37, 10, 37, incident=True)
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_728 = eval_trilinear_point(model, A, B, C, '((b + c - a)**(3))', ID="X_728", classes=["center", "incident"])
            central_line(model, X_1, X_728, 1, 728, incident=True)
            X_65 = eval_trilinear_point(model, A, B, C, 'cos(B) + cos(C)', ID="X_65", classes=["center", "incident"])
            X_1018 = eval_trilinear_point(model, A, B, C, '((b + c)/(b - c))', ID="X_1018", classes=["center", "incident"])
            central_line(model, X_65, X_1018, 65, 1018, incident=True)
            return promote_point_to_top(model, X_3991)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3991 on the given reference triangle."""
            model = create_model('X_3991 = X_1212com[INVERSE(n(INCENTRAL TRIANGLE))]')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03991(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03991.json"))
            write_center_svg(
                model_acute,
                here / "x-03991.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
