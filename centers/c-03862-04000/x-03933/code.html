X3933¶

X32com[INVERSE(n(TANGENTIAL TRIANGLE))]

X3933 = crossdifference of every pair of points on line X2489X3804
"""GEOMETOR construction model for X_3933 = X_32com[INVERSE(n(TANGENTIAL TRIANGLE))]."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            nine_point_center_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03933(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3933 = X_32com[INVERSE(n(TANGENTIAL TRIANGLE))] on reference triangle ABC."""
            X_3933 = eval_barycentric_point(model, A, B, C, '((b**(2) + c**(2))*(b**(2) + c**(2) - a**(2)))', ID="X_3933", classes=["selected", "target"])
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center", "incident"])
            X_69 = eval_trilinear_point(model, A, B, C, '(cos(A))/a**(2)', ID="X_69", classes=["center", "incident"])
            central_line(model, X_3, X_69, 3, 69, incident=True)
            X_5 = nine_point_center_of(model, A, B, C, ID="X_5", classes=["center", "incident"])
            X_76 = eval_trilinear_point(model, A, B, C, '1/a**(3)', ID="X_76", classes=["center", "incident"])
            central_line(model, X_5, X_76, 5, 76, incident=True)
            X_39 = eval_trilinear_point(model, A, B, C, 'a*(b**(2) + c**(2))', ID="X_39", classes=["center", "incident"])
            X_141 = eval_trilinear_point(model, A, B, C, 'b*c*(b**(2) + c**(2))', ID="X_141", classes=["center", "incident"])
            central_line(model, X_39, X_141, 39, 141, incident=True)
            return promote_point_to_top(model, X_3933)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3933 on the given reference triangle."""
            model = create_model('X_3933 = X_32com[INVERSE(n(TANGENTIAL TRIANGLE))]')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03933(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03933.json"))
            write_center_svg(
                model_acute,
                here / "x-03933.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
