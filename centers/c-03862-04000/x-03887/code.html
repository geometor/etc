X3887¶

X2826com[INVERSE(n(3rd EULER TRIANGLE))]

X3887 = crossdifference of every pair of points on line X6X244
"""GEOMETOR construction model for X_3887 = X_2826com[INVERSE(n(3rd EULER TRIANGLE))]."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            feuerbach_point_of,
            isogonal_conjugate,
            point_from_barycentrics,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03887(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3887 = X_2826com[INVERSE(n(3rd EULER TRIANGLE))] on reference triangle ABC."""
            X_1308 = point_from_barycentrics(model, A, B, C, -0.069444444, 0.91666667, 0.15277778, ID="X_1308", classes=["center"])
            X_3887 = isogonal_conjugate(model, A, B, C, X_1308, ID="X_3887", classes=["selected", "target"])
            X_11 = feuerbach_point_of(model, A, B, C, ID="X_11", classes=["center", "incident"])
            X_116 = eval_trilinear_point(model, A, B, C, 'b*c*((b - c)**(2)*(b**(2) + b*c + c**(2) - a*b - a*c))', ID="X_116", classes=["center", "incident"])
            central_line(model, X_11, X_116, 11, 116, incident=True)
            X_100 = eval_trilinear_point(model, A, B, C, '1/(b - c)', ID="X_100", classes=["center", "incident"])
            X_101 = eval_trilinear_point(model, A, B, C, 'a/(b - c)', ID="X_101", classes=["center", "incident"])
            central_line(model, X_100, X_101, 100, 101, incident=True)
            X_103 = eval_trilinear_point(model, A, B, C, 'a/((a - b)*cot(C) + (a - c)*cot(B))', ID="X_103", classes=["center", "incident"])
            X_104 = eval_trilinear_point(model, A, B, C, '1/(-1 + cos(B) + cos(C))', ID="X_104", classes=["center", "incident"])
            central_line(model, X_103, X_104, 103, 104, incident=True)
            return promote_point_to_top(model, X_3887)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3887 on the given reference triangle."""
            model = create_model('X_3887 = X_2826com[INVERSE(n(3rd EULER TRIANGLE))]')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03887(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03887.json"))
            write_center_svg(
                model_acute,
                here / "x-03887.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
