X3689com(INTOUCH TRIANGLE)

X3911 = complement of X908
"""GEOMETOR construction model for X_3911 = X_3689com(INTOUCH TRIANGLE)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            gergonne_point_of,
            isogonal_conjugate,
            point_from_barycentrics,
            promote_point_to_top,
            spieker_center_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03911(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3911 = X_3689com(INTOUCH TRIANGLE) on reference triangle ABC."""
            X_2316 = point_from_barycentrics(model, A, B, C, -5.3030935e-17, 1, -1.0394063e-16, ID="X_2316", classes=["center"])
            X_3911 = isogonal_conjugate(model, A, B, C, X_2316, ID="X_3911", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_7 = gergonne_point_of(model, A, B, C, ID="X_7", classes=["center", "incident"])
            central_line(model, X_2, X_7, 2, 7, incident=True)
            X_10 = spieker_center_of(model, A, B, C, ID="X_10", classes=["center", "incident"])
            X_56 = eval_trilinear_point(model, A, B, C, 'a/(b + c - a)', ID="X_56", classes=["center", "incident"])
            central_line(model, X_10, X_56, 10, 56, incident=True)
            X_36 = eval_trilinear_point(model, A, B, C, '1 - 2*cos(A)', ID="X_36", classes=["center", "incident"])
            X_80 = eval_trilinear_point(model, A, B, C, '1/(1 - 2*cos(A))', ID="X_80", classes=["center", "incident"])
            central_line(model, X_36, X_80, 36, 80, incident=True)
            return promote_point_to_top(model, X_3911)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3911 on the given reference triangle."""
            model = create_model('X_3911 = X_3689com(INTOUCH TRIANGLE)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03911(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03911.json"))
            write_center_svg(
                model_acute,
                here / "x-03911.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
