X3947¶

X3601com[INVERSE(n(INCENTRAL TRIANGLE))]

"""GEOMETOR construction model for X_3947 = X_3601com[INVERSE(n(INCENTRAL TRIANGLE))]."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            promote_point_to_top,
            spieker_center_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03947(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3947 = X_3601com[INVERSE(n(INCENTRAL TRIANGLE))] on reference triangle ABC."""
            X_3947 = eval_barycentric_point(model, A, B, C, '((b + c)*(a + b - c)*(a - b + c)*(a + 3*b + 3*c))', ID="X_3947", classes=["selected", "target"])
            X_10 = spieker_center_of(model, A, B, C, ID="X_10", classes=["center", "incident"])
            X_12 = eval_trilinear_point(model, A, B, C, '1 + cos(B - C)', ID="X_12", classes=["center", "incident"])
            central_line(model, X_10, X_12, 10, 12, incident=True)
            X_495 = eval_trilinear_point(model, A, B, C, '2 + cos(B - C)', ID="X_495", classes=["center", "incident"])
            X_946 = eval_trilinear_point(model, A, B, C, 'b*c*(a**(3)*(b + c) + (b - c)**(2)*(a**(2) - a*b - a*c - b**(2) - c**(2) - 2*b*c))', ID="X_946", classes=["center", "incident"])
            central_line(model, X_495, X_946, 495, 946, incident=True)
            X_142 = eval_trilinear_point(model, A, B, C, 'b + c - ((b - c)**(2))/a', ID="X_142", classes=["center", "incident"])
            X_1329 = eval_trilinear_point(model, A, B, C, '(b*c*(b**(2)/(a - b + c) + c**(2)/(a + b - c)))', ID="X_1329", classes=["center", "incident"])
            central_line(model, X_142, X_1329, 142, 1329, incident=True)
            return promote_point_to_top(model, X_3947)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3947 on the given reference triangle."""
            model = create_model('X_3947 = X_3601com[INVERSE(n(INCENTRAL TRIANGLE))]')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03947(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03947.json"))
            write_center_svg(
                model_acute,
                here / "x-03947.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
