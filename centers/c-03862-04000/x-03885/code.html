X3885¶

X1com[INVERSE(n(4th EULER TRIANGLE))]

"""GEOMETOR construction model for X_3885 = X_1com[INVERSE(n(4th EULER TRIANGLE))]."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            incenter_of,
            nagel_point_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03885(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3885 = X_1com[INVERSE(n(4th EULER TRIANGLE))] on reference triangle ABC."""
            X_3885 = eval_barycentric_point(model, A, B, C, '(a*(b + c - a)*(b**(2) + c**(2) + a*b + a*c - 3*b*c))', ID="X_3885", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_88 = eval_trilinear_point(model, A, B, C, '1/(b + c - 2*a)', ID="X_88", classes=["center", "incident"])
            central_line(model, X_1, X_88, 1, 88, incident=True)
            X_20 = eval_trilinear_point(model, A, B, C, 'cos(A) - cos(B)*cos(C)', ID="X_20", classes=["center", "incident"])
            X_145 = eval_trilinear_point(model, A, B, C, 'b*c*(3*a - b - c)', ID="X_145", classes=["center", "incident"])
            central_line(model, X_20, X_145, 20, 145, incident=True)
            X_8 = nagel_point_of(model, A, B, C, ID="X_8", classes=["center", "incident"])
            X_210 = eval_trilinear_point(model, A, B, C, '(b + c)*(b + c - a)', ID="X_210", classes=["center", "incident"])
            central_line(model, X_8, X_210, 8, 210, incident=True)
            return promote_point_to_top(model, X_3885)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3885 on the given reference triangle."""
            model = create_model('X_3885 = X_1com[INVERSE(n(4th EULER TRIANGLE))]')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03885(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03885.json"))
            write_center_svg(
                model_acute,
                here / "x-03885.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
