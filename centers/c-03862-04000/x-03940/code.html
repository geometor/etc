X3940¶

X999com[INVERSE(n(TANGENTIAL TRIANGLE))]

"""GEOMETOR construction model for X_3940 = X_999com[INVERSE(n(TANGENTIAL TRIANGLE))]."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            incenter_of,
            nagel_point_of,
            nine_point_center_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03940(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3940 = X_999com[INVERSE(n(TANGENTIAL TRIANGLE))] on reference triangle ABC."""
            X_3940 = eval_barycentric_point(model, A, B, C, '(a*(2*b + 2*c - a)*(b**(2) + c**(2) - a**(2)))', ID="X_3940", classes=["selected", "target"])
            X_5 = nine_point_center_of(model, A, B, C, ID="X_5", classes=["center", "incident"])
            X_8 = nagel_point_of(model, A, B, C, ID="X_8", classes=["center", "incident"])
            central_line(model, X_5, X_8, 5, 8, incident=True)
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center", "incident"])
            X_63 = eval_trilinear_point(model, A, B, C, 'cot(A)', ID="X_63", classes=["center", "incident"])
            central_line(model, X_3, X_63, 3, 63, incident=True)
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_210 = eval_trilinear_point(model, A, B, C, '(b + c)*(b + c - a)', ID="X_210", classes=["center", "incident"])
            central_line(model, X_1, X_210, 1, 210, incident=True)
            return promote_point_to_top(model, X_3940)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3940 on the given reference triangle."""
            model = create_model('X_3940 = X_999com[INVERSE(n(TANGENTIAL TRIANGLE))]')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03940(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03940.json"))
            write_center_svg(
                model_acute,
                here / "x-03940.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
