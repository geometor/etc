X3994¶

X899com[INVERSE(n(INCENTRAL TRIANGLE))]

"""GEOMETOR construction model for X_3994 = X_899com[INVERSE(n(INCENTRAL TRIANGLE))]."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            promote_point_to_top,
            spieker_center_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03994(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3994 = X_899com[INVERSE(n(INCENTRAL TRIANGLE))] on reference triangle ABC."""
            X_3994 = eval_barycentric_point(model, A, B, C, '((b + c)*(a*b + a*c - 2*b*c))', ID="X_3994", classes=["selected", "target"])
            X_10 = spieker_center_of(model, A, B, C, ID="X_10", classes=["center", "incident"])
            X_321 = eval_trilinear_point(model, A, B, C, '(b + c)*b**(2)*c**(2)', ID="X_321", classes=["center", "incident"])
            central_line(model, X_10, X_321, 10, 321, incident=True)
            X_38 = eval_trilinear_point(model, A, B, C, 'b**(2) + c**(2)', ID="X_38", classes=["center", "incident"])
            X_312 = eval_trilinear_point(model, A, B, C, '(b + c - a)*b**(2)*c**(2)', ID="X_312", classes=["center", "incident"])
            central_line(model, X_38, X_312, 38, 312, incident=True)
            X_244 = eval_trilinear_point(model, A, B, C, '(b - c)**(2)', ID="X_244", classes=["center", "incident"])
            X_726 = eval_trilinear_point(model, A, B, C, '(a **(-3)*(b **(-1) + c **(-1)) - a **(-2)*(b **(-2) + c **(-2)))', ID="X_726", classes=["center", "incident"])
            central_line(model, X_244, X_726, 244, 726, incident=True)
            return promote_point_to_top(model, X_3994)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3994 on the given reference triangle."""
            model = create_model('X_3994 = X_899com[INVERSE(n(INCENTRAL TRIANGLE))]')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03994(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03994.json"))
            write_center_svg(
                model_acute,
                here / "x-03994.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
