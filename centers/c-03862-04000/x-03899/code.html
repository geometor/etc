X3899¶

X549com[INVERSE(n(4th EULER TRIANGLE))]

"""GEOMETOR construction model for X_3899 = X_549com[INVERSE(n(4th EULER TRIANGLE))]."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03899(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3899 = X_549com[INVERSE(n(4th EULER TRIANGLE))] on reference triangle ABC."""
            X_3899 = eval_barycentric_point(model, A, B, C, '(a*(2*b**(3) + 2*c**(3) - 2*a**(2)*b - 2*a**(2)*c + 3*a*b*c))', ID="X_3899", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_21 = eval_trilinear_point(model, A, B, C, '1/(cos(B) + cos(C))', ID="X_21", classes=["center", "incident"])
            central_line(model, X_1, X_21, 1, 21, incident=True)
            X_210 = eval_trilinear_point(model, A, B, C, '(b + c)*(b + c - a)', ID="X_210", classes=["center", "incident"])
            X_381 = eval_trilinear_point(model, A, B, C, '2*cos(B - C) - cos(A)', ID="X_381", classes=["center", "incident"])
            central_line(model, X_210, X_381, 210, 381, incident=True)
            X_484 = eval_trilinear_point(model, A, B, C, '1 + 2*(cos(A) - cos(B) - cos(C))', ID="X_484", classes=["center", "incident"])
            X_997 = eval_trilinear_point(model, A, B, C, '(a**(3) - a**(2)*(b + c) - a*(b - c)**(2) + (b + c)*(b**(2) + c**(2)))', ID="X_997", classes=["center", "incident"])
            central_line(model, X_484, X_997, 484, 997, incident=True)
            return promote_point_to_top(model, X_3899)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3899 on the given reference triangle."""
            model = create_model('X_3899 = X_549com[INVERSE(n(4th EULER TRIANGLE))]')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03899(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03899.json"))
            write_center_svg(
                model_acute,
                here / "x-03899.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
