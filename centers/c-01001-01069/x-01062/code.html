X1062¶

INTERSECTION OF LINES X1X_3 AND X5X_33

X1062 = homothetic center of intangents triangle and 2nd Euler triangle
"""GEOMETOR construction model for X_1062 = INTERSECTION OF LINES X_1X_3 AND X_5X_33."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            nine_point_center_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01062(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1062 = INTERSECTION OF LINES X_1X_3 AND X_5X_33 on reference triangle ABC."""
            X_1062 = eval_trilinear_point(model, A, B, C, '(2 - sec(B)*sec(C))', ID="X_1062", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center", "incident"])
            central_line(model, X_1, X_3, 1, 3, incident=True)
            X_5 = nine_point_center_of(model, A, B, C, ID="X_5", classes=["center", "incident"])
            X_33 = eval_trilinear_point(model, A, B, C, '1 + sec(A)', ID="X_33", classes=["center", "incident"])
            central_line(model, X_5, X_33, 5, 33, incident=True)
            X_30 = eval_trilinear_point(model, A, B, C, 'cos(A) - 2*cos(B)*cos(C)', ID="X_30", classes=["center", "incident"])
            X_34 = eval_trilinear_point(model, A, B, C, '1 - sec(A)', ID="X_34", classes=["center", "incident"])
            central_line(model, X_30, X_34, 30, 34, incident=True)
            return promote_point_to_top(model, X_1062)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1062 on the given reference triangle."""
            model = create_model('X_1062 = INTERSECTION OF LINES X_1X_3 AND X_5X_33')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01062(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01062.json"))
            write_center_svg(
                model_acute,
                here / "x-01062.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
