X1055¶

POINT ALIOTH

X1055 = crosssum of X2 and X527
"""GEOMETOR construction model for X_1055 = POINT ALIOTH."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01055(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1055 = POINT ALIOTH on reference triangle ABC."""
            X_1055 = eval_trilinear_point(model, A, B, C, '(a*(b**(2) + c**(2) - 2*a**(2) + a*b + a*c - 2*b*c))', ID="X_1055", classes=["selected", "target"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_41 = eval_trilinear_point(model, A, B, C, 'a**(2)*(b + c - a)', ID="X_41", classes=["center", "incident"])
            central_line(model, X_6, X_41, 6, 41, incident=True)
            X_36 = eval_trilinear_point(model, A, B, C, '1 - 2*cos(A)', ID="X_36", classes=["center", "incident"])
            X_101 = eval_trilinear_point(model, A, B, C, 'a/(b - c)', ID="X_101", classes=["center", "incident"])
            central_line(model, X_36, X_101, 36, 101, incident=True)
            X_187 = eval_trilinear_point(model, A, B, C, 'a*(2*a**(2) - b**(2) - c**(2))', ID="X_187", classes=["center", "incident"])
            X_237 = eval_trilinear_point(model, A, B, C, 'a**(2)*cos(A +  omega  )', ID="X_237", classes=["center", "incident"])
            central_line(model, X_187, X_237, 187, 237, incident=True)
            return promote_point_to_top(model, X_1055)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1055 on the given reference triangle."""
            model = create_model('X_1055 = POINT ALIOTH')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01055(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01055.json"))
            write_center_svg(
                model_acute,
                here / "x-01055.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
