X1002¶

ISOGONAL CONJUGATE OF X1001

X1002 is the isogonal conjugate of X1001.
"""GEOMETOR construction model for X_1002 = ISOGONAL CONJUGATE OF X_1001."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            isogonal_conjugate,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01002(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1002 = ISOGONAL CONJUGATE OF X_1001 on reference triangle ABC."""
            X_1001 = eval_trilinear_point(model, A, B, C, 'a**(2) - a*(b + c) - 2*b*c', ID="X_1001", classes=["center"])
            X_1002 = isogonal_conjugate(model, A, B, C, X_1001, ID="X_1002", classes=["selected", "target"])
            X_42 = eval_trilinear_point(model, A, B, C, 'a*(b + c)', ID="X_42", classes=["center", "incident"])
            X_57 = eval_trilinear_point(model, A, B, C, '1/(b + c - a)', ID="X_57", classes=["center", "incident"])
            central_line(model, X_42, X_57, 42, 57, incident=True)
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_105 = eval_trilinear_point(model, A, B, C, '1/(b**(2) + c**(2) - a*(b + c))', ID="X_105", classes=["center", "incident"])
            central_line(model, X_6, X_105, 6, 105, incident=True)
            X_55 = eval_trilinear_point(model, A, B, C, 'a*(b + c - a)', ID="X_55", classes=["center", "incident"])
            X_81 = eval_trilinear_point(model, A, B, C, '1/(b + c)', ID="X_81", classes=["center", "incident"])
            central_line(model, X_55, X_81, 55, 81, incident=True)
            return promote_point_to_top(model, X_1002)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1002 on the given reference triangle."""
            model = create_model('X_1002 = ISOGONAL CONJUGATE OF X_1001')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01002(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01002.json"))
            write_center_svg(
                model_acute,
                here / "x-01002.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
