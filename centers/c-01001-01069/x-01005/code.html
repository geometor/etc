X1005¶

INTERCEPT OF EULER LINE AND LINE X9X_100

X1005 = isogonal conjugate of X1242
"""GEOMETOR construction model for X_1005 = INTERCEPT OF EULER LINE AND LINE X_9X_100."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            intersect_lines,
            mittenpunkt_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01005(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1005 = INTERCEPT OF EULER LINE AND LINE X_9X_100 on reference triangle ABC."""
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center"])
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center"])
            X_9 = mittenpunkt_of(model, A, B, C, ID="X_9", classes=["center"])
            X_100 = eval_trilinear_point(model, A, B, C, '1/(b - c)', ID="X_100", classes=["center"])
            central_line(model, X_2, X_3, 2, 3)
            central_line(model, X_9, X_100, 9, 100)
            X_1005 = intersect_lines(model, X_2, X_3, X_9, X_100, ID="X_1005", classes=["selected", "target"])
            X_55 = eval_trilinear_point(model, A, B, C, 'a*(b + c - a)', ID="X_55", classes=["center", "incident"])
            X_329 = eval_trilinear_point(model, A, B, C, '( -1 - cos(A) + cos(B) + cos(C))*(csc(A))', ID="X_329", classes=["center", "incident"])
            central_line(model, X_55, X_329, 55, 329, incident=True)
            return promote_point_to_top(model, X_1005)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1005 on the given reference triangle."""
            model = create_model('X_1005 = INTERCEPT OF EULER LINE AND LINE X_9X_100')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01005(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01005.json"))
            write_center_svg(
                model_acute,
                here / "x-01005.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
