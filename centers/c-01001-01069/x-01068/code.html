X1068¶

X46-CROSS CONJUGATE OF X4

X1068 = isogonal conjugate of X1069
"""GEOMETOR construction model for X_1068 = X_46-CROSS CONJUGATE OF X_4."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            orthocenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01068(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1068 = X_46-CROSS CONJUGATE OF X_4 on reference triangle ABC."""
            X_1068 = eval_trilinear_point(model, A, B, C, '(1 - sec(A)*(cos(B) + cos(C)))', ID="X_1068", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_4 = orthocenter_of(model, A, B, C, ID="X_4", classes=["center", "incident"])
            central_line(model, X_1, X_4, 1, 4, incident=True)
            X_24 = eval_trilinear_point(model, A, B, C, 'sec(A)*cos(2*A)', ID="X_24", classes=["center", "incident"])
            X_108 = eval_trilinear_point(model, A, B, C, 'a/(sec(B) - sec(C))', ID="X_108", classes=["center", "incident"])
            central_line(model, X_24, X_108, 24, 108, incident=True)
            X_92 = eval_trilinear_point(model, A, B, C, 'csc(2*A)', ID="X_92", classes=["center", "incident"])
            X_406 = eval_trilinear_point(model, A, B, C, '(b*c*(a + b + c) + a*b*c*sec(A))', ID="X_406", classes=["center", "incident"])
            central_line(model, X_92, X_406, 92, 406, incident=True)
            return promote_point_to_top(model, X_1068)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1068 on the given reference triangle."""
            model = create_model('X_1068 = X_46-CROSS CONJUGATE OF X_4')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01068(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01068.json"))
            write_center_svg(
                model_acute,
                here / "x-01068.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
