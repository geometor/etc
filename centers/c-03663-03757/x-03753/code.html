X3753¶

X2com[INVERSE(n(HEXYL TRIANGLE))]

X3753 = X2-of-inverse(n(hexyl triangle))
"""GEOMETOR construction model for X_3753 = X_2com[INVERSE(n(HEXYL TRIANGLE))]."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            promote_point_to_top,
            spieker_center_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03753(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3753 = X_2com[INVERSE(n(HEXYL TRIANGLE))] on reference triangle ABC."""
            X_3753 = eval_barycentric_point(model, A, B, C, '(a*(b + c)*(a**(2) - b**(2) - c**(2) + 4*b*c))', ID="X_3753", classes=["selected", "target"])
            X_10 = spieker_center_of(model, A, B, C, ID="X_10", classes=["center", "incident"])
            X_12 = eval_trilinear_point(model, A, B, C, '1 + cos(B - C)', ID="X_12", classes=["center", "incident"])
            central_line(model, X_10, X_12, 10, 12, incident=True)
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_392 = eval_trilinear_point(model, A, B, C, '(b + c)*(b**(2) + c**(2) - a**(2)) + 4*a*b*c', ID="X_392", classes=["center", "incident"])
            central_line(model, X_2, X_392, 2, 392, incident=True)
            X_40 = eval_trilinear_point(model, A, B, C, 'cos(B) + cos(C) - cos(A) - 1', ID="X_40", classes=["center", "incident"])
            X_405 = eval_trilinear_point(model, A, B, C, 'b*c*(a + b + c) + a*b*c*cos(A)', ID="X_405", classes=["center", "incident"])
            central_line(model, X_40, X_405, 40, 405, incident=True)
            return promote_point_to_top(model, X_3753)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3753 on the given reference triangle."""
            model = create_model('X_3753 = X_2com[INVERSE(n(HEXYL TRIANGLE))]')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03753(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03753.json"))
            write_center_svg(
                model_acute,
                here / "x-03753.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
