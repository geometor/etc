X3742¶

X3175com[n(INCENTRAL)*n(MEDIAL)]

X3742 = reflection of X3740 in X2
"""GEOMETOR construction model for X_3742 = X_3175com[n(INCENTRAL)*n(MEDIAL)]."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
            reflection_in_point,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03742(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3742 = X_3175com[n(INCENTRAL)*n(MEDIAL)] on reference triangle ABC."""
            X_3740 = eval_barycentric_point(model, A, B, C, 'a*(b**(2) + c**(2) - a*b - a*c + 4*b*c)', ID="X_3740", classes=["center"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center"])
            X_3742 = reflection_in_point(model, X_3740, X_2, ID="X_3742", classes=["selected", "target"], guide=False)
            X_210 = eval_trilinear_point(model, A, B, C, '(b + c)*(b + c - a)', ID="X_210", classes=["center", "incident"])
            central_line(model, X_2, X_210, 2, 210, incident=True)
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_474 = eval_trilinear_point(model, A, B, C, 'cos(A) - (a + b + c)/a', ID="X_474", classes=["center", "incident"])
            central_line(model, X_1, X_474, 1, 474, incident=True)
            X_37 = eval_trilinear_point(model, A, B, C, 'b + c', ID="X_37", classes=["center", "incident"])
            X_982 = eval_trilinear_point(model, A, B, C, '(b**(2) - b*c + c**(2))', ID="X_982", classes=["center", "incident"])
            central_line(model, X_37, X_982, 37, 982, incident=True)
            return promote_point_to_top(model, X_3742)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3742 on the given reference triangle."""
            model = create_model('X_3742 = X_3175com[n(INCENTRAL)*n(MEDIAL)]')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03742(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03742.json"))
            write_center_svg(
                model_acute,
                here / "x-03742.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
