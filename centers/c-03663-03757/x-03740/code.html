X3740¶

X210com[n(INCENTRAL)*n(MEDIAL)]

X3740 = reflection of X3742 in X2
"""GEOMETOR construction model for X_3740 = X_210com[n(INCENTRAL)*n(MEDIAL)]."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            mittenpunkt_of,
            nine_point_center_of,
            promote_point_to_top,
            spieker_center_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03740(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3740 = X_210com[n(INCENTRAL)*n(MEDIAL)] on reference triangle ABC."""
            X_3740 = eval_barycentric_point(model, A, B, C, 'a*(b**(2) + c**(2) - a*b - a*c + 4*b*c)', ID="X_3740", classes=["selected", "target"])
            X_5 = nine_point_center_of(model, A, B, C, ID="X_5", classes=["center", "incident"])
            X_10 = spieker_center_of(model, A, B, C, ID="X_10", classes=["center", "incident"])
            central_line(model, X_5, X_10, 5, 10, incident=True)
            X_37 = eval_trilinear_point(model, A, B, C, 'b + c', ID="X_37", classes=["center", "incident"])
            X_43 = eval_trilinear_point(model, A, B, C, 'a*b + a*c - b*c', ID="X_43", classes=["center", "incident"])
            central_line(model, X_37, X_43, 37, 43, incident=True)
            X_9 = mittenpunkt_of(model, A, B, C, ID="X_9", classes=["center", "incident"])
            X_165 = eval_trilinear_point(model, A, B, C, 'tan(B/2) + tan(C/2) - tan(A/2)', ID="X_165", classes=["center", "incident"])
            central_line(model, X_9, X_165, 9, 165, incident=True)
            return promote_point_to_top(model, X_3740)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3740 on the given reference triangle."""
            model = create_model('X_3740 = X_210com[n(INCENTRAL)*n(MEDIAL)]')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03740(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03740.json"))
            write_center_svg(
                model_acute,
                here / "x-03740.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
