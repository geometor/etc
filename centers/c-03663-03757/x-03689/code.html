X3689¶

X3218com(EXTOUCH TRIANGLE)

X3689 = crossdifference of every pair of points on line X57X1022
"""GEOMETOR construction model for X_3689 = X_3218com(EXTOUCH TRIANGLE)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            mittenpunkt_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03689(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3689 = X_3218com(EXTOUCH TRIANGLE) on reference triangle ABC."""
            X_3689 = eval_barycentric_point(model, A, B, C, '(a*(a - b - c)*(2*a - b - c))', ID="X_3689", classes=["selected", "target"])
            X_9 = mittenpunkt_of(model, A, B, C, ID="X_9", classes=["center", "incident"])
            X_55 = eval_trilinear_point(model, A, B, C, 'a*(b + c - a)', ID="X_55", classes=["center", "incident"])
            central_line(model, X_9, X_55, 9, 55, incident=True)
            X_100 = eval_trilinear_point(model, A, B, C, '1/(b - c)', ID="X_100", classes=["center", "incident"])
            X_518 = eval_trilinear_point(model, A, B, C, '(a*b + a*c - b**(2) - c**(2))', ID="X_518", classes=["center", "incident"])
            central_line(model, X_100, X_518, 100, 518, incident=True)
            X_44 = eval_trilinear_point(model, A, B, C, 'b + c - 2*a', ID="X_44", classes=["center", "incident"])
            X_678 = eval_trilinear_point(model, A, B, C, '((b + c - 2*a)**(2))', ID="X_678", classes=["center", "incident"])
            central_line(model, X_44, X_678, 44, 678, incident=True)
            return promote_point_to_top(model, X_3689)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3689 on the given reference triangle."""
            model = create_model('X_3689 = X_3218com(EXTOUCH TRIANGLE)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03689(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03689.json"))
            write_center_svg(
                model_acute,
                here / "x-03689.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
