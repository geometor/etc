X3739¶

X37com[n(INCENTRAL)*n(MEDIAL)]

X3739 = complement of X37
"""GEOMETOR construction model for X_3739 = X_37com[n(INCENTRAL)*n(MEDIAL)]."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            complement,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            feuerbach_point_of,
            promote_point_to_top,
            spieker_center_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03739(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3739 = X_37com[n(INCENTRAL)*n(MEDIAL)] on reference triangle ABC."""
            X_37 = eval_trilinear_point(model, A, B, C, 'b + c', ID="X_37", classes=["center"])
            X_3739 = complement(model, A, B, C, X_37, ID="X_3739", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            central_line(model, X_2, X_37, 2, 37, incident=True)
            X_11 = feuerbach_point_of(model, A, B, C, ID="X_11", classes=["center", "incident"])
            X_126 = eval_trilinear_point(model, A, B, C, '(b*c*(2*a**(2) - b**(2) - c**(2))*(b**(4) + c**(4) + a**(2)*(b**(2) + c**(2)) - 4*b**(2)*c**(2)))', ID="X_126", classes=["center", "incident"])
            central_line(model, X_11, X_126, 11, 126, incident=True)
            X_10 = spieker_center_of(model, A, B, C, ID="X_10", classes=["center", "incident"])
            X_141 = eval_trilinear_point(model, A, B, C, 'b*c*(b**(2) + c**(2))', ID="X_141", classes=["center", "incident"])
            central_line(model, X_10, X_141, 10, 141, incident=True)
            return promote_point_to_top(model, X_3739)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3739 on the given reference triangle."""
            model = create_model('X_3739 = X_37com[n(INCENTRAL)*n(MEDIAL)]')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03739(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03739.json"))
            write_center_svg(
                model_acute,
                here / "x-03739.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
