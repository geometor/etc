X3701¶

X3216com(EXTOUCH TRIANGLE)

X3701 = excentral-to-ABC barycentric image of X3216
"""GEOMETOR construction model for X_3701 = X_3216com(EXTOUCH TRIANGLE)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            isotomic_conjugate,
            nagel_point_of,
            promote_point_to_top,
            spieker_center_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03701(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3701 = X_3216com(EXTOUCH TRIANGLE) on reference triangle ABC."""
            X_1014 = eval_trilinear_point(model, A, B, C, '(1/((b + c)*(b + c - a)))', ID="X_1014", classes=["center"])
            X_3701 = isotomic_conjugate(model, A, B, C, X_1014, ID="X_3701", classes=["selected", "target"])
            X_8 = nagel_point_of(model, A, B, C, ID="X_8", classes=["center", "incident"])
            X_210 = eval_trilinear_point(model, A, B, C, '(b + c)*(b + c - a)', ID="X_210", classes=["center", "incident"])
            central_line(model, X_8, X_210, 8, 210, incident=True)
            X_12 = eval_trilinear_point(model, A, B, C, '1 + cos(B - C)', ID="X_12", classes=["center", "incident"])
            X_313 = eval_trilinear_point(model, A, B, C, '(b + c)*b**(3)*c**(3)', ID="X_313", classes=["center", "incident"])
            central_line(model, X_12, X_313, 12, 313, incident=True)
            X_10 = spieker_center_of(model, A, B, C, ID="X_10", classes=["center", "incident"])
            X_321 = eval_trilinear_point(model, A, B, C, '(b + c)*b**(2)*c**(2)', ID="X_321", classes=["center", "incident"])
            central_line(model, X_10, X_321, 10, 321, incident=True)
            return promote_point_to_top(model, X_3701)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3701 on the given reference triangle."""
            model = create_model('X_3701 = X_3216com(EXTOUCH TRIANGLE)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03701(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03701.json"))
            write_center_svg(
                model_acute,
                here / "x-03701.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
