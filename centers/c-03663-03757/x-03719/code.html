X3719¶

X1711com(EXTOUCH TRIANGLE)

X3719 = excentral-to-ABC barycentric image of X1711
"""GEOMETOR construction model for X_3719 = X_1711com(EXTOUCH TRIANGLE)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            mittenpunkt_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03719(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3719 = X_1711com(EXTOUCH TRIANGLE) on reference triangle ABC."""
            X_3719 = eval_trilinear_point(model, A, B, C, 'cot(A/2)*(cot(A))**(2)', ID="X_3719", classes=["selected", "target"])
            X_63 = eval_trilinear_point(model, A, B, C, 'cot(A)', ID="X_63", classes=["center", "incident"])
            X_69 = eval_trilinear_point(model, A, B, C, '(cos(A))/a**(2)', ID="X_69", classes=["center", "incident"])
            central_line(model, X_63, X_69, 63, 69, incident=True)
            X_78 = eval_trilinear_point(model, A, B, C, '1/(1 - sec(A))', ID="X_78", classes=["center", "incident"])
            X_212 = eval_trilinear_point(model, A, B, C, '(cos(A))*(1 + cos(A))', ID="X_212", classes=["center", "incident"])
            central_line(model, X_78, X_212, 78, 212, incident=True)
            X_9 = mittenpunkt_of(model, A, B, C, ID="X_9", classes=["center", "incident"])
            X_312 = eval_trilinear_point(model, A, B, C, '(b + c - a)*b**(2)*c**(2)', ID="X_312", classes=["center", "incident"])
            central_line(model, X_9, X_312, 9, 312, incident=True)
            return promote_point_to_top(model, X_3719)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3719 on the given reference triangle."""
            model = create_model('X_3719 = X_1711com(EXTOUCH TRIANGLE)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03719(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03719.json"))
            write_center_svg(
                model_acute,
                here / "x-03719.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
