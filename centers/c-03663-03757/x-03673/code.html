X3501com(INTOUCH TRIANGLE)

X3673 = complement of X25242
"""GEOMETOR construction model for X_3673 = X_3501com(INTOUCH TRIANGLE)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            gergonne_point_of,
            incenter_of,
            orthocenter_of,
            promote_point_to_top,
            spieker_center_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03673(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3673 = X_3501com(INTOUCH TRIANGLE) on reference triangle ABC."""
            X_3673 = eval_barycentric_point(model, A, B, C, '(b*c*(a**(2) + b**(2) + c**(2) - 2*b*c))', ID="X_3673", classes=["selected", "target"])
            X_4 = orthocenter_of(model, A, B, C, ID="X_4", classes=["center", "incident"])
            X_7 = gergonne_point_of(model, A, B, C, ID="X_7", classes=["center", "incident"])
            central_line(model, X_4, X_7, 4, 7, incident=True)
            X_10 = spieker_center_of(model, A, B, C, ID="X_10", classes=["center", "incident"])
            X_75 = eval_trilinear_point(model, A, B, C, '1/a**(2)', ID="X_75", classes=["center", "incident"])
            central_line(model, X_10, X_75, 10, 75, incident=True)
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_85 = eval_trilinear_point(model, A, B, C, 'b**(2)*c**(2)/(b + c - a)', ID="X_85", classes=["center", "incident"])
            central_line(model, X_1, X_85, 1, 85, incident=True)
            return promote_point_to_top(model, X_3673)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3673 on the given reference triangle."""
            model = create_model('X_3673 = X_3501com(INTOUCH TRIANGLE)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03673(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03673.json"))
            write_center_svg(
                model_acute,
                here / "x-03673.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
