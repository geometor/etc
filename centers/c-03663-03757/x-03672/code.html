X3672¶

X1449com(INTOUCH TRIANGLE)

X3672 = anticomplement of X2345
"""GEOMETOR construction model for X_3672 = X_1449com(INTOUCH TRIANGLE)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            gergonne_point_of,
            incenter_of,
            isotomic_conjugate,
            point_from_barycentrics,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03672(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3672 = X_1449com(INTOUCH TRIANGLE) on reference triangle ABC."""
            X_1219 = point_from_barycentrics(model, A, B, C, 0.38277044, 0.32452276, 0.2927068, ID="X_1219", classes=["center"])
            X_3672 = isotomic_conjugate(model, A, B, C, X_1219, ID="X_3672", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_7 = gergonne_point_of(model, A, B, C, ID="X_7", classes=["center", "incident"])
            central_line(model, X_1, X_7, 1, 7, incident=True)
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_37 = eval_trilinear_point(model, A, B, C, 'b + c', ID="X_37", classes=["center", "incident"])
            central_line(model, X_2, X_37, 2, 37, incident=True)
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            X_144 = eval_trilinear_point(model, A, B, C, '(csc(A))*(tan(B/2) + tan(C/2) - tan(A/2))', ID="X_144", classes=["center", "incident"])
            central_line(model, X_6, X_144, 6, 144, incident=True)
            return promote_point_to_top(model, X_3672)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3672 on the given reference triangle."""
            model = create_model('X_3672 = X_1449com(INTOUCH TRIANGLE)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03672(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03672.json"))
            write_center_svg(
                model_acute,
                here / "x-03672.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
