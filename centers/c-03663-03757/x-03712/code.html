X3712¶

X896com(EXTOUCH TRIANGLE)

X3712 = excentral-to-ABC barycentric image of X896
"""GEOMETOR construction model for X_3712 = X_896com(EXTOUCH TRIANGLE)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            nagel_point_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03712(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3712 = X_896com(EXTOUCH TRIANGLE) on reference triangle ABC."""
            X_3712 = eval_barycentric_point(model, A, B, C, '((b + c - a)*(b**(2) + c**(2) - 2*a**(2)))', ID="X_3712", classes=["selected", "target"])
            X_8 = nagel_point_of(model, A, B, C, ID="X_8", classes=["center", "incident"])
            X_21 = eval_trilinear_point(model, A, B, C, '1/(cos(B) + cos(C))', ID="X_21", classes=["center", "incident"])
            central_line(model, X_8, X_21, 8, 21, incident=True)
            X_23 = eval_trilinear_point(model, A, B, C, 'a*(b**(4) + c**(4) - a**(4) - b**(2)*c**(2))', ID="X_23", classes=["center", "incident"])
            X_100 = eval_trilinear_point(model, A, B, C, '1/(b - c)', ID="X_100", classes=["center", "incident"])
            central_line(model, X_23, X_100, 23, 100, incident=True)
            X_522 = eval_trilinear_point(model, A, B, C, '(cos(B) - cos(C))*(csc(A))', ID="X_522", classes=["center", "incident"])
            X_650 = eval_trilinear_point(model, A, B, C, 'cos(B) - cos(C)', ID="X_650", classes=["center", "incident"])
            central_line(model, X_522, X_650, 522, 650, incident=True)
            return promote_point_to_top(model, X_3712)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3712 on the given reference triangle."""
            model = create_model('X_3712 = X_896com(EXTOUCH TRIANGLE)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03712(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03712.json"))
            write_center_svg(
                model_acute,
                here / "x-03712.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
