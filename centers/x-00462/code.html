X(462) = EULER LINE INTERCEPT OF LINE X(51)X(397) — Code¶

Barycentrics u sin A tan A : v sin B tan B : w sin C tan C
"""GEOMETOR construction model for X(462) = EULER LINE INTERCEPT OF LINE X(51)X(397)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            intersect_lines,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00462(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X(462) = EULER LINE INTERCEPT OF LINE X(51)X(397) on reference triangle ABC."""
            X_2 = centroid_of(model, A, B, C, ID="X(2)", classes=["center"])
            X_3 = circumcenter_of(model, A, B, C, ID="X(3)", classes=["center"])
            X_51 = eval_trilinear_point(model, A, B, C, 'a**(2)*cos(B - C)', ID="X(51)", classes=["center"])
            X_397 = eval_trilinear_point(model, A, B, C, 'cos(B - C) - 2*cos(A + pi /3)', ID="X(397)", classes=["center"])
            central_line(model, X_2, X_3, 2, 3)
            central_line(model, X_51, X_397, 51, 397)
            X_462 = intersect_lines(model, X_2, X_3, X_51, X_397, ID="X(462)", classes=["selected", "target"])
            X_184 = eval_trilinear_point(model, A, B, C, 'a**(2)*cos(A)', ID="X(184)", classes=["center", "incident"])
            X_398 = eval_trilinear_point(model, A, B, C, 'cos(B - C) - 2*cos(A - pi /3)', ID="X(398)", classes=["center", "incident"])
            central_line(model, X_184, X_398, 184, 398, incident=True)
            return promote_point_to_top(model, X_462)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X(462) on the given reference triangle."""
            model = create_model('X(462) = EULER LINE INTERCEPT OF LINE X(51)X(397)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00462(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent

            model_acute.save(str(here / "x-00462.json"))
            write_center_svg(
                model_acute,
                here / "x-00462.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
