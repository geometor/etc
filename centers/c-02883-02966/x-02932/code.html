X2932¶

TANGENTIAL-ISOGONAL CONJUGATE OF X36

X2932 is the tangential-isogonal conjugate of X36.
"""GEOMETOR construction model for X_2932 = TANGENTIAL-ISOGONAL CONJUGATE OF X_36."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            nagel_point_of,
            promote_point_to_top,
            reflection_in_point,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02932(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2932 = TANGENTIAL-ISOGONAL CONJUGATE OF X_36 on reference triangle ABC."""
            X_149 = eval_trilinear_point(model, A, B, C, '(b*c*(b**(3) + c**(3) - a**(3) + (a**(2) - b*c)*(b + c) + a*(b*c - b**(2) - c**(2))))', ID="X_149", classes=["center"])
            X_496 = eval_trilinear_point(model, A, B, C, '2 - cos(B - C)', ID="X_496", classes=["center"])
            X_2932 = reflection_in_point(model, X_149, X_496, ID="X_2932", classes=["selected", "target"], guide=False)
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center", "incident"])
            X_8 = nagel_point_of(model, A, B, C, ID="X_8", classes=["center", "incident"])
            central_line(model, X_3, X_8, 3, 8, incident=True)
            X_55 = eval_trilinear_point(model, A, B, C, 'a*(b + c - a)', ID="X_55", classes=["center", "incident"])
            X_215 = eval_trilinear_point(model, A, B, C, '((sin(3*A))*(sin(3*B) + sin(3*C) - sin(3*A)))', ID="X_215", classes=["center", "incident"])
            central_line(model, X_55, X_215, 55, 215, incident=True)
            X_35 = eval_trilinear_point(model, A, B, C, '1 + 2*cos(A)', ID="X_35", classes=["center", "incident"])
            X_392 = eval_trilinear_point(model, A, B, C, '(b + c)*(b**(2) + c**(2) - a**(2)) + 4*a*b*c', ID="X_392", classes=["center", "incident"])
            central_line(model, X_35, X_392, 35, 392, incident=True)
            return promote_point_to_top(model, X_2932)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2932 on the given reference triangle."""
            model = create_model('X_2932 = TANGENTIAL-ISOGONAL CONJUGATE OF X_36')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02932(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02932.json"))
            write_center_svg(
                model_acute,
                here / "x-02932.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
