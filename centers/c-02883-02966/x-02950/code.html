X2950¶

EXCENTRAL-ISOGONAL CONJUGATE OF X36

X2950 is the excentral-isogonal conjugate of X36.
"""GEOMETOR construction model for X_2950 = EXCENTRAL-ISOGONAL CONJUGATE OF X_36."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            intersect_lines,
            mittenpunkt_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02950(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2950 = EXCENTRAL-ISOGONAL CONJUGATE OF X_36 on reference triangle ABC."""
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center"])
            X_104 = eval_trilinear_point(model, A, B, C, '1/(-1 + cos(B) + cos(C))', ID="X_104", classes=["center"])
            X_9 = mittenpunkt_of(model, A, B, C, ID="X_9", classes=["center"])
            X_119 = eval_trilinear_point(model, A, B, C, '(csc(A))*(-1 + cos(B) + cos(C))*(sin(2*B) + sin(2*C) + 2*(-1 + cos(A))*(sin(B) + sin(C)))', ID="X_119", classes=["center"])
            central_line(model, X_1, X_104, 1, 104)
            central_line(model, X_9, X_119, 9, 119)
            X_2950 = intersect_lines(model, X_1, X_104, X_9, X_119, ID="X_2950", classes=["selected", "target"])
            X_84 = eval_trilinear_point(model, A, B, C, '1/(cos(B) + cos(C) - cos(A) - 1)', ID="X_84", classes=["center", "incident"])
            X_952 = eval_trilinear_point(model, A, B, C, 'b*c*(2*a**(4) - 2*a**(3)*(b + c) - a**(2)*(b**(2) - 4*b*c + c**(2)) + (2*a - b - c)*(b - c)*(b**(2) - c**(2)))', ID="X_952", classes=["center", "incident"])
            central_line(model, X_84, X_952, 84, 952, incident=True)
            return promote_point_to_top(model, X_2950)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2950 on the given reference triangle."""
            model = create_model('X_2950 = EXCENTRAL-ISOGONAL CONJUGATE OF X_36')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02950(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02950.json"))
            write_center_svg(
                model_acute,
                here / "x-02950.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
