X2891¶

ANTICOMPLEMENTARY CONJUGATE OF X10

X2891 = anticomplement of X1126
"""GEOMETOR construction model for X_2891 = ANTICOMPLEMENTARY CONJUGATE OF X_10."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            anticomplement,
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_02891(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_2891 = ANTICOMPLEMENTARY CONJUGATE OF X_10 on reference triangle ABC."""
            X_1126 = eval_trilinear_point(model, A, B, C, 'a/(2*a+b+c)', ID="X_1126", classes=["center"])
            X_2891 = anticomplement(model, A, B, C, X_1126, ID="X_2891", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            central_line(model, X_2, X_1126, 2, 1126, incident=True)
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_1654 = eval_trilinear_point(model, A, B, C, '(b*c*(-a**(2) + b**(2) + c**(2) + b*c + c*a + a*b))', ID="X_1654", classes=["center", "incident"])
            central_line(model, X_1, X_1654, 1, 1654, incident=True)
            X_1051 = eval_trilinear_point(model, A, B, C, '(3*a**(2) + b**(2) + c**(2) + b*c + 5*a*b + 5*a*c)', ID="X_1051", classes=["center", "incident"])
            X_1125 = eval_trilinear_point(model, A, B, C, '(2*a+b+c)/a', ID="X_1125", classes=["center", "incident"])
            central_line(model, X_1051, X_1125, 1051, 1125, incident=True)
            return promote_point_to_top(model, X_2891)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_2891 on the given reference triangle."""
            model = create_model('X_2891 = ANTICOMPLEMENTARY CONJUGATE OF X_10')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_02891(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-02891.json"))
            write_center_svg(
                model_acute,
                here / "x-02891.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
