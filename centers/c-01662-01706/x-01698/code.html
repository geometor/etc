X1698¶

INSIMILICENTER(BEVAN CIRCLE, NINE-POINTCIRCLE)

X1698 is the insimilicenter(Bevan circle, nine-pointcircle).
"""GEOMETOR construction model for X_1698 = INSIMILICENTER(BEVAN CIRCLE, NINE-POINTCIRCLE)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            mittenpunkt_of,
            nine_point_center_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01698(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1698 = INSIMILICENTER(BEVAN CIRCLE, NINE-POINTCIRCLE) on reference triangle ABC."""
            X_1698 = eval_trilinear_point(model, A, B, C, '2*cos(B - C) + cos(A) - cos(B) - cos(C) + 1', ID="X_1698", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            central_line(model, X_1, X_2, 1, 2, incident=True)
            X_5 = nine_point_center_of(model, A, B, C, ID="X_5", classes=["center", "incident"])
            X_40 = eval_trilinear_point(model, A, B, C, 'cos(B) + cos(C) - cos(A) - 1', ID="X_40", classes=["center", "incident"])
            central_line(model, X_5, X_40, 5, 40, incident=True)
            X_9 = mittenpunkt_of(model, A, B, C, ID="X_9", classes=["center", "incident"])
            X_46 = eval_trilinear_point(model, A, B, C, 'cos(B) + cos(C) - cos(A)', ID="X_46", classes=["center", "incident"])
            central_line(model, X_9, X_46, 9, 46, incident=True)
            return promote_point_to_top(model, X_1698)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1698 on the given reference triangle."""
            model = create_model('X_1698 = INSIMILICENTER(BEVAN CIRCLE, NINE-POINTCIRCLE)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01698(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01698.json"))
            write_center_svg(
                model_acute,
                here / "x-01698.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
