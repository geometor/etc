X1699¶

EXSIMILICENTER(BEVAN CIRCLE, NINE-POINT CIRCLE)

X1699 is the exsimilicenter(Bevan circle, nine-point circle).
"""GEOMETOR construction model for X_1699 = EXSIMILICENTER(BEVAN CIRCLE, NINE-POINT CIRCLE)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            feuerbach_point_of,
            incenter_of,
            nine_point_center_of,
            orthocenter_of,
            promote_point_to_top,
            reflection_in_point,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01699(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1699 = EXSIMILICENTER(BEVAN CIRCLE, NINE-POINT CIRCLE) on reference triangle ABC."""
            X_165 = eval_trilinear_point(model, A, B, C, 'tan(B/2) + tan(C/2) - tan(A/2)', ID="X_165", classes=["center"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center"])
            X_1699 = reflection_in_point(model, X_165, X_2, ID="X_1699", classes=["selected", "target"], guide=False)
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_4 = orthocenter_of(model, A, B, C, ID="X_4", classes=["center", "incident"])
            central_line(model, X_1, X_4, 1, 4, incident=True)
            X_5 = nine_point_center_of(model, A, B, C, ID="X_5", classes=["center", "incident"])
            X_40 = eval_trilinear_point(model, A, B, C, 'cos(B) + cos(C) - cos(A) - 1', ID="X_40", classes=["center", "incident"])
            central_line(model, X_5, X_40, 5, 40, incident=True)
            X_11 = feuerbach_point_of(model, A, B, C, ID="X_11", classes=["center", "incident"])
            X_57 = eval_trilinear_point(model, A, B, C, '1/(b + c - a)', ID="X_57", classes=["center", "incident"])
            central_line(model, X_11, X_57, 11, 57, incident=True)
            return promote_point_to_top(model, X_1699)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1699 on the given reference triangle."""
            model = create_model('X_1699 = EXSIMILICENTER(BEVAN CIRCLE, NINE-POINT CIRCLE)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01699(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01699.json"))
            write_center_svg(
                model_acute,
                here / "x-01699.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
