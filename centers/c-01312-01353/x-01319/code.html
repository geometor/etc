X1319¶

BEVAN-SCHRÖDER POINT

X1319 = circumcircle-inverse of X56
"""GEOMETOR construction model for X_1319 = BEVAN-SCHRÖDER POINT."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            feuerbach_point_of,
            incenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01319(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1319 = BEVAN-SCHRÖDER POINT on reference triangle ABC."""
            X_1319 = eval_trilinear_point(model, A, B, C, '(b + c - 2*a)/ (b + c - a)', ID="X_1319", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center", "incident"])
            central_line(model, X_1, X_3, 1, 3, incident=True)
            X_11 = feuerbach_point_of(model, A, B, C, ID="X_11", classes=["center", "incident"])
            X_515 = eval_trilinear_point(model, A, B, C, '(b + c)*sec(A) - b*sec(B) - c*sec(C)', ID="X_515", classes=["center", "incident"])
            central_line(model, X_11, X_515, 11, 515, incident=True)
            X_59 = eval_trilinear_point(model, A, B, C, '1/(1 - cos(B - C))', ID="X_59", classes=["center", "incident"])
            X_518 = eval_trilinear_point(model, A, B, C, '(a*b + a*c - b**(2) - c**(2))', ID="X_518", classes=["center", "incident"])
            central_line(model, X_59, X_518, 59, 518, incident=True)
            return promote_point_to_top(model, X_1319)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1319 on the given reference triangle."""
            model = create_model('X_1319 = BEVAN-SCHRÖDER POINT')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01319(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01319.json"))
            write_center_svg(
                model_acute,
                here / "x-01319.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
