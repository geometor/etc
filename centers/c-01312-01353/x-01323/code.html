X1323¶

FLETCHER POINT

X1323 is the point of intersection of the line X1X7 and the trilinear polar of X7. These two lines are orthogonal.
"""GEOMETOR construction model for X_1323 = FLETCHER POINT."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            gergonne_point_of,
            incenter_of,
            promote_point_to_top,
            spieker_center_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01323(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1323 = FLETCHER POINT on reference triangle ABC."""
            X_1323 = eval_trilinear_point(model, A, B, C, '(((sec(A/2))**(2))*(2*(cos(A/2))**(2) - (cos(B/2))**(2) - (cos(C/2))**(2)))', ID="X_1323", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_7 = gergonne_point_of(model, A, B, C, ID="X_7", classes=["center", "incident"])
            central_line(model, X_1, X_7, 1, 7, incident=True)
            X_10 = spieker_center_of(model, A, B, C, ID="X_10", classes=["center", "incident"])
            X_348 = eval_trilinear_point(model, A, B, C, 'cot(A)*(sec(A/2))**(2)', ID="X_348", classes=["center", "incident"])
            central_line(model, X_10, X_348, 10, 348, incident=True)
            X_165 = eval_trilinear_point(model, A, B, C, 'tan(B/2) + tan(C/2) - tan(A/2)', ID="X_165", classes=["center", "incident"])
            X_479 = eval_trilinear_point(model, A, B, C, '(tan(A/2)*sec(A/2))**(2)', ID="X_479", classes=["center", "incident"])
            central_line(model, X_165, X_479, 165, 479, incident=True)
            return promote_point_to_top(model, X_1323)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1323 on the given reference triangle."""
            model = create_model('X_1323 = FLETCHER POINT')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01323(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01323.json"))
            write_center_svg(
                model_acute,
                here / "x-01323.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
