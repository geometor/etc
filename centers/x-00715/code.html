X(715) = EVEN (- 3, - 1) CIRCUMCIRCLE POINT — Code¶

X(715) = isogonal conjugate of X(714)
"""GEOMETOR construction model for X(715) = EVEN (- 3, - 1) CIRCUMCIRCLE POINT."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            isogonal_conjugate,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00715(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X(715) = EVEN (- 3, - 1) CIRCUMCIRCLE POINT on reference triangle ABC."""
            X_714 = eval_trilinear_point(model, A, B, C, 'a**(-4)*(b**(-1)+ c**(-1)) - a**(-2)*(b**(-3)+ c**(-3))', ID="X(714)", classes=["center"])
            X_715 = isogonal_conjugate(model, A, B, C, X_714, ID="X(715)", classes=["selected", "target"])
            X_31 = eval_trilinear_point(model, A, B, C, 'a**(2)', ID="X(31)", classes=["center", "incident"])
            X_99 = eval_trilinear_point(model, A, B, C, 'b*c/(b**(2) - c**(2))', ID="X(99)", classes=["center", "incident"])
            central_line(model, X_31, X_99, 31, 99, incident=True)
            X_110 = eval_trilinear_point(model, A, B, C, 'csc(B - C)', ID="X(110)", classes=["center", "incident"])
            X_560 = eval_trilinear_point(model, A, B, C, 'a**(4)', ID="X(560)", classes=["center", "incident"])
            central_line(model, X_110, X_560, 110, 560, incident=True)
            X_82 = eval_trilinear_point(model, A, B, C, '1/(b**(2) + c**(2))', ID="X(82)", classes=["center", "incident"])
            X_689 = eval_trilinear_point(model, A, B, C, '1/(a**(3)*(b**(4) - c**(4)))', ID="X(689)", classes=["center", "incident"])
            central_line(model, X_82, X_689, 82, 689, incident=True)
            return promote_point_to_top(model, X_715)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X(715) on the given reference triangle."""
            model = create_model('X(715) = EVEN (- 3, - 1) CIRCUMCIRCLE POINT')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00715(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00715.json"))
            write_center_svg(
                model_acute,
                here / "x-00715.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
