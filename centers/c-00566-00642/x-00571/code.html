ORTHOHARMONIC OF X52

X571 = inverse-in-Brocard-circle of X570
"""GEOMETOR construction model for X_571 = ORTHOHARMONIC OF X_52."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            ceva_conjugate,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            orthocenter_of,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00571(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_571 = ORTHOHARMONIC OF X_52 on reference triangle ABC."""
            X_4 = orthocenter_of(model, A, B, C, ID="X_4", classes=["center"])
            X_184 = eval_trilinear_point(model, A, B, C, 'a**(2)*cos(A)', ID="X_184", classes=["center"])
            X_571 = ceva_conjugate(model, A, B, C, X_4, X_184, ID="X_571", classes=["selected", "target"])
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center", "incident"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            central_line(model, X_3, X_6, 3, 6, incident=True)
            X_96 = eval_trilinear_point(model, A, B, C, 'sec(2*A)*sec(B - C)', ID="X_96", classes=["center", "incident"])
            central_line(model, X_4, X_96, 4, 96, incident=True)
            X_66 = eval_trilinear_point(model, A, B, C, 'b*c/(b**(4) + c**(4) - a**(4))', ID="X_66", classes=["center", "incident"])
            X_248 = eval_trilinear_point(model, A, B, C, 'sin(2*A)*sec(A +  omega  )', ID="X_248", classes=["center", "incident"])
            central_line(model, X_66, X_248, 66, 248, incident=True)
            return promote_point_to_top(model, X_571)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_571 on the given reference triangle."""
            model = create_model('X_571 = ORTHOHARMONIC OF X_52')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00571(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00571.json"))
            write_center_svg(
                model_acute,
                here / "x-00571.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
