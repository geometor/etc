X(599) = ISOTOMIC CONJUGATE OF X(598) — Code¶

Let OA be the circle centered at the A-vertex of the Artzt triangle and passing through A; define OB and OC cyclically. X(599) is the radical center of OA, OB, OC. (Randy Hutson, August 30, 2020)
"""GEOMETOR construction model for X(599) = ISOTOMIC CONJUGATE OF X(598)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            midpoint,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_00599(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X(599) = ISOTOMIC CONJUGATE OF X(598) on reference triangle ABC."""
            X_2 = centroid_of(model, A, B, C, ID="X(2)", classes=["center"])
            X_69 = eval_trilinear_point(model, A, B, C, '(cos(A))/a**(2)', ID="X(69)", classes=["center"])
            X_599 = midpoint(model, X_2, X_69, ID="X(599)", classes=["selected", "target"], guide=False)
            X_6 = symmedian_point_of(model, A, B, C, ID="X(6)", classes=["center", "incident"])
            central_line(model, X_2, X_6, 2, 6, incident=True)
            X_3 = circumcenter_of(model, A, B, C, ID="X(3)", classes=["center", "incident"])
            X_67 = eval_trilinear_point(model, A, B, C, 'b*c/(b**(4) + c**(4) - a**(4) - b**(2)*c**(2))', ID="X(67)", classes=["center", "incident"])
            central_line(model, X_3, X_67, 3, 67, incident=True)
            X_76 = eval_trilinear_point(model, A, B, C, '1/a**(3)', ID="X(76)", classes=["center", "incident"])
            X_338 = eval_trilinear_point(model, A, B, C, '(b**(2) - c**(2))**(2)/a**(3)', ID="X(338)", classes=["center", "incident"])
            central_line(model, X_76, X_338, 76, 338, incident=True)
            return promote_point_to_top(model, X_599)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X(599) on the given reference triangle."""
            model = create_model('X(599) = ISOTOMIC CONJUGATE OF X(598)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_00599(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-00599.json"))
            write_center_svg(
                model_acute,
                here / "x-00599.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
