X1082¶

POINT ALPHERATZ

Coordinates for X1082 are obtained from those of X559 by changing π/3 to - π/3; contributed by Edward Brisse.
"""GEOMETOR construction model for X_1082 = POINT ALPHERATZ."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            gergonne_point_of,
            incenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01082(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1082 = POINT ALPHERATZ on reference triangle ABC."""
            X_1082 = eval_trilinear_point(model, A, B, C, '(sec(A/2))*sin(A/2 - pi /3)', ID="X_1082", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center", "incident"])
            central_line(model, X_1, X_3, 1, 3, incident=True)
            X_13 = eval_trilinear_point(model, A, B, C, 'csc(A + pi /3)', ID="X_13", classes=["center", "incident"])
            X_226 = eval_trilinear_point(model, A, B, C, '(csc(A))*(cos(B) + cos(C))', ID="X_226", classes=["center", "incident"])
            central_line(model, X_13, X_226, 13, 226, incident=True)
            X_7 = gergonne_point_of(model, A, B, C, ID="X_7", classes=["center", "incident"])
            X_554 = eval_trilinear_point(model, A, B, C, 'sec(A/2)*csc(A/2 + pi /3)', ID="X_554", classes=["center", "incident"])
            central_line(model, X_7, X_554, 7, 554, incident=True)
            return promote_point_to_top(model, X_1082)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1082 on the given reference triangle."""
            model = create_model('X_1082 = POINT ALPHERATZ')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01082(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01082.json"))
            write_center_svg(
                model_acute,
                here / "x-01082.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
