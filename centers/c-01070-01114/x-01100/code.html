X1100¶

COMPLEMENT OF X319

X1100 is the midpoint of the bicentric pair \(y : z : x\) and \(z : x : y\), where \(x : y : z\) = X37
"""GEOMETOR construction model for X_1100 = COMPLEMENT OF X_319."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_01100(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_1100 = COMPLEMENT OF X_319 on reference triangle ABC."""
            X_1100 = eval_trilinear_point(model, A, B, C, '2*a + b + c', ID="X_1100", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            central_line(model, X_1, X_6, 1, 6, incident=True)
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_319 = eval_trilinear_point(model, A, B, C, '(1 + 2*cos(A))/a**(2)', ID="X_319", classes=["center", "incident"])
            central_line(model, X_2, X_319, 2, 319, incident=True)
            X_86 = eval_trilinear_point(model, A, B, C, 'b*c/(b + c)', ID="X_86", classes=["center", "incident"])
            X_239 = eval_trilinear_point(model, A, B, C, 'b*c*(a**(2) - b*c)', ID="X_239", classes=["center", "incident"])
            central_line(model, X_86, X_239, 86, 239, incident=True)
            return promote_point_to_top(model, X_1100)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_1100 on the given reference triangle."""
            model = create_model('X_1100 = COMPLEMENT OF X_319')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_01100(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-01100.json"))
            write_center_svg(
                model_acute,
                here / "x-01100.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
