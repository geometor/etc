X3787¶

X1196com[INVERSE(nT(a², a² - c²))]

"""GEOMETOR construction model for X_3787 = X_1196com[INVERSE(nT(a², a² - c²))]."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03787(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3787 = X_1196com[INVERSE(nT(a², a² - c²))] on reference triangle ABC."""
            X_3787 = eval_barycentric_point(model, A, B, C, '(a**(2)*(b**(2) + c**(2))*(b**(2) + c**(2) - 3*a**(2)))', ID="X_3787", classes=["selected", "target"])
            X_184 = eval_trilinear_point(model, A, B, C, 'a**(2)*cos(A)', ID="X_184", classes=["center", "incident"])
            X_187 = eval_trilinear_point(model, A, B, C, 'a*(2*a**(2) - b**(2) - c**(2))', ID="X_187", classes=["center", "incident"])
            central_line(model, X_184, X_187, 184, 187, incident=True)
            X_32 = eval_trilinear_point(model, A, B, C, 'a**(3)', ID="X_32", classes=["center", "incident"])
            X_394 = eval_trilinear_point(model, A, B, C, 'cos(A)*cot(A)', ID="X_394", classes=["center", "incident"])
            central_line(model, X_32, X_394, 32, 394, incident=True)
            X_511 = eval_trilinear_point(model, A, B, C, 'cos(A +  omega  )', ID="X_511", classes=["center", "incident"])
            X_1196 = eval_trilinear_point(model, A, B, C, '(a*(b**(4) + c**(4) + a**(2)*b**(2) + a**(2)*c**(2) - 2*b**(2)*c**(2)))', ID="X_1196", classes=["center", "incident"])
            central_line(model, X_511, X_1196, 511, 1196, incident=True)
            return promote_point_to_top(model, X_3787)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3787 on the given reference triangle."""
            model = create_model('X_3787 = X_1196com[INVERSE(nT(a², a² - c²))]')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03787(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03787.json"))
            write_center_svg(
                model_acute,
                here / "x-03787.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
