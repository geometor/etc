X3771¶

X1707com[T(a², a² - b²)]

"""GEOMETOR construction model for X_3771 = X_1707com[T(a², a² - b²)]."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03771(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3771 = X_1707com[T(a², a² - b²)] on reference triangle ABC."""
            X_3771 = eval_barycentric_point(model, A, B, C, '(a**(3) + b**(3) + c**(3) - a**(2)*b - a**(2)*c)', ID="X_3771", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            central_line(model, X_1, X_2, 1, 2, incident=True)
            X_345 = eval_trilinear_point(model, A, B, C, '(csc(A))/(1 - sec(A))', ID="X_345", classes=["center", "incident"])
            X_726 = eval_trilinear_point(model, A, B, C, '(a **(-3)*(b **(-1) + c **(-1)) - a **(-2)*(b **(-2) + c **(-2)))', ID="X_726", classes=["center", "incident"])
            central_line(model, X_345, X_726, 345, 726, incident=True)
            X_55 = eval_trilinear_point(model, A, B, C, 'a*(b + c - a)', ID="X_55", classes=["center", "incident"])
            X_2887 = eval_trilinear_point(model, A, B, C, '(b*c*(b + c)*(b**(2) + c**(2) - b*c))', ID="X_2887", classes=["center", "incident"])
            central_line(model, X_55, X_2887, 55, 2887, incident=True)
            return promote_point_to_top(model, X_3771)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3771 on the given reference triangle."""
            model = create_model('X_3771 = X_1707com[T(a², a² - b²)]')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03771(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03771.json"))
            write_center_svg(
                model_acute,
                here / "x-03771.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
