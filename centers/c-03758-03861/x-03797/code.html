X3797¶

X2com[INVERSE(nT(bc, b²))]

"""GEOMETOR construction model for X_3797 = X_2com[INVERSE(nT(bc, b²))]."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03797(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3797 = X_2com[INVERSE(nT(bc, b²))] on reference triangle ABC."""
            X_3797 = eval_barycentric_point(model, A, B, C, '((a**(2) - b*c)*(b**(2) + c**(2) + b*c))', ID="X_3797", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_37 = eval_trilinear_point(model, A, B, C, 'b + c', ID="X_37", classes=["center", "incident"])
            central_line(model, X_2, X_37, 2, 37, incident=True)
            X_238 = eval_trilinear_point(model, A, B, C, 'a**(2) - b*c', ID="X_238", classes=["center", "incident"])
            X_239 = eval_trilinear_point(model, A, B, C, 'b*c*(a**(2) - b*c)', ID="X_239", classes=["center", "incident"])
            central_line(model, X_238, X_239, 238, 239, incident=True)
            X_194 = eval_trilinear_point(model, A, B, C, 'b*c*(a**(2)*b**(2) + a**(2)*c**(2) - b**(2)*c**(2))', ID="X_194", classes=["center", "incident"])
            X_304 = eval_trilinear_point(model, A, B, C, '(cot(A))*(csc(A))**(2)', ID="X_304", classes=["center", "incident"])
            central_line(model, X_194, X_304, 194, 304, incident=True)
            return promote_point_to_top(model, X_3797)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3797 on the given reference triangle."""
            model = create_model('X_3797 = X_2com[INVERSE(nT(bc, b²))]')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03797(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03797.json"))
            write_center_svg(
                model_acute,
                here / "x-03797.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
