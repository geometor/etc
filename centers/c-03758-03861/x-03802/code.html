X3802¶

X1909com[INVERSE(nT(bc, c²))]

"""GEOMETOR construction model for X_3802 = X_1909com[INVERSE(nT(bc, c²))]."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03802(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3802 = X_1909com[INVERSE(nT(bc, c²))] on reference triangle ABC."""
            X_3802 = eval_barycentric_point(model, A, B, C, '(a*(a**(2) - b*c)**(2)*(b**(2) + c**(2) + b*c))', ID="X_3802", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_39 = eval_trilinear_point(model, A, B, C, 'a*(b**(2) + c**(2))', ID="X_39", classes=["center", "incident"])
            central_line(model, X_1, X_39, 1, 39, incident=True)
            X_238 = eval_trilinear_point(model, A, B, C, 'a**(2) - b*c', ID="X_238", classes=["center", "incident"])
            X_239 = eval_trilinear_point(model, A, B, C, 'b*c*(a**(2) - b*c)', ID="X_239", classes=["center", "incident"])
            central_line(model, X_238, X_239, 238, 239, incident=True)
            X_1682 = eval_trilinear_point(model, A, B, C, '(s*cos(A/2) - r*sin(A/2))**(2)', ID="X_1682", classes=["center", "incident"])
            X_3022 = eval_trilinear_point(model, A, B, C, '(a*(b - c)**(2)*(b + c - a)**(3))', ID="X_3022", classes=["center", "incident"])
            central_line(model, X_1682, X_3022, 1682, 3022, incident=True)
            return promote_point_to_top(model, X_3802)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3802 on the given reference triangle."""
            model = create_model('X_3802 = X_1909com[INVERSE(nT(bc, c²))]')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03802(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03802.json"))
            write_center_svg(
                model_acute,
                here / "x-03802.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
