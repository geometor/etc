X3835¶

X693com(3rd EULER TRIANGLE)

X3835 = complement of X649
"""GEOMETOR construction model for X_3835 = X_693com(3rd EULER TRIANGLE)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            complement,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03835(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3835 = X_693com(3rd EULER TRIANGLE) on reference triangle ABC."""
            X_649 = eval_trilinear_point(model, A, B, C, 'a*(b - c)', ID="X_649", classes=["center"])
            X_3835 = complement(model, A, B, C, X_649, ID="X_3835", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            central_line(model, X_2, X_649, 2, 649, incident=True)
            X_512 = eval_trilinear_point(model, A, B, C, 'a*(b**(2) - c**(2))', ID="X_512", classes=["center", "incident"])
            X_625 = eval_trilinear_point(model, A, B, C, '(2*(b**(4) + c**(4) - b**(2)*c**(2)) - a**(2)*(b**(2) + c**(2)))/a', ID="X_625", classes=["center", "incident"])
            central_line(model, X_512, X_625, 512, 625, incident=True)
            X_514 = eval_trilinear_point(model, A, B, C, '(b - c)/a', ID="X_514", classes=["center", "incident"])
            X_661 = eval_trilinear_point(model, A, B, C, 'cot(B) - cot(C)', ID="X_661", classes=["center", "incident"])
            central_line(model, X_514, X_661, 514, 661, incident=True)
            return promote_point_to_top(model, X_3835)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3835 on the given reference triangle."""
            model = create_model('X_3835 = X_693com(3rd EULER TRIANGLE)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03835(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03835.json"))
            write_center_svg(
                model_acute,
                here / "x-03835.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
