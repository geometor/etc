X3794¶

X2com[INVERSE(nT(-bc, c²))]

"""GEOMETOR construction model for X_3794 = X_2com[INVERSE(nT(-bc, c²))]."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            incenter_of,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03794(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3794 = X_2com[INVERSE(nT(-bc, c²))] on reference triangle ABC."""
            X_3794 = eval_barycentric_point(model, A, B, C, '(a*(a + b)*(a + c)*(b + c - a)*(b**(2) + c**(2) - b*c))', ID="X_3794", classes=["selected", "target"])
            X_1 = incenter_of(model, A, B, C, ID="X_1", classes=["center", "incident"])
            X_21 = eval_trilinear_point(model, A, B, C, '1/(cos(B) + cos(C))', ID="X_21", classes=["center", "incident"])
            central_line(model, X_1, X_21, 1, 21, incident=True)
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_51 = eval_trilinear_point(model, A, B, C, 'a**(2)*cos(B - C)', ID="X_51", classes=["center", "incident"])
            central_line(model, X_2, X_51, 2, 51, incident=True)
            X_210 = eval_trilinear_point(model, A, B, C, '(b + c)*(b + c - a)', ID="X_210", classes=["center", "incident"])
            X_333 = eval_trilinear_point(model, A, B, C, 'b*c*(b + c - a)/(b + c)', ID="X_333", classes=["center", "incident"])
            central_line(model, X_210, X_333, 210, 333, incident=True)
            return promote_point_to_top(model, X_3794)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3794 on the given reference triangle."""
            model = create_model('X_3794 = X_2com[INVERSE(nT(-bc, c²))]')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03794(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03794.json"))
            write_center_svg(
                model_acute,
                here / "x-03794.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
