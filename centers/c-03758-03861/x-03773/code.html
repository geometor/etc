X3773¶

X141com[INVERSE(nT(bc, c²))]

"""GEOMETOR construction model for X_3773 = X_141com[INVERSE(nT(bc, c²))]."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            construct_baseline_triangle,
            create_model,
            eval_barycentric_point,
            eval_trilinear_point,
            nagel_point_of,
            promote_point_to_top,
            spieker_center_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03773(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3773 = X_141com[INVERSE(nT(bc, c²))] on reference triangle ABC."""
            X_3773 = eval_barycentric_point(model, A, B, C, '((b + c)*(b**(2) + c**(2) + b*c))', ID="X_3773", classes=["selected", "target"])
            X_10 = spieker_center_of(model, A, B, C, ID="X_10", classes=["center", "incident"])
            X_37 = eval_trilinear_point(model, A, B, C, 'b + c', ID="X_37", classes=["center", "incident"])
            central_line(model, X_10, X_37, 10, 37, incident=True)
            X_8 = nagel_point_of(model, A, B, C, ID="X_8", classes=["center", "incident"])
            X_238 = eval_trilinear_point(model, A, B, C, 'a**(2) - b*c', ID="X_238", classes=["center", "incident"])
            central_line(model, X_8, X_238, 8, 238, incident=True)
            X_141 = eval_trilinear_point(model, A, B, C, 'b*c*(b**(2) + c**(2))', ID="X_141", classes=["center", "incident"])
            X_726 = eval_trilinear_point(model, A, B, C, '(a **(-3)*(b **(-1) + c **(-1)) - a **(-2)*(b **(-2) + c **(-2)))', ID="X_726", classes=["center", "incident"])
            central_line(model, X_141, X_726, 141, 726, incident=True)
            return promote_point_to_top(model, X_3773)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3773 on the given reference triangle."""
            model = create_model('X_3773 = X_141com[INVERSE(nT(bc, c²))]')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03773(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03773.json"))
            write_center_svg(
                model_acute,
                here / "x-03773.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
