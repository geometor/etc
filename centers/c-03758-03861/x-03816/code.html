X3816¶

X57com(3rd EULER TRIANGLE)

X3816 = complement of X1376
"""GEOMETOR construction model for X_3816 = X_57com(3rd EULER TRIANGLE)."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            complement,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            feuerbach_point_of,
            nine_point_center_of,
            promote_point_to_top,
            spieker_center_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03816(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3816 = X_57com(3rd EULER TRIANGLE) on reference triangle ABC."""
            X_1376 = eval_trilinear_point(model, A, B, C, 'a**(2) - a*b - a*c + 2*b*c', ID="X_1376", classes=["center"])
            X_3816 = complement(model, A, B, C, X_1376, ID="X_3816", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_11 = feuerbach_point_of(model, A, B, C, ID="X_11", classes=["center", "incident"])
            central_line(model, X_2, X_11, 2, 11, incident=True)
            X_10 = spieker_center_of(model, A, B, C, ID="X_10", classes=["center", "incident"])
            X_496 = eval_trilinear_point(model, A, B, C, '2 - cos(B - C)', ID="X_496", classes=["center", "incident"])
            central_line(model, X_10, X_496, 10, 496, incident=True)
            X_5 = nine_point_center_of(model, A, B, C, ID="X_5", classes=["center", "incident"])
            X_515 = eval_trilinear_point(model, A, B, C, '(b + c)*sec(A) - b*sec(B) - c*sec(C)', ID="X_515", classes=["center", "incident"])
            central_line(model, X_5, X_515, 5, 515, incident=True)
            return promote_point_to_top(model, X_3816)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3816 on the given reference triangle."""
            model = create_model('X_3816 = X_57com(3rd EULER TRIANGLE)')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03816(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03816.json"))
            write_center_svg(
                model_acute,
                here / "x-03816.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
