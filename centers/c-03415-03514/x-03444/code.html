X3444¶

X37-VERTEX CONJUGATE OF X37

X3444 = isogonal conjugate of X2895
"""GEOMETOR construction model for X_3444 = X_37-VERTEX CONJUGATE OF X_37."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            isogonal_conjugate,
            point_from_barycentrics,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03444(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3444 = X_37-VERTEX CONJUGATE OF X_37 on reference triangle ABC."""
            X_2895 = point_from_barycentrics(model, A, B, C, 0.49425287, 0.34252874, 0.16321839, ID="X_2895", classes=["center"])
            X_3444 = isogonal_conjugate(model, A, B, C, X_2895, ID="X_3444", classes=["selected", "target"])
            X_35 = eval_trilinear_point(model, A, B, C, '1 + 2*cos(A)', ID="X_35", classes=["center", "incident"])
            X_37 = eval_trilinear_point(model, A, B, C, 'b + c', ID="X_37", classes=["center", "incident"])
            central_line(model, X_35, X_37, 35, 37, incident=True)
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_1029 = eval_trilinear_point(model, A, B, C, '(1/(a + 2*(a + b + c)*cos(A)))', ID="X_1029", classes=["center", "incident"])
            central_line(model, X_2, X_1029, 2, 1029, incident=True)
            X_42 = eval_trilinear_point(model, A, B, C, 'a*(b + c)', ID="X_42", classes=["center", "incident"])
            X_2174 = eval_trilinear_point(model, A, B, C, '(sin(A) + sin(2*A))', ID="X_2174", classes=["center", "incident"])
            central_line(model, X_42, X_2174, 42, 2174, incident=True)
            return promote_point_to_top(model, X_3444)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3444 on the given reference triangle."""
            model = create_model('X_3444 = X_37-VERTEX CONJUGATE OF X_37')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03444(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03444.json"))
            write_center_svg(
                model_acute,
                here / "x-03444.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
