X3458¶

X14-VERTEX CONJUGATE OF X16

Let A' be the free vertex of the equilateral triangle constructed inwardly on BC, and define B' and C' cyclically. X3458 is the barycentric product A'*B'*C'. (Randy Hutson, July 31 2018)
"""GEOMETOR construction model for X_3458 = X_14-VERTEX CONJUGATE OF X_16."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            centroid_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            isogonal_conjugate,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03458(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3458 = X_14-VERTEX CONJUGATE OF X_16 on reference triangle ABC."""
            X_299 = eval_trilinear_point(model, A, B, C, '(csc(A))**(2)*sin(A -  pi  /3)', ID="X_299", classes=["center"])
            X_3458 = isogonal_conjugate(model, A, B, C, X_299, ID="X_3458", classes=["selected", "target"])
            X_2 = centroid_of(model, A, B, C, ID="X_2", classes=["center", "incident"])
            X_14 = eval_trilinear_point(model, A, B, C, 'csc(A -  pi  /3)', ID="X_14", classes=["center", "incident"])
            central_line(model, X_2, X_14, 2, 14, incident=True)
            X_61 = eval_trilinear_point(model, A, B, C, 'sin(A +  pi  /6)', ID="X_61", classes=["center", "incident"])
            X_110 = eval_trilinear_point(model, A, B, C, 'csc(B - C)', ID="X_110", classes=["center", "incident"])
            central_line(model, X_61, X_110, 61, 110, incident=True)
            X_301 = eval_trilinear_point(model, A, B, C, '(csc(A))**(2)*csc(A -  pi  /3)', ID="X_301", classes=["center", "incident"])
            X_308 = eval_trilinear_point(model, A, B, C, 'b**(3)*c**(3)/(b**(2) + c**(2))', ID="X_308", classes=["center", "incident"])
            central_line(model, X_301, X_308, 301, 308, incident=True)
            return promote_point_to_top(model, X_3458)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3458 on the given reference triangle."""
            model = create_model('X_3458 = X_14-VERTEX CONJUGATE OF X_16')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03458(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03458.json"))
            write_center_svg(
                model_acute,
                here / "x-03458.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
