X3442¶

X17-VERTEX CONJUGATE OF X17

X3442 = isogonal conjugate of X633
"""GEOMETOR construction model for X_3442 = X_17-VERTEX CONJUGATE OF X_17."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            isogonal_conjugate,
            point_from_barycentrics,
            promote_point_to_top,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03442(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3442 = X_17-VERTEX CONJUGATE OF X_17 on reference triangle ABC."""
            X_633 = eval_trilinear_point(model, A, B, C, '((S*B*S*C + 2*sqrt(3)*SA*area )/a)', ID="X_633", classes=["center"])
            X_3442 = isogonal_conjugate(model, A, B, C, X_633, ID="X_3442", classes=["selected", "target"])
            X_62 = eval_trilinear_point(model, A, B, C, 'sin(A -  pi  /6)', ID="X_62", classes=["center", "incident"])
            X_184 = eval_trilinear_point(model, A, B, C, 'a**(2)*cos(A)', ID="X_184", classes=["center", "incident"])
            central_line(model, X_62, X_184, 62, 184, incident=True)
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center", "incident"])
            X_303 = eval_trilinear_point(model, A, B, C, '(csc(A))**(2)*sin(A -  pi  /6)', ID="X_303", classes=["center", "incident"])
            central_line(model, X_3, X_303, 3, 303, incident=True)
            X_17 = eval_trilinear_point(model, A, B, C, 'csc(A +  pi  /6)', ID="X_17", classes=["center", "incident"])
            X_1607 = point_from_barycentrics(model, A, B, C, 0.43222192, 0.34376902, 0.22400906, ID="X_1607", classes=["center", "incident"])
            central_line(model, X_17, X_1607, 17, 1607, incident=True)
            return promote_point_to_top(model, X_3442)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3442 on the given reference triangle."""
            model = create_model('X_3442 = X_17-VERTEX CONJUGATE OF X_17')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03442(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03442.json"))
            write_center_svg(
                model_acute,
                here / "x-03442.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
