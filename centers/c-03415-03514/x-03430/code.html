X3430¶

ISOGONAL CONJUGATE OF X3429

X3430 is the isogonal conjugate of X3429.
"""GEOMETOR construction model for X_3430 = ISOGONAL CONJUGATE OF X_3429."""

        from __future__ import annotations

        from pathlib import Path

        import sympy.geometry as spg
        from geometor.model import Model

        from geometor.etc.ops import (
            central_line,
            circumcenter_of,
            construct_baseline_triangle,
            create_model,
            eval_trilinear_point,
            midpoint,
            promote_point_to_top,
            symmedian_point_of,
        )
        from geometor.etc.render import write_center_svg


        def construct_x_03430(
            model: Model, A: spg.Point, B: spg.Point, C: spg.Point
        ) -> spg.Point:
            """Construct X_3430 = ISOGONAL CONJUGATE OF X_3429 on reference triangle ABC."""
            X_20 = eval_trilinear_point(model, A, B, C, 'cos(A) - cos(B)*cos(C)', ID="X_20", classes=["center"])
            X_1330 = eval_trilinear_point(model, A, B, C, '(b*c*(b**(2)/(a + c) + c**(2)/(a + b) - a**(2)/(b + c)))', ID="X_1330", classes=["center"])
            X_3430 = midpoint(model, X_20, X_1330, ID="X_3430", classes=["selected", "target"], guide=False)
            X_3 = circumcenter_of(model, A, B, C, ID="X_3", classes=["center", "incident"])
            X_6 = symmedian_point_of(model, A, B, C, ID="X_6", classes=["center", "incident"])
            central_line(model, X_3, X_6, 3, 6, incident=True)
            X_169 = eval_trilinear_point(model, A, B, C, '- (sin(A))*(cos(A/2))**(2) + (sin(B))*(cos(B/2))**(2) + (sin(C))*(cos(C/2))**(2)', ID="X_169", classes=["center", "incident"])
            X_936 = eval_trilinear_point(model, A, B, C, 'a**(3) - a**(2)*(b + c) - a*(b - c)**(2) + (b + c)**(3)', ID="X_936", classes=["center", "incident"])
            central_line(model, X_169, X_936, 169, 936, incident=True)
            central_line(model, X_20, X_1330, 20, 1330, incident=True)
            return promote_point_to_top(model, X_3430)


        def build_model(
            sides: tuple[float, float, float] = (5.0, 6.0, 7.0),
            show_guides: bool = False,
        ) -> Model:
            """Build a `geometor.model.Model` for X_3430 on the given reference triangle."""
            model = create_model('X_3430 = ISOGONAL CONJUGATE OF X_3429')
            A, B, C = construct_baseline_triangle(model, *sides, show_guides=show_guides)
            construct_x_03430(model, A, B, C)
            return model


        if __name__ == "__main__":
            here = Path(__file__).resolve().parent
            model_acute = build_model((5.0, 6.0, 7.0))
            model_acute.save(str(here / "x-03430.json"))
            write_center_svg(
                model_acute,
                here / "x-03430.svg",
                subtitle="(a, b, c) = (5, 6, 7) | AB = 1",
            )
