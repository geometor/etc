      a^2\*(c^2\*(a^2 + b^2 - c^2)\*p\*q + a^2\*c^2\*q^2 - b^2\*(a^2 - b^2 + c^2)\*p\*r - a^2\*b^2\*r^2) : : ,

      (b^2\*c^2\*(-(c^2\*(a^2 + b^2 - c^2)\*p\*q) - a^2\*c^2\*q^2 + b^2\*(a^2 - b^2 + c^2)\*p\*r + a^2\*b^2\*r^2))/((p + q + r)\*(-(b^2\*c^2\*(b^2 - c^2)\*p) + a^2\*c^2\*(a^2 - c^2)\*q - a^2\*b^2\*(a^2 - b^2)\*r)).
